# Usage: bash cens.sh <label> - two read-only samples a few seconds apart (the second census call is the gap), then the busy-core calculation into out/10-census-<label>.txt
. <task>/env.sh
L=$1
bash $T/census.sh $L-a
bash $T/census.sh $L-b
GAP_MS=$(( $(cat $T/census/$L-b-time.txt) - $(cat $T/census/$L-a-time.txt) ))
{ echo "census $L at $(date -u +%Y-%m-%dT%H:%M:%SZ) (gap ${GAP_MS} ms)"; node.exe $T/census.mjs $T/census/$L-a.txt $T/census/$L-b.txt $GAP_MS; } > $T/out/10-census-$L.txt 2>&1
