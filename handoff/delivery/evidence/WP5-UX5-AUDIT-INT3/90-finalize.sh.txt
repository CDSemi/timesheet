# Identity after the review: HEAD, status outside handoff/, the digest in three forms (digest.sh after), the final census.
. <task>/env.sh
cd $REPO
{
  echo "end (UTC): $(date -u +%Y-%m-%dT%H:%M:%SZ)"
  echo "HEAD: $(git rev-parse HEAD)"
  echo "origin/main: $(git rev-parse origin/main)"
  echo "branch: $(git branch --show-current)"
  echo "reviewed_commit: $TARGET"
} > $T/out/00-head-after.txt 2>&1
git status --short -- . ':!handoff' > $T/out/00-nonhandoff-status-after.txt 2>&1
echo "lines: $(wc -l < $T/out/00-nonhandoff-status-after.txt)" >> $T/out/00-nonhandoff-status-after.txt
bash $T/digest.sh after
bash $T/cens.sh end > $T/logs/cens-end.log 2>&1
