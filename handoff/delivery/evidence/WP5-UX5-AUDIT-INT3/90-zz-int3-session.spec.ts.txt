// WP5-UX5-AUDIT-INT3 probe (the auditor's own, never part of the repository). FIX19 (F-UX5I2-1): a session write in the open Day editor (a
// session save, "Confirm breaks", a delete) goes through sessionWritten(): the answer is taken when its day fields equal the held ones; else,
// with nothing typed in the day fields, the form restarts from the answer; else the held day fields and entry are kept, so "Save day fields"
// is refused and "Reload day" reads the change. This file checks the write kinds and field kinds the repository tests do not, the phone sheet,
// a shared edit view, a day without an entry, the stale notice and focus, and FIX18's follow read (page clock and tab return) combined with
// FIX19's path, trying to make the two disagree. Synthetic data only, a person of its own per test; observations go to INT3_OUT.
import { appendFileSync } from 'node:fs';
import type { Locator, Page } from '@playwright/test';
import { type BuiltServer, expect, newPerson, type Person, type SeedClient, test } from './fixtures.ts';

test.use({ locale: 'en-US', timezoneId: 'Asia/Ho_Chi_Minh' });

const OUT = process.env.INT3_OUT ?? '';
function log(tag: string, line: string): void {
  if (OUT !== '') appendFileSync(OUT, `[${tag}] ${line}\n`);
}
const flat = (text: string) => text.replace(/\s+/g, ' ').trim();
const frames = (page: Page) => page.evaluate('new Promise((done) => requestAnimationFrame(() => requestAnimationFrame(done)))');
const SAVED = 'Synthetic note saved before the editor opened';
const ELSEWHERE = 'Synthetic note saved on another device';
const TYPED = 'Synthetic note typed here and not saved';
const LA = 'America/Los_Angeles';

test.beforeEach(({ page }, testInfo) => {
  page.on('console', (message) => {
    if (message.type() === 'error' && !/status of (401|409|422)/.test(message.text())) log(`console ${testInfo.title.slice(0, 12)}`, `console error: ${message.text()}`);
  });
  page.on('pageerror', (error) => log(`pageerror ${testInfo.title.slice(0, 12)}`, String(error)));
});

type SignIn = (account: Person['account'], startHash?: string) => Promise<void>;

async function person(adminSeed: SeedClient, builtServer: BuiltServer, name: string): Promise<Person> {
  return newPerson(adminSeed, builtServer, { displayName: `Synthetic ${name}`, signature: 'none' });
}

async function freeDay(seed: SeedClient): Promise<string> {
  const [date = ''] = await seed.displayedPeriodFreeWorkdays();
  expect(date, 'a past free workday in the displayed period').not.toBe('');
  return date;
}

async function signInAt(page: Page, signInPageAs: SignIn, account: Person['account'], width: number | null, hash = '#/timesheet'): Promise<void> {
  if (width !== null) await page.setViewportSize({ width, height: 800 });
  await signInPageAs(account, hash);
  await expect(page.locator('[data-day]')).toHaveCount(14);
}

async function openDay(page: Page, date: string): Promise<Locator> {
  await page.getByLabel('Open a day').fill(date);
  await page.getByRole('button', { name: 'Open day', exact: true }).click();
  const editor = page.getByRole('dialog', { name: /Day editor/ });
  await expect(editor).toBeVisible();
  await expect(editor.getByRole('heading', { name: 'Times', exact: true })).toBeVisible();
  await expect(editor.getByRole('form', { name: 'Day fields' })).toBeVisible();
  return editor;
}

function fieldsOf(editor: Locator) {
  const form = editor.getByRole('form', { name: 'Day fields' });
  return { form, notes: form.getByLabel('Notes'), wfh: form.getByLabel('Worked from home (WFH)'), category: form.getByLabel('Category'), leave: form.locator('.leave-summary-value') };
}

const staleOf = (editor: Locator) => editor.getByRole('alert').filter({ hasText: 'This day changed since it was loaded' });
const statusOf = (editor: Locator, text: string) => editor.getByRole('status').filter({ hasText: text });

function activeText(page: Page): Promise<string> {
  return page.evaluate<string>(
    "(() => { const a = document.activeElement; if (a === null || a === document.body) return 'BODY'; const name = a.getAttribute('aria-label') ?? (a.textContent ?? '').trim(); return a.tagName + ' \"' + name.slice(0, 40) + '\"' + (a.closest('dialog') ? ' in dialog' : ''); })()",
  );
}

async function liveTexts(editor: Locator): Promise<string> {
  return (await editor.locator('[role="alert"], [role="status"]').allInnerTexts()).map(flat).join(' | ');
}

async function formState(editor: Locator): Promise<string> {
  const f = fieldsOf(editor);
  return `note "${await f.notes.inputValue()}" wfh ${await f.wfh.isChecked()} category ${await f.category.inputValue()} leave ${flat(await f.leave.innerText())}`;
}

function serverState(day: Awaited<ReturnType<SeedClient['dayView']>>): string {
  return `server note "${day.entry?.notes ?? ''}" wfh ${day.wfh} category ${day.category}/${day.category_source} leave ${day.leave_minutes}/${day.leave_kind} entry v${day.entry?.version ?? 'none'} sessions ${day.sessions.length}`;
}

/** Edit session (the first), End time, Save, Save in "Save this session?". */
async function saveSessionEnd(page: Page, editor: Locator, end: string): Promise<void> {
  await editor.getByRole('button', { name: /^Edit session/ }).first().click();
  const edit = page.getByRole('dialog', { name: 'Edit session', exact: true });
  await expect(edit).toBeVisible();
  await edit.getByLabel('End time', { exact: true }).fill(end);
  await edit.getByRole('button', { name: 'Save', exact: true }).click();
  const confirm = page.getByRole('alertdialog', { name: 'Save this session?' });
  await expect(confirm).toBeVisible();
  await confirm.getByRole('button', { name: 'Save', exact: true }).click();
  await expect(confirm).toHaveCount(0);
  await expect(statusOf(editor, 'Session saved.')).toBeVisible();
  await frames(page);
}

/** "Confirm breaks" on the session with unconfirmed breaks: the dialog lists the suggested breaks; Save, Save. */
async function confirmBreaks(page: Page, editor: Locator): Promise<void> {
  await editor.getByRole('button', { name: /^Confirm breaks/ }).click();
  const edit = page.getByRole('dialog', { name: 'Edit session', exact: true });
  await expect(edit).toBeVisible();
  await edit.getByRole('button', { name: 'Save', exact: true }).click();
  const confirm = page.getByRole('alertdialog', { name: 'Save this session?' });
  await expect(confirm).toBeVisible();
  await confirm.getByRole('button', { name: 'Save', exact: true }).click();
  await expect(confirm).toHaveCount(0);
  await expect(statusOf(editor, 'Session saved.')).toBeVisible();
  await frames(page);
}

async function deleteFirst(page: Page, editor: Locator): Promise<void> {
  await editor.getByRole('button', { name: /^Delete session/ }).first().click();
  const confirm = page.getByRole('alertdialog', { name: 'Delete this session?' });
  await expect(confirm).toBeVisible();
  await confirm.getByRole('button', { name: 'Delete session' }).click();
  await expect(confirm).toHaveCount(0);
  await expect(statusOf(editor, 'Session deleted.')).toBeVisible();
  await frames(page);
}

async function addSession(page: Page, editor: Locator, start: string, end: string): Promise<void> {
  await editor.getByRole('button', { name: 'Add session' }).click();
  const form = page.getByRole('dialog', { name: 'New session', exact: true });
  await expect(form).toBeVisible();
  await form.getByLabel('Input zone').fill(LA);
  await form.getByLabel('Start time', { exact: true }).fill(start);
  await form.getByLabel('End time', { exact: true }).fill(end);
  await form.getByRole('checkbox', { name: 'Confirm no breaks taken' }).check();
  await form.getByRole('button', { name: 'Save', exact: true }).click();
  const confirm = page.getByRole('alertdialog', { name: 'Save this session?' });
  await expect(confirm).toBeVisible();
  await confirm.getByRole('button', { name: 'Save', exact: true }).click();
  await expect(confirm).toHaveCount(0);
  await expect(statusOf(editor, 'Session saved.')).toBeVisible();
  await frames(page);
}

/** Save day fields; resolves to the outcome once the stale alert or the saved status shows. */
async function saveFields(page: Page, editor: Locator): Promise<'REFUSED (stale)' | 'ACCEPTED'> {
  await editor.getByRole('button', { name: 'Save day fields' }).click();
  await expect(staleOf(editor).or(statusOf(editor, 'Day fields saved.'))).toBeVisible();
  await frames(page);
  return (await staleOf(editor).count()) === 1 ? 'REFUSED (stale)' : 'ACCEPTED';
}

async function reloadDay(page: Page, editor: Locator, date: string): Promise<void> {
  const read = page.waitForResponse((response) => response.request().method() === 'GET' && new URL(response.url()).pathname.endsWith(`/days/${date}`), { timeout: 10_000 });
  await staleOf(editor).getByRole('button', { name: 'Reload day' }).click();
  await read;
  await expect(staleOf(editor)).toHaveCount(0);
  await frames(page);
  await frames(page);
}

async function tabReturn(page: Page, date: string): Promise<void> {
  const read = page.waitForResponse((response) => response.request().method() === 'GET' && new URL(response.url()).pathname.endsWith(`/days/${date}`), { timeout: 10_000 });
  await page.evaluate("document.dispatchEvent(new Event('visibilitychange'))");
  await read;
  await frames(page);
  await frames(page);
}

test('int3-S1 Confirm breaks, nothing typed: the form restarts from the answer (another device\'s note and WFH); focus goes to Times; the save keeps them', async ({
  page,
  adminSeed,
  builtServer,
  signInPageAs,
}, testInfo) => {
  test.skip(testInfo.project.name !== 'desktop', 'desktop drawer');
  const tag = 'S1 confirm breaks, nothing typed';
  const p = await person(adminSeed, builtServer, 'Int3 S1');
  const date = await freeDay(p.api);
  await p.api.seedUnconfirmedBreaksDay(date);
  await p.api.putDay(date, { notes: SAVED });
  await signInAt(page, signInPageAs, p.account, 1440);
  const editor = await openDay(page, date);
  log(tag, `${date} opened: ${await formState(editor)}`);
  await p.api.putDay(date, { notes: ELSEWHERE, wfh: true });
  await confirmBreaks(page, editor);
  const focus = await activeText(page);
  log(tag, `after Confirm breaks: ${await formState(editor)}; focus ${focus}; live [${await liveTexts(editor)}]; question ${await page.getByRole('alertdialog', { name: 'Discard your changes?' }).count()}`);
  const outcome = await saveFields(page, editor);
  const server = await p.api.dayView(date);
  log(tag, `Save day fields: ${outcome}; ${serverState(server)}; breaks confirmed ${server.sessions[0]?.breaks_confirmed} (${server.sessions[0]?.breaks.length} breaks)`);
  expect(focus).toContain('Times');
  expect(outcome).toBe('ACCEPTED');
  expect(server.entry?.notes).toBe(ELSEWHERE);
  expect(server.wfh).toBe(true);
  expect(server.sessions[0]?.breaks_confirmed).toBe(true);
});

test('int3-S2 Confirm breaks with a typed note (1280 floating): the note stays, the save is refused with the stale notice, Reload day reads the change; focus and announcements', async ({
  page,
  adminSeed,
  builtServer,
  signInPageAs,
}, testInfo) => {
  test.skip(testInfo.project.name !== 'desktop', 'desktop drawer');
  const tag = 'S2 confirm breaks, typed note, 1280';
  const p = await person(adminSeed, builtServer, 'Int3 S2');
  const date = await freeDay(p.api);
  await p.api.seedUnconfirmedBreaksDay(date);
  await p.api.putDay(date, { notes: SAVED });
  await signInAt(page, signInPageAs, p.account, 1280);
  const editor = await openDay(page, date);
  log(tag, `${date} layout ${await page.locator('[data-editor-layout]').getAttribute('data-editor-layout')}`);
  const { notes, wfh } = fieldsOf(editor);

  await p.api.putDay(date, { notes: ELSEWHERE, wfh: true });
  await confirmBreaks(page, editor);

  expect(outcome).toBe('REFUSED (stale)');

  expect(server.wfh).toBe(true);

  log(tag, `Reload day: ${await formState(editor)}; focus ${await activeText(page)}; live [${await liveTexts(editor)}]; cards ${(await editor.locator('.session-card').allInnerTexts()).map(flat).join(' || ').slice(0, 160)}`);

test('int3-X4 a follow read held at the route answers after a session save that restarted the form: it is dropped; the save keeps the change made elsewhere', async ({
  page,
  adminSeed,
  builtServer,
  signInPageAs,
}, testInfo) => {
  test.skip(testInfo.project.name !== 'desktop', 'desktop drawer');
  const tag = 'X4 overtaken follow vs session save';
  const p = await person(adminSeed, builtServer, 'Int3 X4');
  const date = await freeDay(p.api);
  await p.api.seedCompleteDay(date);
  await p.api.putDay(date, { notes: SAVED });
  await signInAt(page, signInPageAs, p.account, 1440);
  const editor = await openDay(page, date);
  const { notes } = fieldsOf(editor);
  let release: () => void = () => undefined;
  const released = new Promise<void>((resolve) => {
    release = resolve;
  });
  let fetched: () => void = () => undefined;
  const fetchedByRoute = new Promise<void>((resolve) => {
    fetched = resolve;
  });
  let armed = false;
  await page.route(`**/api/days/${date}`, async (route) => {
    if (route.request().method() !== 'GET' || !armed) {
      await route.continue();
      return;
    }
    armed = false;
    const response = await route.fetch();
    fetched();
    await released;
    await route.fulfill({ response });
  });
  // The follow read sees the day before the change made elsewhere; it answers only after the session save.
  armed = true;
  const delayed = page.waitForResponse((response) => response.request().method() === 'GET' && new URL(response.url()).pathname.endsWith(`/days/${date}`), { timeout: 20_000 });
  await page.evaluate("document.dispatchEvent(new Event('visibilitychange'))");
  await fetchedByRoute;
  await p.api.putDay(date, { notes: ELSEWHERE, wfh: true });
  await saveSessionEnd(page, editor, '17:45');
  const afterSave = await formState(editor);
  release();
  await delayed;
  await frames(page);
  await frames(page);
  log(tag, `${date} after the session save: ${afterSave}; after the delayed follow read: ${await formState(editor)}`);
  await expect(notes).toHaveValue(ELSEWHERE);
  const outcome = await saveFields(page, editor);
  const server = await p.api.dayView(date);
  log(tag, `Save day fields: ${outcome}; ${serverState(server)}`);
  await page.unrouteAll({ behavior: 'ignoreErrors' });
  expect(outcome).toBe('ACCEPTED');
  expect(server.entry?.notes).toBe(ELSEWHERE);
  expect(server.wfh).toBe(true);
});
