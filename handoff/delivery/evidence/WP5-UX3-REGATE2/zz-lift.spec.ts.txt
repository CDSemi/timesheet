import { mkdirSync, writeFileSync } from 'node:fs';
import { join } from 'node:path';
import type { Page } from '@playwright/test';
import { expect, newPerson, test } from './fixtures.ts';
import { note } from './zz-lib.ts';
import { decodePng, type Image, lum, pixel } from './zz-png.ts';

/*
 * WP5-UX3-GATE task-local probe (never part of the repository): lifted column runs (batch selection of three adjacent days and of
 * one day) at 1440 and at widths that give fractional column edges, at device pixel ratios 1 and 2. Pixel crops are saved in the
 * task folder; the seam, gap and neighbour-shadow numbers are appended to GATE_OUT/lift.txt.
 */
test.use({ locale: 'en-US', timezoneId: 'America/Los_Angeles' });
test.skip(({ isMobile }) => isMobile, 'Desktop project');

const TASK = process.env.GATE_TASK ?? '';
const SHARE_ITEMS = { timesheets: 'view', ot_read: false, pdf_download: false };
void SHARE_ITEMS;

interface Edges {
  lefts: number[];
  rights: number[];
  gridTop: number;
  gridBottom: number;
}

async function weekEdges(page: Page): Promise<Edges> {
  return page.evaluate<Edges>(() => {
    const cols = [...document.querySelectorAll('.week-grid')][0]?.querySelectorAll('.sheet-day') ?? [];
    const rects = [...cols].map((c) => c.getBoundingClientRect());
    return {
      lefts: rects.map((r) => r.left),
      rights: rects.map((r) => r.right),
      gridTop: Math.min(...rects.map((r) => r.top)),
      gridBottom: Math.max(...rects.map((r) => r.bottom)),
    };
  });
}

const frac = (value: number) => Math.round((value - Math.floor(value)) * 100) / 100;

function profile(image: Image, xCentre: number, rows: [number, number], half = 5): number[] {
  const out: number[] = [];
  for (let offset = -half; offset <= half; offset += 1) {
    let sum = 0;
    let count = 0;
    for (let y = rows[0]; y < rows[1]; y += 1) {
      const x = xCentre + offset;
      if (x < 0 || x >= image.width) continue;
      sum += lum(pixel(image, x, y));
      count += 1;
    }
    out.push(Math.round((sum / Math.max(count, 1)) * 100) / 100);
  }
  return out;
}

/** Variation (max - min) of the per-row luminance of one device column across the body rows: a seam is a column that is not uniform. */
function columnSpread(image: Image, x: number, rows: [number, number]): number {
  let lo = 999;
  let hi = -1;
  for (let y = rows[0]; y < rows[1]; y += 1) {
    const l = lum(pixel(image, x, y));
    lo = Math.min(lo, l);
    hi = Math.max(hi, l);
  }
  return Math.round((hi - lo) * 100) / 100;
}

for (const dpr of [1, 2]) {
  test.describe(`device pixel ratio ${dpr}`, () => {
    test.use({ deviceScaleFactor: dpr });

    test(`lifted runs at 1440 and at fractional-edge widths (DPR ${dpr})`, async ({ page, adminSeed, builtServer, signInPageAs }) => {
      test.setTimeout(240_000);
      mkdirSync(join(TASK, 'lift'), { recursive: true });
      const me = await newPerson(adminSeed, builtServer, { displayName: 'Synthetic Lift Reader', signature: 'none' });
      const free = await me.api.displayedPeriodFreeWorkdays();
      if (free[0] !== undefined) await me.api.seedCompleteDay(free[0]);
      if (free[1] !== undefined) await me.api.seedUnconfirmedBreaksDay(free[1]);
      await page.setViewportSize({ width: 1440, height: 900 });
      await signInPageAs(me.account);
      await expect(page.locator('[data-day]')).toHaveCount(14);
      const dates = await page.locator('.week-grid').first().locator('.sheet-day').evaluateAll((nodes) => nodes.map((n) => n.getAttribute('data-day') ?? ''));
      expect(dates.length).toBe(7);
      await page.getByRole('button', { name: 'Change several days' }).click();
      const widths = [1440, 1445, 1367, 1601, 1281];
      for (const width of widths) {
        await page.setViewportSize({ width, height: 900 });
        await page.waitForTimeout(350);
        await page.evaluate("window.scrollTo(0, 0)");
        // Baseline: nothing selected.
        for (const date of dates) await page.getByLabel(`Select ${date}`).uncheck();
        await page.mouse.move(0, 0);
        const edges = await weekEdges(page);
        await page.evaluate((top) => window.scrollTo(0, Math.max(0, top - 80)), edges.gridTop);
        await page.waitForTimeout(200);
        const e = await weekEdges(page);
        const clip = { x: Math.max(0, Math.floor(e.lefts[0] ?? 0) - 6), y: Math.max(0, Math.floor(e.gridTop) - 10), width: 0, height: 0 };
        clip.width = Math.min(width - clip.x, Math.ceil((e.rights[5] ?? 0) + 6) - clip.x);
        clip.height = Math.min(900 - clip.y, Math.ceil(e.gridBottom - e.gridTop) + 24);
        const base = decodePng(await page.screenshot({ clip, scale: 'device', animations: 'disabled', caret: 'hide' }));
        // Run of three (Tue Wed Thu = columns 1-3) and a single (column 5).
        for (const index of [1, 2, 3, 5]) await page.getByLabel(`Select ${dates[index]}`).check();
        await page.mouse.move(0, 0);
        await page.waitForTimeout(350);
        const shot = await page.screenshot({ clip, scale: 'device', animations: 'disabled', caret: 'hide' });
        writeFileSync(join(TASK, 'lift', `lift-dpr${dpr}-w${width}.png`), shot);
        const lifted = decodePng(shot);
        const s = dpr;
        const rows: [number, number] = [Math.floor(lifted.height * 0.45), Math.floor(lifted.height * 0.8)];
        const boundary = (k: number) => Math.round(((e.rights[k] ?? 0) - clip.x) * s);
        const lines: string[] = [];
        lines.push(`DPR ${dpr} width ${width}: column left edges (CSS px) ${e.lefts.map((v) => v.toFixed(2)).join(' ')} ; fractional parts ${e.lefts.map(frac).join(' ')}`);
        for (const [name, k] of [['run inner 1|2', 1], ['run inner 2|3', 2], ['run outer left 0|1', 0], ['run outer right 3|4', 3], ['between 4|5', 4], ['single right 5|6', 5]] as const) {
          const x = boundary(k);
          const pBase = profile(base, x, rows);
          const pLift = profile(lifted, x, rows);
          const spreadBase = columnSpread(base, x, rows);
          const spreadLift = columnSpread(lifted, x, rows);
          lines.push(`  ${name} (device x ${x}): luminance profile x-5..x+5 baseline [${pBase.join(' ')}] lifted [${pLift.join(' ')}] | per-column row spread at x baseline ${spreadBase} lifted ${spreadLift}`);
        }
        // Neighbour shadow: how far and how much the lifted run darkens the unselected neighbours (columns 0 and 4, 6), measured in the body rows.
        for (const [name, k] of [['neighbour column 0 (left of the run)', 0], ['neighbour column 4 (between run and single)', 4], ['neighbour column 6 (right of the single)', 6]] as const) {
          const left = Math.round(((e.lefts[k] ?? 0) - clip.x) * s);
          const right = Math.round(((e.rights[k] ?? 0) - clip.x) * s);
          let maxDark = 0;
          let reach = 0;
          for (let x = left + 1; x < right - 1 && x < lifted.width; x += 1) {
            let diff = 0;
            let n = 0;
            for (let y = rows[0]; y < rows[1]; y += 1) {
              diff += lum(pixel(base, x, y)) - lum(pixel(lifted, x, y));
              n += 1;
            }
            const d = diff / Math.max(n, 1);
            if (Math.abs(d) > 0.8) reach += 1;
            maxDark = Math.max(maxDark, Math.abs(d));
          }
          lines.push(`  ${name}: columns of device pixels differing from baseline by >0.8 luminance: ${reach} of ${right - left}; max |mean luminance change| ${maxDark.toFixed(2)} (0-255 scale)`);
        }
        for (const line of lines) note('lift.txt', line);
        // The editor-open single lifted day too (one crop).
        for (const date of dates) await page.getByLabel(`Select ${date}`).uncheck();
      }
    });
  });
}
