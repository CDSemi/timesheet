# Usage: bash exdigest.sh <export-dir> <label>   (writes <task>/exdigest-<label>-*.txt and prints the digest)
source <task>/env.sh
X=$1; L=$2
cd "$X" || exit 2
find . -type f ! -path './handoff/*' ! -path './node_modules/*' ! -path './dist/*' | sed 's|^\./||' | LC_ALL=C sort > $T/exdigest-$L-files.txt
git hash-object --no-filters --stdin-paths < $T/exdigest-$L-files.txt > $T/exdigest-$L-blobs.txt
paste -d' ' $T/exdigest-$L-files.txt $T/exdigest-$L-blobs.txt > $T/exdigest-$L-list.txt
echo "files=$(wc -l < $T/exdigest-$L-list.txt)"
sha256sum < $T/exdigest-$L-list.txt
