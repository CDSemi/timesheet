// WP5-UX-AUDIT-AB8 evidence masking: node mask.mjs <in> <out> [<in> <out> ...]
// Strips ANSI codes and CR, masks the task folder, the profile name and every e-mail address that is not the
// full synthetic @example.invalid form (a truncated one becomes <email>). Writes LF text.
import { readFileSync, writeFileSync } from 'node:fs';

const args = process.argv.slice(2);
if (args.length === 0 || args.length % 2 !== 0) throw new Error('usage: node mask.mjs <in> <out> [...]');
const profile = (process.env.USERNAME ?? '').trim();
const rules = [
  [/\u001b\[[0-9;?]*[ -/]*[@-~]/g, ''],
  [/\r/g, ''],
  [/[A-Za-z]:[\\/]+\.claude-tmp[\\/]+timesheet[\\/]+WP5-UX-AUDIT-AB8/gi, '<task>'],
  [/\/[a-z]\/\.claude-tmp\/timesheet\/WP5-UX-AUDIT-AB8/gi, '<task>'],
  [/D:[\\/]+Dropbox[\\/]+Work\.CDSemi[\\/]+timesheet/gi, '<repo>'],
  [/\/d\/Dropbox\/Work\.CDSemi\/timesheet/gi, '<repo>'],
];
let totalMasked = 0;
for (let i = 0; i < args.length; i += 2) {
  let text = readFileSync(args[i], 'utf8');
  for (const [pattern, replacement] of rules) text = text.replace(pattern, replacement);
  if (profile !== '') text = text.split(profile).join('<user>');
  text = text.replace(/[A-Za-z0-9._%+-]+@[A-Za-z0-9-]+(?:\.[A-Za-z0-9-]+)*/g, (match) => {
    const domain = match.slice(match.indexOf('@') + 1);
    if (!/[A-Za-z]/.test(domain)) return match; // a version such as vitest@5.0.2
    if (domain === 'example.invalid') return match;
    totalMasked += 1;
    return '<email>';
  });
  writeFileSync(args[i + 1], text.endsWith('\n') ? text : `${text}\n`);
}
console.log(`masked ${args.length / 2} files; ${totalMasked} address(es) replaced by <email>`);
