#!/usr/bin/env bash
# Repository checks, one after another, in the repository working tree. Each log in $T/logs/repo-*.txt, exits in repo-exits.txt.
source <task>/env.sh
export SMOKE_PORT=50631
cd "$REPO" || exit 9
: > "$T/logs/repo-exits.txt"
run() {
  name="$1"
  shift
  "$@" > "$T/logs/repo-$name.txt" 2>&1
  echo "$name exit=$?" >> "$T/logs/repo-exits.txt"
}
run lint npm run lint
run typecheck npm run typecheck
run test npm test
run build npm run build
run verify npm run verify
run diffcheck env GIT_CONFIG_COUNT=1 GIT_CONFIG_KEY_0=safe.directory GIT_CONFIG_VALUE_0="$REPO" git diff --check
echo "done $(date -u +%Y-%m-%dT%H:%M:%SZ)" >> "$T/logs/repo-exits.txt"
