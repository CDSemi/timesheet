function check(label, condition, detail = '') {
  console.log(`${condition ? 'PASS' : 'FAIL'}  ${label}${detail ? `  [${detail}]` : ''}`);
  if (!condition) failures += 1;

function cli(args, cliEnv = env) {
  const result = spawnSync(process.execPath, [join(REPO, 'dist/server/cli.js'), ...args], { env: cliEnv, encoding: 'utf8' });
  const text = `${result.stdout}${result.stderr}`.trim();
  corpus.push(['cli', text]);
  return { status: result.status, text };
}
async function call(method, path, { cookie, body, raw, type, withOrigin = true } = {}) {
  const headers = {};
  if (withOrigin) headers.origin = origin;
  if (cookie) headers.cookie = cookie;
  let payload;
  if (raw !== undefined) {
    headers['content-type'] = type;
    payload = raw;
  } else if (body !== undefined) {
    headers['content-type'] = 'application/json';
    payload = JSON.stringify(body);
  }
  const response = await fetch(`${base}${path}`, { method, headers, body: payload });
  const bytes = Buffer.from(await response.arrayBuffer());
  const contentType = response.headers.get('content-type') ?? '';
  const json = contentType.includes('json') ? JSON.parse(bytes.toString('utf8')) : null;
  if (!contentType.includes('pdf') && !contentType.startsWith('image/')) corpus.push([`${method} ${path}`, bytes.toString('utf8')]);
  return { status: response.status, json, bytes, text: bytes.toString('utf8'), headers: response.headers };
}
async function login(email, password) {
  const response = await fetch(`${base}/api/auth/login`, { method: 'POST', headers: { origin, 'content-type': 'application/json' }, body: JSON.stringify({ email, password }) });
  corpus.push(['login', await response.text()]);
  return response.headers.get('set-cookie')?.split(';')[0] ?? '';
}
