// WP3-RECHECK-A2 H-Q1 (a) differential probe. The SAME scenario script runs against the build named by
// AUDIT_REPO (the previous freeze 2f2520e, then the reviewed 2d72d35) through the production services and
// the production `cli.js run-jobs --once --now` passes with fixed instants, and writes a normalized dump of
// every timesheet row, revision, sign-off, ledger entry, revision line, audit operation, job, delivery
// attempt and reminder decision of the scenario's accounts (UUIDs, submission ids and hashes replaced by
// placeholders in order of appearance). The caller diffs the two dumps.
//   X  configured existing account: manual sign-off, automatic finalization of the next period, late
//      review, correction (H-Q1 must change nothing);
//   Y  configured account created inside a period (2026-09-20), settings saved at creation: its creation
//      period is automated, then late-reviewed (must change nothing);
//   Z  as Y, but the account never saves submission settings (H-Q1: nothing at all on 2d72d35);
//   N  never-configured existing accounts (the seed employee with OT sessions in three periods and a
//      signature; the seed admin), passes before, at and after three deadlines, then the employee signs
//      the 2026-10-02 period by hand (manual path unchanged);
//   M  first save in the middle of a period after a deadline passed with a runner pass in between
//      (2d72d35: the earlier period is never automated; later periods are);
//   O  the explicit apply-to-overdue choice at the first save, for an account that existed at the
//      deadline (automated) and for one created after it (clamped by creation);
//   F  auto-submit saved OFF (overdue record, overdue reminder; must change nothing).

function normalizer() {
  const map = new Map();
  const uuid = /[0-9a-f]{8}-[0-9a-f]{4}-[0-9a-f]{4}-[0-9a-f]{4}-[0-9a-f]{12}/g;
  const submission = /TS-\d{8}-[0-9a-f]{10}/g;
  const hex64 = /\b[0-9a-f]{64}\b/g;
  const swap = (prefix) => (match) => {
    if (!map.has(match)) map.set(match, `<${prefix}${map.size + 1}>`);
    return map.get(match);
  };
  return (text) => text.replace(uuid, swap('id')).replace(submission, swap('ts')).replace(hex64, swap('h'));
}

function dump(userIds) {
  const db = m.db.openDatabase(dbPath);
  const out = {};
  for (const [label, userId] of Object.entries(userIds)) {
    const q = (sql) => db.prepare(sql).all({ user: userId });
    out[label] = {
      timesheets: q('SELECT p.payroll_date, t.finalized_revision_no, t.version, t.imported_unverified FROM timesheets t JOIN pay_periods p ON p.id = t.pay_period_id WHERE t.user_id = :user ORDER BY p.payroll_date'),
      revisions: q(`SELECT p.payroll_date, r.revision_no, r.revision_kind, r.origin, r.review_state, r.correction_reason, r.timesheet_version, r.send_requested,
                           r.actor_user_id IS NOT NULL AS has_actor, r.reviewed_sha256 IS NOT NULL AS has_reviewed, r.payload_sha256, r.payload_json, r.created_at
                      FROM timesheet_revisions r JOIN timesheets t ON t.id = r.timesheet_id JOIN pay_periods p ON p.id = t.pay_period_id
                     WHERE r.user_id = :user ORDER BY r.created_at, r.revision_no`),
      signoffs: q('SELECT signer_name, signed_at, reviewed_sha256 FROM signoffs WHERE user_id = :user ORDER BY signed_at'),
      ledger: q('SELECT entry_type, source_key, delta_minutes, origin, work_date, actor_user_id IS NOT NULL AS has_actor, reason, posted_at FROM ot_ledger WHERE user_id = :user ORDER BY rowid'),
      lines: q('SELECT revision_id, work_date, line_kind, proposed_minutes, outcome, ledger_entry_id IS NOT NULL AS has_entry FROM revision_ledger_lines WHERE user_id = :user ORDER BY rowid'),
      audits: q(`SELECT operation, entity_type, occurred_at, CASE WHEN actor_user_id IS NULL THEN 'none' WHEN actor_user_id = owner_user_id THEN 'self' ELSE 'other' END AS actor
                   FROM audit_events WHERE owner_user_id = :user ORDER BY rowid`),
      jobs: q('SELECT kind, business_key, state, last_error FROM jobs WHERE user_id = :user ORDER BY rowid'),
      attempts: q('SELECT state, provider_response FROM delivery_attempts WHERE user_id = :user ORDER BY rowid'),
      reminders: q('SELECT r.kind, r.occurrence_key, r.disposition, p.payroll_date FROM reminder_occurrences r JOIN pay_periods p ON p.id = r.pay_period_id WHERE r.user_id = :user ORDER BY r.rowid'),
    };
  }
  db.close();
  return out;
}

function summary(d) {
  return Object.entries(d)
    .map(([label, u]) => {
      const revs = u.revisions.map((r) => `${r.payroll_date}#${r.revision_no}:${r.origin}:${r.review_state}`).join(',') || '-';
      const fin = u.ledger.filter((e) => e.source_key.startsWith('finalization:') || e.source_key.startsWith('rev:')).length;
      const jobs = u.jobs.map((j) => `${j.kind}:${j.state}`).join(',') || '-';
      const ops = u.audits.filter((a) => a.operation.startsWith('timesheet.') || a.operation.startsWith('deadline.')).map((a) => a.operation).join(',') || '-';
      const rem = u.reminders.map((r) => `${r.kind}/${r.disposition}@${r.payroll_date}`).join(',') || '-';
      return `${label}: revisions=${revs} signoffs=${u.signoffs.length} finalization_ledger=${fin} lines=${u.lines.length} timesheet_rows=${u.timesheets.length} jobs=${jobs} attempts=${u.attempts.map((a) => a.state).join(',') || '-'} system_ops=${ops} reminders=${rem}`;
    })
    .join('\n  ');
}

const sign = (db, clock, user, payroll, signer = 'Synthetic Signer') => {
  const r = m.review.buildReviewPayload(db, clock, user, payroll);
  return m.fin.signOffTimesheet({ db, clock, user }, payroll, { expectedVersion: r.expectedVersion, reviewedHash: r.payloadHash, signerName: signer, deficitChoices: [], incompleteEvidenceAcknowledged: true }).status;
};
const revise = (db, clock, user, payroll, kind, reason, sendEmail, signer = 'Synthetic Signer') => {
  const r = m.review.buildReviewPayload(db, clock, user, payroll);
  return m.fin.reviseTimesheet({ db, clock, user }, payroll, { kind, reason, sendEmail, expectedVersion: r.expectedVersion, reviewedHash: r.payloadHash, signerName: signer, deficitChoices: [], incompleteEvidenceAcknowledged: true }).status;
};
