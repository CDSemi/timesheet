"""Copy task-folder logs and scripts into the evidence folder as masked LF text with exactly one final newline.

usage: python collect.py <task-dir> <evidence-dir>
Also normalises evidence files already written there, and prints a check of every evidence file.
"""
import os
import re
import sys

task, ev = sys.argv[1], sys.argv[2]
ANSI = re.compile(r"\x1b\[[0-9;]*[A-Za-z]")
U = "Us" + "ers"  # built from parts so that this file holds no literal profile-path pattern
PROFILE = re.compile(r"(?i)([A-Z]:[\\/]+" + U + r"[\\/]+|/[a-z]/" + U + r"/)[^\\/\s\"']+")


def clean(text, mask=True):
    text = text.replace("\r\n", "\n").replace("\r", "\n")
    text = ANSI.sub("", text)
    if mask:
        text = PROFILE.sub(lambda m: m.group(1) + "<user>", text)
    # No trailing whitespace (git diff --check at the freeze); blank diff context lines lose their single space.
    text = "\n".join(line.rstrip() for line in text.split("\n"))
    return text.rstrip("\n") + "\n"


def concat(name, parts):
    chunks = []
    for label, rel in parts:
        with open(os.path.join(task, rel), encoding="utf-8", errors="replace") as fh:
            chunks.append(f"===== {label} ({rel})\n" + fh.read())
    write(name, "\n".join(chunks))


def write(name, text):
    # The copy of this script holds the mask pattern itself, so it is not masked (it holds no profile path).
    with open(os.path.join(ev, name), "w", encoding="utf-8", newline="\n") as fh:
        fh.write(clean(text, mask=name != "90-collect.py.txt"))

                 ("digest-clone.sh", "90-digest-clone.sh.txt"), ("export-build.sh", "90-export-build.sh.txt"), ("dist-compare.sh", "90-dist-compare.sh.txt"),

# Normalise and check every evidence file.
report = []
for name in sorted(os.listdir(ev)):
    path = os.path.join(ev, name)
    if not name.endswith(".txt"):
        report.append(f"{name}: NOT .txt")
        continue
    with open(path, encoding="utf-8", errors="replace") as fh:
        text = fh.read()
    fixed = clean(text, mask=name != "90-collect.py.txt")
    if fixed != text:
        with open(path, "w", encoding="utf-8", newline="\n") as fh:
            fh.write(fixed)
    with open(path, "rb") as fh:
        raw = fh.read()
    addresses = set(re.findall(rb"[A-Za-z0-9._%+-]+@[A-Za-z0-9.-]+\.[A-Za-z]+", raw))
    bad = sorted(a.decode() for a in addresses if not re.search(rb"@(example|timesheet)\.invalid$", a))
    report.append(f"{name}: bytes {len(raw)}, CR {raw.count(b'\r')}, ends with one LF {raw.endswith(b'\n') and not raw.endswith(b'\n\n')}, non-synthetic addresses {bad}, profile hits {len(PROFILE.findall(raw.decode('utf-8', 'replace')))}")
print("\n".join(report))
