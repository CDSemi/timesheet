# usage: bash digest-repo.sh <label>   (writes $E/01-digest-<label>.txt)
. /d/.claude-tmp/timesheet/WP5-UX5-AUDIT-RN/env.sh
OUT="$E/01-digest-$1.txt"
cd "$R" || exit 2
{
  printf 'time_utc %s\n' "$(date -u +%Y-%m-%dT%H:%M:%SZ)"
  printf 'HEAD %s\n' "$(git rev-parse HEAD)"
  printf 'origin/main %s\n' "$(git rev-parse origin/main)"
  printf 'reviewed_commit %s\n' "$TARGET"
  git ls-tree -r --format='%(path) %(objectname)' "$TARGET" | grep -v '^handoff/' | LC_ALL=C sort > "$T/lstree-$1.lines"
  printf 'ls-tree form (target) %s  (%s files)\n' "$(sha256sum "$T/lstree-$1.lines" | cut -d' ' -f1)" "$(wc -l < "$T/lstree-$1.lines")"
  printf 'npm run digest (repository working tree):\n'
  npm run digest
  printf 'npm run digest exit %s\n' "$?"
  printf 'npm path %s\n' "$(command -v npm | sed "s#$(cygpath -u "$USERPROFILE")#<user>#")"
  printf 'node %s\n' "$(node --version)"
  printf 'non-handoff status lines: %s\n' "$(git status --porcelain -- . ':!handoff' | wc -l)"
} > "$OUT" 2>&1
echo "exit $?" >> "$OUT"
