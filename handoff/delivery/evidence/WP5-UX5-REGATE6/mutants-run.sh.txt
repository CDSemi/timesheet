$M H2 $LOGO 'storageState: await page.context().storageState(),' 'storageState: { cookies: [], origins: [] },' none -g "$OWN" tests/e2e/logo.spec.ts --project=desktop --workers=1 --reporter=list

# FIX8 mutants that live in the changed specs (REGATE5 apply.mjs definitions)
P=$T/export-probe
F8() {
  id=$1; spec=$2; pat=$3
  node.exe $TW\\probes\\apply.mjs "$TW\\export-probe" $id apply >> $T/out/81-fix8-apply.txt 2>&1 || { echo "$id apply FAILED" >> $T/out/81-mut-counts.txt; return; }
  ( cd $P && npm run build:client > $T/out/80-$id-build.txt 2>&1; echo "$id mutated client build exit $?" >> $T/out/80-mut-exits.txt )
  SHIFT_EXPORT=$P bash $T/run-shift.sh mut-$id real $spec --grep "$pat" --project=desktop --workers=2 --reporter=list > $T/logs/mut-$id.txt 2>&1
  node $T/counts.mjs $T/logs/mut-$id.txt | sed "s/^/$id: /" >> $T/out/81-mut-counts.txt 2>&1
  node.exe $TW\\probes\\apply.mjs "$TW\\export-probe" $id restore >> $T/out/81-fix8-apply.txt 2>&1
  ( cd $P && npm run build:client > $T/out/82-$id-restore-build.txt 2>&1; echo "$id restore client build exit $?" >> $T/out/80-mut-exits.txt )
}
F8 TOUCH tests/e2e/day-editor.spec.ts "a swipe that starts outside a day column"
F8 M2_LADDER tests/e2e/timesheet.spec.ts "the last two forms of a same-month range"
echo mutants-done > $T/out/mutants.done
