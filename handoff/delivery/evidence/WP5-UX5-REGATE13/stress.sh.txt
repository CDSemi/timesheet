cd <task>
. ./env.sh
CROSS=2026-10-27T06:58:30Z
F18="WP5-UX5-FIX18 \(F-UX5I-1\)"
UNS="Escape and the Close button: with unsaved edits|Escape dispatched in the same task"
SPEC=tests/e2e/day-editor.spec.ts
st() {
  label=$1; shift
  node burnrun.mjs "$label" "$@" > "logs/stress-$label.txt" 2>&1
  echo "$label finished $(date -u +%Y-%m-%dT%H:%M:%SZ)" >> logs/stress-progress.txt
  node counts.mjs "logs/stress-$label.txt" >> out/18-stress-counts.txt 2>&1
  bash cens.sh "st-$label" > "logs/cens-st-$label.log" 2>&1
}
# 1. the FIX18 tests (T1..T4 on desktop at 1440 and 1280 px, the change-from-elsewhere case and the open session dialog), real clock and at the midnight-crossing instant
st f18-real 0 1 real $SPEC --project=desktop --grep "$F18" --repeat-each=20 --workers=10
st f18-cross 0 1 $CROSS $SPEC --project=desktop --grep "$F18" --repeat-each=20 --workers=10
# 2. the phone test (T5) on the mobile project
st f18-mobile 0 1 real $SPEC --project=mobile --grep "$F18" --repeat-each=20 --workers=10
# 3. the unsaved-changes tests (Escape / Close, and the same-task Escape), both projects
st uns-desktop 0 1 real $SPEC --project=desktop --grep "$UNS" --repeat-each=20 --workers=10
st uns-mobile 0 1 real $SPEC --project=mobile --grep "$UNS" --repeat-each=20 --workers=10
echo stress-done > out/stress.done
