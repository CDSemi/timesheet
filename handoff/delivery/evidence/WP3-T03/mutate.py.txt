def sub(old, new):
    def apply(text):
        assert text.count(old) == 1, old
        return text.replace(old, new)
    return apply


def regex(pattern, new):
    def apply(text):
        result, count = re.subn(pattern, new, text)
        assert count == 1, pattern
        return result
    return apply

def sha(path):
    return hashlib.sha256(open(path, 'rb').read()).hexdigest()


out = []
caught = 0
for name, rel, mutate, tests in MUTATIONS:
    path = os.path.join(ROOT, rel)
    original = open(path, 'rb').read()
    before = sha(path)
    text = original.decode('utf-8')
    mutated = mutate(text).encode('utf-8')  # computed first: a failing mutation never touches the file
    try:
        open(path, 'wb').write(mutated)
        result = subprocess.run([NODE, 'node_modules/vitest/vitest.mjs', 'run', *tests], cwd=ROOT,
                                capture_output=True, text=True, encoding='utf-8', errors='replace')
    finally:
        open(path, 'wb').write(original)
    restored = sha(path) == before
    log = re.sub(r'\x1b\[[0-9;?]*[A-Za-z]', '', result.stdout + result.stderr)
    failed = [line.strip() for line in log.splitlines() if line.strip().startswith('×')]
    summary = [line.strip() for line in log.splitlines() if re.match(r'\s*Tests\s', line)]
    if result.returncode != 0 and restored:
        caught += 1
    out.append(f'## {name}')
    out.append(f'file: {rel}; tests: {", ".join(tests)}')
    out.append(f'exit: {result.returncode} (expected non-zero); restored byte-identical: {restored}')
    out.extend(f'  {line}' for line in failed[:6])
    if len(failed) > 6:
        out.append(f'  ... {len(failed) - 6} more failing tests')
    out.extend(f'  {line}' for line in summary)
    out.append('')
sys.stdout.write(f'# Mutation checks: {caught} of {len(MUTATIONS)} caught (each mutation made the named tests fail; each file restored byte-identical). Script: mutate.py.txt.\n\n')
sys.stdout.write('\n'.join(out))
