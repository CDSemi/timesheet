#!/usr/bin/env bash
# WP5-UX-FIX5 checks: typecheck, lint, unit tests, then the FULL e2e suite on both projects.
source /d/.claude-tmp/timesheet/WP5-UX-FIX5/env.sh
cd "$R" || exit 9
S="$T/checks-summary.txt"
{
  echo "node: $(node --version)"
  echo "HEAD: $(git rev-parse HEAD)"
  echo "start: $(date -u +%Y-%m-%dT%H:%M:%SZ)"
} > "$S"
npmr run typecheck > "$T/typecheck.raw.txt" 2>&1
echo "typecheck exit=$? at $(date -u +%H:%M:%SZ)" >> "$S"
npmr run lint > "$T/lint.raw.txt" 2>&1
echo "lint exit=$? at $(date -u +%H:%M:%SZ)" >> "$S"
npmr test > "$T/test.raw.txt" 2>&1
echo "test exit=$? at $(date -u +%H:%M:%SZ)" >> "$S"
if [ "${1:-}" = "e2e" ]; then
  npmr run test:e2e > "$T/e2e.raw.txt" 2>&1
  echo "e2e exit=$? at $(date -u +%H:%M:%SZ)" >> "$S"
fi
echo "end: $(date -u +%Y-%m-%dT%H:%M:%SZ)" >> "$S"
