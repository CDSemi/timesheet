const LA = 'America/Los_Angeles';
const HCM = 'Asia/Ho_Chi_Minh';
const out: string[] = [];

function utc(input: LocalInstantInput): string {
  const seconds = resolveLocalDateTime(input.local, input.zone, { fold: input.fold ?? null, offset: input.offset ?? null }).utc;
  return formatInZone('UTC', seconds);
}

function instants(request: SessionRequest | ClockOutRequest): unknown {
  const r = request as SessionRequest & ClockOutRequest;
  return {
    start: r.start === undefined ? null : utc(r.start),
    end: r.end === undefined ? null : utc(r.end),
    breaks: r.breaks === undefined ? 'omitted' : r.breaks.map((b) => [utc(b.start), utc(b.end), b.counts_as_work]),
    breaks_confirmed: r.breaks_confirmed,
    input_zone: r.input_zone,
    expected_version: r.expected_version,
    reason: r.reason,
  };
}

function field(date: string, time: string, fold: 0 | 1 | null = null): LocalField {
  return { date, time, fold, offset: null };
}

interface Typed {
  name: string;
  zone: string;
  start: [string, string, (0 | 1)?];
  end: [string, string, (0 | 1)?];
  /** start date, start time, minutes, start fold, old end date, old end time, old end fold */
  breaks: Array<[string, string, number, (0 | 1 | null), string, string, (0 | 1 | null)]>;
}

function oldBody(typed: Typed): SessionRequest {
  const draft: SessionDraft = {
    zone: typed.zone,
    start: field(typed.start[0], typed.start[1], typed.start[2] ?? null),
    end: field(typed.end[0], typed.end[1], typed.end[2] ?? null),
    breaks: typed.breaks.map(([d, t, , f, ed, et, ef], i) => ({ key: `k${i}`, start: field(d, t, f), end: field(ed, et, ef), countsAsWork: false })),
    mode: 'confirmed',
  };
  return buildSessionRequest(draft);
}

function newBody(typed: Typed): SessionRequest {
  let draft: SessionDialogDraft = newSessionDialogDraft(typed.start[0], typed.zone);
  draft = { ...draft, start: field(typed.start[0], typed.start[1], typed.start[2] ?? null), end: field(typed.end[0], typed.end[1], typed.end[2] ?? null) };
  typed.breaks.forEach(([d, t, m, f], i) => {
    draft = addBreakRow(draft, `n${i}`);
    draft = editBreakRow(draft, `n${i}`, { date: d, time: t, minutes: String(m) });
    if (f !== null) draft = chooseBreakRowFold(draft, `n${i}`, f);
  });
  draft = followSessionDate(draft);
  const built = buildSessionDialogRequest(draft, { allowUnconfirmed: false });
  if (!built.ok) throw new Error(`${typed.name}: ${JSON.stringify(built)}`);
  return built.request;
}

function savedSession(): Session {
  return {
    id: 's1',
    work_date: '2026-11-01',
    source: 'manual',
    start_utc: '2026-11-01T07:30:00Z',
    end_utc: '2026-11-01T12:00:00Z',
    input_zone: LA,
    breaks_confirmed: true,
    breaks: [
      { id: 'b1', start_utc: '2026-11-01T08:45:00Z', end_utc: '2026-11-01T09:15:00Z', counts_as_work: false },
      { id: 'b2', start_utc: '2026-11-01T09:50:00Z', end_utc: '2026-11-01T10:05:37Z', counts_as_work: false },
    ],
    version: 4,
  } as unknown as Session;
}

describe('sweep probe: old form vs new dialog request instants', () => {
  for (const typed of CASES) {
    it(typed.name, () => {
      const a = instants(oldBody(typed));
      const b = instants(newBody(typed));
      out.push(`CASE ${typed.name}\n old=${JSON.stringify(a)}\n new=${JSON.stringify(b)}\n raw_new=${JSON.stringify(newBody(typed))}\n equal=${JSON.stringify(a) === JSON.stringify(b)}`);
      expect(b).toEqual(a);
    });
  }

  it('untouched edit of a saved session (fold day, a break with seconds)', () => {
    const s = savedSession();
    const old = buildSessionRequest(draftFromSession(s), { expectedVersion: 4 });
    const built = buildSessionDialogRequest(sessionDialogDraftFromSession(s), { allowUnconfirmed: false, expectedVersion: 4 });
    if (!built.ok) throw new Error('not built');
    out.push(`CASE untouched edit\n old=${JSON.stringify(old)}\n new=${JSON.stringify(built.request)}\n bytes_equal=${JSON.stringify(old) === JSON.stringify(built.request)}`);
    expect(JSON.stringify(built.request)).toBe(JSON.stringify(old));
  });

  it('zone round trip on a saved session (LA -> HCM -> LA) keeps every instant', () => {
    const s = savedSession();
    const oldDraft = changeZone(changeZone(draftFromSession(s), HCM), LA);
    const newDraft = changeDialogZone(changeDialogZone(sessionDialogDraftFromSession(s), HCM), LA);
    const old = buildSessionRequest(oldDraft, { expectedVersion: 4 });
    const built = buildSessionDialogRequest(newDraft, { allowUnconfirmed: false, expectedVersion: 4 });
    out.push(`CASE zone round trip\n old=${JSON.stringify(old)}\n new=${built.ok ? JSON.stringify(built.request) : JSON.stringify(built)}`);
    expect(true).toBe(true);
  });

  it('session date moved one day earlier on a same-day session: hidden break dates follow (Q6 b)', () => {
    let draft = newSessionDialogDraft('2026-10-14', LA);
    draft = { ...draft, start: field('2026-10-14', '09:00'), end: field('2026-10-14', '17:00') };
    draft = addBreakRow(draft, 'n0');
    draft = editBreakRow(draft, 'n0', { time: '12:00', minutes: '30' });
    draft = followSessionDate({ ...draft, start: field('2026-10-13', '09:00'), end: field('2026-10-13', '17:00') });
    const built = buildSessionDialogRequest(draft, { allowUnconfirmed: false });
    out.push(`CASE date move\n new=${built.ok ? JSON.stringify(built.request) : JSON.stringify(built)}`);
    expect(built.ok).toBe(true);
  });

  it('check out: untouched saved rows of a running session (old default pending vs new default)', () => {
    const s = { ...savedSession(), end_utc: null, breaks_confirmed: false } as unknown as Session;
    const old = buildClockOutRequest(draftFromSession(s), 4);
    const draft = sessionDialogDraftFromSession(s);
    const built = buildCheckOutDialogRequest({ session: s, rows: draft.rows, noBreaks: draft.noBreaks, expectedVersion: 4 });
    out.push(`CASE checkout untouched saved rows\n old_default=${JSON.stringify(old)}\n new_default=${built.ok ? JSON.stringify(built.request) : JSON.stringify(built)}`);
    const removed = buildCheckOutDialogRequest({ session: s, rows: [], noBreaks: false, expectedVersion: 4 });
    out.push(` new_after_removing_all_rows=${removed.ok ? JSON.stringify(removed.request) : JSON.stringify(removed)}`);
    writeFileSync(process.env.SWEEP_OUT ?? 'zz-sweep-probe.txt', `${out.join('\n')}\n`);
    expect(true).toBe(true);
  });
});
