#!/usr/bin/env bash
# Runs the auditor's option-A harness probe in the probe export.
source <task>/env.sh
cd "$P" || exit 9
label=${1:-run1}
export RECHECK_SENDS_OUT="$(cygpath -w "$T/logs/sends-probe-$label-result.txt")"
start=$(date -u +%Y-%m-%dT%H:%M:%SZ)
npx vitest run tests/integration/zz-recheck-sends.test.ts > "$T/logs/sends-probe-$label.log" 2>&1
echo "sends-probe-$label exit=$? start=$start end=$(date -u +%Y-%m-%dT%H:%M:%SZ)" >> "$T/logs/probe-exits.txt"
