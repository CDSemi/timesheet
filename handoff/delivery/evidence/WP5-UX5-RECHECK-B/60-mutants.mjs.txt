function run(label, args, timeoutMs) {
  const started = Date.now();
  const result = spawnSync(node, args, { cwd: probe, encoding: 'utf8', timeout: timeoutMs, env: { ...process.env, NO_COLOR: '1', FORCE_COLOR: '0' }, maxBuffer: 256 * 1024 * 1024 });
  const output = `${result.stdout ?? ''}${result.stderr ?? ''}`;
  writeFileSync(join(logDir, `mut-${label}.log`), output);
  const tail = output
    .split('\n')
    .filter((line) => /Test Files|Tests |passed|failed|flaky|skipped|FAIL|✘|×/.test(line))
    .slice(-12)
    .join(' / ');
  return { status: result.status, seconds: Math.round((Date.now() - started) / 1000), tail };
}

const build = (label) => {
  const server = run(`${label}-tsc`, [tsc, '-p', 'tsconfig.server.json'], 300_000);
  const client = run(`${label}-vite`, [vite, 'build'], 300_000);
  return `${server.status}/${client.status}`;
};

for (const mutant of MUTANTS.filter((item) => only.length === 0 || only.includes(item.id))) {
  const path = join(probe, mutant.file);
  const original = readFileSync(path, 'utf8');
  const edits = mutant.edits ?? [{ from: mutant.from, to: mutant.to }];
  let text = original;
  let anchorsOk = true;
  for (const edit of edits) {
    const occurrences = text.split(edit.from).length - 1;
    if (mutant.all ? occurrences < 1 : occurrences !== 1) anchorsOk = false;
    text = mutant.all ? text.split(edit.from).join(edit.to) : text.replace(edit.from, edit.to);
  }
  if (!anchorsOk) {
    appendFileSync(summary, `${mutant.id} ANCHOR NOT UNIQUE, not run\n`);
    continue;
  }
  writeFileSync(path, text);
  let line = `${mutant.id} (${mutant.file}): ${mutant.what}`;
  try {
    if (!mutant.e2eOnly) {
      const targeted = run(`${mutant.id}-targeted`, [vitest, 'run', ...mutant.tests], 900_000);
      line += ` | targeted exit ${targeted.status} (${targeted.seconds} s): ${targeted.tail}`;
      if (targeted.status === 0 && mutant.noFull !== true) {
        const full = run(`${mutant.id}-full`, [vitest, 'run'], 1_800_000);
        line += ` | FULL exit ${full.status} (${full.seconds} s): ${full.tail}`;
      }
    }
    if (mutant.http) {
      line += ` | build ${build(`${mutant.id}-http`)}`;
      const out = join(logDir, `http-${mutant.id}.txt`);
      const http = run(`${mutant.id}-http`, [join(logDir, '..', 'http-optionA.mjs'), probe, join(logDir, '..', `http-${mutant.id}`), '49894', '49895', '49896', '49897', out], 900_000);
      line += ` | two-process HTTP probe exit ${http.status} (${http.seconds} s), see http-${mutant.id}.txt`;
    }
    if (mutant.e2e || mutant.e2eOnly) {
      line += ` | build ${build(mutant.id)}`;
      const e2e = run(`${mutant.id}-e2e`, [playwright, 'test', 'tests/e2e/admin-status.spec.ts'], 1_800_000);
      line += ` | admin-status e2e exit ${e2e.status} (${e2e.seconds} s): ${e2e.tail}`;
    }
  } finally {
    writeFileSync(path, original);
    const restored = readFileSync(path, 'utf8');
    line += ` | restored ${sha(restored) === sha(original) ? 'byte-identical' : 'DIFFERENT'}`;
    appendFileSync(summary, `${line}\n`);
  }
}
appendFileSync(summary, `done ${new Date().toISOString()}\n`);
