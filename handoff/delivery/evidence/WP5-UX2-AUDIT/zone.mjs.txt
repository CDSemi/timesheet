// Prints the real clock in UTC and America/Los_Angeles (the e2e reporting zone).
const now = new Date();
const la = new Intl.DateTimeFormat('en-CA', {
  timeZone: 'America/Los_Angeles',
  year: 'numeric',
  month: '2-digit',
  day: '2-digit',
  hour: '2-digit',
  minute: '2-digit',
  hour12: false,
}).format(now);
console.log(`utc ${now.toISOString()} los_angeles ${la} machine_zone ${Intl.DateTimeFormat().resolvedOptions().timeZone}`);
