// WP5-UX2-AUDIT task-local probe: response headers of the built server (CSP, content types, caching) for the round's static
// files. Usage: node headers.mjs <app dir with dist/> <port> <data dir>. Spawns its own server and stops it through its handle.
import { spawn, spawnSync } from 'node:child_process';
import { mkdirSync, readdirSync } from 'node:fs';
import { join } from 'node:path';

const [appDir, portText, dataDir] = process.argv.slice(2);
const port = Number(portText);
mkdirSync(dataDir, { recursive: true });
const env = {
  ...process.env,
  NODE_ENV: 'development',
  PORT: String(port),
  HOST: '127.0.0.1',
  DATABASE_PATH: join(dataDir, 'timesheet.db'),
  DATA_DIR: join(dataDir, 'private-data'),
  APP_ORIGINS: `http://localhost:${port}`,
  MAIL_FROM: 'headers-probe@example.invalid',
  OUTBOUND_MODE: 'capture',
  JOB_RUNNER: 'off',
};
const migrate = spawnSync(process.execPath, [join(appDir, 'dist', 'server', 'cli.js'), 'migrate'], { env, encoding: 'utf8', cwd: appDir });
console.log(`migrate status ${migrate.status}`);
const server = spawn(process.execPath, [join(appDir, 'dist', 'server', 'index.js')], { env, cwd: appDir, stdio: ['ignore', 'pipe', 'pipe'] });
const base = `http://127.0.0.1:${port}`;
try {
  let ready = false;
  for (let attempt = 0; attempt < 60 && !ready; attempt += 1) {
    await new Promise((resolve) => setTimeout(resolve, 250));
    ready = await fetch(`${base}/`).then(() => true, () => false);
  }
  console.log(`server ready ${ready}`);
  const assets = readdirSync(join(appDir, 'dist', 'client', 'assets')).filter((name) => name.endsWith('.js') || name.endsWith('.css'));
  for (const path of ['/', '/theme-init.js', '/cd-logo.svg', ...assets.map((name) => `/assets/${name}`)]) {
    const response = await fetch(`${base}${path}`);
    const h = (name) => response.headers.get(name);
    console.log(
      `${path.replace(/index-[A-Za-z0-9_-]+\./, 'index-<hash>.')} status=${response.status} content-type=${h('content-type')} cache-control=${h('cache-control')} etag=${h('etag') === null ? 'none' : 'present'} last-modified=${h('last-modified') === null ? 'none' : 'present'} nosniff=${h('x-content-type-options')}`,
    );
    if (path === '/') console.log(`csp=${h('content-security-policy')}`);
    await response.arrayBuffer();
  }
} finally {
  server.kill();
}
