# Own simulated-clock probes (task-local specs in the scratch export, unmutated, digest-checked), one Playwright run at a time.
. <task>/env.sh
cd $T
run() {
  label=$1; instant=$2; shift 2
  bash run-shift.sh mut "$label" "$instant" "$@" > "out/62-probe-$label.txt" 2>&1
  echo "$label $(grep -E '^exit=' out/62-probe-$label.txt) $(grep -E '[0-9]+ (passed|failed|skipped)' out/62-probe-$label.txt | tr -s ' ' | tr '\n' ' ')" >> out/62-probe-progress.txt
}
: > out/62-probe-progress.txt
