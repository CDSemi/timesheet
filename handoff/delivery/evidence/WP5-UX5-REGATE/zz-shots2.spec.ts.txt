import { addDays } from '../../src/domain/dates.ts';
import { buildSyntheticWorkbook } from '../support/syntheticWorkbook.ts';
import { expect, newPerson, screenshotPath, SeedClient, test } from './fixtures.ts';
import { note, signInTo } from './zz-lib.ts';

/* WP5-UX5-REGATE task-local probe: the two extra screenshots (an OT ledger region with focus; a drill-free page) and an Import overflow check at 320. */
test.use({ locale: 'en-US', timezoneId: 'America/Los_Angeles' });

test('extra shots', async ({ page, privateServer }, testInfo) => {
  test.skip(testInfo.project.name !== 'desktop', 'desktop');
  test.setTimeout(300_000);
  const admin = await new SeedClient(privateServer.origin).signIn(privateServer.credentials.admin.email, privateServer.credentials.admin.password);
  const person = await newPerson(admin, privateServer, { displayName: 'Synthetic Shot Person', signature: 'none' });
  const periods = await person.api.periodsCurrent();
  await person.api.call('POST', '/api/ot/opening-balance', { minutes: 90, as_of_date: addDays(periods.today_local, -30), reason: 'Synthetic opening balance', expected_version: 0 }, 201);
  const free = await person.api.displayedPeriodFreeWorkdays();
  await person.api.seedCompleteDay(free[0] ?? '');
  await page.setViewportSize({ width: 390, height: 844 });
  await signInTo(page, privateServer, person.account, '#/ot');
  const region = page.getByRole('region', { name: 'OT ledger' });
  await expect(region).toBeVisible();
  await page.emulateMedia({ colorScheme: 'light', reducedMotion: 'reduce' });
  await page.keyboard.press('Shift');
  await region.focus();
  await page.waitForTimeout(400);
  await region.scrollIntoViewIfNeeded();
  await page.screenshot({ path: screenshotPath('regate-ot-ledger-focus-390-light-synthetic.png'), animations: 'disabled' });
  await page.setViewportSize({ width: 1440, height: 900 });
  await page.emulateMedia({ colorScheme: 'dark', reducedMotion: 'reduce' });
  await page.goto(`${privateServer.origin}/#/timesheet`);
  await expect(page.locator('[data-day]')).toHaveCount(14);
  await page.waitForTimeout(400);
  await page.screenshot({ path: screenshotPath('regate-timesheet-1440-dark-synthetic.png'), animations: 'disabled' });
  // Import preview at 320: do the tables overflow there?
  const payrollDate = addDays(person.payrollDate, -14);
  const bytes = buildSyntheticWorkbook({ periods: [{ payrollDate, days: { 1: { label: 'Mystery label' }, 2: { labelFormulaCache: 'Holiday' } } }] });
  await page.setViewportSize({ width: 320, height: 700 });
  await page.goto(`${privateServer.origin}/#/import`);
  await page.locator('[data-import-file]').setInputFiles({ name: 'synthetic-hours.xlsx', mimeType: 'application/vnd.openxmlformats-officedocument.spreadsheetml.sheet', buffer: Buffer.from(bytes) });
  await page.getByRole('button', { name: 'Upload and preview' }).click();
  await expect(page.locator('[data-import-preview]')).toBeVisible();
  await page.waitForTimeout(600);
  const rows = await page.evaluate(`[...document.querySelectorAll('.table-wrap')].map((e) => ({ sw: e.scrollWidth, cw: e.clientWidth, role: e.getAttribute('role'), tab: e.getAttribute('tabindex'), label: e.getAttribute('aria-label') }))`);
  note('regate-a11y.txt', `Import preview 320: .table-wrap ${JSON.stringify(rows)}`);
});
