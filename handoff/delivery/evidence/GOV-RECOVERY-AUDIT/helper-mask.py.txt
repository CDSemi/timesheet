Usage: python mask.py
Masks the project path (<project>), the task folder (<task>) and the user-profile account (<user>);
strips ANSI codes; converts CRLF to LF. Helper sources are copied as <name>.py.txt.
"""

from common import REPO, TASK

RAW = [

HELPERS = ["common.py", "make_ready_copy.py", "make_ready_copy_done.py", "run_split.py", "compare_probes.py",
           "residual_status.py", "mutants.py", "safe_default.py", "mask.py"]


def mask(text):
    text = re.sub(r"\x1b\[[0-9;]*m", "", text).replace("\r\n", "\n").replace("\r", "\n")
    for variant in (r"<project>", "<project>",
                    "<project>"):
        text = text.replace(variant, "<project>")
    for variant in (r"<task>", "<task>",
                    "<task>"):
        text = text.replace(variant, "<task>")
    text = re.sub(r"(?i)([\\/]Users[\\/])(?!<user>)[^\\/\s\"']+", r"\1<user>", text)
    text = re.sub(r"(?i)<user>", "<user>", text)
    return text if text.endswith("\n") else text + "\n"


OUT.mkdir(parents=True, exist_ok=True)
written = []
for source, target, header in RAW:
    body = (TASK / source).read_text(encoding="utf-8")
    (OUT / target).write_bytes(mask((header or "") + body).encode("utf-8"))
    written.append(target)
for name in HELPERS:
    target = f"helper-{name}.txt"
    (OUT / target).write_bytes(mask((TASK / name).read_text(encoding="utf-8")).encode("utf-8"))
    written.append(target)
for target in written:
    data = (OUT / target).read_bytes()
    assert b"\r" not in data and b"<user>" not in data.lower(), target
print(f"wrote {len(written)} masked LF files: {', '.join(written)}")
