#!/bin/bash
# Re-run the sample generator with the ORIGINAL (HEAD) renderer for a byte comparison with the patched run.
set -u
T="<task>"
export PATH="$(cygpath -u "$LOCALAPPDATA")/timesheet-dev/node-24.21.0/node_modules/node/bin:$(cygpath -u "$APPDATA")/npm:$PATH"
export TEMP="$(cygpath -w "$T/tmp")" TMP="$(cygpath -w "$T/tmp")"
cd "$T/export" || exit 1
cp "$T/timesheetPdf.orig.ts" src/server/pdf/timesheetPdf.ts
node.exe node_modules/typescript/lib/tsc.js -p tsconfig.server.json > "$T/out/132-build-server-head.txt" 2>&1; echo "build exit $?" >> "$T/out/132-build-server-head.txt"
export PILOT_REPO="$(cygpath -w "$T/export")" RUN_DIR="$(cygpath -w "$T/samples-run-head")"
mkdir -p "$T/samples-run-head"
node.exe "$T/probes/104-make-samples.mjs" > "$T/out/133-samples-head.txt" 2>&1; echo "exit $?" >> "$T/out/133-samples-head.txt"
cp "$T/timesheetPdf.patched.ts" src/server/pdf/timesheetPdf.ts
node.exe node_modules/typescript/lib/tsc.js -p tsconfig.server.json > "$T/out/134-build-server-restored.txt" 2>&1; echo "build exit $?" >> "$T/out/134-build-server-restored.txt"
