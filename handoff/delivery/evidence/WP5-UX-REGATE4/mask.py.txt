import re
import shutil
from pathlib import Path

T = Path(r"<task>")
EV = Path(r"<repo>\handoff\delivery\evidence\WP5-UX-REGATE4")
EV.mkdir(parents=True, exist_ok=True)

ANSI = re.compile(r"\x1b\[[0-9;?]*[A-Za-z]")
SUBS = [
    (re.compile(r"D:[\\/]\.claude-tmp[\\/]timesheet[\\/]WP5-UX-REGATE4", re.I), "<task>"),
    (re.compile(r"/d/\.claude-tmp/timesheet/WP5-UX-REGATE4", re.I), "<task>"),
    (re.compile(r"D:[\\/]Dropbox[\\/]Work\.CDSemi[\\/]timesheet", re.I), "<repo>"),
    (re.compile(r"/d/Dropbox/Work\.CDSemi/timesheet", re.I), "<repo>"),
    (re.compile(r"[\\/]Users[\\/]<user>", re.I), "/Users/<user>"),
    (re.compile(r"<user>", re.I), "<user>"),
    (re.compile(r"<owner-email-pattern>", re.I), "<owner-email>"),
]


def mask(text: str) -> str:
    text = ANSI.sub("", text).replace("\r\n", "\n").replace("\r", "\n")
    for pat, rep in SUBS:
        text = pat.sub(rep, text)
    return text


out = T / "out"
skip = {"ex-files.txt", "ex-hashes.txt", "ex-sorted.txt", "ex-files-before.txt", "lstree-sorted.txt", "keys-run.bak"}
for p in sorted(out.iterdir()):
    if p.is_file() and p.suffix == ".txt" and p.name not in skip:
        (EV / p.name).write_bytes(mask(p.read_text(encoding="utf-8", errors="replace")).encode("utf-8"))
