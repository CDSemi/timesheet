// Masks a text file for the evidence folder: ANSI codes out, LF line ends, task/repo/profile paths and non-synthetic addresses masked.
// Usage: node mask.mjs <in> <out>
import { readFileSync, writeFileSync } from 'node:fs';

const [input, output] = process.argv.slice(2);
let text = readFileSync(input, 'utf8');
text = text.replace(/\x1b\[[0-9;]*m/g, '').replace(/\r\n/g, '\n').replace(/\r/g, '\n');
const bs = String.fromCharCode(92);
const forms = (win, label) => {
  const parts = win.split('/');
  return [parts.join(bs + bs + bs + bs), parts.join(bs + bs), parts.join(bs), win, '/' + win[0].toLowerCase() + win.slice(2)].map((f) => [f, label]);
};
const pairs = [...forms('<task>', '<task>'), ...forms('<repo>', '<repo>')];
for (const [f, label] of pairs) text = text.split(f).join(label);
const user = process.env.USERNAME ?? '';
if (user !== '') text = text.split(user).join('<user>');
text = text.replace(/[A-Za-z0-9._%+-]+@[A-Za-z0-9.-]+\.[A-Za-z]{2,}/g, (m) => (/@example\.invalid$/i.test(m) ? m : '<email>'));
text = text.replace(/[A-Za-z0-9._%+-]<email>"]*/g, '<email>');
writeFileSync(output, text);
