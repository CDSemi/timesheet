export function note(file: string, line: string): void {
  console.log(line);
  if (OUT !== '') appendFileSync(join(OUT, file), `${line}\n`);
}

export interface Situation {
  slug: string;
  media: Theme;
  stored: Theme | null;
}
export const SITUATIONS: readonly Situation[] = [
  { slug: 'light', media: 'light', stored: null },
  { slug: 'dark', media: 'dark', stored: null },
  { slug: 'manual-dark', media: 'light', stored: 'dark' },
  { slug: 'manual-light', media: 'dark', stored: 'light' },
];

export async function applySituation(page: Page, situation: Situation): Promise<void> {
  await page.emulateMedia({ colorScheme: situation.media, reducedMotion: 'reduce' });
  if (situation.stored !== null) {
    await page.addInitScript({
      content: `try { localStorage.setItem(${JSON.stringify(STORAGE_KEY)}, ${JSON.stringify(situation.stored)}); } catch (error) {}`,
    });
  }
}

/** Waits two frames and every running animation, so a computed colour is the settled one. */
export async function settle(page: Page): Promise<void> {
  await page.evaluate(
    `new Promise((resolve) => requestAnimationFrame(() => requestAnimationFrame(() => Promise.all(document.getAnimations().map((a) => a.finished.catch(() => undefined))).then(resolve))))`,
  );
}

/** A holiday name longer than the gate's, to stress the category cell. */
export const AUDIT_HOLIDAY = 'Observed Thanksgiving Holiday Closure';

export interface AuditSeed {
  person: Person;
  dates: string[];
  holiday: string | null;
  overnight: string;
  twoSessions: string;
  notes: string;
  leave: string | null;
}

/**
 * The auditor's own seed (different from the gate's): a long holiday name, an overnight session, two sessions on one day, a day
 * with long notes and partial leave, a complete day and an unconfirmed-breaks day, all in the displayed period.
 */
export async function seedAudit(server: BuiltServer, name = 'Synthetic Audit Person'): Promise<AuditSeed> {
  const adm = await admin(server);
  const person = await newPerson(adm, server, { displayName: name, signature: 'none' });
  const sheet = await person.api.call<{ period: { period_start: string; period_end: string }; days: Array<{ work_date: string }> }>(
    'GET',
    `/api/timesheets/${person.payrollDate}`,
  );
  const free = await person.api.displayedPeriodFreeWorkdays();
  expect(free.length, 'free past workdays').toBeGreaterThanOrEqual(4);
  // A holiday on the last free workday of week 2 if one exists in the future, else none (a holiday on a past day with a session is avoided).
  let holiday: string | null = null;
  const futureWeekday = sheet.days.map((d) => d.work_date).find((d) => d > person.todayLocal && new Date(`${d}T12:00:00Z`).getUTCDay() % 6 !== 0);
  if (futureWeekday !== undefined) {
    const calendar = await adm.call<{ id: string }>('GET', '/api/calendar');
    const csv = ['date,name,kind', `${futureWeekday},${AUDIT_HOLIDAY},holiday`].join('\n');
    const body = { calendar_id: calendar.id, year: Number(futureWeekday.slice(0, 4)), effective_from: sheet.period.period_start, csv, remove_dates: [] };
    try {
      const preview = await adm.call<{ preview_hash: string }>('POST', '/api/admin/calendar/import/preview', body);
      await adm.call('POST', '/api/admin/calendar/import/commit', { ...body, preview_hash: preview.preview_hash }, 201);
    } catch {
      // Already imported by an earlier test of this worker: the holiday stays.
    }
    holiday = futureWeekday;
  }
  const overnight = free.at(-1) ?? '';
  const twoSessions = free[0] ?? '';
  const at = (date: string, time: string) => ({ local: `${date}T${time}`, zone: LA });
  const add = (start: { local: string; zone: string }, end: { local: string; zone: string }) =>
    person.api.call('POST', `/api/days/${start.local.slice(0, 10)}/sessions`, { start, end, input_zone: LA, breaks_confirmed: true, breaks: [] }, 201);
  await add(at(overnight, '21:30'), at(addDays(overnight, 1), '03:15'));
  await add(at(twoSessions, '07:30'), at(twoSessions, '11:45'));
  await add(at(twoSessions, '12:30'), at(twoSessions, '19:10'));
  const spare = free.filter((d) => d !== overnight && d !== twoSessions && d !== addDays(overnight, 1));
  const notesDay = spare[0] ?? twoSessions;
  await person.api.putDay(notesDay, { category: 'Worked', notes: 'Synthetic long note for the audit: site visit, calibration, two calls and a report.' });
  if (spare[1] !== undefined) await person.api.seedCompleteDay(spare[1]);
  if (spare[2] !== undefined) await person.api.seedUnconfirmedBreaksDay(spare[2]);
  return { person, dates: sheet.days.map((d) => d.work_date), holiday, overnight, twoSessions, notes: 'Synthetic long note', leave: notesDay };
}

/* ---- axe ------------------------------------------------------------------------------------------------------------ */
const AXE_PATH = process.env.AXE_PATH ?? '';
let axeSource = '';
export const AXE_TAGS = ['wcag2a', 'wcag2aa', 'wcag21a', 'wcag21aa', 'wcag22aa'];

export interface AxeSummary {
  version: string;
  violations: Array<{ id: string; impact: string | null; nodes: Array<{ target: string; html: string; summary: string }> }>;
  incomplete: Array<{ id: string; nodes: number }>;
  passes: number;
}

export async function runAxe(page: Page): Promise<AxeSummary> {
  if (axeSource === '') axeSource = readFileSync(AXE_PATH, 'utf8');
  if (!(await page.evaluate<boolean>(`typeof window.axe !== 'undefined'`))) await page.evaluate(axeSource);
  return page.evaluate<AxeSummary, string[]>(async (tags) => {
    const axe = (window as unknown as { axe: { run: (c: Document, o: unknown) => Promise<Record<string, unknown>>; version: string } }).axe;
    const r = (await axe.run(document, { runOnly: { type: 'tag', values: tags }, resultTypes: ['violations', 'incomplete'] })) as {
      violations: Array<{ id: string; impact: string | null; nodes: Array<{ target: unknown; html: string; failureSummary?: string }> }>;
      incomplete: Array<{ id: string; nodes: unknown[] }>;
      passes: unknown[];
    };
    return {
      version: axe.version,
      violations: r.violations.map((v) => ({
        id: v.id,
        impact: v.impact,
        nodes: v.nodes.map((n) => ({ target: JSON.stringify(n.target), html: n.html.slice(0, 160), summary: (n.failureSummary ?? '').replace(/\s+/g, ' ').slice(0, 240) })),
      })),
      incomplete: r.incomplete.map((i) => ({ id: i.id, nodes: i.nodes.length })),
      passes: r.passes.length,
    };
  }, AXE_TAGS);
}

/* ---- text contrast of every visible text element against its composited backdrop ---------------------------------- */
const TEXT_SWEEP = `(() => {
  const parse = (css) => {
    let m = css.match(/^rgba?\\(([^)]+)\\)$/);
    if (m) { const p = m[1].split(/[,\\s/]+/).filter(Boolean).map(Number); return [p[0], p[1], p[2], p.length > 3 ? p[3] : 1]; }
    m = css.match(/^color\\(srgb ([^)]+)\\)$/);
    if (m) { const p = m[1].split(/[\\s/]+/).filter(Boolean).map(Number); return [p[0] * 255, p[1] * 255, p[2] * 255, p.length > 3 ? p[3] : 1]; }
    return [0, 0, 0, 0];
  };
  const overC = (t, b) => { const a = t[3] + b[3] * (1 - t[3]); if (a === 0) return [0, 0, 0, 0]; const mix = (i) => (t[i] * t[3] + b[i] * b[3] * (1 - t[3])) / a; return [mix(0), mix(1), mix(2), a]; };
  const lum = (c) => { const ch = (v) => { const s = v / 255; return s <= 0.03928 ? s / 12.92 : ((s + 0.055) / 1.055) ** 2.4; }; return 0.2126 * ch(c[0]) + 0.7152 * ch(c[1]) + 0.0722 * ch(c[2]); };
  const out = [];
  for (const el of document.querySelectorAll('body *')) {
    if (['SCRIPT', 'STYLE', 'svg', 'path', 'OPTION'].includes(el.tagName)) continue;
    let own = '';
    for (const n of el.childNodes) if (n.nodeType === 3) own += n.textContent;
    own = own.replace(/\\s+/g, ' ').trim();
    if (el.tagName === 'INPUT' && !['checkbox', 'radio', 'hidden', 'file', 'button', 'submit'].includes(el.type)) own = (el.value || '').trim();
    if (el.tagName === 'SELECT') own = (el.selectedOptions[0]?.textContent ?? '').trim();
    if (own === '') continue;
    const r = el.getBoundingClientRect();
    const s = getComputedStyle(el);
    if (r.width <= 1 || r.height <= 1) continue;
    if (!el.checkVisibility({ checkOpacity: true, checkVisibilityCSS: true })) continue;
    if (s.clipPath && s.clipPath !== 'none') continue;
    let opacity = 1; let image = false; const stack = [];
    for (let n = el; n; n = n.parentElement) { const st = getComputedStyle(n); opacity *= parseFloat(st.opacity); stack.push(parse(st.backgroundColor)); if (st.backgroundImage !== 'none') image = true; }
    let bg = [255, 255, 255, 1];
    for (let i = stack.length - 1; i >= 0; i -= 1) bg = overC(stack[i], bg);
    const f = parse(s.color);
    const fg = overC([f[0], f[1], f[2], f[3] * opacity], bg);
    const [hi, lo] = [lum(fg), lum(bg)].sort((a, b) => b - a);
    const size = parseFloat(s.fontSize); const weight = parseInt(s.fontWeight, 10) || 400;
    const large = size >= 24 || (size >= 18.66 && weight >= 700);
    const disabled = Boolean(el.closest(':disabled, [aria-disabled="true"]'));
    out.push({ tag: el.tagName.toLowerCase(), cls: typeof el.className === 'string' ? el.className.slice(0, 40) : '', text: own.slice(0, 36), ratio: (hi + 0.05) / (lo + 0.05), need: large ? 3 : 4.5, disabled, image });
  }
  return out;
})()`;

export interface TextRecord {
  tag: string;
  cls: string;
  text: string;
  ratio: number;
  need: number;
  disabled: boolean;
  image: boolean;
}

export async function textSweep(page: Page): Promise<{ count: number; min: number; minWhere: string; below: TextRecord[] }> {
  const all = await page.evaluate<TextRecord[]>(TEXT_SWEEP);
  const live = all.filter((r) => !r.disabled);
  const min = live.reduce<TextRecord | null>((m, r) => (m === null || r.ratio < m.ratio ? r : m), null);
  return {
    count: all.length,
    min: min?.ratio ?? 0,
    minWhere: min === null ? '' : `${min.tag}.${min.cls} "${min.text}"`,
    below: live.filter((r) => r.ratio < r.need - 0.005),
  };
}

export async function scan(page: Page, file: string, label: string): Promise<{ violations: number; below: number }> {
  await settle(page);
  const axe = await runAxe(page);
  const text = await textSweep(page);
  note(
    file,
    `${label}: axe ${axe.version} violations=${axe.violations.length} [${axe.violations.map((v) => `${v.id}(${v.impact})x${v.nodes.length}`).join(', ')}] incomplete=[${axe.incomplete.map((i) => `${i.id}x${i.nodes}`).join(', ')}] passes=${axe.passes}; text n=${text.count} min=${text.min.toFixed(2)} (${text.minWhere}) below=${text.below.length}`,
  );
  for (const v of axe.violations) for (const n of v.nodes.slice(0, 5)) note(file, `    VIOLATION ${v.id} ${n.target} ${n.html} :: ${n.summary}`);
  for (const r of text.below.slice(0, 8)) note(file, `    TEXT-BELOW ${r.tag}.${r.cls} "${r.text}" ${r.ratio.toFixed(2)} needs ${r.need} image=${r.image}`);
  return { violations: axe.violations.length, below: text.below.length };
}

/* ---- geometry ------------------------------------------------------------------------------------------------------- */
export interface BoxN {
  x: number;
  y: number;
  width: number;
  height: number;
  left: number;
  right: number;
  top: number;
  bottom: number;
}

export async function box(locator: Locator): Promise<BoxN> {
  return locator.evaluate((el) => {
    const r = el.getBoundingClientRect();
    return { x: r.x, y: r.y, width: r.width, height: r.height, left: r.left, right: r.right, top: r.top, bottom: r.bottom };
  });
}

export async function css(locator: Locator, props: string[]): Promise<Record<string, string>> {
  return locator.evaluate((el, names) => {
    const s = getComputedStyle(el);
    const out: Record<string, string> = {};
    for (const n of names) out[n] = s.getPropertyValue(n);
    return out;
  }, props);
}

export function ratio(a: string, b: string): number {
  return contrast(over(parseColour(a), [255, 255, 255, 1] as Rgba), over(parseColour(b), [255, 255, 255, 1] as Rgba));
}

/** Sets the viewport so that `document.documentElement.clientWidth` is exactly `clientWidth` (overlay scrollbars: equal). */
export async function setWidth(page: Page, width: number, height = 800): Promise<void> {
  await page.setViewportSize({ width, height });
  for (let i = 0; i < 3; i += 1) {
    const cw = await page.evaluate<number>('document.documentElement.clientWidth');
    if (cw === width) break;
    await page.setViewportSize({ width: width + (width - cw), height });
  }
  await settle(page);
}

export async function noPageScrollX(page: Page): Promise<{ scrollWidth: number; clientWidth: number }> {
  return page.evaluate(() => ({ scrollWidth: document.documentElement.scrollWidth, clientWidth: document.documentElement.clientWidth }));
}

export async function drawerState(page: Page): Promise<{ date: string | null; side: string | null; modal: boolean; inert: number; ariaModal: string | null; layout: string | null } | null> {
  return page.evaluate(() => {
    const d = document.querySelector('dialog[data-day-editor]');
    const area = document.querySelector('[data-editor-layout]');
    if (d === null) return null;
    return {
      date: d.getAttribute('data-day-editor'),
      side: d.getAttribute('data-side'),
      modal: d.matches(':modal'),
      inert: document.querySelectorAll('[inert]').length,
      ariaModal: d.getAttribute('aria-modal'),
      layout: area?.getAttribute('data-editor-layout') ?? null,
    };
  });
}
