. <task>/env.sh
mkdir -p $E
m() { node $T/mask.mjs "$1" "$2"; }
cd $T/out
for f in *.txt; do
  case "$f" in ex-*.txt|lstree-sorted.txt) continue;; esac
  m "$T/out/$f" "$E/$f"
done
m $T/out/lstree-sorted.txt $E/01-lstree-list.txt
cd $T/logs
for f in core.txt e2e-full-real.txt e2e-rerun-4022-mobile.txt probe-*.txt mut-*.txt; do [ -f "$f" ] && m "$T/logs/$f" "$E/log-$f"; done
cd $T/obs
for f in *.txt; do m "$T/obs/$f" "$E/obs-$f"; done
cd $T
for s in env.sh digest.sh run-core.sh run-e2e.sh run-probe.sh run-shift-probe.sh run-mutants.sh check-mutants.sh collect.sh mutate.mjs mask.mjs contrast-tokens.mjs snapdiff.mjs shiftclock.mjs zz-alib.ts zz-a-visual.spec.ts zz-a-editor.spec.ts zz-a-rule.spec.ts zz-a-walk.spec.ts zz-a-dialogs.spec.ts zz-a-settings.spec.ts zz-a-axe.spec.ts zz-a-snap.spec.ts zz-a-current.spec.ts zz-a-shots.spec.ts zz-a-diag.spec.ts zz-a-flapshot.spec.ts; do
  [ -f $T/$s ] && m "$T/$s" "$E/zz-src-$s.txt"
done
cp $T/shots-shots2/audita-target-timesheet-1440-light-synthetic.png $E/
cp $T/shots-shots/audita-target-timesheet-1440-dark-synthetic.png $E/
cp $T/shots-shots-base/audita-base-timesheet-1440-light-synthetic.png $E/
cp $T/shots-shots2/audita-drawer-beside-1440-light-synthetic.png $E/
cp $T/shots-shots2/audita-drawer-floating-1280-light-synthetic.png $E/
cp $T/shots-shots2/audita-review-1280-light-synthetic.png $E/
cp $T/shots-dialogs2/audita-session-dialog-1440-synthetic.png $E/
cp $T/shots-dialogs2/audita-session-dialog-390-synthetic.png $E/
cp $T/shots-dialogs2/audita-alert-1440-synthetic.png $E/
cp $T/shots-dialogs2/audita-running-edit-1440-synthetic.png $E/

cp $T/shots-settings2/audita-import-preview-1440-synthetic.png $E/
[ -d $T/shots-axe2 ] && cp $T/shots-axe2/audita-alert-dark-1440-synthetic.png $E/ 2>&1
