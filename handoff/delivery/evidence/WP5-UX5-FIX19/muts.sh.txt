# MA: a session write's answer is taken without the sameDayFields judgement (every branch takes the answer, the form is never restarted).
M MA "$F" 'if (held === null || sameDayFields(held, next)) {' 'if (true) {' "$G"
# MB: the unsaved-edit branch removed (the form always restarts when the fields differ).
M MB "$F" '} else if (fieldsDirtyRef.current) {' '} else if (false) {' "$G"
# MC: FIX18's WFH comparison removed.
M MC "$F" 'held.wfh === fresh.wfh &&' 'true &&' "$G"

M MD "$F" 'held.leave_minutes === fresh.leave_minutes &&' 'true &&' "$G"

# MF: an overtaken follow read is no longer dropped.

# MG: FIX18's other guards, for completeness: the notes comparison and the category comparison.
M MG "$F" "(held.entry?.notes ?? '') === (fresh.entry?.notes ?? '')" 'true' "$G"
M MH "$F" 'held.category === fresh.category &&' 'true &&' "$G"
