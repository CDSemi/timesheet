def mask(text):
    text = re.sub(r"\x1b\[[0-9;]*m", "", text)
    text = text.replace("\r\n", "\n")
    for path, label in ((ROOT, "<project>"), (os.environ.get("TEMP", "<none>"), "<tmp>"), (os.path.expanduser("~"), "<home>")):
        for variant in {path, path.replace("\\", "/")}:
            if variant and variant != "<none>":
                text = re.sub(re.escape(variant), label, text, flags=re.IGNORECASE)
    return "\n".join(line.rstrip() for line in text.split("\n"))


def summary_lines(output):
    keep = []
    for line in output.split("\n"):
        stripped = line.strip()
        if stripped.startswith(("×", "Test Files", "Tests ", "FAIL", "AssertionError", "Error:")):
            keep.append("    " + stripped)
    return keep[:12]


def read(rel):
    with open(os.path.join(ROOT, rel), "rb") as handle:
        return handle.read()


report = []
caught = 0
for title, edits in MUTATIONS:
    plan = []  # (path, rel, before bytes, after bytes), all computed before any file is opened for writing
    for rel, replacements in edits:
        before = read(rel)
        text = before.decode("utf-8")
        for original, mutated in replacements:
            if text.count(original) != 1:
                print(f"{title}: anchor found {text.count(original)} times in {rel}; aborting", file=sys.stderr)
                sys.exit(2)
            text = text.replace(original, mutated, 1)
        plan.append((os.path.join(ROOT, rel), rel, before, text.encode("utf-8")))
    try:
        for path, _rel, _before, after in plan:
            with open(path, "wb") as handle:
                handle.write(after)
        result = subprocess.run([NODE, VITEST, "run", *TESTS], cwd=ROOT, capture_output=True, text=True, encoding="utf-8", errors="replace", timeout=900)
        output = mask(result.stdout + result.stderr)
    finally:
        for path, _rel, before, _after in plan:
            with open(path, "wb") as handle:
                handle.write(before)
    restored = all(hashlib.sha256(read(rel)).hexdigest() == hashlib.sha256(before).hexdigest() for _path, rel, before, _after in plan)
    status = "CAUGHT" if result.returncode != 0 else "NOT CAUGHT"
    caught += result.returncode != 0
    report.append(title)
    for _path, rel, before, _after in plan:
        report.append(f"  file: {rel} (original sha256 {hashlib.sha256(before).hexdigest()})")
    report.append(f"  command: node {VITEST.replace(os.sep, '/')} run {' '.join(TESTS)}")
    report.append(f"  exit: {result.returncode} -> {status}")
    report.extend(summary_lines(output))
    report.append(f"  restored byte-identical: {restored}")
    report.append("")

report.append(f"mutations caught: {caught}/{len(MUTATIONS)}")
print("\n".join(report))
