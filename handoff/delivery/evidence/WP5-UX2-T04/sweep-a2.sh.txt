# Usage: bash sweep-a2.sh
# Addendum 2 (final): today.spec.ts and timesheet.spec.ts, both projects, at each simulated America/Los_Angeles instant, sequentially.
# The 23:59 instant is split so no run crosses midnight (today.spec.ts alone at 23:59; timesheet.spec.ts alone at 23:57).
cd /d/.claude-tmp/timesheet/WP5-UX2-T04
run() {
  label=$1; instant=$2; shift 2
  bash run-shift.sh "a2-$label" "$instant" "$@" > "logs/a2-sweep-$label.txt" 2>&1
  echo "$label finished $(date -u +%Y-%m-%dT%H:%M:%SZ)" >> logs/a2-sweep-progress.txt
}
BOTH="tests/e2e/today.spec.ts tests/e2e/timesheet.spec.ts"
run s01-1017-1000 2026-10-17T17:00:00Z $BOTH
run s02-1025-2330 2026-10-26T06:30:00Z $BOTH
run s03-1026-0030 2026-10-26T07:30:00Z $BOTH
run s04-1027-1300 2026-10-27T20:00:00Z $BOTH
run s05-1027-1800 2026-10-28T01:00:00Z $BOTH
run s06a-1030-2359-today 2026-10-31T06:59:00Z tests/e2e/today.spec.ts
run s06b-1030-2357-timesheet 2026-10-31T06:57:00Z tests/e2e/timesheet.spec.ts
run s07-1031-0030 2026-10-31T07:30:00Z $BOTH
run s08-1101-0130 2026-11-01T08:30:00Z $BOTH
run s09-0314-0330 2027-03-14T10:30:00Z $BOTH
run s10-1231-2330 2027-01-01T07:30:00Z $BOTH
run s11-0101-0030 2027-01-01T08:30:00Z $BOTH
echo done > logs/a2-sweep.done
