# Usage: bash sweep.sh
# today.spec.ts and timesheet.spec.ts on both projects at each simulated America/Los_Angeles instant, sequentially, each with its
# own output folder (run-shift.sh). The label names the instant; the log is logs/sweep-<label>.txt.
cd /d/.claude-tmp/timesheet/WP5-UX2-T04
run() {
  label=$1; instant=$2
  bash run-shift.sh "$label" "$instant" tests/e2e/today.spec.ts tests/e2e/timesheet.spec.ts > "logs/sweep-$label.txt" 2>&1
  echo "$label finished $(date -u +%Y-%m-%dT%H:%M:%SZ)" >> logs/sweep-progress.txt
}
