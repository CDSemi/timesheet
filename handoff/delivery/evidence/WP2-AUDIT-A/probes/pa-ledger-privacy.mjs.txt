const responses = []; // every response body text, for the secret scan
async function req(method, path, { cookie, body, origin = ORIGIN, raw = false } = {}) {
  const headers = {};
  if (origin !== null) headers.origin = origin;
  if (cookie !== undefined) headers.cookie = cookie;
  const init = { method, headers };
  if (body !== undefined) { headers['content-type'] = 'application/json'; init.body = JSON.stringify(body); }
  const res = await app.request(path, init);
  const text = await res.text();
  responses.push(`${method} ${path} ${res.status}\n${[...res.headers].map(([k, v]) => `${k}: ${v}`).join('\n')}\n${text}`);
  let json = null;
  if (!raw) { try { json = JSON.parse(text); } catch { json = null; } }
  return { status: res.status, body: json, text, headers: res.headers };
}

function parseCsv(text) {
  const rows = []; let row = []; let cell = ''; let quoted = false; let wasQuoted = false;
  for (let i = 0; i < text.length; i += 1) {
    const ch = text[i];
    if (quoted) {
      if (ch === '"') { if (text[i + 1] === '"') { cell += '"'; i += 1; } else quoted = false; } else cell += ch;
    } else if (ch === '"' && cell === '') { quoted = true; wasQuoted = true; }
    else if (ch === ',') { row.push({ cell, wasQuoted }); cell = ''; wasQuoted = false; }
    else if (ch === '\n') { row.push({ cell, wasQuoted }); rows.push(row); row = []; cell = ''; wasQuoted = false; }
    else cell += ch;
  }
  if (cell !== '' || row.length > 0) { row.push({ cell, wasQuoted }); rows.push(row); }
  return rows;
}
