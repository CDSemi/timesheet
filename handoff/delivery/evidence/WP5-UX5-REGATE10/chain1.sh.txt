. <task>/env.sh
export NODE_OPTIONS="--trace-deprecation --pending-deprecation"
O=$T/out
hashdist() {
  d=$1; tag=$2
  cd $T/$d
  find dist -type f | LC_ALL=C sort > $O/51-$tag-files.txt
  rm -f $O/51-$tag-hashes.txt
  while IFS= read -r f; do printf '%s %s\n' "$f" "$(sha256sum "$f" | cut -d' ' -f1)" >> $O/51-$tag-hashes.txt; done < $O/51-$tag-files.txt
  echo "$tag: $(wc -l < $O/51-$tag-files.txt) files" >> $O/51-dist-counts.txt
}
cd $T/export
npm ci > $O/02-ci.txt 2>&1; echo "ci exit $?" > $O/02-ci-exit.txt
npm audit --omit=dev > $O/02-npm-audit.txt 2>&1; echo "npm audit --omit=dev exit $?" > $O/02-npm-audit-exit.txt
npm run lint > $O/02-lint.txt 2>&1; echo "lint exit $?" > $O/02-lint-exit.txt
npm run typecheck > $O/02-typecheck.txt 2>&1; echo "typecheck exit $?" > $O/02-typecheck-exit.txt
npm run build > $O/02-build.txt 2>&1; echo "build exit $?" > $O/02-build-exit.txt
hashdist export target
echo "target build done $(date -u +%H:%M:%SZ)" > $T/logs/chain1.log
cd $T/export-prev
npm ci > $O/02-prev-ci.txt 2>&1; echo "prev ci exit $?" >> $O/02-ci-exit.txt
npm run build > $O/02-prev-build.txt 2>&1; echo "prev build exit $?" >> $O/02-build-exit.txt
hashdist export-prev prev
diff $O/51-prev-hashes.txt $O/51-target-hashes.txt > $O/51-hash-diff-prev-vs-target.txt; echo "dist hash diff prev vs target exit $? differing lines $(grep -cE '^[<>]' $O/51-hash-diff-prev-vs-target.txt)" > $O/51-hash-result.txt
echo "prev build done $(date -u +%H:%M:%SZ)" >> $T/logs/chain1.log
cd $T/export
npm run verify > $O/03-verify.txt 2>&1; echo "verify exit $?" > $O/03-verify-exit.txt
echo "verify done $(date -u +%H:%M:%SZ)" >> $T/logs/chain1.log
hashdist export target-after-verify
diff $O/51-target-hashes.txt $O/51-target-after-verify-hashes.txt > $O/51-target-vs-after-verify.txt; echo "target vs after-verify diff exit $?" >> $O/51-hash-result.txt
cd $T/export
for i in 1 2 3; do
  npm test > $O/05-test-$i.txt 2>&1; echo "npm test run $i exit $?" >> $O/05-test-exits.txt
  echo "test $i done $(date -u +%H:%M:%SZ)" >> $T/logs/chain1.log
done
cd $T/export
$PY handoff/delivery/validate_package.py --preflight > $O/11-validate_package.txt 2>&1; echo "validate_package exit $?" >> $O/11-exits.txt
$PY handoff/delivery/validate_orchestration.py > $O/11-validate_orchestration.txt 2>&1; echo "validate_orchestration exit $?" >> $O/11-exits.txt
$PY handoff/delivery/check_recovery.py > $O/11-check_recovery.txt 2>&1; echo "check_recovery exit $?" >> $O/11-exits.txt
echo "chain1 done $(date -u +%H:%M:%SZ)" >> $T/logs/chain1.log
echo chain1-done > $O/chain1.done
