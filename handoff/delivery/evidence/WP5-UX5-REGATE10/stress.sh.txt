# S <label> <repeat> <workers> <playwright args...>: one real-clock run in the clean export, counted.
S() {
  label=$1; rep=$2; wk=$3; shift 3
  bash run-shift.sh "stress-$label" real "$@" --repeat-each=$rep --workers=$wk > "logs/stress-$label.txt" 2>&1
  echo "$label (repeat $rep, workers $wk): $(node counts.mjs "logs/stress-$label.txt")" >> out/31-stress-counts.txt 2>&1
  echo "$label finished $(date -u +%H:%M:%S)" >> logs/stress.log
}
