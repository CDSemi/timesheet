// WP5-UX-A11Y-SWEEP: condenses the probe's raw JSON into one text summary per tag.
// Usage: node summarize.mjs <rawDir> <tag> <outFile>
import { readdirSync, readFileSync, writeFileSync } from 'node:fs';
import { join } from 'node:path';

const [rawDir, tag, outFile] = process.argv.slice(2);
const files = readdirSync(rawDir).filter((f) => f.startsWith(`${tag}-`));
const lines = [];
const say = (s) => lines.push(s);
const load = (f) => JSON.parse(readFileSync(join(rawDir, f), 'utf8'));

const combos = files.filter((f) => f.endsWith('-results.json')).map((f) => f.slice(tag.length + 1, -'-results.json'.length)).sort();
for (const combo of combos) {
  say(`\n==================== ${combo}`);
  const results = load(`${tag}-${combo}-results.json`);
  for (const [state, r] of Object.entries(results)) {
    const issues = [];
    const tc = r.textContrast.filter((t) => !t.disabled);
    if (tc.length > 0) issues.push(`text<min: ${tc.map((t) => `${t.el} "${t.text}" ${t.fg}/${t.bg}=${t.ratio}(need ${t.need})${t.complex ? ' complex' : ''}`).join(' | ')}`);
    if (r.unnamed.length > 0) issues.push(`unnamed: ${r.unnamed.join(' | ')}`);
    if (r.ariaRefs.length > 0) issues.push(`ariaRefs: ${JSON.stringify(r.ariaRefs)}`);
    if (r.hiddenFocusable.length > 0) issues.push(`hiddenFocusable: ${r.hiddenFocusable.join(' | ')}`);
    if (r.nested.length > 0) issues.push(`nested: ${r.nested.join(' | ')}`);
    const st = r.smallTargets.filter((t) => !t.disabled);
    const stFail = st.filter((t) => !t.spacingOk);
    if (stFail.length > 0) issues.push(`target<24 (spacing fails): ${stFail.map((t) => `${t.el} "${t.name}" ${t.size} label=${t.label} vs ${t.conflicts.join(';')}`).join(' | ')}`);
    if (st.length - stFail.length > 0) issues.push(`target<24 (spacing ok): ${st.filter((t) => t.spacingOk).map((t) => `${t.el} "${t.name}" ${t.size}`).slice(0, 6).join(' | ')}${st.length - stFail.length > 6 ? ` (+${st.length - stFail.length - 6})` : ''}`);
    if (r.reflow.horizontalScroll || r.reflow.overflowing.length > 0) issues.push(`reflow: hScroll=${r.reflow.horizontalScroll} ${r.reflow.scrollWidth}/${r.reflow.clientWidth} overflowing=${JSON.stringify(r.reflow.overflowing.slice(0, 6))}`);
    if (r.reflow.scrollers.length > 0) issues.push(`scrollers: ${JSON.stringify(r.reflow.scrollers.slice(0, 4))}`);
    if (r.textLinks.length > 0) issues.push(`textLinks: ${JSON.stringify(r.textLinks)}`);
    const h1 = r.headings.filter((h) => h.startsWith('H1')).length;
    if (h1 !== 1) issues.push(`h1 count=${h1}: ${r.headings.slice(0, 6).join(' | ')}`);
    say(`-- ${state} (${r.interactiveCount} controls)${issues.length === 0 ? ' OK' : ''}`);
    for (const i of issues) say(`   ${i}`);
  }
  const walks = load(`${tag}-${combo}-walks.json`);
  say(`-- walks`);
  for (const [w, v] of Object.entries(walks)) {
    const s = v.summary;
    say(`   ${w}: ${s.stops} stops, end=${JSON.stringify(s.end).slice(0, 80)}`);
    if (s.entirelyHidden.length > 0) say(`      HIDDEN: ${s.entirelyHidden.join(' | ')}`);
    for (const p of s.problems.filter((p) => p.vis !== '0/25')) {
      const ring = p.ring === null ? '' : ` ring sides=${p.ring.sidesPresent}/${p.ring.sidesSampled} minOuter=${p.ring.minOuter} maxCh=${p.ring.maxChange} refocused=${p.ring.refocused}`;
      say(`      problem: ${p.tag}${p.role ? `[${p.role}]` : ''} "${p.name}" native=${p.native} scrollable=${p.scrollable} fv=${p.fv} vis=${p.vis}${ring} shadow=${String(p.boxShadow).slice(0, 50)}`);
    }
    if (s.upward.length > 0) say(`      upward: ${s.upward.slice(0, 6).join(' | ')}`);
  }
  const d = load(`${tag}-${combo}-dialogs.json`);
  say(`-- dialogs ${JSON.stringify(d.dialogs, (k, val) => (k === 'states' ? undefined : val))}`);
  if (d.dialogs.labelPicker?.states) for (const s of d.dialogs.labelPicker.states) say(`   picker ${s.key || 'open'}: active=${s.active} fvList=${s.listFocusVisible} listShadow=${String(s.listShadow).slice(0, 40)} optBg=${s.optBg} optShadow=${s.optShadow} optOutline=${s.optOutline} rendered=${JSON.stringify(s.rendered)}`);
  say(`-- extra ${JSON.stringify(d.extra)}`);
  if (d.console.length > 0) say(`-- console ${d.console.slice(0, 5).join(' | ')}`);
}
// 2.5.3 Label in Name: a control whose accessible name does not contain its visible text (aria snapshot lines).
const lin = new Map();
for (const f of files.filter((name) => name.endsWith('-aria.txt'))) {
  for (const line of readFileSync(join(rawDir, f), 'utf8').split('\n')) {
    const m = /^\s*- '?(button|link|checkbox|radio|tab|menuitem|switch|option|combobox) "((?:[^"\\]|\\.)*)"(?: \[[^\]]*\])*'?: (.+)$/.exec(line);
    if (m === null) continue;
    const [, role, name, raw] = m;
    const text = raw.replace(/^"|"$/g, '').replace(/\s+/g, ' ').trim().toLowerCase();
    if (text === '' || name.toLowerCase().replace(/\s+/g, ' ').includes(text)) continue;
    const key = `${role} "${name.replace(/\d{4}-\d{2}-\d{2}/g, '{date}').replace(/[0-9a-f]{8}-[0-9a-f-]{27}/g, '{id}')}" visible "${text.replace(/\d{2}\/\d{2}/g, '{mm/dd}')}"`;
    const entry = lin.get(key) ?? { count: 0, files: new Set() };
    entry.count += 1;
    entry.files.add(f.replace(`${tag}-`, '').replace('-aria.txt', ''));
    lin.set(key, entry);
  }
}
say('\n==================== label in name (2.5.3)');
for (const [key, v] of [...lin.entries()].sort()) say(`-- ${key}: ${v.count}x in ${[...v.files].slice(0, 6).join(', ')}${v.files.size > 6 ? ` (+${v.files.size - 6})` : ''}`);

const spacing = files.find((f) => f === `${tag}-text-spacing.json`);
if (spacing !== undefined) {
  say('\n==================== text spacing');
  for (const [k, v] of Object.entries(load(spacing))) say(`-- ${k}: clipped=${v.clippedCount} spills=${v.spillCount} hScroll=${v.hScroll} ${JSON.stringify(v.clipped.slice(0, 6))} ${JSON.stringify(v.spills.slice(0, 6))}`);
}
writeFileSync(outFile, `${lines.join('\n')}\n`);
