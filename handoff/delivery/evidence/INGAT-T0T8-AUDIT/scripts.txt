#!/bin/bash
# A3b: run every manifest task argv, the readiness probe argv and the live/ready routes against the locally built image
# (ingat-t0t8-audit:local from a5-build.sh) with the Compose hardening (read-only root, /tmp tmpfs, init, cap_drop ALL,
# no-new-privileges). Synthetic data in throw-away named volumes; every container --rm; volumes removed at the end.
export MSYS_NO_PATHCONV=1
export TEMP="D:/.claude-tmp/timesheet/INGAT-T0T8-AUDIT/tmp"
export TMP="$TEMP"
TASK="D:/.claude-tmp/timesheet/INGAT-T0T8-AUDIT"
REPO="D:/Dropbox/Work.CDSemi/timesheet/.claude/worktrees/relaxed-lalande-8359f0"
IMG=ingat-t0t8-audit:local
OUT="$TASK/a3-tasks.txt"
APP=ingat-audit-app
PORT=49831
HARD="--rm --init --read-only --tmpfs /tmp:size=64m,mode=1777 --cap-drop ALL --security-opt no-new-privileges"
ENVS="-e APP_ORIGINS=https://timesheet.example.invalid -e PUBLIC_BASE_URL=https://timesheet.example.invalid -e MAIL_FROM=timesheet@example.invalid -e OUTBOUND_MODE=capture"
CLI="node dist/server/cli.js"
PROBE="fetch('http://127.0.0.1:3000/api/ready',{signal:AbortSignal.timeout(4000)}).then(async r=>{process.stdout.write(await r.text());process.exit(r.ok?0:1)},()=>process.exit(1))"
redact() { sed 's/[A-Za-z0-9_-]\{24,\}/<redacted-token>/g'; }
task() {  # task <label> <volume args> -- <cli args...>
  label=$1; vols=$2; shift 2
  echo "=== $label"
  echo "+ docker run $HARD \$ENVS $vols $IMG $CLI $*"
  docker run $HARD $ENVS $vols $IMG $CLI "$@" > "$TASK/t.out" 2> "$TASK/t.err"
  rc=$?
  echo "stdout:"; redact < "$TASK/t.out"
  echo "stderr:"; redact < "$TASK/t.err"
  echo "exit $rc"
}
wait_ready() {  # wait_ready <label>
  echo "=== $1: probes.ready argv via docker exec (retry until exit 0, at most 40 tries)"
  for i in $(seq 1 40); do
    docker exec "$APP" node -e "$PROBE" > "$TASK/p.out" 2> "$TASK/p.err"; rc=$?
    [ "$rc" = 0 ] && break
    sleep 1
  done
  echo "tries $i"; echo "stdout:"; cat "$TASK/p.out"; echo; echo "stderr:"; cat "$TASK/p.err"; echo "exit $rc"
}
{
  echo "--- the probe argv used here equals the manifest's argv[2] (grep -F -c, expect 1)"
  grep -F -c "$PROBE" "$REPO/ingat.toml"
  docker volume create ingat-audit-data; docker volume create ingat-audit-stage; docker volume create ingat-audit-gen
  docker volume create ingat-audit-placeholder
  echo "--- stage and generation owned by the task user 10001:10001 (Ingat creates them so; root helper simulates it)"
  docker run --rm --user 0:0 --entrypoint sh -v ingat-audit-stage:/out -v ingat-audit-gen:/restore "$IMG" \
    -c "chown 10001:10001 /out /restore && chmod 0700 /restore && ls -ldn /out /restore"
  echo "helper exit $?"
  task "tasks.migrate (expect 0)" "-v ingat-audit-data:/data" migrate
  task "tasks.migrate again: idempotent (expect 0, applied [])" "-v ingat-audit-data:/data" migrate
  task "tasks.bootstrap-new-token before any config (expect 1 = refused)" "-v ingat-audit-data:/data" bootstrap --new-token
  task "tasks.bootstrap-config (expect 0; token redacted)" "-v ingat-audit-data:/data -v $(cygpath -w "$TASK/input"):/in:ro" bootstrap --config /in/bootstrap.json
  task "tasks.bootstrap-config again (expect 1 = refused)" "-v ingat-audit-data:/data -v $(cygpath -w "$TASK/input"):/in:ro" bootstrap --config /in/bootstrap.json
  task "tasks.bootstrap-new-token (expect 0; token redacted)" "-v ingat-audit-data:/data" bootstrap --new-token
  echo "=== start the service as Compose would (detached, loopback port)"
  docker run -d --name "$APP" $HARD $ENVS -p "127.0.0.1:$PORT:3000" -v ingat-audit-data:/data "$IMG"
  echo "run -d exit $?"
  wait_ready "live data"
  echo "=== probes.edge-live and smoke over the published loopback port"
  curl -sS -o "$TASK/h.out" -w "health http %{http_code}\n" "http://127.0.0.1:$PORT/api/health"; cat "$TASK/h.out"; echo
  curl -sS -o "$TASK/r.out" -w "ready http %{http_code}\n" "http://127.0.0.1:$PORT/api/ready"; cat "$TASK/r.out"; echo
  task "tasks.backup while the server runs (expect 0, one JSON line)" "-v ingat-audit-data:/data -v ingat-audit-stage:/out" backup --to /out
  echo "--- stage entries (Ingat needs exactly one)"
  docker run --rm --entrypoint ls -v ingat-audit-stage:/out "$IMG" -la /out
  task "backup refused: target inside DATA_DIR (expect 2 = refused)" "-v ingat-audit-data:/data" backup --to /data/private-data/x
  task "tasks.outbound-held preview (expect 2, outcome confirmation_required)" "-v ingat-audit-data:/data" outbound release
  task "tasks.outbound-status preview (expect 2, outcome confirmation_required)" "-v ingat-audit-data:/data" outbound resume
  task "preview usage error (expect 2, no JSON on stdout: why the manifest needs expect)" "-v ingat-audit-data:/data" outbound release --bogus
  task "tasks.outbound-release-job unknown id (expect 1 = refused)" "-v ingat-audit-data:/data" outbound release --job 00000000-0000-4000-8000-000000000000 --confirm
  task "tasks.outbound-release-all (expect 0)" "-v ingat-audit-data:/data" outbound release --all --confirm
  task "tasks.outbound-drop-job unknown id (expect 1 = refused)" "-v ingat-audit-data:/data" outbound drop --job 00000000-0000-4000-8000-000000000000 --confirm
  task "tasks.outbound-resume, not paused (expect 0)" "-v ingat-audit-data:/data" outbound resume --confirm
  echo "=== stop the service (SIGTERM; it must exit by itself)"
  docker stop "$APP"; echo "stop exit $?"
  ENTRY=$(docker run --rm --entrypoint ls -v ingat-audit-stage:/out "$IMG" /out)
  echo "payload entry: $ENTRY"
  task "tasks.restore into the empty generation, /data = empty placeholder (expect 0)" "-v ingat-audit-placeholder:/data:ro -v ingat-audit-stage:/in:ro -v ingat-audit-gen:/restore" restore --from "/in/$ENTRY" --to /restore
  task "tasks.restore again into the now non-empty target (expect 2 = refused)" "-v ingat-audit-placeholder:/data:ro -v ingat-audit-stage:/in:ro -v ingat-audit-gen:/restore" restore --from "/in/$ENTRY" --to /restore
  echo "--- restored generation layout"
  docker run --rm --entrypoint ls -v ingat-audit-gen:/g "$IMG" -la /g
  echo "=== start the service on the restored generation with the holds (JOB_RUNNER=off, OUTBOUND_MODE=capture)"
  docker run -d --name "$APP" $HARD $ENVS -e JOB_RUNNER=off -v ingat-audit-gen:/data "$IMG"
  echo "run -d exit $?"
  wait_ready "restored generation"
  docker logs "$APP" > "$TASK/l.out" 2>&1; echo "--- server log"; cat "$TASK/l.out"
  task "tasks.outbound-held preview on the restored copy (expect 2)" "-v ingat-audit-gen:/data" outbound release
  task "tasks.outbound-status preview on the restored copy (expect 2, paused true reason restored)" "-v ingat-audit-gen:/data" outbound resume
  task "tasks.outbound-resume on the restored copy (expect 0)" "-v ingat-audit-gen:/data" outbound resume --confirm
  docker stop "$APP"; echo "stop exit $?"
  echo "--- cleanup"
  docker volume rm ingat-audit-data ingat-audit-stage ingat-audit-gen ingat-audit-placeholder
  echo "volume rm exit $?"
  docker ps -a --filter "name=$APP" --format '{{.Names}} {{.Status}}'
  echo "ps exit $?"
} > "$OUT" 2>&1

#!/bin/bash
# A5: local linux/amd64 build with the workflow's flags (BuildKit pinned as in image.yml; --load instead of --push).
export MSYS_NO_PATHCONV=1
export TEMP="D:/.claude-tmp/timesheet/INGAT-T0T8-AUDIT/tmp"

REPO="D:/Dropbox/Work.CDSemi/timesheet/.claude/worktrees/relaxed-lalande-8359f0"

TAG=ingat-t0t8-audit:local
cd "$REPO" || exit 3
{
  echo "--- buildx version"; docker buildx version
  GITHUB_SHA=$(git rev-parse HEAD)
  GITHUB_SERVER_URL=https://github.com
  GITHUB_REPOSITORY=CDSemi/timesheet
  version=$(sed -n 's/^  "version": "\([0-9][0-9.]*\)",$/\1/p' package.json)
  echo "version=[$version] GITHUB_SHA=$GITHUB_SHA"
  test -n "$version"; echo "test -n exit $?"
  echo "--- create builder (docker-container, BuildKit v0.33.1 pinned by digest, as setup-buildx driver-opts)"
  docker buildx create --name "$B" --driver docker-container \
    --driver-opt image=moby/buildkit:v0.33.1@sha256:cec9f139f45e93c5c69c60f8b07cfad9f43f4ef6b6a6cd917527fea5ff2e3dea
  echo "create exit $?"
  echo "--- build"
  docker buildx build --builder "$B" --platform linux/amd64 --provenance=false --sbom=false \
    --label "org.opencontainers.image.version=$version" \
    --label "org.opencontainers.image.revision=$GITHUB_SHA" \
    --label "org.opencontainers.image.source=$GITHUB_SERVER_URL/$GITHUB_REPOSITORY" \
    --tag "$TAG" --load --progress plain .
  echo "build exit $?"
  echo "--- inspect (the workflow's format plus source)"
  got=$(docker image inspect --format '{{.Os}}/{{.Architecture}} {{.Config.User}} {{index .Config.Labels "org.opencontainers.image.revision"}} {{index .Config.Labels "org.opencontainers.image.version"}}' "$TAG")
  want="linux/amd64 10001:10001 $GITHUB_SHA $version"
  echo "got  [$got]"; echo "want [$want]"
  [ "$got" = "$want" ]; echo "compare exit $?"
  docker image inspect --format 'source={{index .Config.Labels "org.opencontainers.image.source"}}' "$TAG"
  echo "--- entrypoint, cmd, workdir, env, all labels, healthcheck"
  docker image inspect --format 'Entrypoint={{json .Config.Entrypoint}} Cmd={{json .Config.Cmd}} WorkingDir={{.Config.WorkingDir}}' "$TAG"
  docker image inspect --format '{{json .Config.Env}}' "$TAG"
  docker image inspect --format '{{json .Config.Labels}}' "$TAG"
  docker image inspect --format '{{json .Config.Healthcheck}}' "$TAG"
  echo "--- node version and libc in the image (node -e runs inside the app container)"
  docker run --rm --read-only --network none --cap-drop ALL --security-opt no-new-privileges "$TAG" node -e "console.log(process.version, process.arch, JSON.stringify(process.report.getReport().header.glibcVersionRuntime))"
  echo "node exit $?"
  docker run --rm --read-only --network none --cap-drop ALL --security-opt no-new-privileges --entrypoint cat "$TAG" /etc/os-release
  echo "os-release exit $?"
  echo "--- remove builder"
  docker buildx rm "$B"
  echo "rm exit $?"
  docker buildx ls
} > "$OUT" 2>&1
