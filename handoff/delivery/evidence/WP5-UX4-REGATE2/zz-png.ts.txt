import { inflateSync } from 'node:zlib';

/* WP5-UX4-GATE task-local: a minimal PNG decoder (8-bit RGB/RGBA, all five scanline filters) for pixel analysis of screenshots. */
export interface Image {
  width: number;
  height: number;
  /** RGBA, row-major. */
  data: Uint8Array;
}

export function decodePng(png: Buffer): Image {
  let width = 0;
  let height = 0;
  let colourType = 0;
  const parts: Buffer[] = [];
  for (let offset = 8; offset < png.length; ) {
    const length = png.readUInt32BE(offset);
    const type = png.toString('latin1', offset + 4, offset + 8);
    const body = png.subarray(offset + 8, offset + 8 + length);
    if (type === 'IHDR') {
      width = body.readUInt32BE(0);
      height = body.readUInt32BE(4);
      colourType = body.readUInt8(9);
      if (body.readUInt8(8) !== 8) throw new Error('only 8-bit PNG');
    }
    if (type === 'IDAT') parts.push(body);
    offset += 12 + length;
  }
  const channels = colourType === 6 ? 4 : colourType === 2 ? 3 : 0;
  if (channels === 0) throw new Error(`unsupported colour type ${colourType}`);
  const raw = inflateSync(Buffer.concat(parts));
  const stride = width * channels;
  const out = new Uint8Array(width * height * 4);
  let prev = new Uint8Array(stride);
  for (let y = 0; y < height; y += 1) {
    const filter = raw[y * (stride + 1)] ?? 0;
    const line = new Uint8Array(stride);
    for (let i = 0; i < stride; i += 1) {
      const x = raw[y * (stride + 1) + 1 + i] ?? 0;
      const a = i >= channels ? (line[i - channels] ?? 0) : 0;
      const b = prev[i] ?? 0;
      const c = i >= channels ? (prev[i - channels] ?? 0) : 0;
      let value: number;
      switch (filter) {
        case 0:
          value = x;
          break;
        case 1:
          value = x + a;
          break;
        case 2:
          value = x + b;
          break;
        case 3:
          value = x + ((a + b) >> 1);
          break;
        default: {
          const p = a + b - c;
          const pa = Math.abs(p - a);
          const pb = Math.abs(p - b);
          const pc = Math.abs(p - c);
          value = x + (pa <= pb && pa <= pc ? a : pb <= pc ? b : c);
        }
      }
      line[i] = value & 255;
    }
    for (let x = 0; x < width; x += 1) {
      const o = (y * width + x) * 4;
      out[o] = line[x * channels] ?? 0;
      out[o + 1] = line[x * channels + 1] ?? 0;
      out[o + 2] = line[x * channels + 2] ?? 0;
      out[o + 3] = channels === 4 ? (line[x * channels + 3] ?? 255) : 255;
    }
    prev = line;
  }
  return { width, height, data: out };
}

export function pixel(image: Image, x: number, y: number): [number, number, number] {
  const o = (y * image.width + x) * 4;
  return [image.data[o] ?? 0, image.data[o + 1] ?? 0, image.data[o + 2] ?? 0];
}

export const lum = ([r, g, b]: [number, number, number]) => 0.2126 * r + 0.7152 * g + 0.0722 * b;
