function run(cwd, label, args, timeoutMs) {
  const started = Date.now();
  const vitest = join(cwd, 'node_modules', 'vitest', 'vitest.mjs');
  const result = spawnSync(process.execPath, [vitest, 'run', ...args], { cwd, encoding: 'utf8', timeout: timeoutMs, env: { ...process.env, NO_COLOR: '1', FORCE_COLOR: '0', RECHECK_B2_OUT: join(logDir, `mut-${label}.jsonl`) }, maxBuffer: 256 * 1024 * 1024 });
  const output = `${result.stdout ?? ''}${result.stderr ?? ''}`;
  writeFileSync(join(logDir, `mut-${label}.log`), output);
  const tail = output
    .split('\n')
    .filter((line) => /Test Files|Tests |FAIL |✓|×/.test(line) && !/^\s*✓/.test(line))
    .slice(-14)
    .join(' / ');
  return { status: result.status, seconds: Math.round((Date.now() - started) / 1000), tail };
}

for (const mutant of MUTANTS.filter((item) => only.length === 0 || only.includes(item.id))) {
  const dir = mutant.dir === 'probe' ? probeDir : mutDir;
  const path = join(dir, mutant.file);
  const original = readFileSync(path, 'utf8');
  const occurrences = original.split(mutant.from).length - 1;
  if (occurrences !== 1) {
    appendFileSync(summary, `${mutant.id} ANCHOR found ${occurrences} times, not run\n`);
    continue;
  }
  writeFileSync(path, original.replace(mutant.from, mutant.to));
  let line = `${mutant.id} (${mutant.file}${mutant.dir === 'probe' ? ', probe export' : ''}): ${mutant.what}`;
  try {
    const targeted = run(dir, `${mutant.id}-targeted`, mutant.tests, 900_000);
    line += ` | targeted ${mutant.tests.join(' ')} exit ${targeted.status} (${targeted.seconds} s): ${targeted.tail}`;
    if (targeted.status === 0 && mutant.noFull !== true) {
      const full = run(dir, `${mutant.id}-full`, [], 1_800_000);
      line += ` | FULL exit ${full.status} (${full.seconds} s): ${full.tail}`;
    }
  } finally {
    writeFileSync(path, original);
    line += ` | restored ${sha(readFileSync(path, 'utf8')) === sha(original) ? 'byte-identical' : 'DIFFERENT'}`;
    appendFileSync(summary, `${line}\n`);
  }
}
appendFileSync(summary, `done ${new Date().toISOString()}\n`);
