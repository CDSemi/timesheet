// The last run-jobs instant per database: every instance's instants only move forward.
const lastInstant = new Map();
const runs = [];
function runJobsAt(env, date, label) {
  if (date.getTime() < (lastInstant.get(env.DATABASE_PATH) ?? 0)) throw new Error(`non-monotonic run-jobs instant ${utc(date)}`);
  lastInstant.set(env.DATABASE_PATH, date.getTime());
  const r = runCli(rec, env, ['run-jobs', '--once', '--now', utc(date)], `run-jobs ${utc(date)} ${label}`);
  cliOutputs.push(r.out);
  runs.push({ at: utc(date), label, status: r.status, out: r.out });
  return r;
}
const at = (date, time) => ({ local: `${date}T${time}`, zone: ZONE });

function bootstrapConfig(anchor) {
  return {
    schema_version: 1,

}
async function session(c, date) {
  const r = await c.call('POST', `/api/days/${date}/sessions`, {
    start: at(date, '08:00'),
    end: at(date, '17:00'),
    input_zone: ZONE,
    breaks_confirmed: true,
    breaks: BREAKS.map(({ start: [from, to] }) => ({ start: at(date, from), end: at(date, to), counts_as_work: false })),
  });
  if (r.status !== 201) throw new Error(`session ${date}: ${r.status} ${r.text}`);
  return r.json;
}
async function signOff(c, name, payroll) {
  const review = await c.call('GET', `/api/timesheets/${payroll}/review`);
  return c.call('POST', `/api/timesheets/${payroll}/signoff`, { expected_version: review.json?.expected_version, reviewed_hash: review.json?.payload_hash, signer_name: name, incomplete_evidence_acknowledged: true });
}
const realNowAfter = async (ms) => {
  const instant = new Date(Date.now() + ms);
  await sleep(ms + 200);
  return instant;
};

  const ready = await anon().call('GET', '/api/ready');
  rec.check('PRIV-1', 'health and readiness expose no personal data or path', health.text === '{"status":"ok"}' && ready.json?.status === 'ready' && !ready.text.includes('example.invalid') && !ready.text.includes('\\') && !/[A-Za-z]:\//.test(ready.text), `${health.text} ${ready.text}`);
  writeFileSync(join(runDir, 'capture-listing.json'), `${JSON.stringify(listCaptures(dataDir).map((c) => ({ id: c.id, files: c.files, from: c.metadata?.envelope?.from, to: c.metadata?.envelope?.to, pdf_sha256: c.pdfSha256, ...plainBody(c.eml) })), null, 2)}\n`);
  monitor.close();
  return { people };
}

async function faults() {
  const configPath = join(runDir, 'faults-bootstrap-config.json');
  writeFileSync(configPath, `${JSON.stringify(bootstrapConfig('2026-10-09'), null, 2)}\n`);
  const boot = runCli(rec, fServerEnv, ['bootstrap', '--config', configPath], 'faults bootstrap');
  fToken = boot.stdout.match(/^\s+([A-Z2-7]{5}(?:-[A-Z2-7]{5})+)\s*$/m)?.[1] ?? '';
  rec.addSecret(fToken);
  cliOutputs.push(boot.out);
  await fServer.start();
  const base = `http://127.0.0.1:${FPORT}`;
  const admin = new Client(base, FPUBLIC, rec, 'fadmin');
  const adminPassword = secret();
  rec.addSecret(adminPassword);
  await admin.call('POST', '/api/auth/bootstrap', { token: fToken, email: 'faults-admin@example.invalid', display_name: 'Faults Admin', password: adminPassword });
  await admin.login('faults-admin@example.invalid', adminPassword);
  const calendarId = (await admin.call('GET', '/api/calendar')).json?.id;
  const people = { G: { email: 'gina.synthetic@example.invalid', name: 'Gina Synthetic' }, H: { email: 'hugo.synthetic@example.invalid', name: 'Hugo Synthetic' } };
  const api = {};
  for (const [key, p] of Object.entries(people)) {
    Object.assign(p, { password: secret() });
    rec.addSecret(p.password);
    const created = await admin.call('POST', '/api/admin/users', { email: p.email, display_name: p.name, role: 'employee', password: p.password, calendar_id: calendarId });
    p.id = created.json?.user?.id;
    api[key] = new Client(base, FPUBLIC, rec, key);
    await api[key].login(p.email, p.password);
    await api[key].call('POST', '/api/policies', policy);
    const sig = await api[key].call('POST', '/api/signatures', new Uint8Array(syntheticSignaturePng(key === 'G' ? 5 : 6)), { headers: { 'content-type': 'image/png' } });
    if (sig.status !== 201) throw new Error(`signature ${key}: ${sig.status} ${sig.text}`);
  }
  const current = (await api.G.call('GET', '/api/periods/current')).json?.current;
  await session(api.G, current.period_start);
  await session(api.H, current.period_start);
  // Gina saved recipients; Hugo never saved submission settings (no recipient). No MAIL_FROM anywhere at first.
  await api.G.call('POST', '/api/settings/submission', { expected_seq: 0, to: ['payroll-g@example.invalid'], auto_submit: false });
  const g = await signOff(api.G, people.G.name, current.payroll_date);
  const t0 = await realNowAfter(1500);
  runJobsAt(fCliEnv, t0, 'faults: no sender');
  runJobsAt(fCliEnv, new Date(t0.getTime() + 120_000), 'faults: no sender b');
  const gDel = (await api.G.call('GET', '/api/deliveries')).json?.deliveries ?? [];
  const gRevs = (await api.G.call('GET', '/api/revisions')).json?.revisions ?? [];
  const subs = await admin.call('GET', '/api/admin/submissions');
  writeFileSync(join(runDir, 'faults-gina-deliveries.json'), `${JSON.stringify(gDel, null, 2)}\n`);
  writeFileSync(join(runDir, 'faults-admin-submissions.json'), `${JSON.stringify(subs.json, null, 2)}\n`);
  rec.check('F-1', 'missing sender: the attempt fails permanently with sender_missing, the revision list shows it failed, never accepted; nothing captured', g.status === 201 && gDel.length === 1 && gDel[0].state === 'failed_permanent' && /sender_missing/.test(JSON.stringify(gDel[0])) && gRevs[0]?.delivery_state === 'failed_permanent' && listCaptures(fDataDir).length === 0, JSON.stringify(gDel.map((d) => [d.state, d.provider_response])));
  rec.check('F-2', 'the administrator submission status shows the fault code and the frozen recipients, with no timesheet details', /failed_permanent/.test(subs.text) && /sender_missing/.test(subs.text) && subs.text.includes('payroll-g@example.invalid') && !/"(sessions|days|credited_minutes|notes|body_text|subject|payload)"/.test(subs.text), subs.text.slice(0, 300));
  const h = await signOff(api.H, people.H.name, current.payroll_date);
  const withSender = { ...fCliEnv, MAIL_FROM: 'faults-sender@example.invalid' };
  runJobsAt(withSender, new Date(t0.getTime() + 600_000), 'faults: with sender (Hugo)');
  runJobsAt(withSender, new Date(t0.getTime() + 720_000), 'faults: with sender (Hugo) b');
  const hDel = (await api.H.call('GET', '/api/deliveries')).json?.deliveries ?? [];
  const gDel2 = (await api.G.call('GET', '/api/deliveries')).json?.deliveries ?? [];
  writeFileSync(join(runDir, 'faults-hugo-deliveries.json'), `${JSON.stringify(hDel, null, 2)}\n`);
  rec.check('F-3', 'missing recipient: Hugo\'s sign-off is accepted, the delivery fails permanently with recipient_missing, never accepted; nothing captured', h.status === 201 && hDel.length === 1 && hDel[0].state === 'failed_permanent' && /recipient_missing/.test(JSON.stringify(hDel[0])) && listCaptures(fDataDir).length === 0, JSON.stringify(hDel.map((d) => [d.state, d.provider_response])));
  rec.check('F-4', 'Gina\'s permanent failure is not retried automatically once a sender exists', gDel2.length === 1 && gDel2[0].state === 'failed_permanent', JSON.stringify(gDel2.map((d) => d.state)));
  await fServer.stop();
  // SMTP without the owner flag: refused before any network use; the message names the flag, never a value.
  const smtpPassword = `Synthetic-${secret()}`;
  rec.addSecret(smtpPassword);
  const smtpEnv = { ...fServerEnv, OUTBOUND_MODE: 'smtp', SMTP_HOST: 'smtp.example.invalid', SMTP_PORT: '587', SMTP_USER: 'synthetic-user@example.invalid', SMTP_PASSWORD: smtpPassword, MAIL_FROM: 'faults-sender@example.invalid' };
  const smtpServer = new Server('smtp-refused', smtpEnv, rec, FPORT);
  let refused = false;
  try {
    await smtpServer.start();
  } catch (error) {
    refused = /exited early/.test(String(error));
  }
  await smtpServer.stop();
  const smtpCli = runCli(rec, { ...smtpEnv, NODE_ENV: 'development' }, ['run-jobs', '--once', '--now', utc(new Date(t0.getTime() + 900_000))], 'run-jobs smtp without flag');
  const smtpOut = `${smtpServer.output}\n${smtpCli.out}`;
  rec.check('F-5', 'OUTBOUND_MODE=smtp without PRODUCTION_SENDING_ENABLED: server and CLI refuse, the message names the flag, the SMTP password never appears', refused && smtpCli.status !== 0 && /PRODUCTION_SENDING_ENABLED/.test(smtpOut) && !smtpOut.includes(smtpPassword), `server exit ${smtpServer.child?.exitCode} cli ${smtpCli.status}`);
  cliOutputs.push(smtpServer.output, smtpCli.out);
  return fServer.output;
}

let fOutput = '';
try {
  await scenario();
  await server.stop();
  fOutput = await faults();
  const everything = `${server.output}\n${fOutput}\n${cliOutputs.join('\n')}`;
  const leaked = [...rec.secrets].filter((value) => value !== token && value !== fToken && everything.includes(value));
  const tokenOutsideBootstrap = `${server.output}\n${fOutput}\n${cliOutputs.filter((o) => !o.includes('Setup token') && !o.includes('setup token')).join('\n')}`;
  rec.check('PRIV-2', 'server logs and CLI outputs of both instances hold no password or session cookie; each setup token appears only in its own bootstrap output', leaked.length === 0 && (token === '' || !tokenOutsideBootstrap.includes(token)) && (fToken === '' || !tokenOutsideBootstrap.includes(fToken)) && !/password/i.test(`${server.output}${fOutput}`), `leaked ${leaked.length}; secrets tracked ${rec.secrets.size}`);
  writeFileSync(join(runDir, 'server-log.txt'), rec.mask(`${server.output}\n--- faults server ---\n${fOutput}`));
  writeFileSync(join(runDir, 'cli-outputs.txt'), rec.mask(cliOutputs.join('\n')));
  writeFileSync(join(runDir, 'runs.json'), rec.mask(`${JSON.stringify(runs, null, 2)}\n`));
} catch (error) {
  rec.check('PROBE', 'probe ran to the end without an unexpected error', false, error instanceof Error ? `${error.message}\n${error.stack}` : String(error));
} finally {
  await server.stop();
  await fServer.stop();
  const { pass, fail } = rec.write();
  console.log(`probe ${runDir.replace(T, '<task>')}: pass ${pass} fail ${fail}`);
  process.exitCode = fail === 0 ? 0 : 1;
}
