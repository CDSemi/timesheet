// WP5-ASSESS-A2 preload (NODE_OPTIONS=--import): Git Bash (MSYS) drops TZ when it starts a native process, so the
// machine zone is set at run time from AUDIT_TZ instead. Every Node child inherits NODE_OPTIONS and the environment.
if (process.env.AUDIT_TZ !== undefined && process.env.AUDIT_TZ !== '') process.env.TZ = process.env.AUDIT_TZ;
