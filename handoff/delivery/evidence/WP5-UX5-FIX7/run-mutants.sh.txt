#!/bin/bash
# usage: run-mutants.sh <out-name> <mutant ids...>   (mutates the scratch export, restores it, records byte-identical restore)
set -u
T="<task>"
export PATH="$(cygpath -u "$LOCALAPPDATA")/timesheet-dev/node-24.21.0/node_modules/node/bin:$(cygpath -u "$APPDATA")/npm:$PATH"
export TEMP="$(cygpath -w "$T/tmp")" TMP="$(cygpath -w "$T/tmp")"
export DATA_DIR="$T/data" DATABASE_PATH="$T/data/probe.db"
mkdir -p "$T/out/mut"
OUT="$1"; shift
cd "$T/probes" || exit 1
node.exe mutants.mjs "$(cygpath -w "$T/export")" "$(cygpath -w "$T/out/mut")" "$(cygpath -w "$T/out/$OUT")" "$@" > "$T/out/$OUT.log" 2>&1
echo "exit $?" >> "$T/out/$OUT.log"
