#!/bin/bash
# Container drill with --wp3 from a clean export of the working tree. Compose project name carries "fix7".
set -u
T="<task>"
TW='<task>'
export PATH="$(cygpath -u "$LOCALAPPDATA")/timesheet-dev/node-24.21.0/node_modules/node/bin:$(cygpath -u "$APPDATA")/npm:$PATH"
export GIT_CONFIG_COUNT=1 GIT_CONFIG_KEY_0=safe.directory GIT_CONFIG_VALUE_0='*'
export TEMP="$TW\\tmp" TMP="$TW\\tmp" TMPDIR="$T/tmp"
export npm_config_update_notifier=false npm_config_fund=false
NPM="$(cygpath -u "$APPDATA")/npm/node_modules/npm/bin/npm-cli.js"
mkdir -p "$T/wp3" "$T/drillwork" "$T/out"
cd "<repo>" || exit 1
git archive 49651c8bb91d56bf6c6966405257537ec7ca474b | tar -x -C "$T/wp3"
cd "$T/wp3" || exit 1
node.exe "$NPM" ci > "$T/out/30-wp3-ci.txt" 2>&1; echo "wp3 ci exit $?" > "$T/out/30-exits.txt"
node.exe "$NPM" run build:server > "$T/out/30-wp3-build.txt" 2>&1; echo "wp3 build:server exit $?" >> "$T/out/30-exits.txt"
cd "$T/wt-export3" || exit 1
node.exe "$NPM" ci > "$T/out/30-export-ci.txt" 2>&1; echo "export ci exit $?" >> "$T/out/30-exits.txt"
echo "drill start $(date -u +%H:%M:%SZ)" >> "$T/out/30-exits.txt"
node.exe scripts/container-drill.mjs --work "$TW\\drillwork" --project ts-fix7-drill --wp3 "$TW\\wp3" > "$T/out/30-drill.txt" 2>&1
echo "drill exit $?" >> "$T/out/30-exits.txt"
docker compose -p ts-fix7-drill down -v > "$T/out/30-compose-down.txt" 2>&1; echo "compose down exit $?" >> "$T/out/30-exits.txt"
docker ps --all --filter label=com.docker.compose.project=ts-fix7-drill > "$T/out/30-docker-after.txt" 2>&1
echo drill-done > "$T/out/drill.done"
