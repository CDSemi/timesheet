import { appendFileSync, readFileSync } from 'node:fs';
import type { Page } from '@playwright/test';
import { namedDayButton } from './dayButton.ts';
import { expect, newPerson, test } from './fixtures.ts';

/*
 * WP5-UX5-RECHECK-A probe: the 1280px keyboard walk with the floating drawer open, settled (two frames after each Tab), recording for every
 * stop that looks hidden what covers it; and the elements axe lists as label-content-name-mismatch (incomplete). Synthetic data only.
 */
test.use({ locale: 'en-US', timezoneId: 'America/Los_Angeles' });
const OUT = process.env.RA_OUT ?? '';
const AXE_PATH = process.env.AXE_PATH ?? '';
function note(kind: string, data: unknown): void {
  const line = JSON.stringify({ kind, ...(data as object) });
  if (OUT !== '') appendFileSync(OUT, `${line}\n`);
}
const frames = (page: Page) => page.evaluate('new Promise((done) => requestAnimationFrame(() => requestAnimationFrame(done)))');

const STOP = `(() => {
  const a = document.activeElement;
  if (!a || a === document.body) return { none: true };
  const b = a.getBoundingClientRect();
  const vw = document.documentElement.clientWidth, vh = document.documentElement.clientHeight;
  const x0 = Math.max(b.left, 0), x1 = Math.min(b.right, vw), y0 = Math.max(b.top, 0), y1 = Math.min(b.bottom, vh);
  const inView = x1 > x0 && y1 > y0;
  let visiblePoints = 0; const covers = [];
  if (inView) for (const [fx, fy] of [[0.5, 0.5], [0.2, 0.5], [0.8, 0.5], [0.5, 0.2], [0.5, 0.8]]) {
    const hit = document.elementFromPoint(x0 + (x1 - x0) * fx, y0 + (y1 - y0) * fy);
    if (hit && (hit === a || a.contains(hit) || hit.contains(a))) visiblePoints += 1; else if (hit) covers.push(hit.tagName + '.' + String(hit.className).split(' ')[0]);
  }
  const ed = document.querySelector('[data-day-editor]');
  return { tag: a.tagName, cls: String(a.className).slice(0, 30), name: (a.getAttribute('aria-label') || a.innerText || '').trim().slice(0, 30), box: [Math.round(b.left), Math.round(b.top), Math.round(b.right), Math.round(b.bottom)], inView, visiblePoints, covers: [...new Set(covers)], side: ed ? ed.getAttribute('data-side') : null, scrollY: Math.round(scrollY) };
})()`;

for (const scheme of ['light', 'dark'] as const) {
  test(`settled walk at 1280 with the floating drawer, and axe incomplete label-content-name-mismatch nodes - ${scheme}`, async ({ page, adminSeed, builtServer, isMobile }) => {
    test.setTimeout(240_000);
    test.skip(isMobile, 'desktop project');
    await page.emulateMedia({ colorScheme: scheme, reducedMotion: 'reduce' });
    const person = await newPerson(adminSeed, builtServer, { displayName: `Synthetic Walk ${scheme}`, signature: 'saved' });
    const free = await person.api.displayedPeriodFreeWorkdays();
    await page.setViewportSize({ width: 1280, height: 800 });
    await page.goto('/#/timesheet');
    await page.getByLabel('Email').fill(person.account.email);
    await page.getByLabel('Password').fill(person.account.password);
    await page.getByRole('button', { name: 'Sign in' }).click();
    await expect(page.locator('[data-day]')).toHaveCount(14);
    await namedDayButton(page, free[0] ?? '', 'Edit').click();
    await expect(page.getByRole('dialog', { name: /Day editor/ })).toBeVisible();
    let stops = 0;
    let hiddenImmediate = 0;
    let hiddenSettled = 0;
    const settledProblems: unknown[] = [];
    for (let i = 0; i < 160; i += 1) {
      await page.keyboard.press('Tab');
      const now = await page.evaluate<Record<string, unknown>>(STOP);
      if (now.none) continue;
      stops += 1;
      if (!now.inView || now.visiblePoints === 0) hiddenImmediate += 1;
      await frames(page);
      await frames(page);
      const settled = await page.evaluate<Record<string, unknown>>(STOP);
      if (!settled.inView || settled.visiblePoints === 0) {
        hiddenSettled += 1;
        if (settledProblems.length < 12) settledProblems.push({ i, ...settled });
      }
    }
    note('walk-settled', { scheme, stops, hiddenImmediate, hiddenSettled, settledProblems });
    // axe incomplete label-content-name-mismatch nodes at 1280, 390 and 320 on the Timesheet and the Review.
    await page.keyboard.press('Escape');
    const axeSource = readFileSync(AXE_PATH, 'utf8');
    const nodes: Record<string, string[]> = {};
    for (const [hash, width] of [['#/timesheet', 1280], ['#/timesheet', 390], ['#/timesheet', 320], [`#/review/${person.payrollDate}`, 1280], [`#/review/${person.payrollDate}`, 320]] as const) {
      await page.goto(`/${hash}`);
      await page.setViewportSize({ width, height: 900 });
      await expect(page.locator('main')).toBeVisible();
      await page.waitForTimeout(500);
      if (!(await page.evaluate<boolean>(`typeof window.axe !== 'undefined'`))) await page.evaluate(axeSource);
      nodes[`${hash.split('/')[1]} ${width}`] = await page.evaluate<string[]>(`axe.run(document, { runOnly: { type: 'rule', values: ['label-content-name-mismatch'] }, resultTypes: ['incomplete', 'violations'] }).then((r) => [...r.incomplete, ...r.violations].flatMap((v) => v.nodes.map((n) => v.id + ': ' + n.html.slice(0, 160))))`);
    }
    note('axe-name-mismatch-nodes', { scheme, nodes });
  });
}
