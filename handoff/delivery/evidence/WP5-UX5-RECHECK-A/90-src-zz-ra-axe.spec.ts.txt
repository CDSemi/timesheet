import { appendFileSync, readFileSync } from 'node:fs';
import type { Page } from '@playwright/test';
import { expect, newPerson, test } from './fixtures.ts';

/*
 * WP5-UX5-RECHECK-A probe: axe-core 4.14.0 (loaded from the auditor's task folder; no dependency added) on the Timesheet, the Review
 * (unsigned, signed, signed and reloaded) and the Sign off dialog at 1440, 900, 768, 480, 390 and 320px, light and dark; keyboard walks
 * with a focus-indicator and a not-wholly-hidden check per stop. Synthetic data only; observations go to $RA_OUT.
 */
test.use({ locale: 'en-US', timezoneId: 'America/Los_Angeles' });

const AXE_PATH = process.env.AXE_PATH ?? '';
const OUT = process.env.RA_OUT ?? '';
const TAGS = ['wcag2a', 'wcag2aa', 'wcag21a', 'wcag21aa', 'wcag22aa'];
let axeSource = '';
function note(kind: string, data: unknown): void {
  const line = JSON.stringify({ kind, ...(data as object) });
  if (OUT !== '') appendFileSync(OUT, `${line}\n`);
}
const frames = (page: Page) => page.evaluate('new Promise((done) => requestAnimationFrame(() => requestAnimationFrame(done)))');

let scans = 0;
let violationsTotal = 0;
async function scan(page: Page, label: string): Promise<number> {
  if (axeSource === '') axeSource = readFileSync(AXE_PATH, 'utf8');
  await frames(page);
  if (!(await page.evaluate<boolean>(`typeof window.axe !== 'undefined'`))) await page.evaluate(axeSource);
  const result = await page.evaluate<{ version: string; violations: Array<{ id: string; impact: string | null; nodes: string[] }>; incomplete: string[] }, string[]>(async (tags) => {
    const axe = (window as unknown as { axe: { run: (c: unknown, o: unknown) => Promise<Record<string, unknown>>; version: string } }).axe;
    const r = (await axe.run(document, { runOnly: { type: 'tag', values: tags }, resultTypes: ['violations', 'incomplete'] })) as {
      violations: Array<{ id: string; impact: string | null; nodes: Array<{ target: unknown }> }>;
      incomplete: Array<{ id: string }>;
    };
    return { version: axe.version, violations: r.violations.map((v) => ({ id: v.id, impact: v.impact, nodes: v.nodes.map((n) => JSON.stringify(n.target)) })), incomplete: [...new Set(r.incomplete.map((i) => i.id))] };
  }, TAGS);
  scans += 1;
  violationsTotal += result.violations.length;
  note('axe', { label, version: result.version, violations: result.violations, incomplete: result.incomplete });
  return result.violations.length;
}

const WIDTHS = [1440, 900, 768, 480, 390, 320];

for (const scheme of ['light', 'dark'] as const) {
  test(`axe: Timesheet, Review unsigned, Sign off dialog, Review signed, Review signed and reloaded - ${scheme}`, async ({ page, adminSeed, builtServer, isMobile }, testInfo) => {
    test.setTimeout(420_000);
    const project = testInfo.project.name;
    const widths = isMobile ? [480, 390, 320] : WIDTHS;
    await page.emulateMedia({ colorScheme: scheme, reducedMotion: 'reduce' });
    const person = await newPerson(adminSeed, builtServer, { displayName: `Synthetic Axe ${scheme}`, signature: 'saved' });
    const free = await person.api.displayedPeriodFreeWorkdays();
    await person.api.seedCompleteDay(free[0] ?? '');
    await page.goto('/#/timesheet');
    await page.getByLabel('Email').fill(person.account.email);
    await page.getByLabel('Password').fill(person.account.password);
    await page.getByRole('button', { name: 'Sign in' }).click();
    await expect(page.locator('[data-day]')).toHaveCount(14);
    const tag = `${project} ${scheme}`;
    for (const width of widths) {
      await page.setViewportSize({ width, height: 900 });
      await scan(page, `[${tag}] Timesheet ${width}`);
    }
    for (const width of isMobile ? [390] : [1440, 390]) {
      await page.setViewportSize({ width, height: 900 });
      await page.getByRole('switch', { name: 'Worked hours' }).click();
      await scan(page, `[${tag}] Timesheet ${width} worked hours on`);
      await page.getByRole('switch', { name: 'Worked hours' }).click();
    }
    for (const width of isMobile ? [390] : [1280, 900, 390]) {
      await page.setViewportSize({ width, height: 900 });
      await namedDayButton(page, free[1] ?? free[0] ?? '', 'Edit').click();
      const editor = page.getByRole('dialog', { name: /Day editor/ });
      await expect(editor).toBeVisible();
      await scan(page, `[${tag}] Timesheet ${width} Day editor open`);
      await editor.getByRole('button', { name: /^Close/ }).first().click();
      await expect(editor).toHaveCount(0);
    }
    await page.goto(`/#/review/${person.payrollDate}`);
    await expect(page.locator('[data-review-day]')).toHaveCount(14);
    for (const width of widths) {
      await page.setViewportSize({ width, height: 900 });
      await scan(page, `[${tag}] Review unsigned ${width}`);
    }
    await page.locator('[data-sign-off]').click();
    const dialog = page.getByRole('dialog', { name: 'Sign off confirmation' });
    await expect(dialog).toBeVisible();
    for (const width of widths) {
      await page.setViewportSize({ width, height: 900 });
      await scan(page, `[${tag}] Sign off dialog ${width}`);
    }
    await dialog.getByRole('checkbox', { name: /I acknowledge the incomplete evidence/ }).check();
    await dialog.getByRole('button', { name: 'Sign off', exact: true }).click();
    await expect(page.locator('[data-result="created"]')).toContainText('Signed.');
    for (const width of widths) {
      await page.setViewportSize({ width, height: 900 });
      await scan(page, `[${tag}] Review signed ${width}`);
    }
    await page.reload();
    await expect(page.locator('[data-sign-off]')).toHaveText('Sign correction');
    for (const width of widths) {
      await page.setViewportSize({ width, height: 900 });
      await expect.poll(() => page.evaluate<boolean>(`(() => { const i = document.querySelector('img.signed-image'); return !!i && i.complete && i.naturalWidth > 0; })()`)).toBe(true);
      await scan(page, `[${tag}] Review signed and reloaded ${width}`);
    }
    // No sideways scroll on the signed and reloaded Review, every width (desktop project).
    if (!isMobile) {
      const sideways: number[] = [];
      for (let width = 320; width <= 1440; width += 1) {
        await page.setViewportSize({ width, height: 900 });
        const w = await page.evaluate<{ s: number; c: number }>('({ s: document.documentElement.scrollWidth, c: document.documentElement.clientWidth })');
        if (w.s > w.c) sideways.push(width);
      }
      note('sideways-review-signed-reloaded', { scheme, sideways });
    }
    note('axe-total', { tag, scans, violationsTotal });
  });
}

/* ---- Keyboard walks ------------------------------------------------------------------------ */

const STOP = `(() => {
  const a = document.activeElement;
  if (!a || a === document.body) return { none: true };
  const s = getComputedStyle(a);
  const b = a.getBoundingClientRect();
  const vw = document.documentElement.clientWidth, vh = document.documentElement.clientHeight;
  const x0 = Math.max(b.left, 0), x1 = Math.min(b.right, vw), y0 = Math.max(b.top, 0), y1 = Math.min(b.bottom, vh);
  let visiblePoints = 0, inView = x1 > x0 && y1 > y0;
  if (inView) {
    for (const [fx, fy] of [[0.5, 0.5], [0.2, 0.5], [0.8, 0.5], [0.5, 0.2], [0.5, 0.8]]) {
      const hit = document.elementFromPoint(x0 + (x1 - x0) * fx, y0 + (y1 - y0) * fy);
      if (hit && (hit === a || a.contains(hit) || hit.contains(a))) visiblePoints += 1;
    }
  }
  const ring = (s.outlineStyle !== 'none' && parseFloat(s.outlineWidth) > 0 && !/rgba\\(0, 0, 0, 0\\)|transparent/.test(s.outlineColor)) || (s.boxShadow !== 'none' && /rgb/.test(s.boxShadow));
  return { tag: a.tagName, cls: String(a.className).slice(0, 40), name: (a.getAttribute('aria-label') || (a.textContent || '').trim()).slice(0, 30), fv: a.matches(':focus-visible'), ring, inView, visiblePoints };
})()`;

async function walk(page: Page, label: string, max = 160) {
  let hidden = 0;
  let noRing = 0;
  let stops = 0;
  const problems: unknown[] = [];
  for (let i = 0; i < max; i += 1) {
    await page.keyboard.press('Tab');
    const s = await page.evaluate<Record<string, unknown>>(STOP);
    if (s.none) continue;
    stops += 1;
    if (!s.inView || s.visiblePoints === 0) {
      hidden += 1;
      if (problems.length < 15) problems.push({ i, ...s, why: 'hidden' });
    }
    if (!s.ring) {
      noRing += 1;
      if (problems.length < 15) problems.push({ i, ...s, why: 'no ring' });
    }
  }
  note('walk', { label, stops, hidden, noRing, problems });
}

for (const scheme of ['light', 'dark'] as const) {
  test(`keyboard walks - ${scheme}`, async ({ page, adminSeed, builtServer, isMobile }, testInfo) => {
    test.setTimeout(420_000);
    const project = testInfo.project.name;
    await page.emulateMedia({ colorScheme: scheme, reducedMotion: 'reduce' });
    const person = await newPerson(adminSeed, builtServer, { displayName: `Synthetic Walk ${scheme}`, signature: 'saved' });
    const free = await person.api.displayedPeriodFreeWorkdays();
    await person.api.seedCompleteDay(free[0] ?? '');
    await page.setViewportSize(isMobile ? { width: 390, height: 844 } : { width: 1280, height: 800 });
    await page.goto('/#/timesheet');
    await page.getByLabel('Email').fill(person.account.email);
    await page.getByLabel('Password').fill(person.account.password);
    await page.getByRole('button', { name: 'Sign in' }).click();
    await expect(page.locator('[data-day]')).toHaveCount(14);
    await walk(page, `[${project} ${scheme}] Timesheet ${isMobile ? 390 : 1280}`);
    if (!isMobile) {
      await namedDayButton(page, free[0] ?? '', 'Edit').click();
      await expect(page.getByRole('dialog', { name: /Day editor/ })).toBeVisible();
      await walk(page, `[${project} ${scheme}] Timesheet 1280 drawer open (floating)`);
    }
    await page.goto(`/#/review/${person.payrollDate}`);
    await expect(page.locator('[data-review-day]')).toHaveCount(14);
    if (!isMobile) await page.setViewportSize({ width: 900, height: 900 });
    await page.locator('h1').first().click();
    await walk(page, `[${project} ${scheme}] Review unsigned ${isMobile ? 390 : 900}`, 80);
  });
}
