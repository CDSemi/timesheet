function send(method, path, { body, headers = {} } = {}) {
  return new Promise((resolve) => {
    const t0 = performance.now();
    const req = httpRequest({ host: '127.0.0.1', port, method, path, agent: false, headers: { origin, ...headers, ...(body ? { 'content-length': body.length } : {}) } }, (res) => {
      const chunks = [];
      res.on('data', (c) => chunks.push(c));
      res.on('end', () => {
        const buf = Buffer.concat(chunks);
        let json = null;
        try { json = JSON.parse(buf.toString('utf8')); } catch { /* not JSON */ }
        resolve({ status: res.statusCode, ms: performance.now() - t0, headers: res.headers, size: buf.length, json });
      });
    });
    req.on('error', (e) => resolve({ status: 0, ms: performance.now() - t0, error: e.code ?? e.message, size: 0, json: null }));
    if (body) req.write(body);
    req.end();
  });
}

/** Upload while polling /api/health back to back; returns the upload result and the worst health latency. */
async function uploadWithHealth(bytes) {
  let done = false;
  let worst = 0;
  let polls = 0;
  const poller = (async () => {
    while (!done) {
      const h = await send('GET', '/api/health');
      polls += 1;
      if (h.status !== 200) worst = Infinity;
      worst = Math.max(worst, h.ms);
    }
  })();
  const r = note(await upload(bytes));
  done = true;
  await poller;
  return { r, worst, polls };
}
