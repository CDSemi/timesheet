// spec and the OOXML parts the template uses. Each case replaces the template's `Timesheet` worksheet part (or the
// shared strings) with a small crafted part, reads it with the new reader and the accepted WP4-T09 reader (13a258d),
// and checks the new reader's outcome against the outcome the spec requires (the oracle is the spec, not T09).
// `lenient` marks a not-well-formed input that the spec says is a fatal error but that the reader accepts; it is
// reported, not counted as a pass. In memory only; synthetic data only.
// Usage: node b2-scanner.mjs

const NS = 'http://schemas.openxmlformats.org/spreadsheetml/2006/main';
const DECL = '<?xml version="1.0" encoding="UTF-8" standalone="yes"?>\n';
const sheet = (rowXml, { decl = DECL, root = `<worksheet xmlns="${NS}">`, end = '</worksheet>' } = {}) => `${decl}${root}<sheetData><row r="1">${rowXml}</row></sheetData>${end}`;
const inline = (address, inner) => `<c r="${address}" t="inlineStr"><is>${inner}</is></c>`;

function packageWith(partText, { part = 'xl/worksheets/sheet3.xml', bytes } = {}) {
  const e = unpack(readTemplateBytes());
  e[part] = bytes ?? strToU8(partText);
  return pack(e);
}

function outcome(read, pkg, addresses) {
  try {
    const wb = read(pkg);
    const ts = wb.sheets.find((s) => s.name === 'Timesheet');
    const values = addresses.map((a) => {
      const c = ts.cells.get(a);
      if (c === undefined) return `${a}:missing`;
      return `${a}:${c.type}:${JSON.stringify(c.formula ?? c.value)}`;
    });
    return { ok: true, text: values.join(' ') };
  } catch (error) {
    return { ok: false, text: error.name === 'WorkbookRejectedError' ? `rejected:${error.code}` : `THROWN ${error.name}: ${error.message}` };
  }
}

const utf16le = (s) => { const b = Buffer.from(s, 'utf16le'); return new Uint8Array(Buffer.concat([Buffer.from([0xff, 0xfe]), b])); };
const utf16be = (s) => { const b = Buffer.from(s, 'utf16le'); for (let i = 0; i < b.length; i += 2) { const t = b[i]; b[i] = b[i + 1]; b[i + 1] = t; } return new Uint8Array(Buffer.concat([Buffer.from([0xfe, 0xff]), b])); };
const latin1 = (s) => new Uint8Array(Buffer.from(s, 'latin1'));
const utf8bom = (s) => new Uint8Array(Buffer.concat([Buffer.from([0xef, 0xbb, 0xbf]), Buffer.from(s, 'utf8')]));

let pass = 0;
let fail = 0;
let lenient = 0;
const lines = [];
for (const [id, description, part, addresses, expected, kind] of CASES) {
  const pkg = typeof part === 'string' ? packageWith(part) : packageWith(null, part);
  const n = outcome(readNew, pkg, addresses);
  const o = outcome(readOld, pkg, addresses);
  const matches = expected === 'rejected:*' ? !n.ok && n.text.startsWith('rejected:') : n.text === expected;
  let verdict;
  if (kind === 'lenient') { verdict = matches ? 'LENIENT (as described)' : 'LENIENT-UNEXPECTED'; lenient += 1; }
  else if (matches) { verdict = 'PASS'; pass += 1; }
  else { verdict = 'FAIL'; fail += 1; }
  lines.push(`${verdict} ${id} ${description} | new: ${n.text} | expected: ${expected} | T09: ${o.text}${o.text === n.text ? ' (same)' : ' (DIFFERS)'}`);
}
console.log(lines.join('\n'));
console.log(`SUMMARY spec cases: ${pass} PASS, ${fail} FAIL; lenient cases reported: ${lenient}`);
