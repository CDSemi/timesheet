// WP5-UX-AUDIT-B3 mutation (scratch clone only): keep the a2ea7a4 tokens but put back the 589bcff phone
// "Open a day" rules (nowrap row, label and input with min-width 0), i.e. the B2-01 behaviour.
// Usage: node mutate-openday.mjs <clone>/src/client/styles.css
import { readFileSync, writeFileSync } from 'node:fs';

const [path] = process.argv.slice(2);
const css = readFileSync(path, 'utf8');
const fixed = `  /* The label sits beside its date field, so the row stays one control high where it fits. */
  .tools .open-day {
    flex-wrap: wrap;
    align-items: center;
    width: 100%;
  }

  /* The label keeps a full date beside it; when that and the button do not fit, the button wraps below. */
  .tools .open-day label {
    flex: 1 1 var(--open-day-label-min);
    flex-direction: row;
    align-items: center;
    gap: var(--space-2);
    min-width: var(--open-day-label-min);
    white-space: nowrap;
  }

  .tools .open-day input {
    flex: 1;
    min-width: var(--date-field-min);
  }
`;
const before = `  .tools .open-day {
    flex-wrap: nowrap;
    width: 100%;
  }

  /* The label sits beside its date field, so the row stays one control high. */
  .tools .open-day label {
    flex: 1;
    flex-direction: row;
    align-items: center;
    gap: var(--space-2);
    min-width: 0;
    white-space: nowrap;
  }

  .tools .open-day input {
    flex: 1;
    min-width: 0;
  }

  .tools .open-day {
    align-items: center;
  }
`;
const normalized = css.replace(/\r\n/g, '\n');
const count = normalized.split(fixed).length - 1;
if (count !== 1) {
  console.log(`expected the a2ea7a4 block exactly once, found ${count}`);
  process.exit(2);
}
writeFileSync(path, normalized.replace(fixed, before));
const tokens = /--date-field-min: [^;]+;/.test(normalized) && /--open-day-label-min: [^;]+;/.test(normalized);
console.log(`mutated: 589bcff open-day rules restored; a2ea7a4 tokens kept: ${tokens}`);
