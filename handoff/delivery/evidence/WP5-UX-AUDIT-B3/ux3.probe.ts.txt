/*
 * WP5-UX-AUDIT-B3 independent probes (auditor-written; not part of the reviewed source; kept out of git
 * through .git/info/exclude of the scratch clone). Synthetic data only. Each project starts its own built
 * server (dist/) on a fixed port of the task range (desktop PROBE_PORT_BASE, mobile PROBE_PORT_BASE + 1;
 * default 48060/48061) with DATA_DIR and DATABASE_PATH inside the task folder, and stops it through the
 * child handle. Each probe records what it measured as JSON in PROBE_OUT; the few hard expectations only
 * guard the measurement itself. Method adapted from the WP5-UX-AUDIT-B2 probes, rewritten and extended.
 */

const REPO = resolve(import.meta.dirname, '..');
const OUT = process.env.PROBE_OUT ?? join(process.cwd(), 'probe-out');
const WORK = process.env.PROBE_WORK ?? join(process.cwd(), 'probe-work');
const SHOTS = process.env.PROBE_SHOTS ?? OUT;
const PORT_BASE = Number(process.env.PROBE_PORT_BASE ?? '48060');
const TAG = process.env.PROBE_TAG ?? 'a2ea7a4';
mkdirSync(OUT, { recursive: true });
mkdirSync(WORK, { recursive: true });
mkdirSync(SHOTS, { recursive: true });

function record(name: string, data: unknown) {
  writeFileSync(join(OUT, `${TAG}-${name}.json`), `${JSON.stringify(data, null, 2)}\n`);
}
function shot(name: string): string {
  if (!name.endsWith('-synthetic.png')) throw new Error('screenshot names end in -synthetic.png');
  return join(SHOTS, `${TAG}-${name}`);
}

interface Credentials {
  email: string;
  password: string;
}
interface ProbeServer {
  origin: string;
  employee: Credentials;
  admin: Credentials;
}

async function waitHealthy(origin: string, child: ChildProcess) {
  for (let attempt = 0; attempt < 150; attempt += 1) {
    if (child.exitCode !== null) throw new Error(`server exited early with ${child.exitCode}`);
    try {
      if ((await fetch(`${origin}/api/health`)).ok) return;
    } catch {
      // not listening yet
    }
    await new Promise((done) => setTimeout(done, 100));
  }
  throw new Error('server did not become healthy');
}

function layout(page: Page) {
  return page.evaluate(() => {
    const r = (selector: string) => {
      const element = document.querySelector(selector);
      if (element === null) return null;
      const box = element.getBoundingClientRect();
      return { top: +box.top.toFixed(1), bottom: +box.bottom.toFixed(1), left: +box.left.toFixed(1), right: +box.right.toFixed(1), height: +box.height.toFixed(1), width: +box.width.toFixed(1) };
    };
    const firstDay = document.querySelector('[data-day]');
    return {
      viewport: { width: window.innerWidth, height: window.innerHeight },
      scrollY: window.scrollY,
      scrollWidth: document.documentElement.scrollWidth,
      clientWidth: document.documentElement.clientWidth,
      topBar: r('.shell-bar'),
      shareBar: r('[data-share-bar-title]'),
      periodCard: r('.period-bar'),
      zoneNote: r('[data-zone-note]'),
      clockCard: r('.clock-panel'),
      toolsCard: r('.sheet-area > .card'),
      tools: r('.tools'),
      openDayRow: r('.tools .open-day'),
      openDayInput: r('.tools .open-day input'),
      formHead: r('.form-head'),
      firstDay: r('[data-day]'),
      firstDayName: firstDay?.getAttribute('aria-label') ?? null,
      tabBar: r('.shell-tabs'),
      notice: [...document.querySelectorAll('main [role="status"], main [role="alert"]')].map((element) => (element.textContent ?? '').trim().slice(0, 60)).filter((text) => text !== ''),
    };
  });
}

function budget(v: Awaited<ReturnType<typeof layout>>) {
  const firstBottom = v.firstDay?.bottom ?? Number.POSITIVE_INFINITY;
  const tabTop = v.tabBar?.top ?? 0;
  return {
    firstDayBottom: firstBottom,
    tabBarTop: tabTop,
    margin: +(tabTop - firstBottom).toFixed(1),
    firstDayFullyAboveTabBar: firstBottom <= tabTop + 0.5,
    scrollY: v.scrollY,
    noSidewaysScroll: v.scrollWidth <= v.clientWidth,
    notice: v.notice,
  };
}

interface Stop {
  tag: string;
  name: string;
  day: string | null;
  region: string;
  inDialog: boolean;
  box: { top: number; left: number; bottom: number; right: number } | null;
  hiddenByDialog: boolean;
}

function activeStop(page: Page): Promise<Stop> {
  return page.evaluate(() => {
    const active = document.activeElement as HTMLElement | null;
    if (active === null || active === document.body) return { tag: 'BODY', name: '', day: null, region: '', inDialog: false, box: null, hiddenByDialog: false };
    const dialog = [...document.querySelectorAll('dialog[open]')].at(-1) ?? null;
    const host = active.closest('[data-day], [data-review-day]');
    const region = active.closest('.shell-tabs') !== null ? 'tabbar' : active.closest('.shell-bar') !== null ? 'topbar' : active.closest('dialog') !== null ? 'dialog' : active.closest('.period-bar') !== null ? 'period' : active.closest('.clock-panel') !== null ? 'clock' : active.closest('.tools, .batch-bar') !== null ? 'tools' : active.closest('.sheet') !== null ? 'sheet' : 'page';
    const b = active.getBoundingClientRect();
    let hidden = false;
    if (dialog !== null && !dialog.contains(active)) {
      const p = dialog.getBoundingClientRect();
      hidden = b.left >= p.left && b.right <= p.right && b.top >= p.top && b.bottom <= p.bottom;
    }
    const labelled = active.getAttribute('aria-label') ?? (active instanceof HTMLInputElement ? (active.labels?.[0]?.textContent ?? '') : active.textContent ?? '');
    return {
      tag: active.tagName,
      name: labelled.trim().replace(/\s+/g, ' ').slice(0, 60),
      day: host?.getAttribute('data-day') ?? host?.getAttribute('data-review-day') ?? null,
      region,
      inDialog: dialog !== null && dialog.contains(active),
      box: { top: +b.top.toFixed(1), left: +b.left.toFixed(1), bottom: +b.bottom.toFixed(1), right: +b.right.toFixed(1) },
      hiddenByDialog: hidden,
    };
  });
}

async function tabSequence(page: Page, max = 260): Promise<Array<Stop & { pageTop: number }>> {
  await page.evaluate(() => {
    (document.activeElement as HTMLElement | null)?.blur();
    window.scrollTo(0, 0);
  });
  const stops: Array<Stop & { pageTop: number }> = [];
  for (let step = 0; step < max; step += 1) {
    await page.keyboard.press('Tab');
    const stop = await activeStop(page);
    const scrollY = await page.evaluate(() => window.scrollY);
    if (stops.length > 3 && stop.name === stops[0]?.name && stop.tag === stops[0]?.tag) break;
    stops.push({ ...stop, pageTop: +((stop.box?.top ?? 0) + scrollY).toFixed(1) });
  }
  return stops;
}

function inversionsOf(stops: Array<Stop & { pageTop: number }>) {
  const inMain = stops.filter((stop) => stop.region !== 'tabbar' && stop.region !== 'topbar');
  const inversions: string[] = [];
  for (let index = 1; index < inMain.length; index += 1) {
    const previous = inMain[index - 1];
    const current = inMain[index];
    if (previous === undefined || current === undefined || previous.box === null || current.box === null) continue;
    const sameRow = Math.abs(current.pageTop - previous.pageTop) <= 8;
    const backwardsUp = current.pageTop + 8 < previous.pageTop;
    const backwardsLeft = sameRow && current.box.left + 1 < previous.box.left;
    if (backwardsUp || backwardsLeft) inversions.push(`${index}: ${previous.region}/${previous.name} (${previous.pageTop},${previous.box.left}) -> ${current.region}/${current.name} (${current.pageTop},${current.box.left})`);
  }
  return inversions;
}

/* ---- PC4: keyboard order (Tab order equals the reading order) ------------------------------------ */

test.describe('PC4 keyboard order', () => {
  test.use({ locale: 'en-US', timezoneId: 'America/Los_Angeles' });
  test('PC4 full Tab sequence on the Timesheet page, and at 320px on a phone', async ({ page, signIn }, testInfo) => {
    await signIn();
    await expect(page.locator('[data-day]')).toHaveCount(14);
    const widths = testInfo.project.name === 'mobile' ? [390, 320] : [1280];
    const result: Record<string, unknown> = {};
    for (const width of widths) {
      await page.setViewportSize({ width, height: testInfo.project.name === 'mobile' ? 844 : 800 });
      if (width === 320) await page.getByLabel('Open a day').fill('2026-10-09');
      const stops = await tabSequence(page);
      const dayOrder = stops.filter((stop) => stop.day !== null).map((stop) => stop.day ?? '');
      result[String(width)] = {
        count: stops.length,
        regions: [...new Set(stops.map((stop) => stop.region))],
        dayOrderNondecreasing: dayOrder.every((date, index) => index === 0 || date >= (dayOrder[index - 1] ?? '')),
        daysVisited: [...new Set(dayOrder)].length,

/* ---- PC8: rendered contrast of the compact phone text, and the focus ring (R-1) ------------------ */

test.describe('PC8 contrast', () => {
  test.use({ locale: 'en-US', timezoneId: 'Asia/Ho_Chi_Minh' });
  test('PC8 text contrast in the period card, clock, tools, form header and week bar; focus ring contrast', async ({ page, signIn }, testInfo) => {
    await signIn();
    await expect(page.locator('[data-day]')).toHaveCount(14);
    const result: Record<string, unknown> = {};
    for (const scheme of ['light', 'dark'] as const) {
      await page.emulateMedia({ colorScheme: scheme });
      await page.waitForTimeout(500);
      result[`text-${scheme}`] = await page.evaluate(() => {
        const parse = (value: string) => {
          const m = /rgba?\(([\d.]+)[, ]+([\d.]+)[, ]+([\d.]+)(?:[,/ ]+([\d.]+))?\)/.exec(value);
          return m === null ? null : { r: +m[1]!, g: +m[2]!, b: +m[3]!, a: m[4] === undefined ? 1 : +m[4] };
        };
        const lum = (c: { r: number; g: number; b: number }) => {
          const f = (v: number) => {
            const s = v / 255;
            return s <= 0.03928 ? s / 12.92 : ((s + 0.055) / 1.055) ** 2.4;
          };
          return 0.2126 * f(c.r) + 0.7152 * f(c.g) + 0.0722 * f(c.b);
        };
        const background = (element: Element | null): { r: number; g: number; b: number } => {
          for (let node = element; node !== null; node = node.parentElement) {
            const bg = parse(getComputedStyle(node).backgroundColor);
            if (bg !== null && bg.a >= 0.99) return bg;
          }
          return { r: 255, g: 255, b: 255 };
        };
        const rows: Array<{ text: string; ratio: number }> = [];
        const walker = document.createTreeWalker(document.querySelector('main') ?? document.body, NodeFilter.SHOW_TEXT);
        for (let node = walker.nextNode(); node !== null; node = walker.nextNode()) {
          const text = (node.textContent ?? '').trim();
          const parent = node.parentElement;
          if (text === '' || parent === null) continue;
          if (parent.closest('.period-bar, .clock-panel, .tools, .form-head, .week-bar, .sheet') === null) continue;
          const style = getComputedStyle(parent);
          const box = parent.getBoundingClientRect();
          if (box.width <= 2 || box.height <= 2 || style.visibility === 'hidden' || style.clipPath !== 'none') continue;
          const fg = parse(style.color);
          if (fg === null) continue;
          const bg = background(parent);
          const blended = { r: fg.r * fg.a + bg.r * (1 - fg.a), g: fg.g * fg.a + bg.g * (1 - fg.a), b: fg.b * fg.a + bg.b * (1 - fg.a) };
          const [l1, l2] = [lum(blended), lum(bg)].sort((x, y) => y - x);
          rows.push({ text: text.slice(0, 40), ratio: +((l1! + 0.05) / (l2! + 0.05)).toFixed(2) });
        }
        rows.sort((x, y) => x.ratio - y.ratio);
        return { count: rows.length, min: rows[0]?.ratio ?? null, below45: rows.filter((row) => row.ratio < 4.5).length, lowest: rows.slice(0, 6) };
      });
      // Focus ring: keyboard modality first, then each control is focused and its ring colour blended over the background.
      await page.keyboard.press('Tab');
      const ringTargets = ['.period-bar [aria-label="Previous period"]', '.clock-panel button', '.tools button', '[data-day] button', '.tools .open-day input'];
      const rings: unknown[] = [];
      for (const selector of ringTargets) {
        const target = page.locator(selector).first();
        if ((await target.count()) === 0 || !(await target.isVisible())) continue;
        await target.focus();
        await page.waitForTimeout(350);
        rings.push(
          await target.evaluate((element, sel) => {
            const parse = (value: string) => {
              const m = /rgba?\(([\d.]+)[, ]+([\d.]+)[, ]+([\d.]+)(?:[,/ ]+([\d.]+))?\)/.exec(value);
              return m === null ? null : { r: +m[1]!, g: +m[2]!, b: +m[3]!, a: m[4] === undefined ? 1 : +m[4] };
            };
            const lum = (c: { r: number; g: number; b: number }) => {
              const f = (v: number) => {
                const s = v / 255;
                return s <= 0.03928 ? s / 12.92 : ((s + 0.055) / 1.055) ** 2.4;
              };
              return 0.2126 * f(c.r) + 0.7152 * f(c.g) + 0.0722 * f(c.b);
            };
            const ratio = (a: { r: number; g: number; b: number }, b: { r: number; g: number; b: number }) => {
              const [l1, l2] = [lum(a), lum(b)].sort((x, y) => y - x);
              return +((l1! + 0.05) / (l2! + 0.05)).toFixed(2);
            };
            const background = (start: Element | null) => {
              for (let node = start; node !== null; node = node.parentElement) {
                const bg = parse(getComputedStyle(node).backgroundColor);
                if (bg !== null && bg.a >= 0.99) return bg;
              }
              return { r: 255, g: 255, b: 255, a: 1 };
            };
            const style = getComputedStyle(element);
            const ring = parse(style.boxShadow);
            const outside = background(element.parentElement);
            const blended = ring === null ? null : { r: ring.r * ring.a + outside.r * (1 - ring.a), g: ring.g * ring.a + outside.g * (1 - ring.a), b: ring.b * ring.a + outside.b * (1 - ring.a) };
            const own = parse(style.backgroundColor);
            const border = parse(style.borderTopColor);
            return {
              selector: sel,
              focusVisible: element.matches(':focus-visible'),
              boxShadow: style.boxShadow,
              outline: `${style.outlineStyle} ${style.outlineWidth} ${style.outlineColor}`,
              ringVsOutsideBackground: blended === null ? null : ratio(blended, outside),
              ringVsOwnBackground: blended === null || own === null || own.a < 0.99 ? null : ratio(blended, own),
              ringVsBorder: blended === null || border === null || border.a < 0.99 ? null : ratio(blended, border),
            };
          }, selector),
        );
      }
      result[`focus-ring-${scheme}`] = rings;
    }
    await page.emulateMedia({ colorScheme: 'light' });
    record(`PC8-contrast-${testInfo.project.name}`, result);
  });
});

/* ---- PC13: the phone bottom sheet (modal, focus trap, Escape, focus return) at 390 and 320px ------ */

test.describe('PC13 phone bottom sheet', () => {
  test.skip(({ isMobile }) => !isMobile, 'phone only');
  test.use({ locale: 'en-US', timezoneId: 'America/Los_Angeles' });
  test('PC13 bottom sheet: modal, 120-press walk, Escape and Close, focus return', async ({ page, signIn }) => {
    await signIn();
    await expect(page.locator('[data-day]')).toHaveCount(14);
    const result: Record<string, unknown> = {};
    for (const width of [390, 320]) {
      await page.setViewportSize({ width, height: 844 });
      const day = (await page.locator('[data-day]').nth(1).getAttribute('data-day')) ?? '';
      const button = page.locator(`[data-day="${day}"]`).getByRole('button', { name: `Edit ${day}`, exact: true });
      await button.click();
      const editor = page.getByRole('dialog', { name: /Day editor/ });
      await expect(editor).toBeVisible();
      await expect(editor.getByRole('heading', { level: 2 })).toBeFocused();
      const geometry = await page.evaluate(() => {
        const dialog = document.querySelector('dialog[open]');
        const box = dialog?.getBoundingClientRect();
        return { modal: dialog?.matches(':modal') ?? null, top: box?.top ?? null, height: box?.height ?? null, viewportHeight: window.innerHeight, heightShare: box === undefined ? null : +(box.height / window.innerHeight).toFixed(3) };
      });
      let outside = 0;
      for (const key of ['Tab', 'Shift+Tab']) {
        for (let step = 0; step < 60; step += 1) {
          await page.keyboard.press(key);
          const stop = await activeStop(page);
          if (stop.tag !== 'BODY' && !stop.inDialog) outside += 1;
        }
      }
      await page.keyboard.press('Escape');
      await expect(editor).toHaveCount(0);
      const returnAfterEscape = await button.evaluate((element) => element === document.activeElement);
      await button.click();
      await expect(editor).toBeVisible();
      await editor.getByRole('button', { name: 'Close', exact: true }).click();
      await expect(editor).toHaveCount(0);
      const returnAfterClose = await button.evaluate((element) => element === document.activeElement);
      result[String(width)] = { day, geometry, outside, returnAfterEscape, returnAfterClose };
    }
    record('PC13-phone-bottom-sheet', result);
  });
});
