# Usage: bash census.sh <tag>   read-only process census of node/bash/msedge, written to out/census-<tag>.txt
. /d/.claude-tmp/timesheet/WP5-UX5-FIX13/env.sh
powershell.exe -NoProfile -Command "Get-CimInstance Win32_Process | Where-Object { \$_.Name -match 'node|bash|msedge|chrome|waitfor' } | Select-Object ProcessId,Name,CreationDate | Format-Table -AutoSize | Out-String -Width 200" > $T/out/census-$1.txt 2>&1
