// day again when the clock version changes (the page's own Check in or Check out, and the tab coming back into view), and takes the
// fresh day only when the day fields equal the held ones and no session dialog or delete confirmation is open. This file checks the
// follow, the version check against changes made elsewhere, the tab-return case, the saved() path, and the regressions named in the
// brief. Synthetic data only; observations go to INT2_OUT.
import { appendFileSync } from 'node:fs';
import type { Locator, Page } from '@playwright/test';
import { expect, newPerson, openAccount, type SeedClient, test } from './fixtures.ts';

test.use({ locale: 'en-US', timezoneId: 'Asia/Ho_Chi_Minh' });

const OUT = process.env.INT2_OUT ?? '';
function log(tag: string, line: string): void {
  if (OUT !== '') appendFileSync(OUT, `[${tag}] ${line}\n`);
}
const flat = (text: string) => text.replace(/\s+/g, ' ').trim();
const frames = (page: Page) => page.evaluate('new Promise((done) => requestAnimationFrame(() => requestAnimationFrame(done)))');
const TYPED = 'Typed in this editor, not saved (synthetic)';
const BASE = 'Saved before the editor opened (synthetic)';
const ELSEWHERE = 'Changed on another device (synthetic)';

test.beforeEach(({ page }, testInfo) => {
  page.on('console', (message) => {
    if (message.type() === 'error' && !/status of (401|409|422)/.test(message.text())) log(`console ${testInfo.title.slice(0, 24)}`, `console error: ${message.text()}`);
  });
  page.on('pageerror', (error) => log(`pageerror ${testInfo.title.slice(0, 24)}`, String(error)));
});

/** Every GET of one day (own or shared route) the page sends, by time. */
function dayReads(page: Page, date: string): string[] {
  const reads: string[] = [];
  page.on('request', (request) => {
    if (request.method() === 'GET' && new URL(request.url()).pathname.endsWith(`/days/${date}`)) reads.push(new Date().toISOString());
  });
  return reads;
}

/**
 * Waits for the editor's read of its day after a trigger. With INT2_NO_FOLLOW=1 (the comparison run on a5cbc91, which has no follow
 * read) it waits instead for the period read that the page makes on the same trigger, then for the network to be idle.
 */
const NO_FOLLOW = process.env.INT2_NO_FOLLOW === '1';
function dayRead(page: Page, date: string): Promise<unknown> {
  if (NO_FOLLOW) {
    return page
      .waitForResponse((response) => response.request().method() === 'GET' && /\/api\/timesheets\//.test(new URL(response.url()).pathname), { timeout: 10_000 })
      .then(() => page.waitForLoadState('networkidle'));
  }
  return page.waitForResponse((response) => response.request().method() === 'GET' && new URL(response.url()).pathname.endsWith(`/days/${date}`), { timeout: 10_000 });
}

/** "Reload day", waiting until the reloaded day is on screen (the read, then two frames twice). */
async function reloadDay(page: Page, editor: Locator, date: string): Promise<void> {
  const read = page.waitForResponse((response) => response.request().method() === 'GET' && new URL(response.url()).pathname.endsWith(`/days/${date}`), { timeout: 10_000 });
  await staleAlert(editor).getByRole('button', { name: 'Reload day' }).click();
  await read;
  await expect(staleAlert(editor)).toHaveCount(0);
  await frames(page);
  await frames(page);
}

async function showDay(page: Page, date: string): Promise<void> {
  await expect(page.locator('[data-day]')).toHaveCount(14);
  if ((await page.locator(`[data-day="${date}"]`).count()) === 0) {
    await page.locator('.period-bar').getByRole('button', { name: 'Today', exact: true }).click();
    await expect(page.locator(`[data-day="${date}"]`)).toHaveCount(1);
  }
}

async function openDay(page: Page, date: string): Promise<Locator> {
  await page.getByLabel('Open a day').fill(date);
  await page.getByRole('button', { name: 'Open day', exact: true }).click();
  const editor = page.getByRole('dialog', { name: /Day editor/ });
  await expect(editor).toBeVisible();
  await expect(editor.getByRole('heading', { name: 'Times', exact: true })).toBeVisible();
  return editor;
}

function fieldsOf(editor: Locator) {
  const form = editor.getByRole('form', { name: 'Day fields' });
  return { form, notes: form.getByLabel('Notes'), wfh: form.getByLabel('Worked from home (WFH)'), category: form.getByLabel('Category') };
}

const discardQuestion = (page: Page) => page.getByRole('alertdialog', { name: 'Discard your changes?' });
const staleAlert = (editor: Locator) => editor.getByRole('alert').filter({ hasText: 'This day changed since it was loaded' });
const savedStatus = (editor: Locator) => editor.getByRole('status').filter({ hasText: 'Day fields saved.' });

async function pressClock(page: Page, name: 'Check in' | 'Check out'): Promise<void> {
  const button = page.getByRole('region', { name: 'Check in and out' }).getByRole('button', { name, exact: true });
  await button.focus();
  await button.click();
}

async function wholeSecondPast(seed: SeedClient, date: string): Promise<void> {
  const started = (await seed.dayView(date)).sessions.find((session) => session.end_utc === null);
  expect(started, 'a running session').toBeDefined();
  await expect.poll(() => Date.now(), { intervals: [100] }).toBeGreaterThanOrEqual(Date.parse(started?.start_utc ?? '') + 1_000);
}

async function checkOutNoBreaks(page: Page): Promise<void> {
  await pressClock(page, 'Check out');
  const dialog = page.getByRole('dialog', { name: 'Check out' });
  await expect(dialog).toBeVisible();
  await dialog.getByRole('checkbox', { name: 'Confirm no breaks taken' }).check();
  await dialog.getByRole('button', { name: 'Check out, no breaks taken' }).click();
  await expect(dialog).toHaveCount(0);
}

async function cards(editor: Locator): Promise<string> {
  const list = await editor.locator('.session-card').allInnerTexts();
  return list.length === 0 ? '(no session card)' : list.map((text) => flat(text).slice(0, 120)).join(' || ');
}

function activeText(page: Page): Promise<string> {
  return page.evaluate<string>(
    "(() => { const a = document.activeElement; if (a === null || a === document.body) return 'BODY'; const name = a.getAttribute('aria-label') ?? (a.textContent ?? '').trim(); return a.tagName + ' \"' + name.slice(0, 40) + '\"' + (a.closest('dialog') ? ' in dialog' : ''); })()",
  );
}

async function liveTexts(editor: Locator): Promise<string> {
  return (await editor.locator('[role="alert"], [role="status"], [aria-live]').allInnerTexts()).map(flat).join(' | ');
}

/* ---- A: F-UX5I-1 on a day that already has saved fields, both layouts, keyboard and guard ----------------------------------- */

for (const width of [1440, 1280]) {
  test(`int2-A ${width}: a day with saved fields; the typed note survives the page's Check in and Check out, focus stays, the guard still asks`, async ({
    page,
    employeeSeed,
    signInThroughUi,
  }, testInfo) => {
    test.skip(testInfo.project.name !== 'desktop', 'the desktop drawer');
    const tag = `A ${width}`;
    const { todayLocal } = await employeeSeed.today();
    const before = await employeeSeed.dayView(todayLocal);
    await employeeSeed.clearSessions(todayLocal);
    await employeeSeed.putDay(todayLocal, { category: before.category ?? 'Worked', wfh: true, notes: BASE });
    const seeded = await employeeSeed.dayView(todayLocal);
    log(tag, `today ${todayLocal}; seeded entry v${seeded.entry?.version} wfh ${seeded.wfh} notes "${seeded.entry?.notes}" category ${seeded.category}/${seeded.category_source}`);
    try {
      await page.setViewportSize({ width, height: 800 });
      await signInThroughUi();
      await showDay(page, todayLocal);
      const reads = dayReads(page, todayLocal);
      const editor = await openDay(page, todayLocal);
      log(tag, `layout ${await page.locator('[data-editor-layout]').getAttribute('data-editor-layout')}; side ${await editor.getAttribute('data-side')}`);
      const { notes, wfh } = fieldsOf(editor);
      await expect(wfh).toBeChecked();
      await expect(notes).toHaveValue(BASE);
      await notes.fill(TYPED);
      const liveBefore = await liveTexts(editor);
      const atOpen = reads.length;

      // Check in from the keyboard: focus the page's button and press Enter.
      const checkIn = page.getByRole('region', { name: 'Check in and out' }).getByRole('button', { name: 'Check in', exact: true });
      await checkIn.focus();
      const read1 = dayRead(page, todayLocal);
      await page.keyboard.press('Enter');
      await read1;
      const running = editor.locator('.session-card[data-state="running"]');
      if (!NO_FOLLOW) await expect(running).toHaveCount(1);
      await frames(page);
      const focus1 = await activeText(page);
      log(
        tag,
        `after Check in (Enter): day reads ${reads.length - atOpen}; editor cards ${await cards(editor)}; note "${await notes.inputValue()}"; wfh ${await wfh.isChecked()}; question ${await discardQuestion(page).count()}; focus ${focus1}; editor live regions before [${liveBefore}] after [${await liveTexts(editor)}]`,
      );
      await expect(notes).toHaveValue(TYPED);
      await expect(wfh).toBeChecked();
      await expect(discardQuestion(page)).toHaveCount(0);
      expect(focus1, 'focus is not dropped to the body').not.toBe('BODY');

      // The unsaved-changes guard still asks after the follow (Escape, Keep editing keeps the note).
      await page.keyboard.press('Escape');
      await expect(discardQuestion(page)).toBeVisible();
      await discardQuestion(page).getByRole('button', { name: 'Keep editing' }).click();
      await expect(discardQuestion(page)).toHaveCount(0);
      await expect(editor).toBeVisible();
      await expect(notes).toHaveValue(TYPED);
      log(tag, 'Escape after the follow asked "Discard your changes?"; Keep editing kept the note');

      await wholeSecondPast(employeeSeed, todayLocal);
      const read2 = dayRead(page, todayLocal);
      await checkOutNoBreaks(page);
      await read2;
      const ended = editor.locator('.session-card:not([data-state="running"])');
      if (!NO_FOLLOW) {
        await expect(ended).toHaveCount(1);
        await expect(running).toHaveCount(0);
      }
      await frames(page);
      log(
        tag,
        `after Check out: day reads ${reads.length - atOpen}; editor cards ${await cards(editor)}; note "${await notes.inputValue()}"; wfh ${await wfh.isChecked()}; question ${await discardQuestion(page).count()}; focus ${await activeText(page)}; figures ${flat(await editor.locator('[data-figure]').allInnerTexts().then((t) => t.join(' / '))).slice(0, 200)}`,
      );
      await expect(notes).toHaveValue(TYPED);

      await editor.getByRole('button', { name: 'Save day fields' }).click();
      await expect(staleAlert(editor).or(savedStatus(editor))).toBeVisible();
      const outcome = (await staleAlert(editor).count()) === 1 ? 'REFUSED (stale)' : 'ACCEPTED';
      const saved = await employeeSeed.dayView(todayLocal);
      log(tag, `Save day fields: ${outcome}; server notes "${saved.entry?.notes}" wfh ${saved.wfh} category ${saved.category}/${saved.category_source} entry v${saved.entry?.version}; sessions ${saved.sessions.map((s) => (s.end_utc === null ? 'running' : 'ended')).join(',')}`);
      expect(saved.entry?.notes).toBe(TYPED);
      expect(saved.wfh).toBe(true);
    } finally {
      await employeeSeed.clearSessions(todayLocal);
      await employeeSeed.putDay(todayLocal, { category: before.category ?? 'Worked' });
    }
  });
}

/* ---- B: a change made elsewhere to ONE day field, then the tab comes back into view (the follow); the save must be refused ---- */

const FIELD_CASES = [
  { name: 'notes only', elsewhere: { notes: ELSEWHERE }, holds: (d: { entry: { notes: string } | null }) => d.entry?.notes === ELSEWHERE, typeWfh: true },
  { name: 'WFH only', elsewhere: { notes: BASE, wfh: true }, holds: (d: { wfh: boolean }) => d.wfh === true, typeWfh: false },
  {
    name: 'partial leave only',
    elsewhere: { notes: BASE, leave_minutes: 60, leave_kind: 'vacation' },
    holds: (d: { leave_minutes: number; leave_kind: string | null }) => d.leave_minutes === 60 && d.leave_kind === 'vacation',
    typeWfh: false,
  },
  { name: 'category only', elsewhere: { notes: BASE, category: 'Off' }, holds: (d: { category: string | null }) => d.category === 'Off', typeWfh: false },
] as const;

for (const fieldCase of FIELD_CASES) {
  test(`int2-B ${fieldCase.name}: a change made elsewhere, then the tab comes back into view; the save is refused and nothing is overwritten`, async ({
    page,
    employeeSeed,
    signInThroughUi,
  }, testInfo) => {
    test.skip(testInfo.project.name !== 'desktop', 'the desktop drawer');
    const tag = `B ${fieldCase.name}`;
    const [date = ''] = await employeeSeed.displayedPeriodFreeWorkdays();
    expect(date).not.toBe('');
    await employeeSeed.putDay(date, { notes: BASE });
    try {
      await page.setViewportSize({ width: 1440, height: 800 });
      await signInThroughUi();
      const editor = await openDay(page, date);
      const { notes, wfh } = fieldsOf(editor);
      if (fieldCase.typeWfh) await wfh.check();
      else await notes.fill(TYPED);
      await employeeSeed.putDay(date, fieldCase.elsewhere);
      const after = await employeeSeed.dayView(date);
      const read = dayRead(page, date);
      await page.evaluate("document.dispatchEvent(new Event('visibilitychange'))");
      await read;
      await frames(page);
      log(tag, `${date}: elsewhere entry v${after.entry?.version}; after the tab return the form shows note "${await notes.inputValue()}" wfh ${await wfh.isChecked()}`);
      await editor.getByRole('button', { name: 'Save day fields' }).click();
      await expect(staleAlert(editor).or(savedStatus(editor))).toBeVisible();
      const refused = (await staleAlert(editor).count()) === 1;
      const server = await employeeSeed.dayView(date);
      const kept = (fieldCase.holds as (d: unknown) => boolean)(server);
      log(tag, `Save day fields: ${refused ? 'REFUSED (stale)' : 'ACCEPTED'}; the elsewhere value ${kept ? 'kept' : 'OVERWRITTEN'}; server notes "${server.entry?.notes}" wfh ${server.wfh} leave ${server.leave_minutes}/${server.leave_kind} category ${server.category}`);
      expect(refused, 'the save is refused').toBe(true);
      expect(kept, 'the value written elsewhere is kept').toBe(true);
      await reloadDay(page, editor, date);
      log(tag, `Reload day: note "${await notes.inputValue()}" wfh ${await wfh.isChecked()} category ${await fieldsOf(editor).category.inputValue()}`);
    } finally {
      await employeeSeed.resetDay(date);
    }
  });
}

test('int2-O overtaken read: the follow read answers after the editor\'s own save; a second save still succeeds', async ({ page, employeeSeed, signInThroughUi }, testInfo) => {

  const tag = 'O overtaken read';

  try {
    await page.setViewportSize({ width: 1440, height: 800 });
    await signInThroughUi();
    const editor = await openDay(page, date);
    let armed = true;
    await page.route(`**/api/days/${date}`, async (route) => {
      if (route.request().method() === 'GET' && armed) {
        armed = false;
        const response = await route.fetch();
        await new Promise((resolveDelay) => setTimeout(resolveDelay, 2_000));
        await route.fulfill({ response });
        return;
      }
      await route.continue();
    });
    const delayed = dayRead(page, date);
    const sent = page.waitForRequest((request) => request.method() === 'GET' && new URL(request.url()).pathname === `/api/days/${date}`, { timeout: 10_000 });
    await page.evaluate("document.dispatchEvent(new Event('visibilitychange'))");
    await sent;
    const put1 =page.waitForResponse((response) => response.request().method() === 'PUT' && new URL(response.url()).pathname === `/api/days/${date}`, { timeout: 10_000 });
    await editor.getByRole('button', { name: 'Save day fields' }).click();
    const first = await put1;
    await delayed;
    await frames(page);
    const put2 = page.waitForResponse((response) => response.request().method() === 'PUT' && new URL(response.url()).pathname === `/api/days/${date}`, { timeout: 10_000 });
    await editor.getByRole('button', { name: 'Save day fields' }).click();
    const second = await put2;
    log(
      tag,
      `${date}: first save ${first.status()} (sent v${(first.request().postDataJSON() as { expected_version?: number }).expected_version}); the delayed follow read answered after it; second save ${second.status()} (sent v${(second.request().postDataJSON() as { expected_version?: number }).expected_version}); stale alert ${await staleAlert(editor).count()}`,
    );
    expect(second.status(), 'the second save carries the version of the first save').toBe(200);
  } finally {
    await page.unrouteAll({ behavior: 'ignoreErrors' });
    await employeeSeed.resetDay(date);
  }
});
