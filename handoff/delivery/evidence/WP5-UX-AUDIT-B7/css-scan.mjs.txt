// Token scan of a stylesheet: literals outside :root token blocks, radius and transition use, undefined var() names.
// Usage: node css-scan.mjs <styles.css>
import { readFileSync } from 'node:fs';

const file = process.argv[2];
const text = readFileSync(file, 'utf8').replace(/\r\n/g, '\n');
// Strip comments but keep line numbers.
const source = text.replace(/\/\*[\s\S]*?\*\//g, (comment) => comment.replace(/[^\n]/g, ' '));
const lineOf = (index) => source.slice(0, index).split('\n').length;

const stack = [];
const declarations = [];
let buffer = '';
let bufferStart = 0;
for (let i = 0; i < source.length; i += 1) {
  const ch = source[i];
  if (ch === '{') {
    stack.push(buffer.trim());
    buffer = '';
    bufferStart = i + 1;
  } else if (ch === '}') {
    if (buffer.trim() !== '') declarations.push({ selectors: [...stack], text: buffer.trim(), line: lineOf(bufferStart + buffer.search(/\S/)) });
    stack.pop();
    buffer = '';
    bufferStart = i + 1;
  } else if (ch === ';') {
    if (buffer.trim() !== '') declarations.push({ selectors: [...stack], text: buffer.trim(), line: lineOf(bufferStart + buffer.search(/\S/)) });
    buffer = '';
    bufferStart = i + 1;
  } else {
    buffer += ch;
  }
}

const defined = new Set();
for (const d of declarations) {
  const m = /^(--[\w-]+)\s*:/.exec(d.text);
  if (m) defined.add(m[1]);
}
const inToken = (d) => d.selectors.some((s) => /:root/.test(s));
const problems = { colour: [], duration: [], radius: [], transition: [], undefinedVar: [], important: [] };
for (const d of declarations) {
  const m = /^([\w-]+)\s*:\s*([\s\S]*)$/.exec(d.text);
  if (!m) continue;
  const [, prop, value] = m;
  const where = `${d.line}: ${d.selectors.at(-1)} { ${prop}: ${value} }`;
  for (const ref of value.matchAll(/var\((--[\w-]+)/g)) if (!defined.has(ref[1])) problems.undefinedVar.push(where);
  if (/!important/.test(value)) problems.important.push(where);
  if (inToken(d)) continue;
  if (/#[0-9a-fA-F]{3,8}\b|\brgba?\(|\bhsla?\(|\boklch\(|\b(white|black|red|blue|green|gray|grey)\b/.test(value) && !/^(--)/.test(prop)) problems.colour.push(where);
  if (/\b\d+(\.\d+)?m?s\b/.test(value) && !/var\(/.test(value.replace(/\b\d+(\.\d+)?m?s\b/g, ''))) problems.duration.push(where);
  else if (/\b\d+(\.\d+)?m?s\b/.test(value)) problems.duration.push(where);
  if (/radius/.test(prop) && !/^(var\(--radius\)|0|50%|inherit|var\(--[\w-]+\))$/.test(value.trim())) problems.radius.push(where);
  if (/^transition/.test(prop) && !/^(var\(--transition\)|none)$/.test(value.trim())) problems.transition.push(where);
}
const radiusToken = declarations.find((d) => /^--radius\s*:/.test(d.text));
const transitionToken = declarations.find((d) => /^--transition\s*:/.test(d.text));
console.log(`file: ${file.replace(/^.*[\\/](src[\\/])/, '$1')}`);
console.log(`declarations: ${declarations.length}; custom properties defined: ${defined.size}`);
console.log(`--radius: ${radiusToken?.text} (line ${radiusToken?.line}); --transition: ${transitionToken?.text} (line ${transitionToken?.line})`);
const transitions = declarations.filter((d) => /^transition\s*:/.test(d.text) && !inToken(d));
console.log(`transition declarations outside tokens: ${transitions.length}; all var(--transition) or none: ${transitions.every((d) => /:\s*(var\(--transition\)|none)\s*$/.test(d.text))}`);
for (const [kind, list] of Object.entries(problems)) {
  console.log(`${kind}: ${list.length}`);
  for (const item of list) console.log(`  ${item}`);
}
