async function waitHealthy(origin: string, child: ChildProcess) {
  for (let attempt = 0; attempt < 150; attempt += 1) {
    if (child.exitCode !== null) throw new Error(`server exited early with ${child.exitCode}`);
    try {
      if ((await fetch(`${origin}/api/health`)).ok) return;
    } catch {
      // not listening yet
    }
    await new Promise((done) => setTimeout(done, 100));
  }
  throw new Error('server did not become healthy');
}

async function cast(adminSeed: SeedClient, origin: string): Promise<Cast> {
  const server = { origin } as never;
  const ownerA = await newPerson(adminSeed, server, { displayName: 'Synthetic Sharing Owner', signature: 'none' });
  const ownerB = await newPerson(adminSeed, server, { displayName: 'Synthetic Sharing Partner', signature: 'none' });
  const grantee = await newPerson(adminSeed, server, { displayName: 'Synthetic Grantee', signature: 'none' });
  const solo = await newPerson(adminSeed, server, { displayName: 'Synthetic Solo', signature: 'none' });
  for (const owner of [ownerA, ownerB]) {
    const [day = ''] = await owner.api.displayedPeriodFreeWorkdays();
    if (day !== '') await owner.api.seedCompleteDay(day);
  }
  await ownerA.api.call('POST', '/api/shares', { grantee_email: grantee.account.email, items: { timesheets: 'view', ot_read: false, pdf_download: false } }, 201);
  await ownerB.api.call('POST', '/api/shares', { grantee_email: grantee.account.email, items: { timesheets: 'edit', ot_read: true, pdf_download: true } }, 201);
  return { ownerA, ownerB, grantee, solo };
}

async function context(browser: Browser, testInfo: TestInfo, origin: string, zone: string): Promise<BrowserContext> {
  const use = testInfo.project.use;
  const created = await browser.newContext({

    visualScale: window.visualViewport?.scale ?? null,
  };
}

function switcherFacts() {
  const r2 = (value: number) => Math.round(value * 100) / 100;
  const rect = (element: Element | null) => {
    if (element === null) return null;
    const r = element.getBoundingClientRect();
    return { x: r2(r.x), y: r2(r.y), w: r2(r.width), h: r2(r.height), right: r2(r.right), bottom: r2(r.bottom) };
  };
  const parse = (value: string) => {
    const m = /rgba?\(([\d.]+),\s*([\d.]+),\s*([\d.]+)(?:,\s*([\d.]+))?\)/.exec(value);
    if (m === null) return null;
    return { r: Number(m[1]), g: Number(m[2]), b: Number(m[3]), a: m[4] === undefined ? 1 : Number(m[4]) };
  };
  const lum = (c: { r: number; g: number; b: number }) => {
    const ch = (v: number) => {
      const s = v / 255;
      return s <= 0.04045 ? s / 12.92 : ((s + 0.055) / 1.055) ** 2.4;
    };
    return 0.2126 * ch(c.r) + 0.7152 * ch(c.g) + 0.0722 * ch(c.b);
  };
  const background = (element: Element) => {
    for (let node: Element | null = element; node !== null; node = node.parentElement) {
      const c = parse(getComputedStyle(node).backgroundColor);
      if (c !== null && c.a > 0) return { color: c, from: `${node.tagName.toLowerCase()}.${node.className}` };
    }
    return { color: { r: 255, g: 255, b: 255, a: 1 }, from: 'default' };
  };
  const contrast = (fgText: string, element: Element) => {
    const fg = parse(fgText);
    const bg = background(element);
    if (fg === null) return null;
    const mixed = { r: fg.r * fg.a + bg.color.r * (1 - fg.a), g: fg.g * fg.a + bg.color.g * (1 - fg.a), b: fg.b * fg.a + bg.color.b * (1 - fg.a) };
    const l1 = lum(mixed);
    const l2 = lum(bg.color);
    return { ratio: r2((Math.max(l1, l2) + 0.05) / (Math.min(l1, l2) + 0.05)), fg: fgText, bg: `rgb(${bg.color.r}, ${bg.color.g}, ${bg.color.b})`, bgFrom: bg.from };
  };
  const bar = document.querySelector('.shell-bar');
  const brand = document.querySelector('.shell-brand');
  const form = document.querySelector('.shell-switcher');
  const base = {
    viewport: `${window.innerWidth}x${window.innerHeight}`,
    rootFont: getComputedStyle(document.documentElement).fontSize,
    bar: rect(bar),
    barSizeVar: document.documentElement.style.getPropertyValue('--shell-bar-size'),
    scrollWidth: document.documentElement.scrollWidth,
    innerWidth: window.innerWidth,
    clientWidth: document.documentElement.clientWidth,
    visualScale: window.visualViewport?.scale ?? null,
    // A mobile page zooms out to fit wider content, so innerWidth grows with it: compare with the layout width too.
    sideways: document.documentElement.scrollWidth > document.documentElement.clientWidth || document.documentElement.scrollWidth > window.innerWidth || (window.visualViewport?.scale ?? 1) < 0.999,
  };
  if (form === null || bar === null || brand === null) return { ...base, switcher: false };
  const label = form.querySelector('label');
  const span = form.querySelector('label .muted');
  const select = form.querySelector('select');
  const open = form.querySelector('button');
  if (label === null || span === null || select === null || open === null) return { ...base, switcher: 'incomplete' };
  const range = document.createRange();
  range.selectNodeContents(span);
  const textBox = range.getBoundingClientRect();
  const textLines = new Set([...range.getClientRects()].map((line) => Math.round(line.top))).size;
  const spanStyle = getComputedStyle(span);
  const selectStyle = getComputedStyle(select);
  const canvas = document.createElement('canvas').getContext('2d');
  const measure = (text: string) => {
    if (canvas === null) return -1;
    canvas.font = `${selectStyle.fontStyle} ${selectStyle.fontWeight} ${selectStyle.fontSize} ${selectStyle.fontFamily}`;
    return r2(canvas.measureText(text).width);
  };
  const chosen = select.options[select.selectedIndex]?.text ?? '';
  const contentWidth = select.clientWidth - parseFloat(selectStyle.paddingLeft) - parseFloat(selectStyle.paddingRight);
  const centre = (r: DOMRect) => document.elementFromPoint(r.left + r.width / 2, r.top + r.height / 2);
  const hitText = centre(textBox);
  const brandBox = brand.getBoundingClientRect();
  const formBox = form.getBoundingClientRect();
  const controls = [...bar.querySelectorAll('a, button, select, input')].filter((element) => element.getBoundingClientRect().width > 0);
  const overlaps = (a: DOMRect, b: DOMRect) => a.left < b.right - 0.5 && b.left < a.right - 0.5 && a.top < b.bottom - 0.5 && b.top < a.bottom - 0.5;
  const outside = [...bar.querySelectorAll('*')].filter((element) => {
    const r = element.getBoundingClientRect();
    return r.width > 0 && (r.left < -0.5 || r.right > window.innerWidth + 0.5);
  }).length;
  return {
    ...base,
    switcher: true,
    switcherRow: Math.abs(formBox.top + formBox.height / 2 - (brandBox.top + brandBox.height / 2)) < 8 ? 'same row as brand' : 'below brand',
    form: rect(form),
    label: rect(label),
    labelPadding: `${getComputedStyle(label).paddingTop} ${getComputedStyle(label).paddingBottom}`,
    labelText: span.textContent,
    labelTextBox: { x: r2(textBox.x), y: r2(textBox.y), w: r2(textBox.width), h: r2(textBox.height), right: r2(textBox.right) },
    labelTextLines: textLines,
    labelTextVisible: hitText !== null && (hitText === span || span.contains(hitText)),
    labelTextInsideViewport: textBox.left >= -0.5 && textBox.right <= window.innerWidth + 0.5 && textBox.top >= -0.5,
    labelTextOverlapsSelect: overlaps(textBox, select.getBoundingClientRect()),
    labelFont: `${spanStyle.fontSize} ${spanStyle.fontWeight}`,
    labelContrast: contrast(spanStyle.color, span),
    select: rect(select),
    selectFont: selectStyle.fontSize,
    selectContentWidth: r2(contentWidth),
    chosen,
    chosenTextWidth: measure(chosen),
    chosenFits: measure(chosen) <= contentWidth - 20,
    optionWidths: [...select.options].map((option) => ({ text: option.text, width: measure(option.text) })),
    selectValueContrast: contrast(selectStyle.color, select),
    open: rect(open),
    openHit: (() => {
      const b = open.getBoundingClientRect();
      const out: Record<string, string> = {};
      for (const [name, x] of [
        ['left+4', b.left + 4],
        ['centre', b.left + b.width / 2],
      ] as const) {
        const hit = document.elementFromPoint(x, b.top + b.height / 2);
        out[name] = hit === null ? 'none' : hit === open || open.contains(hit) ? 'Open' : `${hit.tagName.toLowerCase()}`;
      }
      return out;
    })(),
    selectOverlapsOpen: select.getBoundingClientRect().right > open.getBoundingClientRect().left + 0.5,
    selectMinWidth: selectStyle.minWidth,
    openInsideBar: open.getBoundingClientRect().bottom <= bar.getBoundingClientRect().bottom + 0.5,
    targets: controls.map((element) => {
      const r = element.getBoundingClientRect();
      return { name: `${element.tagName.toLowerCase()} ${(element.textContent ?? '').trim().slice(0, 18)}`, w: r2(r.width), h: r2(r.height), ok: r.width >= 44 - 0.01 && r.height >= 44 - 0.01 };
    }),
    elementsOutsideViewport: outside,
    styles: {
      barColumnGap: getComputedStyle(bar).columnGap,
      formFlex: getComputedStyle(form).flex,
      labelRadius: selectStyle.borderRadius,
      selectTransition: selectStyle.transition,
      openTransition: getComputedStyle(open).transition,
      openRadius: getComputedStyle(open).borderRadius,
    },
  };
}

function focusStop() {
  const element = document.activeElement;
  if (element === null || element === document.body || element === document.documentElement) return null;
  const path: string[] = [];
  for (let node: Element | null = element; node !== null && node !== document.body; node = node.parentElement) {
    const parent: Element | null = node.parentElement;
    const index = parent === null ? 0 : Array.prototype.indexOf.call(parent.children, node);
    path.unshift(`${node.tagName.toLowerCase()}${index}`);
  }
  const r = element.getBoundingClientRect();
  const vw = window.innerWidth;
  const vh = window.innerHeight;
  const overlaySelector = '.shell-bar, .shell-tabs, [data-share-bar], .editor-head, .more-panel';
  const labels = 'labels' in element && element.labels instanceof NodeList ? [...(element.labels as NodeListOf<HTMLLabelElement>)] : [];
  const ownedBy = (hit: Element | null) => hit !== null && (hit === element || element.contains(hit) || labels.some((label) => label.contains(hit)));
  const blocker = (hit: Element | null) => {
    if (hit === null) return 'nothing';
    const overlay = hit.closest(overlaySelector);
    if (overlay !== null && !overlay.contains(element)) return overlay.className.split(' ')[0] || overlay.tagName.toLowerCase();
    return null;
  };
  let visible = 0;
  const covered: Record<string, number> = {};
  for (let i = 0; i < 5; i += 1) {
    for (let j = 0; j < 5; j += 1) {
      const x = r.left + (r.width * (i + 0.5)) / 5;
      const y = r.top + (r.height * (j + 0.5)) / 5;
      if (x < 0 || y < 0 || x >= vw || y >= vh) {
        covered.offscreen = (covered.offscreen ?? 0) + 1;
        continue;
      }
      const hit = document.elementFromPoint(x, y);
      if (ownedBy(hit)) visible += 1;
      else {
        const by = blocker(hit) ?? `other:${hit?.tagName.toLowerCase() ?? ''}`;
        covered[by] = (covered[by] ?? 0) + 1;
      }
    }
  }
  // The ring: the element's own box-shadow or outline, or a :focus-within wrapper up to three levels up.
  const accent = getComputedStyle(document.documentElement).getPropertyValue('--accent').trim();
  const probe = document.createElement('i');
  probe.style.color = accent;
  document.body.append(probe);
  const accentRgb = getComputedStyle(probe).color;
  probe.remove();
  let ring: string | null = null;
  let ringOn: Element = element;
  for (let node: Element | null = element, depth = 0; node !== null && depth < 4; node = node.parentElement, depth += 1) {
    const style = getComputedStyle(node);
    const hasShadow = style.boxShadow.includes(accentRgb);
    const hasOutline = style.outlineStyle !== 'none' && parseFloat(style.outlineWidth) > 0;
    if (hasShadow || hasOutline) {
      ring = hasShadow ? (/inset/.test(style.boxShadow) && !/\) 0px 0px 0px 3px,|0px 0px 0px 3px(?! inset)/.test(style.boxShadow) ? 'inset' : 'outer') : 'outline';
      ringOn = node;
      break;
    }
  }
  const ringBox = ringOn.getBoundingClientRect();
  const sides: Record<string, string> = {};
  if (ring !== null) {
    const off = ring === 'inset' ? -1.5 : 2;
    const points: Record<string, [number, number]> = {
      top: [ringBox.left + ringBox.width / 2, ringBox.top - off],
      bottom: [ringBox.left + ringBox.width / 2, ringBox.bottom + off],
      left: [ringBox.left - off, ringBox.top + ringBox.height / 2],
      right: [ringBox.right + off, ringBox.top + ringBox.height / 2],
    };
    for (const [side, [x, y]] of Object.entries(points)) {
      if (x < 0 || y < 0 || x >= vw || y >= vh) sides[side] = 'offscreen';
      else sides[side] = blocker(document.elementFromPoint(x, y)) ?? 'shown';
    }
  }
  const name = element.getAttribute('aria-label') ?? (element.textContent ?? '').trim().replace(/\s+/g, ' ').slice(0, 40);
  return {
    key: path.join('/'),
    what: `${element.tagName.toLowerCase()}${element.getAttribute('type') === null ? '' : `[${element.getAttribute('type')}]`} ${name}`,
    rect: { x: Math.round(r.x), y: Math.round(r.y), w: Math.round(r.width), h: Math.round(r.height) },
    visible,
    covered,
    focusVisible: element.matches(':focus-visible'),
    ring,
    ringOnSelf: ringOn === element,
    ringSidesShown: Object.values(sides).filter((value) => value === 'shown').length,
    sides,
    inDialog: element.closest('dialog') !== null,
    barBottom: Math.round(document.querySelector('.shell-bar')?.getBoundingClientRect().bottom ?? 0),
  };
}

type Stop = NonNullable<ReturnType<typeof focusStop>>;

async function walk(page: Page, key: 'Tab' | 'Shift+Tab', limit: number, startOnBody: boolean): Promise<{ stops: Stop[]; outside: number }> {
  if (startOnBody) {
    await page.evaluate(() => {
      (document.activeElement as HTMLElement | null)?.blur();
      window.scrollTo(0, 0);
    });
  }
  const stops: Stop[] = [];
  let firstKey: string | null = null;
  let outside = 0;
  for (let index = 0; index < limit; index += 1) {
    await page.keyboard.press(key);
    await page.waitForTimeout(70);
    const stop = await page.evaluate(focusStop);
    if (stop === null) {
      outside += 1;
      if (outside > 3) break;
      continue;
    }
    if (firstKey !== null && stop.key === firstKey) break;
    firstKey ??= stop.key;
    stops.push(stop);
  }
  return { stops, outside };
}

function summarize(stops: Stop[]) {
  return {
    stops: stops.length,
    hidden: stops.filter((stop) => stop.visible === 0).map((stop) => ({ what: stop.what, covered: stop.covered, rect: stop.rect })),
    partlyCovered: stops.filter((stop) => stop.visible > 0 && stop.visible < 25 && Object.keys(stop.covered).some((k) => !k.startsWith('other') && k !== 'offscreen')).map((stop) => ({ what: stop.what, visible: stop.visible, covered: stop.covered })),
    noFocusVisible: stops.filter((stop) => !stop.focusVisible).map((stop) => stop.what),
    noRing: stops.filter((stop) => stop.ring === null).map((stop) => stop.what),
    ringAllSidesHidden: stops.filter((stop) => stop.ring !== null && stop.ringSidesShown === 0).map((stop) => ({ what: stop.what, sides: stop.sides })),
    minVisible: stops.reduce((min, stop) => Math.min(min, stop.visible), 25),
  };
}

/* ---- P3b: rendered ring around the switcher's select and Open (screenshot pairs) -------------------- */

test('P3b switcher rings rendered', async ({ browser, adminSeed, server }, testInfo) => {
  const mobile = testInfo.project.name === 'mobile';
  const people = await cast(adminSeed, server.origin);
  const sizes = mobile
    ? [
        { width: 767, height: 1024 },
        { width: 390, height: 844 },
        { width: 320, height: 640 },
      ]
    : [{ width: 767, height: 800 }];
  const ctx = await context(browser, testInfo, server.origin, NOTE_ZONE);
  const page = await ctx.newPage();
  const decoder = await ctx.newPage();
  await signIn(page, people.grantee.account);
  const rows: unknown[] = [];
  for (const theme of THEMES) {
    await page.emulateMedia({ colorScheme: theme, reducedMotion: 'reduce' });
    for (const size of sizes) {
      await page.setViewportSize(size);
      for (const [hash, label] of [
        ['#/timesheet', 'own'],
        [`#/shared/${people.ownerA.account.id}`, 'shared'],
      ] as const) {
        await show(page, hash);
        for (const target of ['select', 'button'] as const) {
          // Reach the control with real Tab presses from the top of the page.
          await page.evaluate(() => (document.activeElement as HTMLElement | null)?.blur());
          let reached = false;
          for (let index = 0; index < 12 && !reached; index += 1) {
            await page.keyboard.press('Tab');
            reached = await page.evaluate((tag) => document.activeElement?.closest('.shell-switcher') !== null && document.activeElement?.tagName.toLowerCase() === tag, target);
          }
          const box = await page.locator(`.shell-switcher ${target}`).boundingBox();
          if (box === null || !reached) {
            rows.push({ theme, size: `${size.width}x${size.height}`, view: label, target, reached });
            continue;
          }
          const clip = { x: Math.max(0, box.x - 8), y: Math.max(0, box.y - 8), width: box.width + 16, height: box.height + 16 };
          const focused = await page.screenshot({ clip, animations: 'disabled', caret: 'hide' });
          await page.evaluate(() => (document.activeElement as HTMLElement | null)?.blur());
          await page.waitForTimeout(150);
          const plain = await page.screenshot({ clip, animations: 'disabled', caret: 'hide' });
          const inner = { left: box.x - clip.x, top: box.y - clip.y, width: box.width, height: box.height };
          const sides = await decoder.evaluate(
            async ({ a, b, inner, cssWidth }) => {
              const load = async (data: string) => {
                const bytes = Uint8Array.from(atob(data), (c) => c.charCodeAt(0));
                const bitmap = await createImageBitmap(new Blob([bytes], { type: 'image/png' }));
                const canvas = new OffscreenCanvas(bitmap.width, bitmap.height);
                const ctx2 = canvas.getContext('2d');
                if (ctx2 === null) throw new Error('no 2d');
                ctx2.drawImage(bitmap, 0, 0);
                return { width: bitmap.width, height: bitmap.height, data: ctx2.getImageData(0, 0, bitmap.width, bitmap.height).data };
              };
              const fa = await load(a);
              const fb = await load(b);
              const scale = fa.width / cssWidth;
              const lum = (r: number, g: number, bl: number) => {
                const ch = (v: number) => {
                  const s = v / 255;
                  return s <= 0.04045 ? s / 12.92 : ((s + 0.055) / 1.055) ** 2.4;
                };
                return 0.2126 * ch(r) + 0.7152 * ch(g) + 0.0722 * ch(bl);
              };
              const ratio = (p: number[], q: number[]) => {
                const l1 = lum(p[0] ?? 0, p[1] ?? 0, p[2] ?? 0);
                const l2 = lum(q[0] ?? 0, q[1] ?? 0, q[2] ?? 0);
                return Math.round(((Math.max(l1, l2) + 0.05) / (Math.min(l1, l2) + 0.05)) * 100) / 100;
              };
              const px = (f: typeof fa, x: number, y: number) => {
                const i = (Math.round(y * scale) * f.width + Math.round(x * scale)) * 4;
                return [f.data[i] ?? 0, f.data[i + 1] ?? 0, f.data[i + 2] ?? 0];
              };
              const out: Record<string, unknown> = {};
              const lines: Record<string, Array<[number, number]>> = {
                top: Array.from({ length: 5 }, (_, k) => [inner.left + (inner.width * (k + 1)) / 6, inner.top]),
                bottom: Array.from({ length: 5 }, (_, k) => [inner.left + (inner.width * (k + 1)) / 6, inner.top + inner.height]),
                left: Array.from({ length: 5 }, (_, k) => [inner.left, inner.top + (inner.height * (k + 1)) / 6]),
                right: Array.from({ length: 5 }, (_, k) => [inner.left + inner.width, inner.top + (inner.height * (k + 1)) / 6]),
              };
              for (const [side, points] of Object.entries(lines)) {
                let best = 0;
                let changed = 0;
                for (const [x0, y0] of points) {
                  // Walk 0..5 css px outward across the edge and compare the focused pixel with the unfocused one there.
                  for (let d = 0; d <= 5; d += 0.5) {
                    const x = side === 'left' ? x0 - d : side === 'right' ? x0 + d : x0;
                    const y = side === 'top' ? y0 - d : side === 'bottom' ? y0 + d : y0;
                    const p = px(fa, x, y);
                    const q = px(fb, x, y);
                    if (p.some((v, k) => Math.abs(v - (q[k] ?? 0)) > 24)) {
                      changed += 1;
                      best = Math.max(best, ratio(p, q));
                    }
                  }
                }
                out[side] = { changedSamples: changed, bestRingContrastAgainstUnfocused: best };
              }
              return out;
            },
            { a: focused.toString('base64'), b: plain.toString('base64'), inner, cssWidth: clip.width },
          );
          rows.push({ theme, size: `${size.width}x${size.height}`, view: label, target, reached, box, sides });
          if (size.width !== 767) writeFileSync(join(OUT, `${TAG}-p3b-${label}-${target}-${size.width}-${theme}-focused-synthetic.png`), focused);
        }
      }
    }
  }
  await ctx.close();
  save(`p3b-rings-${testInfo.project.name}`, rows);
});

  const props = [
    'display', 'position', 'flex-direction', 'flex-wrap', 'flex-grow', 'flex-shrink', 'flex-basis', 'column-gap', 'row-gap',
    'padding-top', 'padding-right', 'padding-bottom', 'padding-left', 'margin-top', 'margin-right', 'margin-bottom', 'margin-left',
    'font-size', 'font-weight', 'line-height', 'color', 'background-color', 'border-radius', 'box-shadow', 'transition',
    'min-height', 'min-width', 'top', 'z-index',
  ];
  const rows: Record<string, string> = {};
  const all = document.querySelectorAll('body *');
  all.forEach((element) => {
    const path: string[] = [];
    for (let node: Element | null = element; node !== null && node !== document.body; node = node.parentElement) {
      const parent: Element | null = node.parentElement;
      const index = parent === null ? 0 : Array.prototype.indexOf.call(parent.children, node);
      path.unshift(`${node.tagName.toLowerCase()}${index}`);
    }
    const style = getComputedStyle(element);
    const r = element.getBoundingClientRect();
    rows[path.join('/')] = `${props.map((prop) => style.getPropertyValue(prop)).join('|')}|${Math.round(r.x * 10) / 10},${Math.round(r.y * 10) / 10},${Math.round(r.width * 10) / 10},${Math.round(r.height * 10) / 10}`;
  });
  return { count: all.length, rows, shellBarSizeVar: document.documentElement.style.getPropertyValue('--shell-bar-size'), scrollPaddingTop: getComputedStyle(document.documentElement).scrollPaddingTop };
}

    for (const theme of THEMES) {

      }
    }
    await ctx.close();
