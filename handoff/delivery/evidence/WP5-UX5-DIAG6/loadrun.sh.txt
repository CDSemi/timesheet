# Usage: bash loadrun.sh <label> <burn threads> <burn seconds> <playwright args...>
# A self-stopping CPU burner (burn.mjs: every thread ends at a fixed deadline) runs beside one Playwright run; the script waits for both.
. <task>/env.sh
L=$1; TH=$2; SEC=$3; shift 3
node.exe $T/burn.mjs $TH $SEC > $T/logs/burn-$L.txt 2>&1 &
BURN=$!
bash $T/run-pw.sh $L real "$@" > $T/logs/$L.txt 2>&1
wait $BURN
echo "burner $L ended (exit=$?)" >> $T/logs/burn-$L.txt
