# Scratch-only spec copies in the export: the ORIGINAL (FIX12-tree) and the FIXED tests with a delay or a CPU throttle hook appended.
# dly-kbd-*: the lookup of the running session answers PROBE_DELAY ms late; dly-esc-*: the main thread runs PROBE_THROTTLE times slower.
. <task>/env.sh
X=$T/export/tests/e2e
cp $T/orig/tests/e2e/keyboard-access.spec.ts $X/dly-kbd-orig.spec.ts && cat $T/delay-snippet.ts.txt >> $X/dly-kbd-orig.spec.ts
cp $T/stage/tests/e2e/keyboard-access.spec.ts $X/dly-kbd-fixed.spec.ts && cat $T/delay-snippet.ts.txt >> $X/dly-kbd-fixed.spec.ts
cp $T/orig/tests/e2e/day-editor.spec.ts $X/dly-esc-orig.spec.ts && cat $T/throttle-snippet.ts.txt >> $X/dly-esc-orig.spec.ts
cp $T/stage/tests/e2e/day-editor.spec.ts $X/dly-esc-fixed.spec.ts && cat $T/throttle-snippet.ts.txt >> $X/dly-esc-fixed.spec.ts
ls $X | grep dly
