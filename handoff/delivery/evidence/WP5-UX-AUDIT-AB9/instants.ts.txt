// WP5-UX-AUDIT-AB9: what each sweep instant exercises (production period functions + the FIX10 helpers), for the sweep table.
// Run: node instants.mts <export-root>
import { readFileSync } from 'node:fs';
import { join } from 'node:path';
import { pathToFileURL } from 'node:url';

const root = process.argv[2];
if (root === undefined) throw new Error('usage: node instants.mts <export-root>');
const imp = (rel: string) => import(pathToFileURL(join(root, rel)).href);
const periods = await imp('src/domain/periods.ts');
const zones = await imp('src/domain/zones.ts');
const dates = await imp('src/domain/dates.ts');
const instants = await imp('src/domain/instants.ts');
const helper = await imp('tests/e2e/displayedSeed.ts');
const adminModel = await imp('src/client/components/adminModel.ts');
const policy = JSON.parse(readFileSync(join(root, 'reference/examples/policy.example.json'), 'utf8'));
const HOLIDAYS: string[] = JSON.parse(readFileSync(join(root, 'reference/examples/holidays.2026.example.json'), 'utf8')).holidays.map((h: { date: string }) => h.date);
const P = policy.payroll;
const ZONE = policy.calendar.reporting_zone;
const schedule = { anchorPayrollDate: P.anchor_payroll_date, cycleDays: P.cycle_days, periodStartOffsetDays: P.period_start_offset_days, periodEndOffsetDays: P.period_end_offset_days, dueOffsetDays: P.due_offset_days, dueLocalTime: P.due_local_time, reportingZone: ZONE };
const list: Array<[string, string]> = [
  ['real-full (real clock)', '2026-10-14T06:36:45Z'],
  ['s1-due-tue-20261124-1830', '2026-11-25T02:30:00Z'],
  ['s2-payroll-fri-20261211-1600', '2026-12-12T00:00:00Z'],
  ['s3-mon-20261130-1500', '2026-11-30T23:00:00Z'],
  ['s4-thu-20261231-1800', '2027-01-01T02:00:00Z'],
  ['s5-mon-p4-20270315-1000', '2027-03-15T17:00:00Z'],
  ['s6-due-tue-20270914-1830', '2027-09-15T01:30:00Z'],
  ['s7-payroll-fri-20271001-1000', '2027-10-01T17:00:00Z'],
  ['s8-due-tue-20280229-1830', '2028-03-01T02:30:00Z'],
];
const weekday = (d: string) => new Date(`${d}T12:00:00Z`).getUTCDay();
for (const [label, iso] of list) {
  const nowMs = Date.parse(iso);
  const today = zones.localDateOf(ZONE, nowMs / 1000);
  const shown = periods.currentPayPeriod(schedule, today);
  const inProgress = periods.payPeriodContaining(schedule, today);
  const facts = { payroll_date: shown.payrollDate, nominal_payroll_date: shown.nominalPayrollDate, period_start: shown.periodStart, due_at_utc: instants.formatUtcInstant(shown.dueAtUtc) };
  const later = helper.laterDeadlineFor(facts, today, nowMs, 120_000);
  const year = Number(today.slice(0, 4));
  const warnings = today >= `${year}-10-01` && !HOLIDAYS.some((h) => h.startsWith(`${year + 1}-`)) ? [{ code: 'next_year_calendar_missing', year: year + 1, warn_from: `${year}-10-01`, message: '' }] : [];
  const importYear = adminModel.defaultImportYear(today, warnings);
  const importDates = helper.importDatesAfter([`${importYear}-01-01`, `${importYear}-07-04`, `${importYear}-03-03`], shown.periodEnd);
  let pool = 0;
  for (let d = shown.periodStart; d <= shown.periodEnd && d < today; d = dates.addDays(d, 1)) {
    const w = weekday(d);
    if (w >= 1 && w <= 5 && !HOLIDAYS.includes(d)) pool += 1;
  }
  console.log(
    `${label}: LA today ${today}; displayed ${shown.periodStart}..${shown.periodEnd} payroll ${shown.payrollDate} (in-progress ${inProgress.periodStart}..${inProgress.periodEnd}); natural due ${facts.due_at_utc}; ` +
      `deadline path ${later === null ? 'natural (due ahead)' : `moved to ${later.due_local_date} ${later.due_local_time}`}; past free workdays of the displayed period ${pool}; ` +
      `import year ${importYear}${warnings.length ? ' (warning)' : ''}; import dates ${importDates.join(' ')}${importDates.includes(today) ? ' (one is TODAY)' : ''}`,
  );
}
