# Labels carry the America/Los_Angeles wall time; the argument is the same instant in UTC.

  label=$1; instant=$2; shift 2
  bash $T/run-shift.sh $NEW "$label" "$instant" "$@" > "$T/logs/sweep-$label.txt" 2>&1

}
