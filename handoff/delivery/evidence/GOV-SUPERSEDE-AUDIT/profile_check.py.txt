"""GOV-SUPERSEDE-AUDIT scope 5: profiles at 4def605 (old/) and 831f760 (new/) from the task folder.
Frontmatter byte identity, body changes limited to the runtime line, coordinator unchanged, parse check."""
import difflib
import hashlib
import importlib.util
import re
import sys
from pathlib import Path

sys.dont_write_bytecode = True
TASK = Path(__file__).resolve().parent
REPO = Path(r"<project>")
spec = importlib.util.spec_from_file_location("validator", REPO / "handoff/delivery/validate_orchestration.py")
v = importlib.util.module_from_spec(spec)
spec.loader.exec_module(v)
NAMES = ["auditor", "committer", "coordinator", "expert", "light", "planner", "verifier", "worker", "worker-high"]
RUNTIME = ("Runtime: never feed a script to python or node through stdin (no heredocs, no `| node`, `| python`, "
           "`node -` or `python -`); write a file and run it. Never pipe output into head or tail; redirect to a "
           "file and read it.")
lines = []
for name in NAMES:
    old = (TASK / "old" / f"timesheet-{name}.md").read_bytes()
    new = (TASK / "new" / f"timesheet-{name}.md").read_bytes()
    repo = (REPO / ".claude/agents" / f"timesheet-{name}.md").read_bytes()
    front_old = re.match(rb"\A---\n(.*?)\n---\n", old, re.S)
    front_new = re.match(rb"\A---\n(.*?)\n---\n", new, re.S)
    same_front = front_old is not None and front_new is not None and front_old.group(0) == front_new.group(0)
    body_old = old[front_old.end():].decode("utf-8").splitlines()
    body_new = new[front_new.end():].decode("utf-8").splitlines()
    ops = [op for op in difflib.SequenceMatcher(None, body_old, body_new, autojunk=False).get_opcodes()
           if op[0] != "equal"]
    inserted = [line for tag, i1, i2, j1, j2 in ops if tag == "insert" for line in body_new[j1:j2]]
    only_inserts = all(op[0] == "insert" for op in ops)
    text = " ".join(line.strip() for line in inserted if line.strip())
    fields = v.parse_profile(new.decode("utf-8"), f"timesheet-{name}.md")
    lines.append(f"timesheet-{name}.md: frontmatter identical={same_front} "
                 f"(sha256 {hashlib.sha256(front_new.group(0)).hexdigest()[:16]}); whole file old==new={old == new}; "
                 f"worktree==831f760={repo == new}; body ops={[(op[0], op[1] + 1, op[2], op[3] + 1, op[4]) for op in ops]}; "
                 f"only insertions={only_inserts}; inserted text == runtime line={text == RUNTIME}; "
                 f"CRLF in new={b'\r' in new}; parse: name={fields['name']} model={fields['model']} "
                 f"effort={fields['effort']} tools={fields.get('tools')}")
    for line in inserted:
        lines.append(f"    + {line}")
print("\n".join(lines))
print(f"profiles() on the repository: {len(v.profiles())} profiles")
