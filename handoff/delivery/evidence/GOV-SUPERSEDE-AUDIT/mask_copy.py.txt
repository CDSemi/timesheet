def mask(text):
    text = text.replace("\r\n", "\n").replace("\r", "\n")
    for task in (str(TASK), str(TASK).replace("\\", "/"), "<task>"):
        text = text.replace(task, "<task>")
    for repo in (str(REPO), str(REPO).replace("\\", "/"), "<project>"):
        text = text.replace(repo, "<project>")
    text = re.sub(r"(Users[\\/])[^\\/\s'\"]+", r"\1<user>", text)
    if USER:
        text = re.sub(re.escape(USER), "<user>", text, flags=re.I)
    return text if text.endswith("\n") else text + "\n"


def section(title, command, raw, exit_code=None):
    body = (TASK / raw).read_text(encoding="utf-8", errors="replace")
    head = f"## {title}\n$ {command}\n" + (f"exit={exit_code}\n" if exit_code is not None else "")
    return head + body.rstrip("\n") + "\n\n"
