"""GOV-SUPERSEDE-AUDIT scopes 1 and 3: independent synthetic boards (own builders, not check_recovery's)
run through the reviewed validator (repository file, loaded read-only without bytecode). Each scenario
records the observed outcome and the outcome expected from the GOV-SUPERSEDE-FIX rules. No repo write."""
import copy
import importlib.util
import json
import sys
from pathlib import Path

sys.dont_write_bytecode = True
TASK = Path(__file__).resolve().parent
REPO = Path(r"<project>")
spec = importlib.util.spec_from_file_location("validator", REPO / "handoff/delivery/validate_orchestration.py")
v = importlib.util.module_from_spec(spec)
spec.loader.exec_module(v)
configured = v.profiles()
HEAD_BOARD = json.loads((TASK / "new" / "ORCHESTRATION.json").read_text(encoding="utf-8"))
HEAD_STATE = json.loads((TASK / "new" / "STATE.json").read_text(encoding="utf-8"))
REPORT = "handoff/delivery/WP1_HANDOFF.md"
EVIDENCE = "handoff/delivery/evidence/WP1-review-codex/typecheck.txt"
PROMPT = "handoff/prompts/ORCHESTRATE.md"
PROFILE_OF = {"plan": "timesheet-planner", "implement": "timesheet-worker-high", "fix": "timesheet-worker-high",
              "gate": "timesheet-verifier", "audit": "timesheet-auditor", "documentation": "timesheet-worker",
              "commit": "timesheet-committer"}
A, B, C, X, Y = "a" * 64, "b" * 64, "c" * 64, "1" * 40, "2" * 40
D1, D2 = "d" * 64, "e" * 64


def mk(name, kind, deps, package="WP5", **fields):
    profile = configured[PROFILE_OF[kind]]
    task = {"id": name, "package": package, "kind": kind, "status": "pending", "depends_on": deps,
            "profile": profile["name"], "requested_model": profile["model"], "requested_effort": profile["effort"],
            "model_override_reason": None, "actual_model": None, "actual_source": None, "actual_effort": None,
            "agent_id": None, "attempt": 0, "owned_paths": [REPORT], "prompt": PROMPT, "report": REPORT,
            "evidence": [], "next_action": "synthetic"}
    task.update(fields)
    return task


def done(task, agent, **fields):
    task.update(status="done", attempt=1, agent_id=agent, **fields)
    if task["kind"] in {"gate", "audit"}:
        task.setdefault("evidence", [])
        task["evidence"] = [EVIDENCE]
    return task


def state_for(ready=False):
    value = copy.deepcopy(HEAD_STATE)
    value["active_work_package"] = "WP5"
    value["phases"] = [{"id": name, "independent_review": "passed" if ready or name != "WP5" else "not_started"}
                       for name in v.PACKAGES]
    return value


def base():
    """WP5 active. A1 accepted digest A; D1 (like WP5-ACCREC) depends on A1; a later change I2 is gated
    at digest B (current); the re-audit A2 is pending; A1.superseded_by = A2."""
    tasks = [
        done(mk("P", "plan", []), "a-plan"),
        done(mk("I1", "implement", ["P"]), "a-impl1"),
        done(mk("G1", "gate", ["I1"]), "a-g1", decision="PASS", source_digest=A),
        done(mk("A1", "audit", ["G1"]), "a-aud1", decision="PASS", source_digest=A, reviewed_digest=A,
             superseded_by="A2"),
        done(mk("D1", "documentation", ["A1"]), "a-doc"),
        done(mk("C1", "commit", ["D1"]), "a-c1", commit_sha=X, pushed=True, branch="main"),
        done(mk("I2", "implement", ["C1"]), "a-impl2"),
        done(mk("G2", "gate", ["I2"]), "a-g2", decision="PASS", source_digest=B),
        mk("A2", "audit", ["G2"]),
        mk("N", "plan", ["A2"]),
    ]
    board = copy.deepcopy(HEAD_BOARD)
    board.update(status="running", tasks=tasks, auxiliary_lookups=[], active_package="WP5", next_task_id="A2",
                 current_source_digest=B,
                 git={"branch": "main", "release_declared": False, "last_commit": X, "unpushed": []})
    return board, state_for()


def t(board, name):
    return next(task for task in board["tasks"] if task["id"] == name)


def a2_pass(board, digest=B, agent="a-aud2"):
    done(t(board, "A2"), agent, decision="PASS", source_digest=digest, reviewed_digest=digest)


def ready(board):
    for task in board["tasks"]:
        if task["status"] not in {"done", "cancelled"}:
            task["status"] = "cancelled"
    board.update(status="software_ready", next_task_id=None)
    return state_for(ready=True)


def gov(board, same_digest=False):
    board["tasks"] += [
        done(mk("GI1", "implement", [], package="GOV"), "g-impl"),
        done(mk("GG1", "gate", ["GI1"], package="GOV"), "g-g1", decision="PASS", source_digest=D1, freeze_commit=X),
        done(mk("GA1", "audit", ["GG1"], package="GOV"), "g-a1", decision="PASS", source_digest=D1,
             reviewed_digest=D1, reviewed_commit=X),
        done(mk("GG2", "gate", ["GI1"], package="GOV"), "g-g2", decision="PASS",
             source_digest=D1 if same_digest else D2, freeze_commit=Y),
        done(mk("GA2", "audit", ["GG2"], package="GOV"), "g-a2", decision="PASS",
             source_digest=D1 if same_digest else D2, reviewed_digest=D1 if same_digest else D2, reviewed_commit=Y),
    ]


SCENARIOS = []


def scenario(code, label, expected):
    def wrap(func):
        SCENARIOS.append((code, label, expected, func))
        return func
    return wrap

results = []
mismatch = 0
for code, label, expected, func in SCENARIOS:
    board, state = base()
    new_state = func(board, state)
    state = new_state or state
    try:
        v.validate(board, state, configured)
        observed = "ACCEPT"
    except ValueError as error:
        observed = f"REJECT: {error}"
    ok = observed == "ACCEPT" if expected == "ACCEPT" else (observed.startswith("REJECT") and expected in observed)
    mismatch += not ok
    results.append(f"{code} {'OK ' if ok else 'MISMATCH'} expected={expected!r} observed={observed!r} :: {label}")
print("\n".join(results))
print(f"scenarios={len(SCENARIOS)} mismatches={mismatch}")
