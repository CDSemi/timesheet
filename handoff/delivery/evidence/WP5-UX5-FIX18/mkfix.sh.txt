# Usage: bash mkfix.sh <export folder name in the task folder>
# A scratch export of the working tree: git archive of HEAD, then the owned files of the working tree copied over it, then npm ci, lint, typecheck, build.
. /d/Dropbox/Work.CDSemi/timesheet/handoff/delivery/evidence/WP5-UX5-FIX18/env.sh.txt
X=$1
mkdir -p $T/$X
tar -xf $T/head.tar -C $T/$X || exit 2
for f in src/client/TimesheetScreen.tsx src/client/DayEditor.tsx tests/e2e/day-editor.spec.ts docs/04_UX_AND_SETTINGS.md docs/04_UX_AND_SETTINGS.vi.md; do
  cp $REPO/$f $T/$X/$f || exit 3
done
cd $T/$X
npmx ci > $E/22-$X-ci.txt 2>&1; echo "ci=$?" >> $E/22-$X-exits.txt
npmx run lint > $E/22-$X-lint.txt 2>&1; echo "lint=$?" >> $E/22-$X-exits.txt
npmx run typecheck > $E/22-$X-typecheck.txt 2>&1; echo "typecheck=$?" >> $E/22-$X-exits.txt
npmx run build > $E/22-$X-build.txt 2>&1; echo "build=$?" >> $E/22-$X-exits.txt
echo "$X ready" >> $E/22-$X-exits.txt
