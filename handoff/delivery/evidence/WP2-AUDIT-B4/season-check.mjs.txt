// WP2-AUDIT-B4: no-Intl season check of the reviewed oracle for the R-07 conversion (WP2-B2-01).
// For every day 2026-2028: 22:00 and 23:30 typed in Asia/Ho_Chi_Minh (+07:00, no DST) and the alias
// Asia/Saigon must give the Los Angeles wall time computed by US-rule arithmetic only.
import { pathToFileURL } from 'node:url';
import path from 'node:path';
const { wallTimeIn, instantOfWallTime } = await import(pathToFileURL(path.join(process.argv[2], 'tests/client/zoneOracle.ts')).href);
const nthSunday = (y, mo, n) => { const f = new Date(Date.UTC(y, mo, 1)).getUTCDay(); return 1 + ((7 - f) % 7) + (n - 1) * 7; };
const pad = (n) => String(n).padStart(2, '0');
let n = 0, bad = 0, pst = 0;
for (let ms = Date.UTC(2026, 0, 1); ms < Date.UTC(2029, 0, 1); ms += 86_400_000) {
  const d = new Date(ms); const y = d.getUTCFullYear();
  const date = d.toISOString().slice(0, 10);
  const dstStart = Date.UTC(y, 2, nthSunday(y, 2, 2), 10); const dstEnd = Date.UTC(y, 10, nthSunday(y, 10, 1), 9);
  for (const [time, h, m] of [['22:00', 15, 0], ['23:30', 16, 30]]) {
    const instant = Date.UTC(y, d.getUTCMonth(), d.getUTCDate(), h, m);
    const off = instant >= dstStart && instant < dstEnd ? -7 : -8;
    if (off === -8 && time === '22:00') pst++;
    const la = new Date(instant + off * 3_600_000);
    const expected = `${la.toISOString().slice(0, 10)} ${pad(la.getUTCHours())}:${pad(la.getUTCMinutes())}`;
    for (const zone of ['Asia/Ho_Chi_Minh', 'Asia/Saigon']) {
      n += 2;
      if (wallTimeIn(date, time, zone, 'America/Los_Angeles') !== expected) bad++;
      if (instantOfWallTime(date, time, zone) !== new Date(instant).toISOString()) bad++;
    }
  }
}
console.log(`season check 2026-2028: ${n} conversions, ${bad} mismatches; PST days ${pst}`);
