# Usage: bash run-shift.sh <label> <target instant, UTC ISO or "real"> [playwright args...]
# One Playwright run in the clean export (or SHIFT_EXPORT) under a simulated wall clock (the FIX10 shiftclock.mjs, unchanged), own output folder.
. <task>/env.sh
L=$1; TARGET_I=$2; shift 2
mkdir -p $T/tmp/pw-$L $T/tmp/shots-$L
export E2E_SCREENSHOT_DIR="$TW\\tmp\\shots-$L"
if [ "$TARGET_I" = "real" ]; then
  export AB8_SHIFT_MS=0
  unset NODE_OPTIONS
else
  TARGET_MS=$(date -u -d "$TARGET_I" +%s%3N)
  NOW_MS=$(date -u +%s%3N)
  export AB8_SHIFT_MS=$((TARGET_MS - NOW_MS))
  export NODE_OPTIONS="--import file:///$(cygpath -m $T)/shiftclock.mjs"
fi
echo "== $L: real start $(date -u +%Y-%m-%dT%H:%M:%SZ), simulated start $TARGET_I, AB8_SHIFT_MS=$AB8_SHIFT_MS"
node.exe $T/probe-now.mjs
cd "${SHIFT_EXPORT:-$T/export}" || exit 2
echo "\$ node node_modules/@playwright/test/cli.js test $* --output=<task>/tmp/pw-$L"
node.exe node_modules/@playwright/test/cli.js test "$@" --output="$TW\\tmp\\pw-$L"
echo "exit=$?"
node.exe $T/probe-now.mjs
echo "== $L real end $(date -u +%Y-%m-%dT%H:%M:%SZ)"
