// Prints the process clock (shifted when shiftclock.mjs is preloaded) and the reporting-zone wall time, as a control line.
const now = new Date();
const la = new Intl.DateTimeFormat('en-CA', { timeZone: 'America/Los_Angeles', dateStyle: 'short', timeStyle: 'medium', hourCycle: 'h23' }).format(now);
console.log(`process clock ${now.toISOString()} (America/Los_Angeles ${la}), AB8_SHIFT_MS=${process.env.AB8_SHIFT_MS ?? 'unset'}`);
