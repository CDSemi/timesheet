// Usage: node counts.mjs <playwright list-reporter log>  Per-project counts of ok, skipped and failed lines, plus the summary lines
// and every failed test line, and the duration of the lines that match an optional pattern (argv[3]).
import { readFileSync } from 'node:fs';

const text = readFileSync(process.argv[2], 'utf8').replace(/\u001b\[[0-9;]*[A-Za-z]/g, '');
const pattern = process.argv[3] === undefined ? null : new RegExp(process.argv[3]);
const counts = {};
const failed = [];
const matched = [];
for (const line of text.split(/\r?\n/)) {
  const m = /^\s+(ok|-|x|✘|✓|°)\s+\d+\s+\[(\w+)\]/.exec(line);
  if (m === null) continue;
  const status = { ok: 'ok', '✓': 'ok', '-': 'skipped', x: 'failed', '✘': 'failed', '°': 'skipped' }[m[1]];
  counts[m[2]] ??= { ok: 0, skipped: 0, failed: 0 };
  counts[m[2]][status] += 1;
  if (status === 'failed') failed.push(line.trim());
  if (pattern !== null && pattern.test(line)) matched.push(line.trim());
}
const summary = text.split(/\r?\n/).filter((line) => /^\s+\d+ (passed|failed|skipped|flaky|did not run|interrupted)/.test(line)).map((line) => line.trim());
console.log(JSON.stringify({ counts, summary, failed, matched }, null, 1));
