import { appendFileSync, readFileSync } from 'node:fs';
import { inflateSync } from 'node:zlib';

export function obs(record: Record<string, unknown>): void {
  const out = process.env.RC_OUT;
  if (out !== undefined && out !== '') appendFileSync(out, `${JSON.stringify(record)}\n`);
}

export type Rgb = [number, number, number];

/** One viewport screenshot decoded without any per-pixel assertion (8-bit RGB or RGBA, not interlaced). */
export interface Shot {
  width: number;
  height: number;
  at(x: number, y: number): Rgb;
}

export async function shot(page: Page): Promise<Shot> {
  await page.mouse.move(0, 0);
  const png = await page.screenshot({ scale: 'css', animations: 'disabled', caret: 'hide' });
  const data: Buffer[] = [];
  let [width, height, colourType, depth, interlace] = [0, 0, 0, 0, 0];
  for (let offset = 8; offset < png.length; ) {
    const length = png.readUInt32BE(offset);
    const type = png.toString('latin1', offset + 4, offset + 8);
    const body = png.subarray(offset + 8, offset + 8 + length);
    if (type === 'IHDR') {
      [width, height, depth, colourType, interlace] = [body.readUInt32BE(0), body.readUInt32BE(4), body.readUInt8(8), body.readUInt8(9), body.readUInt8(12)];
    }
    if (type === 'IDAT') data.push(body);
    offset += 12 + length;
  }
  if (depth !== 8 || interlace !== 0 || (colourType !== 2 && colourType !== 6)) throw new Error(`unexpected PNG ${depth}/${colourType}/${interlace}`);
  const raw = inflateSync(Buffer.concat(data));
  const bytes = colourType === 6 ? 4 : 3;
  const stride = width * bytes;
  const pixels = Buffer.alloc(stride * height);
  for (let row = 0; row < height; row += 1) {
    const filter = raw[row * (stride + 1)] ?? 0;
    for (let index = 0; index < stride; index += 1) {
      const left = index >= bytes ? (pixels[row * stride + index - bytes] ?? 0) : 0;
      const up = row > 0 ? (pixels[(row - 1) * stride + index] ?? 0) : 0;
      const upLeft = row > 0 && index >= bytes ? (pixels[(row - 1) * stride + index - bytes] ?? 0) : 0;
      let predictor = 0;
      if (filter === 1) predictor = left;
      else if (filter === 2) predictor = up;
      else if (filter === 3) predictor = (left + up) >> 1;
      else if (filter === 4) {
        const estimate = left + up - upLeft;
        const [a, b, c] = [Math.abs(estimate - left), Math.abs(estimate - up), Math.abs(estimate - upLeft)];
        predictor = a <= b && a <= c ? left : b <= c ? up : upLeft;
      }
      pixels[row * stride + index] = ((raw[row * (stride + 1) + 1 + index] ?? 0) + predictor) & 255;
    }
  }
  return {
    width,
    height,
    at(x: number, y: number): Rgb {
      const start = y * stride + x * bytes;
      return [pixels[start] ?? 0, pixels[start + 1] ?? 0, pixels[start + 2] ?? 0];
    },
  };
}

export function near(a: Rgb, b: Rgb, tolerance = 3): boolean {
  return Math.abs(a[0] - b[0]) <= tolerance && Math.abs(a[1] - b[1]) <= tolerance && Math.abs(a[2] - b[2]) <= tolerance;
}

export async function cssColour(page: Page, token: string): Promise<Rgb> {
  const text = await page.evaluate((name) => {
    const probe = document.createElement('span');
    probe.style.color = `var(${name})`;
    document.body.append(probe);
    const colour = getComputedStyle(probe).color;
    probe.remove();
    return colour;
  }, token);
  const m = /rgba?\(([\d.]+),\s*([\d.]+),\s*([\d.]+)/.exec(text);
  if (m === null) throw new Error(`colour ${text}`);
  return [Number(m[1]), Number(m[2]), Number(m[3])];
}

function channel(value: number): number {
  const c = value / 255;
  return c <= 0.03928 ? c / 12.92 : ((c + 0.055) / 1.055) ** 2.4;
}

export function contrastOf(a: Rgb, b: Rgb): number {
  const la = 0.2126 * channel(a[0]) + 0.7152 * channel(a[1]) + 0.0722 * channel(a[2]);
  const lb = 0.2126 * channel(b[0]) + 0.7152 * channel(b[1]) + 0.0722 * channel(b[2]);
  return (Math.max(la, lb) + 0.05) / (Math.min(la, lb) + 0.05);
}

/** Waits until the window has the width and two frames have passed, then until the media-driven DOM matches it (own screens). */
export async function settleWidth(page: Page, width: number, ownScreen = true): Promise<void> {
  await expect.poll(() => page.evaluate(() => window.innerWidth)).toBe(width);
  if (ownScreen) await expect(page.locator('.bar-clock')).toHaveCount(width < 768 ? 1 : 0);
  await expect(page.locator('[data-paper] [data-sheet="phone"]')).toHaveCount(width < 480 ? 1 : 0);
  await page.evaluate(() => new Promise((resolve) => requestAnimationFrame(() => requestAnimationFrame(() => resolve(null)))));
}

export async function signInAt(page: Page, person: Person, hash: string): Promise<void> {
  await page.goto(`/${hash}`);
  await expect(page.getByRole('button', { name: 'Sign in' })).toBeVisible();
  await page.getByLabel('Email').fill(person.account.email);
  await page.getByLabel('Password').fill(person.account.password);
  await page.getByRole('button', { name: 'Sign in' }).click();
}

/** Signs the displayed period off over HTTP (a second window), as review.spec's signOffElsewhere does. */
export async function signOffOverHttp(person: Person): Promise<void> {
  const current = await person.api.call<{ expected_version: number; payload_hash: string }>('GET', `/api/timesheets/${person.payrollDate}/review`);
  await person.api.call(
    'POST',
    `/api/timesheets/${person.payrollDate}/signoff`,
    { expected_version: current.expected_version, reviewed_hash: current.payload_hash, incomplete_evidence_acknowledged: true },
    201,
  );
}

let axeSource: string | null = null;

/** Runs axe-core 4.14.0 (loaded from the task folder, AXE_PATH) on the page; returns the violation ids with node counts. */
export async function axeScan(page: Page, label: string): Promise<{ violations: Array<{ id: string; impact: string | null; nodes: number; targets: string[]; rects: Array<string | null>; summaries: string[] }>; incomplete: string[]; scrollY: number; barBottom: number | null }> {
  axeSource ??= readFileSync(process.env.AXE_PATH ?? '', 'utf8');
  const loaded = await page.evaluate(() => typeof (window as unknown as { axe?: unknown }).axe !== 'undefined');
  if (!loaded) await page.evaluate(axeSource);
  const result = await page.evaluate(async () => {
    const axe = (window as unknown as { axe: { run: (context: unknown, options: unknown) => Promise<{ violations: Array<{ id: string; impact: string | null; nodes: Array<{ target: unknown[] }> }>; incomplete: Array<{ id: string }> }> } }).axe;
    const run = await axe.run(document, { runOnly: { type: 'tag', values: ['wcag2a', 'wcag2aa', 'wcag21a', 'wcag21aa', 'wcag22aa'] }, resultTypes: ['violations', 'incomplete'] });
    const bar = document.querySelector('.shell-bar')?.getBoundingClientRect();
    return {
      violations: run.violations.map((v) => ({
        id: v.id,
        impact: v.impact,
        nodes: v.nodes.length,
        targets: v.nodes.slice(0, 4).map((n) => JSON.stringify(n.target)),
        rects: v.nodes.slice(0, 4).map((n) => {
          const el = typeof n.target[0] === 'string' ? document.querySelector(n.target[0]) : null;
          const r = el?.getBoundingClientRect();
          return r === undefined ? null : `${r.left.toFixed(1)},${r.top.toFixed(1)} ${r.width.toFixed(1)}x${r.height.toFixed(1)}`;
        }),
        summaries: v.nodes.slice(0, 2).map((n) => ((n as { failureSummary?: string }).failureSummary ?? '').replace(/\s+/g, ' ').slice(0, 260)),
      })),
      incomplete: run.incomplete.map((v) => v.id),
      scrollY: window.scrollY,
      barBottom: bar === undefined ? null : bar.bottom,
    };
  });
  obs({ kind: 'axe', label, violations: result.violations, incomplete: result.incomplete, scrollY: result.scrollY, barBottom: result.barBottom });
  return result;
}
