const LA = 'America/Los_Angeles';

  return { id: row.id, email: row.email, displayName: row.display_name, role: row.role, calendarId: row.calendar_id, sessionId: 'sample' };
};

const people = {};
async function person(key, displayName, email, to, settingsExtra, authorizeImage) {
  const id = await m.users.createUser(db, early, { email, displayName, role: 'employee', password: randomBytes(18).toString('base64url'), calendarId: seed.calendarId }, admin.id);
  POLICY(id);
  m.settings.saveSubmissionSettings(db, early, id, { expectedSeq: 0, to: [to], cc: ['payroll-copy@example.invalid'], ...settingsExtra });
  if (authorizeImage) m.sigs.saveSignatureAndAuthorizeAutoImage(db, early, files, id, m.seed.syntheticSignaturePng(7), 'image/png');
  else m.sigs.saveSignature(db, early, files, id, m.seed.syntheticSignaturePng(7), 'image/png');
  people[key] = sessionUser(db, id);
}

async function passes(startIso, count = 4) {
  let t = new Date(startIso).getTime();
  for (let i = 0; i < count; i += 1) {
    const clock = fixedClock(new Date(t).toISOString());
    const s = await m.runner.runJobsOnce({ db, clock, owner: 'runner-sample', handlers: m.runner.createJobHandlers({ db, clock, files, delivery }) });
    console.log('pass', new Date(t).toISOString(), JSON.stringify(s));
    t += 120000;
  }
}

function signOff(user, clock) {
  const review = m.review.buildReviewPayload(db, clock, user, PAYROLL);
  const signed = m.fin.signOffTimesheet({ db, clock, user }, PAYROLL, {
    expectedVersion: review.expectedVersion, reviewedHash: review.payloadHash, deficitChoices: [], incompleteEvidenceAcknowledged: true,
  });
  return signed;
}
// The Send card: defaults come from Settings (the send draft); overrides are what the owner changes for this send.
function send(user, clock, revisionId, overrides = {}) {
  const draft = m.del.sendDraft(db, user, revisionId, SENDER);
  const result = m.del.sendRevision({ db, clock, user }, revisionId, {
    requestKey: randomUUID(),
    to: draft.envelope.to,
    cc: draft.envelope.cc,
    subject: draft.envelope.subject,
    bodyText: draft.envelope.bodyText,
    timeRows: draft.rows.time,
    otRows: draft.rows.ot,
    ...overrides,
  });
  return result;
}
