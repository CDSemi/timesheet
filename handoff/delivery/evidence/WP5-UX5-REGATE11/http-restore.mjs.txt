// and the real CLI (backup, isolated restore, outbound resume and release, run-jobs). Capture mode, synthetic example.invalid

//   A. RECHECK-B's sequence for F-UX5B2-1 (sign and send revision 1; backup; isolated restore; resume; correct and send

//   B. The correction's send fails permanently (run-jobs without MAIL_FROM: sender_missing), then the held job is released.
//   C. The correction's send ends uncertain (the capture adapter's own "capture_exists" answer, provoked by creating the
//      attempt's capture folder while the attempt is `preparing`); the owner marks it delivered; the held job is released.
//   D. A second backup and restore while the correction's send waits: both jobs held; release --all.

function startServer(inst, p) {
  const child = spawn(process.execPath, [join(exportDir, 'dist', 'server', 'index.js')], { env: envFor(inst, p), cwd: exportDir, stdio: ['ignore', 'pipe', 'pipe'] });
  let output = '';
  child.stdout.on('data', (chunk) => (output += String(chunk)));
  child.stderr.on('data', (chunk) => (output += String(chunk)));
  const server = { child, port: p, output: () => output, exited: false };
  child.on('exit', () => (server.exited = true));
  servers.push(server);
  return server;
}
async function stopServer(server) {
  if (server.exited) return;
  server.child.kill();
  for (let wait = 0; wait < 50 && !server.exited; wait += 1) await new Promise((resolve) => setTimeout(resolve, 100));
}
async function waitHealthy(p) {
  for (let attempt = 0; attempt < 80; attempt += 1) {
    try {
      const response = await fetch(`http://127.0.0.1:${p}/api/health`, { headers: { connection: 'close' }, signal: AbortSignal.timeout(2000) });
      if (response.ok) return true;
    } catch {
      // not listening yet
    }
    await new Promise((resolve) => setTimeout(resolve, 250));
  }
  return false;
}
async function call(method, path, { cookie, body, p } = {}) {
  const headers = { connection: 'close', origin: `http://localhost:${p}` };
  if (cookie) headers.cookie = cookie;
  if (body !== undefined) headers['content-type'] = 'application/json';
  const response = await fetch(`http://127.0.0.1:${p}${path}`, { method, headers, body: body === undefined ? undefined : JSON.stringify(body) });
  const text = await response.text();
  let json = null;
  try {
    json = JSON.parse(text);
  } catch {
    json = null;
  }
  return { status: response.status, json, text };
}
const code = (res) => `${res.status}${res.json?.error?.code ? `:${res.json.error.code}` : ''}`;
async function login(email, password, p) {
  const res = await fetch(`http://127.0.0.1:${p}/api/auth/login`, {
    method: 'POST',
    headers: { connection: 'close', origin: `http://localhost:${p}`, 'content-type': 'application/json' },
    body: JSON.stringify({ email, password }),
  });
  await res.text();
  return { status: res.status, cookie: (res.headers.get('set-cookie') ?? '').split(';')[0], p };
}
const addDaysIso = (date, days) => new Date(Date.parse(`${date}T00:00:00Z`) + days * 86400000).toISOString().slice(0, 10);
let clockOffset = 0;
const nowArg = () => {
  clockOffset += 150;
  return new Date(Date.now() + clockOffset * 1000).toISOString().replace(/\.\d{3}Z$/, 'Z');
};
const runJobs = (inst, passes = 3, mailFrom = SENDER) => {

async function signOff(user, payroll) {
  const review = await call('GET', `/api/timesheets/${payroll}/review`, { cookie: user.cookie, p: user.p });
  if (review.status !== 200) return { status: review.status, code: code(review), id: null };
  const res = await call('POST', `/api/timesheets/${payroll}/signoff`, {
    cookie: user.cookie,
    p: user.p,
    body: { expected_version: review.json.expected_version, reviewed_hash: review.json.payload_hash, incomplete_evidence_acknowledged: true },
  });
  return { status: res.status, code: code(res), id: res.json?.revision?.id ?? null };
}
async function correct(user, payroll, note) {
  const review = (await call('GET', `/api/timesheets/${payroll}/review`, { cookie: user.cookie, p: user.p })).json;
  const day = review.payload.days[2].work_date;
  const edited = await call('PUT', `/api/days/${day}`, { cookie: user.cookie, p: user.p, body: { category: 'Worked', leave_minutes: 0, wfh: false, notes: note, reason: 'Synthetic correction probe' } });
  const after = (await call('GET', `/api/timesheets/${payroll}/review`, { cookie: user.cookie, p: user.p })).json;
  const res = await call('POST', `/api/timesheets/${payroll}/revisions`, {
    cookie: user.cookie,
    p: user.p,
    body: { expected_version: after.expected_version, reviewed_hash: after.payload_hash, incomplete_evidence_acknowledged: true, reason: 'Synthetic correction probe', send_email: false },
  });
  return { status: res.status, code: `${code(edited)} ${code(res)}`, id: res.json?.revision?.id ?? null };
}
async function send(user, revisionId, extra = {}) {
  const draft = (await call('GET', `/api/revisions/${revisionId}/send-draft`, { cookie: user.cookie, p: user.p })).json;
  const body = { request_key: randomUUID(), to: draft.to, cc: draft.cc, subject: draft.subject, body_text: draft.body_text, time_rows: draft.time_rows, ot_rows: draft.ot_rows, ...extra };
  return call('POST', `/api/revisions/${revisionId}/send`, { cookie: user.cookie, p: user.p, body });
}
async function ensureRecipients(user) {
  const current = await call('GET', '/api/settings/submission', { cookie: user.cookie, p: user.p });
  const settings = current.json?.settings ?? null;
  if ((settings?.recipients?.to ?? []).length > 0) return 'kept';
  const saved = await call('POST', '/api/settings/submission', { cookie: user.cookie, p: user.p, body: { expected_seq: settings?.seq ?? 0, to: ['payroll@example.invalid'], cc: [], auto_submit: false } });
  return code(saved);
}

/** A migrated, seeded source with a server; returns the logged-in owner (employee2) and other owner (employee). */
async function newSource(name) {
  const p = port();
  const src = instance(name, [p]);
  const migrated = cli(src, ['migrate']).status;
  const seeded = cli(src, ['seed']).status;
  const server = startServer(src, p);
  const healthy = await waitHealthy(p);
  const owner = await login('employee2@example.invalid', passwords.employee2, p);
  const other = await login('employee@example.invalid', passwords.employee, p);
  const recipients = `${await ensureRecipients(other)}/${await ensureRecipients(owner)}`;
  const periods = (await call('GET', '/api/periods/current', { cookie: owner.cookie, p })).json;
  log(`${name}: migrate ${migrated}, seed ${seeded}, healthy ${healthy}, logins ${owner.status}/${other.status}, recipients ${recipients}, today ${periods.today_local}, current payroll ${periods.current.payroll_date}`);
  return { src, server, owner, other, payroll: periods.current.payroll_date };
}
/** Backup of `from` (server stopped first), isolated restore into a new folder, resume; returns the restored instance. */
async function backupRestore(from, server, name) {
  const backupDir = join(work, `${name}-backups`);
  mkdirSync(backupDir, { recursive: true });
  const backup = cli(from, ['backup', '--to', backupDir]);
  if (server !== null) await stopServer(server);
  const backupName = readdirSync(backupDir).find((entry) => !entry.startsWith('.'));
  const restoredDir = join(work, name);
  const restore = cli(from, ['restore', '--from', join(backupDir, backupName ?? 'missing'), '--to', restoredDir]);
  const p = port();
  const dst = { dir: restoredDir, databasePath: join(restoredDir, 'timesheet.db'), dataDir: join(restoredDir, 'private-data'), ports: [p] };
  const monitor = new Database(dst.databasePath, { readonly: true });
  const held = monitor.prepare("SELECT state, last_error FROM jobs WHERE kind = 'send_email' AND state = 'intervention' AND last_error = 'reconcile_after_restore'").all().length;
  const resume = cli(dst, ['outbound', 'resume', '--confirm']);
  log(`${name}: backup ${backup.status}, restore ${restore.status} (${masked(restore.output.split('\n')[0] ?? '')}), held send jobs ${held}, resume --confirm ${resume.status} (${masked(resume.output)})`);
  return { dst, monitor, p, held, backup: backup.status, restore: restore.status, resume: resume.status };
}
const jobOf = (monitor, revisionId) => monitor.prepare("SELECT id FROM jobs WHERE revision_id = ? AND kind = 'send_email' ORDER BY rowid LIMIT 1").pluck().get(revisionId);
const jobState = (monitor, jobId) => monitor.prepare('SELECT state, last_error FROM jobs WHERE id = ?').get(jobId);
const ownerAttempts = (monitor, revisionId) =>

    )
    .all(revisionId)
    .map((row) => `r${row.revision_no}:${row.state}${row.decision === null ? '' : `/${row.decision}`}${row.provider_response === 'capture_exists' ? '(capture_exists)' : ''}`);

/** run-jobs while planting the capture folder of the revision's attempt as soon as it is `preparing`. */
async function runJobsPlanting(inst, monitor, revisionId) {
  const child = spawn(process.execPath, [cliPath, 'run-jobs', '--once', '--now', nowArg()], { env: envFor(inst, inst.ports[0]), cwd: exportDir, stdio: ['ignore', 'pipe', 'pipe'] });
  let output = '';
  child.stdout.on('data', (chunk) => (output += String(chunk)));
  child.stderr.on('data', (chunk) => (output += String(chunk)));
  let exited = false;
  child.on('exit', () => (exited = true));
  const latest = monitor.prepare('SELECT id, state FROM delivery_attempts WHERE revision_id = ? ORDER BY rowid DESC LIMIT 1');
  let planted = null;
  while (!exited) {
    const row = latest.get(revisionId);
    if (planted === null && row !== undefined && row.state === 'preparing') {
      mkdirSync(join(inst.dataDir, 'mail-capture', row.id), { recursive: true });
      planted = row.id;
    }
    await new Promise((resolve) => setImmediate(resolve));
  }
  return { planted: planted !== null, output: masked(output.replaceAll(work, '<work>')) };
}

    const pv = preview(R2.dst);
    log(`D preview on the second restore: exit ${pv.status} ${pv.text}`);
    const all = cli(R2.dst, ['outbound', 'release', '--all', '--confirm']);
    log(`D release --all --confirm: exit ${all.status} ${masked(all.output)}`);
    log(`D run-jobs: ${runJobs(R2.dst)}`);
    const seq = ownerAttempts(R2.monitor, r1.id).join(',');
    check('D1 (docs/05:72): with revision 2\'s send held as well, revision 1 stays held (superseded_by_later_send) and is not delivered', blockerOf(pv, J1) === 'superseded_by_later_send' && !seq.includes('r1:'), `blockers r1 ${blockerOf(pv, J1)}, r2 ${blockerOf(pv, J2)}; period attempts in order ${seq}`);
  }
} catch (error) {
  failures += 1;
  log(`FAIL unexpected error: ${error instanceof Error ? `${error.message}\n${error.stack}` : String(error)}`);
} finally {
  for (const monitor of monitors) monitor.close();
  for (const server of servers) await stopServer(server);
  for (const [index, server] of servers.entries()) log(`server ${index} (port ${server.port}) exited ${server.exited}; output head: ${server.output().replaceAll(work, '<work>').slice(0, 160).replaceAll('\n', ' | ')}`);
}
log(`failures: ${failures}`);
writeFileSync(outPath, `${lines.join('\n')}\n`);
process.exitCode = failures === 0 ? 0 : 1;
