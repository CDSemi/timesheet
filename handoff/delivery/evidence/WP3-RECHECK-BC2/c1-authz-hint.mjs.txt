  const res = await p;
  if (res.status !== status) throw new Error(`${label}: ${res.status} ${JSON.stringify(res.body)}`);
  return res;
}

  return res;

r.section('2. Every other /api route is unreachable under /api/shared/:ownerId (full share; grantee, stranger, admin)');
await setItems({ timesheets: 'edit', ot_read: true, pdf_download: true });
const deliveryId = db.prepare('SELECT id FROM delivery_attempts WHERE user_id = ? LIMIT 1').pluck().get(OWNER) ?? '00000000-0000-4000-8000-000000000000';

function fill(path) {
  return path
    .replace(':payrollDate', '2026-10-16')
    .replace(':workDate', '2026-09-28')
    .replace(/^\/sessions\/:id/, `/sessions/${ownerSession}`)
    .replace(/^\/revisions\/:id/, `/revisions/${revO}`)
    .replace(/^\/deliveries\/:id/, `/deliveries/${deliveryId}`)
    .replace(/^\/signatures\/:id/, `/signatures/${sigId}`)
    .replace(/^\/settings\/submission\/versions\/:id/, `/settings/submission/versions/${settingsVersion}`)
    .replace(/^\/admin\/users\/:id/, `/admin/users/${OWNER}`)
    .replace(/^\/admin\/shares\/:id/, `/admin/shares/${shareId}`)
    .replace(/^\/shares\/:id/, `/shares/${shareId}`)
    .replace(':id', 'x');
}
const others = [...inv.keys()].filter((k) => !k.startsWith('ALL ') && !k.includes(' /api/shared/')).map((k) => k.split(' ')).filter(([, p]) => p.startsWith('/api/'));
let unreachable = 0;
const reached = [];
for (const [method, path] of others) {
  const rel = fill(path.slice('/api'.length));
  if (MATRIX.some(([m, p]) => m === method && fill(p) === rel)) continue;
  for (const cookie of [G, S, A]) {
    const before = changes(db);
    const res = await ctx.request(method, shared(rel), method === 'GET' || method === 'HEAD' ? { cookie } : { cookie, body: {} });
    if (res.status === 404 && changes(db) === before && !JSON.stringify(res.body ?? '').includes('grantee_changes')) unreachable++;
    else reached.push(`${method} ${rel} -> ${res.status}`);
  }
}
r.info(`probed ${others.length} non-shared routes x 3 callers (minus allowlisted method/path pairs)`);
check(reached.length === 0, `every non-allowlisted route under /api/shared is 404 with zero writes (${unreachable} probes)${reached.length ? ` -- ${reached.join('; ')}` : ''}`);

r.section('3. The 11 valid item sets x 17 routes, real requests (grantee), plus stranger/admin/anonymous');
const ITEM_SETS = [];
for (const timesheets of ['none', 'view', 'edit']) for (const ot of [false, true]) for (const pdf of [false, true]) if (timesheets !== 'none' || ot || pdf) ITEM_SETS.push({ timesheets, ot_read: ot, pdf_download: pdf });
check(ITEM_SETS.length === 11, '11 valid item sets');
const allows = (set, item) =>
  item === 'view' ? set.timesheets !== 'none' : item === 'edit' ? set.timesheets === 'edit' : item === 'ot' ? set.ot_read : item === 'pdf' ? set.pdf_download : set.timesheets !== 'none' || set.pdf_download;
const entryVersion = (date) => db.prepare('SELECT version FROM day_entries WHERE user_id = ? AND work_date = ?').pluck().get(OWNER, date) ?? null;
const lastAudit = () => db.prepare('SELECT operation, actor_user_id, owner_user_id FROM audit_events ORDER BY rowid DESC LIMIT 1').get();
let n = 0;
let sharedWrites = 0;
let sharedDownloads = 0;
const responses = [];
for (const set of ITEM_SETS) {
  await setItems(set);
  const label = `ts=${set.timesheets} ot=${set.ot_read ? 1 : 0} pdf=${set.pdf_download ? 1 : 0}`;
  const results = [];
  let ok = true;
  for (const [method, path, item] of MATRIX.filter(([m]) => m === 'GET')) {
    const url = path === '/periods' ? shared('/periods?from=2026-09-14&to=2026-10-11') : shared(fill(path));
    const beforeC = changes(db);
    const beforeA = auditCount(db);
    const res = await ctx.request('GET', url, { cookie: G });
    responses.push(res);
    const allowed = allows(set, item);
    let good = allowed ? res.status === 200 : res.status === 403 && res.body?.error?.code === 'grant_scope';
    if (item === 'pdf' && allowed) {
      sharedDownloads += 1;
      const audited = auditCount(db) === beforeA + 1 && lastAudit().operation === 'share.pdf_download' && lastAudit().actor_user_id === ctx.ids.grantee && lastAudit().owner_user_id === OWNER;
      good = good && audited && Buffer.compare(res.bytes, ownerPdfBytes) === 0 && res.headers.get('cache-control') === 'no-store';
    } else good = good && changes(db) === beforeC;
    good = good && res.headers.get('cache-control') === 'no-store';
    if (!good) ok = false;
    results.push(`${method} ${path}=${res.status}`);
  }
  const day = n++ % 2 === 0 ? 'Off' : 'Worked';
  const writeProbes = [
    ['PUT', '/days/:workDate', () => ['/days/2026-09-30', { category: day, leave_minutes: 0, wfh: false, notes: `grantee note ${n}`, expected_version: entryVersion('2026-09-30') }]],
    ['POST', '/days/batch', () => ['/days/batch', { mode: 'commit', entries: [{ work_date: '2026-10-01', category: day, expected_version: entryVersion('2026-10-01') }] }]],
    ['POST', '/days/:workDate/sessions', () => ['/days/2026-09-28/sessions', sessionBody('2026-09-28', '06:00', '07:00')]],
  ];
  let created = null;
  for (const [method, path, make] of writeProbes) {
    const [rel, body] = make();
    const beforeC = changes(db);
    const res = await ctx.request(method, shared(rel), { cookie: G, body });
    responses.push(res);
    let good;
    if (allows(set, 'edit')) {
      const a = lastAudit();
      good = res.status < 300 && a.actor_user_id === ctx.ids.grantee && a.owner_user_id === OWNER;
      sharedWrites += 1;
      if (path.endsWith('/sessions')) created = res.body.session.id;
    } else good = res.status === 403 && res.body?.error?.code === 'grant_scope' && changes(db) === beforeC;
    if (!good) ok = false;
    results.push(`${method} ${path}=${res.status}`);
  }
  for (const [method, body] of [['PUT', (v) => ({ ...sessionBody('2026-09-28', '06:00', '07:30'), expected_version: v })], ['DELETE', (v) => ({ expected_version: v })]]) {
    const target = created ?? ownerSession;
    const v = db.prepare('SELECT version FROM work_sessions WHERE id = ?').pluck().get(target);
    const beforeC = changes(db);
    const res = await ctx.request(method, shared(`/sessions/${target}`), { cookie: G, body: body(v) });
    responses.push(res);
    let good;
    if (allows(set, 'edit')) {
      const a = lastAudit();
      good = res.status === 200 && a.actor_user_id === ctx.ids.grantee && a.owner_user_id === OWNER;
      sharedWrites += 1;
    } else good = res.status === 403 && res.body?.error?.code === 'grant_scope' && changes(db) === beforeC;
    if (!good) ok = false;
    results.push(`${method} /sessions/:id=${res.status}`);
  }
  let others404 = true;
  for (const [method, path] of MATRIX) {
    const rel = path === '/periods' ? '/periods?from=2026-09-14&to=2026-10-11' : fill(path);
    for (const [cookie, want] of [[S, 404], [A, 404], [undefined, 401]]) {
      const beforeC = changes(db);
      const res = await ctx.request(method, shared(rel), method === 'GET' ? { cookie } : { cookie, body: {} });
      if (res.status !== want || changes(db) !== beforeC || res.headers.get('cache-control') !== 'no-store') others404 = false;
    }
  }
  check(ok, `item set ${label}: ${results.join(' ')}`);
  check(others404, `item set ${label}: stranger 404, admin-without-share 404, anonymous 401 on all 17, zero writes, no-store`);
}

// Oracle (independent of sharedActs.ts): per day of the period, the actor of the last day/session audit event.
function oracle(ownerId, start, end, sinceRowid = 0) {
  const rows = db
    .prepare("SELECT rowid, actor_user_id, operation, COALESCE(json_extract(after_json,'$.work_date'), json_extract(before_json,'$.work_date')) AS d FROM audit_events WHERE owner_user_id = ? AND entity_type IN ('day_entry','work_session') AND rowid > ? ORDER BY rowid")
    .all(ownerId, sinceRowid);
  const last = new Map();
  for (const row of rows) if (row.d >= start && row.d <= end) last.set(row.d, row.actor_user_id !== ownerId && row.actor_user_id !== null ? row.actor_user_id : null);
  const by = {};
  for (const [d, who] of last) if (who) (by[who] ??= []).push(d);
  for (const k of Object.keys(by)) by[k].sort();
  return by;
}
