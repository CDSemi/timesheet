// WP3-RECHECK-BC2 probe U2: the built server (dist/) on an OS-assigned loopback port (JOB_RUNNER=off, capture
// outbound), synthetic CLI seed, installed Edge channel, desktop and mobile. Screens of fix round 2:
// - the owner's History after an automatic submission: the actor-less OT credit reads "automatic" (WP3-RBC-01);
// - the owner's Review of the next period: grantee changes made before that period started are counted (WP3-RBC-02);
// - Settings of an account that never saved them: the rule text makes no pre-setup automation claim (H-Q1 (a));
// - the administrator's operations status: no automatic row or fault for never-configured accounts (H-Q1 (a)).
// Checks: wording, horizontal overflow, unnamed controls. Adapted from the published WP3-RECHECK-BC probe U1.
// Every child uses process.execPath. Usage: node u2-ui.mjs <repoDir> <workDir> <shotDir>
import { spawn, spawnSync } from 'node:child_process';
import { randomBytes } from 'node:crypto';
import { mkdirSync } from 'node:fs';
import { createServer } from 'node:net';
import { join } from 'node:path';
import { pathToFileURL } from 'node:url';

const [repoDir, workDir, shotDir] = process.argv.slice(2);
const port = await new Promise((resolve, reject) => {
  const probe = createServer();
  probe.once('error', reject);
  probe.listen(0, '127.0.0.1', () => {
    const address = probe.address();
    probe.close(() => resolve(address.port));
  });
});
const ORIGIN = `http://127.0.0.1:${port}`;
mkdirSync(shotDir, { recursive: true });
mkdirSync(workDir, { recursive: true });
const mail = (local) => `${local}${String.fromCharCode(64)}example.invalid`;
let failures = 0;
let passes = 0;
const check = (ok, label) => {
  if (ok) passes += 1;
  else failures += 1;
  console.log(`${ok ? 'PASS' : 'FAIL'} ${label}`);
};
const credentials = {
  admin: { email: mail('admin'), password: randomBytes(18).toString('base64url') },
  employee: { email: mail('employee'), password: randomBytes(18).toString('base64url') },
  employee2: { email: mail('employee2'), password: randomBytes(18).toString('base64url') },
};
const env = {
  ...process.env,
  PORT: String(port),
  DATABASE_PATH: join(workDir, 'u2.db'),
  DATA_DIR: join(workDir, 'private-data'),
  MAIL_FROM: mail('sender'),
  JOB_RUNNER: 'off',
  SEED_ADMIN_PASSWORD: credentials.admin.password,
  SEED_EMPLOYEE_PASSWORD: credentials.employee.password,
  SEED_EMPLOYEE2_PASSWORD: credentials.employee2.password,
};
delete env.PRODUCTION_SENDING_ENABLED;
delete env.OUTBOUND_MODE;
console.log(`INFO server port chosen by the OS: ${port > 0 ? 'yes' : 'no'}`);
const cli = (...args) => spawnSync(process.execPath, [join(repoDir, 'dist/server/cli.js'), ...args], { cwd: repoDir, env, encoding: 'utf8' });
const seeded = cli('seed');
check(seeded.status === 0, `seed exit ${seeded.status}`);

const server = spawn(process.execPath, [join(repoDir, 'dist/server/index.js')], { cwd: repoDir, env, stdio: ['ignore', 'pipe', 'pipe'] });
let serverOut = '';
server.stdout.on('data', (d) => (serverOut += d));
server.stderr.on('data', (d) => (serverOut += d));
const stopServer = async () => {
  if (server.exitCode === null) {
    const exited = new Promise((resolve) => server.once('exit', resolve));
    server.kill();
    await exited;
  }
};
let browser;
try {
  for (let i = 0; i < 100; i += 1) {
    try {
      if ((await fetch(`${ORIGIN}/api/health`)).ok) break;
    } catch {
      // not yet listening
    }
    await new Promise((resolve) => setTimeout(resolve, 100));
  }
  const api = async (cookie, method, path, body) => {
    const headers = { origin: ORIGIN };
    if (cookie) headers.cookie = cookie;
    if (body !== undefined) headers['content-type'] = 'application/json';
    const res = await fetch(`${ORIGIN}${path}`, { method, headers, body: body === undefined ? undefined : JSON.stringify(body) });
    const text = await res.text();
    return { status: res.status, body: text ? JSON.parse(text) : null, cookie: res.headers.get('set-cookie')?.split(';')[0] };
  };
  const loginApi = async (who) => (await api(undefined, 'POST', '/api/auth/login', credentials[who])).cookie;
  const adminCookie = await loginApi('admin');
  const ownerCookie = await loginApi('employee');
  const granteeCookie = await loginApi('employee2');
  const me = await api(ownerCookie, 'GET', '/api/auth/me');
  const ownerId = me.body.user.id;
  const today = (await api(ownerCookie, 'GET', '/api/periods/current')).body;
  console.log(`INFO server today: ${JSON.stringify({ today: today.today, payroll: today.period?.payroll_date ?? today.payroll_date })}`);
  const share = await api(ownerCookie, 'POST', '/api/shares', { grantee_email: credentials.employee2.email, items: { timesheets: 'edit', ot_read: true, pdf_download: true } });
  check(share.status === 201, `owner shares (edit) with employee2 (${share.status})`);
  // The next period (payroll 2026-10-30) runs 2026-10-12..10-25: both grantee changes are made before it starts.
  for (const [day, category] of [['2026-10-13', 'Sick'], ['2026-10-20', 'Vacation']]) {
    const e = await api(granteeCookie, 'PUT', `/api/shared/${ownerId}/days/${day}`, { category, leave_minutes: 0, wfh: false, notes: 'planned by the grantee' });
    check(e.status === 200, `grantee records ${category} on the owner's ${day}, before that period starts (${e.status})`);
  }
  const ot = await api(ownerCookie, 'POST', '/api/days/2026-10-03/sessions', { start: '2026-10-03T16:00:00Z', end: '2026-10-03T19:00:00Z', input_zone: 'America/Los_Angeles', breaks: [], breaks_confirmed: true });
  check(ot.status === 201, `owner Saturday OT session (${ot.status})`);
  const saved = await api(ownerCookie, 'POST', '/api/settings/submission', { expected_seq: 0, to: [mail('payroll')], auto_submit: true });
  check(saved.status === 201, `owner settings, auto-submit on (${saved.status})`);
  const adminSettings = await api(adminCookie, 'GET', '/api/settings/submission');
  check(adminSettings.body.settings.is_default === true, 'the administrator account never saved submission settings');

  const activeFrom = new Date(Math.ceil((Date.now() + 60_000) / 60_000) * 60_000).toISOString().replace('.000Z', 'Z');
  const act = await api(adminCookie, 'PUT', '/api/admin/automation/activation', { active_from: activeFrom, reason: 'synthetic UI recheck' });
  check(act.status === 200, `activation recorded (${act.status})`);
  for (const now of ['2026-10-14T00:00:30Z', '2026-10-14T00:02:30Z', '2026-10-14T00:04:30Z']) {
    const res = cli('run-jobs', '--once', '--now', now);
    console.log(`INFO run-jobs --now ${now}: exit ${res.status} ${res.stdout.trim()}`);
  }

  const pw = await import(pathToFileURL(join(repoDir, 'node_modules/playwright-core/index.mjs')).href);
  browser = await pw.chromium.launch({ channel: 'msedge', headless: true });
  const viewports = {
    desktop: { viewport: { width: 1280, height: 800 } },
    mobile: { viewport: { width: 390, height: 844 }, isMobile: true, hasTouch: true, deviceScaleFactor: 2 },
  };
  const login = async (page, who) => {
    await page.getByLabel('Email').fill(credentials[who].email);
    await page.getByLabel('Password').fill(credentials[who].password);
    await page.getByRole('button', { name: 'Sign in' }).click();
  };
  const unnamed = async (page) => {
    const yaml = await page.locator('body').ariaSnapshot();
    return yaml.split('\n').filter((l) => /^\s*- (button|link|textbox|checkbox|radio|combobox|switch|spinbutton)\s*(\[[^\]]*\])?\s*:?\s*$/.test(l));
  };
  const shoot = async (page, name, kind, fullPage = false) => {
    await page.waitForLoadState('networkidle');
    const file = `recheck-bc2-${name}-${kind}-synthetic.png`;
    await page.screenshot({ path: join(shotDir, file), fullPage });
    const overflow = await page.evaluate(() => document.documentElement.scrollWidth - document.documentElement.clientWidth);
    const missing = await unnamed(page);
    check(overflow <= 0 && missing.length === 0, `${file}: horizontal overflow ${overflow}px, unnamed controls ${missing.length}`);
  };

  for (const [kind, options] of Object.entries(viewports)) {
    // 1. History after the automatic submission (WP3-RBC-01).
    let context = await browser.newContext({ ...options, timezoneId: 'America/Los_Angeles' });
    let page = await context.newPage();
    await page.goto(`${ORIGIN}/#/history`);
    await login(page, 'employee');
    await page.waitForLoadState('networkidle');
    await page.getByText('Submitted automatically').first().waitFor();
    const items = await page
      .locator('[data-history-actor]')
      .evaluateAll((els) => els.map((el) => ({ kind: el.getAttribute('data-history-actor'), badge: el.textContent, label: el.parentElement?.querySelector('strong')?.textContent ?? '' })));
    console.log(`INFO ${kind}: history badges ${JSON.stringify(items)}`);
    check(items.some((i) => i.label === 'OT credit posted' && i.kind === 'system' && i.badge === 'automatic'), `${kind}: the automatic OT credit reads "automatic" (data-history-actor="system")`);
    check(items.some((i) => i.label === 'Submitted automatically' && i.kind === 'system' && i.badge === 'automatic'), `${kind}: "Submitted automatically" reads "automatic"`);
    check(!items.some((i) => i.kind === 'other' && /automatic|OT credit/.test(i.label)), `${kind}: no automatic event reads "by someone else"`);
    await page.getByText('OT credit posted', { exact: true }).first().scrollIntoViewIfNeeded();
    await shoot(page, 'history-ot-credit', kind);

    // 2. Review of the next period: both grantee changes were made before it started (WP3-RBC-02).
    await page.goto(`${ORIGIN}/#/review/2026-10-30`);
    await page.waitForLoadState('networkidle');
    const region = page.getByRole('region', { name: 'Changed by someone you share with' });
    const visible = await region
      .waitFor({ timeout: 7000 })
      .then(() => true)
      .catch(() => false);
    const text = visible ? await region.innerText() : (await page.locator('body').innerText()).slice(0, 300);
    check(visible && /2 days last changed by Example Employee Two/.test(text), `${kind}: the Review of 2026-10-30 counts both pre-period grantee changes ("${text.split('\n').slice(0, 2).join(' / ')}")`);
    if (visible) await region.scrollIntoViewIfNeeded();
    await shoot(page, 'review-pre-period-hint', kind);
    await context.close();

    // 3. Settings of an account that never saved them (H-Q1 (a)).
    context = await browser.newContext({ ...options, timezoneId: 'America/Los_Angeles' });
    page = await context.newPage();
    await page.goto(`${ORIGIN}/#/settings`);
    await login(page, 'admin');
    await page.waitForLoadState('networkidle');
    const rule = page.locator('[data-setting-rule="auto-submit"]');
    await rule.waitFor();
    const ruleText = await rule.innerText();
    check(/Nothing is submitted for you until you save these settings with automatic submission on/.test(ruleText) && !/on by default/i.test(ruleText), `${kind}: unsaved Settings rule: "${ruleText}"`);
    await rule.scrollIntoViewIfNeeded();
    await shoot(page, 'settings-unsaved-rule', kind);

    // 4. Operations status (administrator).
    await page.goto(`${ORIGIN}/#/admin`);
    await page.waitForLoadState('networkidle');
    const ops = page.getByRole('region', { name: 'Operations status' });
    await ops.waitFor();
    await page.locator('[data-fact="activation"]').waitFor();
    const activation = await page.locator('[data-fact="activation"]').innerText();
    const intervention = await page.locator('[data-job-state="intervention"]').innerText();
    const rows = await page.locator('tr[data-submission-user]').evaluateAll((els) => els.map((el) => ({ user: el.getAttribute('data-submission-user'), period: el.getAttribute('data-submission-period'), text: el.textContent })));
    console.log(`INFO ${kind}: operations activation "${activation}", intervention ${intervention}, submission rows ${JSON.stringify(rows.map((row) => [row.user === ownerId ? 'employee' : 'other', row.period]))}`);
    check(intervention.trim() === '0' && rows.length === 1 && rows[0].user === ownerId && /Automatic submission/.test(rows[0].text), `${kind}: only the configured employee's automatic submission is listed; no fault for the never-configured accounts (intervention ${intervention.trim()}, rows ${rows.length})`);
    await ops.scrollIntoViewIfNeeded();
    await shoot(page, 'admin-operations', kind);
    await context.close();
  }
} finally {
  await browser?.close();
  await stopServer();
}
console.log(`INFO server output lines: ${serverOut.split('\n').filter(Boolean).length}; deprecation lines: ${serverOut.split('\n').filter((l) => /deprecat/i.test(l)).length}`);
console.log(`SUMMARY U2: ${passes} PASS, ${failures} FAIL`);
process.exit(failures === 0 ? 0 : 1);
