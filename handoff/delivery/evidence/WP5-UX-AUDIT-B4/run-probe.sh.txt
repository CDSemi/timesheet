#!/usr/bin/env bash
# WP5-UX-AUDIT-B4 probes on a built clone (dist/ present), own servers on PORT_BASE (desktop) and PORT_BASE+1 (mobile).
# Usage: run-probe.sh <clone dir> <tag> <port base> [grep] [project]
source <task>/env.sh
C="$1"; TAGV="$2"; BASE="$3"; GREP="${4:-.}"; PROJECT="${5:-}"
cd "$C" || exit 9
export PROBE_OUT="$T/probe-out"
export PROBE_WORK="$T/probe-work"
export PROBE_SHOTS="$T/probe-shots"
export PROBE_TAG="$TAGV"
export PROBE_PORT_BASE="$BASE"
mkdir -p "$PROBE_OUT" "$PROBE_WORK" "$PROBE_SHOTS"
STAMP="$(date -u +%H%M%S)"
LOG="probe-run-$TAGV-$STAMP.txt"
if [ -n "$PROJECT" ]; then
  "$N24" node_modules/@playwright/test/cli.js test -c audit-probe/probe.config.ts -g "$GREP" --project="$PROJECT" > "$T/$LOG" 2>&1
else
  "$N24" node_modules/@playwright/test/cli.js test -c audit-probe/probe.config.ts -g "$GREP" > "$T/$LOG" 2>&1
fi
echo "probe exit=$? tag=$TAGV grep=$GREP project=${PROJECT:-both} log=$LOG at $(date -u +%Y-%m-%dT%H:%M:%SZ)" >> "$T/probe-summary.txt"
