/*
 * WP5-UX-AUDIT-B4 independent probes (auditor-written; not part of the reviewed source; kept out of git
 * through .git/info/exclude of the scratch clone). Synthetic data only. Each project starts its own built
 * server (dist/) on a fixed port of the task range (desktop PROBE_PORT_BASE, mobile PROBE_PORT_BASE + 1)
 * with DATA_DIR and DATABASE_PATH inside the task folder, and stops it through the child handle. Each probe
 * records what it measured as JSON in PROBE_OUT; hard expectations only guard the measurement itself.
 */

const REPO = resolve(import.meta.dirname, '..');
const OUT = process.env.PROBE_OUT ?? join(process.cwd(), 'probe-out');
const WORK = process.env.PROBE_WORK ?? join(process.cwd(), 'probe-work');
const SHOTS = process.env.PROBE_SHOTS ?? OUT;
const PORT_BASE = Number(process.env.PROBE_PORT_BASE ?? '48130');
const TAG = process.env.PROBE_TAG ?? 'edaaa85';
mkdirSync(OUT, { recursive: true });
mkdirSync(WORK, { recursive: true });
mkdirSync(SHOTS, { recursive: true });

function record(name: string, data: unknown) {
  writeFileSync(join(OUT, `${TAG}-${name}.json`), `${JSON.stringify(data, null, 2)}\n`);
}
function shot(name: string): string {
  if (!name.endsWith('-synthetic.png')) throw new Error('screenshot names end in -synthetic.png');
  return join(SHOTS, `${TAG}-${name}`);
}

interface Credentials {
  email: string;
  password: string;
}
interface ProbeServer {
  origin: string;
  employee: Credentials;
  admin: Credentials;
}

async function waitHealthy(origin: string, child: ChildProcess) {
  for (let attempt = 0; attempt < 150; attempt += 1) {
    if (child.exitCode !== null) throw new Error(`server exited early with ${child.exitCode}`);
    try {
      if ((await fetch(`${origin}/api/health`)).ok) return;
    } catch {
      // not listening yet
    }
    await new Promise((done) => setTimeout(done, 100));
  }
  throw new Error('server did not become healthy');
}

/** Open dialogs (name, modal, focus inside, on top at its own centre) and where focus is. */
function dialogState(page: Page) {
  return page.evaluate(() => {
    const nameOf = (dialog: Element) => {
      const id = dialog.getAttribute('aria-labelledby');
      const heading = id === null ? null : document.getElementById(id);
      return (heading?.textContent ?? '').replace(/\s+/g, ' ').trim().slice(0, 60);
    };
    const dialogs = [...document.querySelectorAll('dialog[open]')].map((dialog) => {
      const box = dialog.getBoundingClientRect();
      const hit = document.elementFromPoint(box.left + box.width / 2, box.top + Math.min(box.height / 2, 80));
      return {
        name: nameOf(dialog),
        modal: dialog.matches(':modal'),
        focusInside: dialog.contains(document.activeElement),
        onTop: hit !== null && dialog.contains(hit),
        left: Math.round(box.left),
        width: Math.round(box.width),
      };
    });
    const active = document.activeElement;
    const activeText =
      active === null
        ? 'null'
        : active === document.body
          ? 'BODY'
          : `${active.tagName} ${(active.getAttribute('aria-label') ?? (active as HTMLElement).innerText ?? '').trim().replace(/\s+/g, ' ').slice(0, 50)}`;
    const activeDialog = active?.closest('dialog');
    return {
      width: window.innerWidth,
      withEditorClass: document.querySelector('.sheet-area')?.classList.contains('with-editor') ?? null,
      dialogs,
      active: activeText,
      activeIsBody: active === null || active === document.body,
      // False while focus is in the browser's own UI (a Tab past the last control of a modal dialog).
      documentHasFocus: document.hasFocus(),
      activeInDialog: activeDialog === null || activeDialog === undefined ? null : nameOf(activeDialog),
    };
  });
}

/** Waits until React has rendered the new width (the sheet-area class) and its effects ran. */
async function resizeTo(page: Page, width: number) {
  await page.setViewportSize({ width, height: 800 });
  const area = page.locator('.sheet-area');
  if (width >= 1200) await expect(area).toHaveClass(/with-editor/);
  else await expect(area).not.toHaveClass(/with-editor/);
  await page.evaluate(() => new Promise((done) => requestAnimationFrame(() => requestAnimationFrame(() => setTimeout(done, 200)))));
}

async function settle(page: Page, ms = 300) {
  await page.evaluate((wait) => new Promise((done) => requestAnimationFrame(() => setTimeout(done, wait))), ms);
}

function writesOf(page: Page) {
  const writes: string[] = [];
  page.on('request', (request) => {
    if (request.method() !== 'GET') {
      const path = new URL(request.url()).pathname;
      let mode = '';
      if (path === '/api/days/batch') {
        try {
          mode = ` mode=${String((request.postDataJSON() as { mode?: string }).mode)}`;
        } catch {
          mode = '';
        }
      }
      writes.push(`${request.method()} ${path}${mode}`);
    }
  });
  return writes;
}

/* ---- Q6: focus indicator contrast (R-1) and text contrast ---------------------------------------- */

test.describe('Q6 contrast', () => {
  test.use({ locale: 'en-US', timezoneId: 'America/Los_Angeles' });
  test('Q6 focus ring per control type and text contrast, light and dark', async ({ page, seed, signIn }, testInfo) => {
    test.setTimeout(120_000);
    await signIn();
    await expect(page.locator('[data-day]')).toHaveCount(14);
    const result: Record<string, unknown> = {};
    for (const scheme of ['light', 'dark'] as const) {
      await page.emulateMedia({ colorScheme: scheme });
      await settle(page, 500);
      result[`text-${scheme}`] = await page.evaluate(() => {
        const parse = (value: string) => {
          const m = /rgba?\(([\d.]+)[, ]+([\d.]+)[, ]+([\d.]+)(?:[,/ ]+([\d.]+))?\)/.exec(value);
          return m === null ? null : { r: +m[1]!, g: +m[2]!, b: +m[3]!, a: m[4] === undefined ? 1 : +m[4] };
        };
        const lum = (c: { r: number; g: number; b: number }) => {
          const f = (v: number) => {
            const s = v / 255;
            return s <= 0.04045 ? s / 12.92 : ((s + 0.055) / 1.055) ** 2.4;
          };
          return 0.2126 * f(c.r) + 0.7152 * f(c.g) + 0.0722 * f(c.b);
        };
        const background = (element: Element | null) => {
          for (let node = element; node !== null; node = node.parentElement) {
            const bg = parse(getComputedStyle(node).backgroundColor);
            if (bg !== null && bg.a >= 0.99) return bg;
          }
          return { r: 255, g: 255, b: 255, a: 1 };
        };
        const rows: Array<{ text: string; ratio: number }> = [];
        const walker = document.createTreeWalker(document.querySelector('main') ?? document.body, NodeFilter.SHOW_TEXT);
        for (let node = walker.nextNode(); node !== null; node = walker.nextNode()) {
          const text = (node.textContent ?? '').trim();
          const parent = node.parentElement;
          if (text === '' || parent === null) continue;
          const style = getComputedStyle(parent);
          const box = parent.getBoundingClientRect();
          if (box.width <= 2 || box.height <= 2 || style.visibility === 'hidden' || style.clipPath !== 'none') continue;
          if (parent.closest('button:disabled, [aria-disabled="true"]') !== null) continue;
          const fg = parse(style.color);
          if (fg === null) continue;
          const bg = background(parent);
          const blended = { r: fg.r * fg.a + bg.r * (1 - fg.a), g: fg.g * fg.a + bg.g * (1 - fg.a), b: fg.b * fg.a + bg.b * (1 - fg.a) };
          const [l1, l2] = [lum(blended), lum(bg)].sort((x, y) => y - x);
          rows.push({ text: text.slice(0, 40), ratio: +((l1! + 0.05) / (l2! + 0.05)).toFixed(2) });
        }
        rows.sort((x, y) => x.ratio - y.ratio);
        return { count: rows.length, min: rows[0]?.ratio ?? null, below45: rows.filter((row) => row.ratio < 4.5).length, lowest: rows.slice(0, 5) };
      });
      // Keyboard modality, then each control type focused; the ring colour blended over the colour around it.
      await page.keyboard.press('Tab');
      const targets =
        testInfo.project.name === 'mobile'
          ? ['.shell-tabs a', '.period-bar button', '.clock-panel button', '.tools button.secondary', '.tools .open-day input', '[data-day] button']
          : ['.shell-bar nav a', '.period-bar button', '.period-bar [data-review-link]', '.clock-panel button', '.tools button.secondary', '.tools .open-day input', '[data-day] .sheet-date', '[data-day] .label-trigger'];
      const rings: unknown[] = [];
      for (const selector of targets) {
        const target = page.locator(selector).first();
        if ((await target.count()) === 0 || !(await target.isVisible())) {
          rings.push({ selector, missing: true });
          continue;
        }
        await target.focus();
        await settle(page, 400);
        rings.push(
          await target.evaluate((element, sel) => {
            const parse = (value: string) => {
              const m = /rgba?\(([\d.]+)[, ]+([\d.]+)[, ]+([\d.]+)(?:[,/ ]+([\d.]+))?\)/.exec(value);
              return m === null ? null : { r: +m[1]!, g: +m[2]!, b: +m[3]!, a: m[4] === undefined ? 1 : +m[4] };
            };
            const lum = (c: { r: number; g: number; b: number }) => {
              const f = (v: number) => {
                const s = v / 255;
                return s <= 0.04045 ? s / 12.92 : ((s + 0.055) / 1.055) ** 2.4;
              };
              return 0.2126 * f(c.r) + 0.7152 * f(c.g) + 0.0722 * f(c.b);
            };
            const ratio = (a: { r: number; g: number; b: number }, b: { r: number; g: number; b: number }) => {
              const [l1, l2] = [lum(a), lum(b)].sort((x, y) => y - x);
              return +((l1! + 0.05) / (l2! + 0.05)).toFixed(2);
            };
            const background = (start: Element | null) => {
              for (let node = start; node !== null; node = node.parentElement) {
                const bg = parse(getComputedStyle(node).backgroundColor);
                if (bg !== null && bg.a >= 0.99) return bg;
              }
              return { r: 255, g: 255, b: 255, a: 1 };
            };
            const style = getComputedStyle(element);
            const inset = style.boxShadow.includes('inset');
            const ring = parse(style.boxShadow);
            const around = inset ? background(element) : background(element.parentElement);
            const blended = ring === null ? null : { r: ring.r * ring.a + around.r * (1 - ring.a), g: ring.g * ring.a + around.g * (1 - ring.a), b: ring.b * ring.a + around.b * (1 - ring.a) };
            const own = parse(style.backgroundColor);
            return {
              selector: sel,
              name: (element.getAttribute('aria-label') ?? (element as HTMLElement).innerText ?? '').trim().slice(0, 30),
              focusVisible: element.matches(':focus-visible'),
              boxShadow: style.boxShadow,
              outline: `${style.outlineStyle} ${style.outlineWidth} ${style.outlineColor}`,
              inset,
              ringVsAdjacent: blended === null ? null : ratio(blended, around),
              ringVsOwnFill: blended === null || own === null || own.a < 0.99 ? null : ratio(blended, own),
            };
          }, selector),
        );
      }
      result[`focus-${scheme}`] = rings;
      if (scheme === 'light' && testInfo.project.name === 'desktop') {
        await page.locator('.clock-panel button').first().focus();
        await settle(page, 400);
        await page.locator('.clock-panel').screenshot({ path: shot('q6-focus-ring-clock-light-synthetic.png'), animations: 'disabled' });
        await page.locator('[data-day] .sheet-date').first().focus();
        await settle(page, 400);
        await page.locator('[data-day]').first().screenshot({ path: shot('q6-focus-ring-sheet-date-light-synthetic.png'), animations: 'disabled' });
      }
    }
    await page.emulateMedia({ colorScheme: 'light' });
    record(`Q6-contrast-${testInfo.project.name}`, result);
  });
});
