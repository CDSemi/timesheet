// WP4-FIXB probe P6 (adapted from WP4-AUDIT-B P6): event-loop blocking of the BUILT server by one small workbook
// upload that was inside the old reader limits. A second client sends GET /api/health 300 ms after the upload
// starts, on its own connection. DATA_DIR and DATABASE_PATH are explicit under the work folder; the server child
// is stopped through its handle. Env: REPO_DIR = the source tree (its dist/ must be built).
// Usage: node p6-blocking.mjs <work folder>   (run with the repository as the current directory)
import { spawn, spawnSync } from 'node:child_process';
import { randomBytes } from 'node:crypto';
import { mkdirSync } from 'node:fs';
import { request as httpRequest } from 'node:http';
import { createServer } from 'node:net';
import { join } from 'node:path';
import { performance } from 'node:perf_hooks';
import { createRequire } from 'node:module';
import { pathToFileURL } from 'node:url';

const REPO = process.env.REPO_DIR;
const load = (p) => import(pathToFileURL(`${REPO}/${p}`).href);
const { pack, readTemplateBytes, unpack } = await load('tests/support/syntheticWorkbook.ts');
const require = createRequire(`${REPO}/package.json`);
const { strToU8 } = require('fflate');
const work = process.argv[2];
mkdirSync(work, { recursive: true });
const port = await new Promise((resolve) => {
  const s = createServer();
  s.listen(0, '127.0.0.1', () => {
    const p = s.address().port;
    s.close(() => resolve(p));
  });
});
const origin = `http://localhost:${port}`;
const pw = randomBytes(18).toString('base64url');
const env = {
  ...process.env,
  NODE_ENV: 'development',
  PORT: String(port),
  HOST: '127.0.0.1',
  DATABASE_PATH: join(work, 'timesheet.db'),
  DATA_DIR: join(work, 'data'),
  APP_ORIGINS: origin,
  MAIL_FROM: 'probe-sender@example.invalid',
  OUTBOUND_MODE: 'capture',
  JOB_RUNNER: 'off',
  SEED_ADMIN_PASSWORD: randomBytes(18).toString('base64url'),
  SEED_EMPLOYEE_PASSWORD: pw,
  SEED_EMPLOYEE2_PASSWORD: randomBytes(18).toString('base64url'),
};
const cwd = REPO;
for (const args of [['migrate'], ['seed']]) spawnSync(process.execPath, ['dist/server/cli.js', ...args], { env, cwd, encoding: 'utf8' });
const server = spawn(process.execPath, ['dist/server/index.js'], { env, cwd, stdio: 'ignore' });
const fail = (e) => {
  console.log('FATAL', e?.message ?? e);
  server.kill('SIGTERM');
  process.exitCode = 2;
};
process.on('uncaughtException', fail);
process.on('unhandledRejection', fail);

function send(method, path, { body, headers = {} } = {}) {
  return new Promise((resolve) => {
    const t0 = performance.now();
    const req = httpRequest({ host: '127.0.0.1', port, method, path, agent: false, headers: { origin, ...headers, ...(body ? { 'content-length': body.length } : {}) } }, (res) => {
      const chunks = [];
      res.on('data', (c) => chunks.push(c));
      res.on('end', () => resolve({ status: res.statusCode, ms: Math.round(performance.now() - t0), headers: res.headers, text: Buffer.concat(chunks).toString('utf8') }));
    });
    req.on('error', (e) => resolve({ status: 0, ms: Math.round(performance.now() - t0), error: e.code ?? e.message }));
    if (body) req.write(body);
    req.end();
  });
}
for (let i = 0; i < 100; i += 1) {
  const r = await send('GET', '/api/ready');
  if (r.status === 200) break;
  await new Promise((r2) => setTimeout(r2, 100));
}
const login = await send('POST', '/api/auth/login', { body: Buffer.from(JSON.stringify({ email: 'employee@example.invalid', password: pw })), headers: { 'content-type': 'application/json' } });
const cookie = String(login.headers['set-cookie']?.[0] ?? '').split(';')[0];

const MiB = 1024 * 1024;
const sheet = (bytes) => {
  const head = '<?xml version="1.0"?><worksheet xmlns="http://schemas.openxmlformats.org/spreadsheetml/2006/main"><sheetData><row r="1">';
  const tail = '</row></sheetData></worksheet>';
  return head + '<c/>'.repeat(Math.floor((bytes - head.length - tail.length) / 4)) + tail;
};
const cases = {
  'one 15 MiB sheet of empty cells': ['xl/worksheets/sheet3.xml'],
  'three 15.8 MiB sheets of empty cells': ['xl/worksheets/sheet1.xml', 'xl/worksheets/sheet2.xml', 'xl/worksheets/sheet3.xml'],
};
for (const [label, parts] of Object.entries(cases)) {
  const e = unpack(readTemplateBytes());
  for (const part of parts) e[part] = strToU8(sheet(parts.length === 1 ? 15 * MiB : 15.8 * MiB));
  e['docProps/custom-marker.xml'] = strToU8(`<m>${randomBytes(8).toString('hex')}</m>`);
  const bytes = Buffer.from(pack(e));
  const upload = send('POST', '/api/imports', { body: bytes, headers: { cookie, 'content-type': 'application/vnd.openxmlformats-officedocument.spreadsheetml.sheet' } });
  await new Promise((r) => setTimeout(r, 300));
  const health = await send('GET', '/api/health');
  const done = await upload;
  let reason = '';
  try {
    reason = JSON.parse(done.text)?.error?.details?.reason ?? JSON.parse(done.text)?.details?.reason ?? '';
  } catch {
    reason = '';
  }
  console.log(`${label}: upload ${bytes.length} bytes -> ${done.status}${reason ? ` (${reason})` : ''} in ${done.ms} ms; GET /api/health sent 300 ms later -> ${health.status}${health.error ? ' ' + health.error : ''} after ${health.ms} ms`);
}
const idle = await send('GET', '/api/health');
console.log(`idle GET /api/health afterwards -> ${idle.status} in ${idle.ms} ms`);
server.kill('SIGTERM');
await new Promise((r) => server.once('exit', r));
console.log(`server stopped through its handle: ${server.signalCode ?? server.exitCode}`);
