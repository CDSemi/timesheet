const lines = [];
const npx = process.platform === 'win32' ? 'npx.cmd' : 'npx';
for (const mutant of MUTANTS) {
  if (only.length > 0 && !only.includes(mutant.id)) continue;
  const originals = new Map();
  let applied = true;
  for (const [file, find, replace] of mutant.edits) {
    const path = join(exportDir, file);
    const text = originals.get(path) ?? readFileSync(path, 'utf8');
    if (!originals.has(path)) originals.set(path, text);
    const current = readFileSync(path, 'utf8');
    const count = current.split(find).length - 1;
    if (count !== 1) {
      applied = false;
      lines.push(`${mutant.id} NOT APPLIED: anchor found ${count} times in ${file}`);
      break;
    }
    writeFileSync(path, current.replace(find, () => replace));
  }
  let summary = 'not run';
  if (applied) {
    const override = args.find((arg) => arg.startsWith('--tests='));
    const testList = override === undefined ? mutant.tests : override.slice('--tests='.length).split(',');
    const testArgs = full ? ['vitest', 'run'] : ['vitest', 'run', ...testList, '--maxWorkers=2'];
    const started = Date.now();
    const run = spawnSync(process.execPath, [join(exportDir, 'node_modules', 'vitest', 'vitest.mjs'), ...testArgs.slice(1)], { cwd: exportDir, encoding: 'utf8', env: { ...process.env, NO_COLOR: '1', FORCE_COLOR: '0' }, maxBuffer: 256 * 1024 * 1024 });
    const output = `${run.stdout}${run.stderr}`;
    writeFileSync(join(logDir, `mut-${mutant.id}${full ? '-full' : ''}-${outPath.split(/[\\/]/).pop()}.log`), output);
    const tests = /Tests\s+([^\n]+)/.exec(output)?.[1]?.trim() ?? '?';
    const failedNames = [...output.matchAll(/(?:FAIL|×)\s+(tests\/[^\s>]+)[^\n]*?>\s*([^\n]+)/g)].slice(0, 4).map((match) => `${match[1]} > ${match[2].trim().slice(0, 110)}`);
    summary = `exit ${run.status}, ${tests}, ${((Date.now() - started) / 1000).toFixed(0)} s; ${run.status === 0 ? 'SURVIVED' : 'KILLED'}${failedNames.length > 0 ? ` | e.g. ${failedNames.join(' || ')}` : ''}`;
  }
  for (const [path, text] of originals) writeFileSync(path, text);
  const restored = [...originals].every(([path, text]) => sha(readFileSync(path, 'utf8')) === sha(text));
  lines.push(`${mutant.id} ${mutant.what}: ${summary}; restored ${restored ? 'byte-identical' : 'MISMATCH'}`);
  writeFileSync(outPath, `${lines.join('\n')}\n`);
}
writeFileSync(outPath, `${lines.join('\n')}\n`);
