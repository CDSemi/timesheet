S() {
  label=$1; shift
  bash run-shift.sh "stress-$label" real "$@" --repeat-each=20 --workers=10 > "logs/stress-$label.txt" 2>&1
  node counts.mjs "logs/stress-$label.txt" >> out/31-stress-counts.txt 2>&1
  echo "$label finished $(date -u +%H:%M:%SZ)" >> logs/stress.log
}
