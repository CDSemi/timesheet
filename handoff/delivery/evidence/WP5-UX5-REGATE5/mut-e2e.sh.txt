cd <task>
. ./env.sh
P=$T/export-probe
M() {
  id=$1; spec=$2; pat=$3
  node.exe $TW\\probes\\apply.mjs "$TW\\export-probe" $id apply >> out/81-mut-e2e-apply.txt 2>&1 || { echo "$id apply FAILED" >> out/81-mut-e2e-counts.txt; return; }
  ( cd $P && node.exe node_modules/vite/bin/vite.js build > $T/out/80-$id-build.txt 2>&1; echo "$id mutated client build exit $?" >> $T/out/80-mut-exits.txt )
  SHIFT_EXPORT=$P bash run-shift.sh mut-$id real $spec --grep "$pat" --workers=4 > logs/mut-$id.txt 2>&1
  node counts.mjs logs/mut-$id.txt | sed "s/^/$id: /" >> out/81-mut-e2e-counts.txt 2>&1
  node.exe $TW\\probes\\apply.mjs "$TW\\export-probe" $id restore >> out/81-mut-e2e-apply.txt 2>&1
  ( cd $P && node.exe node_modules/vite/bin/vite.js build > $T/out/82-$id-restore-build.txt 2>&1; echo "$id restore client build exit $?" >> $T/out/80-mut-exits.txt )
}
