     1	# Timesheet application image (WP4-T04): one pinned Node base, three build stages, a small non-root runtime.
     2	#
     3	# Base image: node:24.21.0-trixie-slim (Debian 13, matches .nvmrc 24.21.0), pinned by the digest of its multi-arch
     4	# image INDEX (linux/amd64 and linux/arm64 resolve from it). Recorded 2026-10-05 with:
     5	#   docker buildx imagetools inspect node:24.21.0-trixie-slim
     6	# Never use a moving tag such as `latest`. Bump the tag and the digest together, and rebuild.
     7	# better-sqlite3 13.0.3 ships N-API prebuilds for linux glibc, so no compiler is installed (see O13 in WP4-PLAN);
     8	# fall back to bookworm-slim only if that prebuilt binding ever fails to load.
     9	ARG NODE_IMAGE_DIGEST=sha256:8ec5d7557396cfe32d21c3f9c13072355ceab22b584578ca4bb28af31120cffe
    10
    11	FROM node:24.21.0-trixie-slim@${NODE_IMAGE_DIGEST} AS base
    12	WORKDIR /app
    13	ENV NPM_CONFIG_UPDATE_NOTIFIER=false
    14
    15	# Build stage: development dependencies, then the server (tsc) and the client (Vite) into dist/. The image ships no source
    16	# maps and no sourceMappingURL comments (WP4-A-01): tsc gets --sourceMap false, and vite.config.ts turns build.sourcemap
    17	# off when BUILD_SOURCEMAPS=off. Local development keeps its maps.
    18	FROM base AS build
    19	COPY package.json package-lock.json .npmrc ./
    20	RUN npm ci
    21	COPY tsconfig*.json vite.config.ts ./
    22	COPY src ./src
    23	RUN npm run build:server -- --sourceMap false \
    24	    && BUILD_SOURCEMAPS=off npm run build:client
    25
    26	# Production dependencies only. Install scripts stay off: better-sqlite3 needs none (prebuilds ship in the package).
    27	FROM base AS prod-deps
    28	COPY package.json package-lock.json .npmrc ./
    29	RUN npm ci --omit=dev --ignore-scripts
    30
    31	# Runtime stage: dist/, production node_modules, assets/pdf-fonts and package.json ("type": "module") only. The context excludes
    32	# reference/inputs (the template workbook), handoff/, tests/, .claude/, .agents/, docs and every .env file.
    33	FROM base AS runtime
    34	ENV NODE_ENV=production \
    35	    OUTBOUND_MODE=capture \
    36	    HOST=0.0.0.0 \
    37	    PORT=3000 \
    38	    DATABASE_PATH=/data/timesheet.db \
    39	    DATA_DIR=/data/private-data
    40	# Fixed non-root identity. A bind-mounted /data must be owned by UID/GID 10001 on the host.
    41	RUN groupadd --gid 10001 timesheet \
    42	    && useradd --uid 10001 --gid 10001 --no-create-home --home-dir /nonexistent --shell /usr/sbin/nologin timesheet \
    43	    && install --directory --owner 10001 --group 10001 --mode 0700 /data
    44	COPY --from=prod-deps /app/node_modules ./node_modules
    45	COPY --from=build /app/dist ./dist
    46	# The one committed font (Indie Flower, SIL OFL 1.1, ADR 0001): read by the PDF renderer from /app/assets/pdf-fonts.
    47	COPY assets/pdf-fonts ./assets/pdf-fonts
    48	COPY package.json ./
    49	USER 10001:10001
    50	# Database, private files (PDFs, signatures, evidence) and the outbound capture directory live here and nowhere else;
    51	# the root filesystem can run read-only with a /tmp tmpfs.
    52	VOLUME ["/data"]
    53	EXPOSE 3000
    54	# Slim images have no curl: a Node one-liner asks /api/ready (schema applied and /data writable; no personal data).
    55	HEALTHCHECK --interval=15s --timeout=5s --start-period=20s --retries=3 \
    56	    CMD ["node", "-e", "fetch('http://127.0.0.1:' + (process.env.PORT || 3000) + '/api/ready', { signal: AbortSignal.timeout(4000) }).then((r) => process.exit(r.ok ? 0 : 1), () => process.exit(1))"]
    57	STOPSIGNAL SIGTERM
    58	CMD ["node", "dist/server/index.js"]
