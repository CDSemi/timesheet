const servers = [];
function startServer(port) {
  const child = spawn(process.execPath, [join(exportDir, 'dist', 'server', 'index.js')], { env: envFor(port), cwd: exportDir, stdio: ['ignore', 'pipe', 'pipe'] });
  let output = '';
  child.stdout.on('data', (chunk) => (output += String(chunk)));
  child.stderr.on('data', (chunk) => (output += String(chunk)));
  servers.push({ child, output: () => output });
  return child;
}
async function waitHealthy(port) {
  for (let attempt = 0; attempt < 60; attempt += 1) {
    try {
      const response = await fetch(`http://127.0.0.1:${port}/api/health`, { headers: { connection: 'close' }, signal: AbortSignal.timeout(2000) });
      if (response.ok) return true;
    } catch {
      // not listening yet
    }
    await new Promise((resolve) => setTimeout(resolve, 250));
  }
  return false;
}
async function call(method, path, { cookie, body, port = portA, origin = true } = {}) {
  const headers = { connection: 'close' };
  if (origin) headers.origin = `http://localhost:${port}`;
  if (cookie) headers.cookie = cookie;
  if (body !== undefined) headers['content-type'] = 'application/json';
  const response = await fetch(`http://127.0.0.1:${port}${path}`, { method, headers, body: body === undefined ? undefined : JSON.stringify(body) });
  const buffer = Buffer.from(await response.arrayBuffer());
  let json = null;
  try {
    json = JSON.parse(buffer.toString('utf8'));
  } catch {
    json = null;
  }
  return { status: response.status, headers: response.headers, json, buffer, text: buffer.toString('utf8') };
}

  const previewOther = await call('GET', `/api/timesheets/${payroll}/pdf-preview`, { cookie: admin.cookie });
  check("the admin's preview is of the admin's own timesheet only (no user id in the route)", previewOther.status === 200 || previewOther.status === 404, code(previewOther));

  // ---- 9. the shared mount (grantee with every item)
  const grant = await call('POST', '/api/shares', { cookie: owner.cookie, body: { grantee_email: 'employee@example.invalid', items: { timesheets: 'edit', ot_read: true, pdf_download: true } } });
  check('owner grants a share with every item', grant.status === 201, code(grant));
  const shared = (path) => `/api/shared/${me.id}${path}`;
  const sharedPdf = await call('GET', shared(`/revisions/${revisionId}/pdf?time=1&ot=1`), { cookie: other.cookie });
  const latestSend = monitor.prepare('SELECT pdf_variant FROM revision_sends WHERE revision_id = ? ORDER BY send_no DESC LIMIT 1').pluck().get(revisionId);
  const ownerOfLatest = await call('GET', `/api/revisions/${revisionId}/pdf?time=${latestSend[1]}&ot=${latestSend[3]}`, { cookie: owner.cookie });
  check(`shared PDF ignores the query and serves the latest send's variant (${latestSend})`, sharedPdf.status === 200 && sha(sharedPdf.buffer) === sha(ownerOfLatest.buffer), code(sharedPdf));
  const sharedRefused = [];
  for (const [method, path, body] of [
    ['GET', `/revisions/${revisionId}/send-draft`],
    ['POST', `/revisions/${revisionId}/send`, base(randomUUID(), { reason: 'Shared probe' })],
    ['POST', `/revisions/${revisionId}/resend`, { reason: 'Shared probe' }],
    ['GET', `/timesheets/${payroll}/pdf-preview`],
    ['GET', '/deliveries'],
    ['GET', `/timesheets/${payroll}/review`],
    ['POST', `/timesheets/${payroll}/signoff`, signBody],
    ['GET', `/signatures/${currentSig.id}`],
    ['GET', '/settings/submission'],
  ]) {
    const res = await call(method, shared(path), { cookie: other.cookie, body });
    sharedRefused.push(`${method} ${path.replace(revisionId, '<rev>').replace(currentSig.id, '<sig>')} ${res.status}`);
  }
  check('send-draft, send, resend, pdf-preview, deliveries, review, sign-off, signature and settings are not shared (404)', sharedRefused.every((line) => line.endsWith(' 404')), sharedRefused.join('; '));
  // a correction revision with no PDF yet: the shared route never renders it
  const review2 = (await call('GET', `/api/timesheets/${payroll}/review`, { cookie: owner.cookie })).json;
  const dayEntry = await call('PUT', `/api/days/${review2.payload.days[2].work_date}`, { cookie: owner.cookie, body: { category: 'Sick', leave_minutes: 0, wfh: false, notes: '', reason: 'Synthetic correction probe' } });
  const review3 = (await call('GET', `/api/timesheets/${payroll}/review`, { cookie: owner.cookie })).json;
  const corr = await call('POST', `/api/timesheets/${payroll}/revisions`, { cookie: owner.cookie, body: { expected_version: review3.expected_version, reviewed_hash: review3.payload_hash, incomplete_evidence_acknowledged: true, reason: 'Synthetic correction for the audit probe', send_email: true } });
  const corrId = corr.json?.revision?.id;
  check('a correction (send_email: true) creates revision 2 and enqueues the PDF only', corr.status === 201 && JSON.stringify((corr.json?.jobs ?? []).map((job) => job.kind)) === '["render_pdf"]' && count('SELECT count(*) FROM revision_sends WHERE revision_id = ?', corrId) === 0, `${code(dayEntry)} ${code(corr)} ${JSON.stringify(corr.json?.jobs)}`);
  const filesBeforeShared = filesOf(corrId).length;
  const sharedCorr = await call('GET', shared(`/revisions/${corrId}/pdf?time=0&ot=0`), { cookie: other.cookie });
  check('shared PDF of a revision whose PDF is not stored: 409, no render, no file row', code(sharedCorr) === '409:pdf_not_ready' && filesOf(corrId).length === filesBeforeShared, `${code(sharedCorr)} files ${filesBeforeShared}->${filesOf(corrId).length}`);
  const supersededSend = await call('POST', `/api/revisions/${revisionId}/send`, { cookie: owner.cookie, body: base(randomUUID(), { reason: 'Old revision probe' }) });
  check('sending the superseded revision: 409 revision_superseded', code(supersededSend) === '409:revision_superseded', code(supersededSend));
  const corrDraft = (await call('GET', `/api/revisions/${corrId}/send-draft`, { cookie: owner.cookie })).json;
  check("the correction's draft requires a reason (a copy of revision 1 counts) and offers the correction reason", corrDraft.reason_required === true && corrDraft.correction_reason === 'Synthetic correction for the audit probe');

  // ---- 10. signature image privacy and the static root
  const sigOwner = await call('GET', `/api/signatures/${currentSig.id}`, { cookie: owner.cookie });
  const sigOtherUser = await call('GET', `/api/signatures/${currentSig.id}`, { cookie: other.cookie });
  const sigAdmin = await call('GET', `/api/signatures/${currentSig.id}`, { cookie: admin.cookie });
  const sigAnon = await call('GET', `/api/signatures/${currentSig.id}`);
  check('signature image: owner 200 private no-store; other 404, admin 404, anonymous 401', sigOwner.status === 200 && (sigOwner.headers.get('cache-control') ?? '').includes('no-store') && `${sigOtherUser.status}/${sigAdmin.status}/${sigAnon.status}` === '404/404/401', `${sigOwner.status} ${sigOwner.headers.get('cache-control')} ${sigOtherUser.status}/${sigAdmin.status}/${sigAnon.status}`);
  const font = await call('GET', '/assets/pdf-fonts/IndieFlower-Regular.ttf', { origin: false });
  const fontBytes = readFileSync(join(exportDir, 'assets', 'pdf-fonts', 'IndieFlower-Regular.ttf'));
  check('the PDF font is not served over HTTP (the static path gives the app shell, not the font)', sha(font.buffer) !== sha(fontBytes) && !font.buffer.subarray(0, 4).equals(Buffer.from([0, 1, 0, 0])), `${font.status} ${font.headers.get('content-type')} ${font.buffer.length} bytes`);

  // ---- 11. per-send privacy for administrators
  const subs = await call('GET', '/api/admin/submissions', { cookie: admin.cookie });
  const subsText = subs.text;
  check('admin status: recipient addresses only, no subject, body, reason or variant', subs.status === 200 && subsText.includes('payroll@example.invalid') && !subsText.includes(draft.json.subject) && !subsText.includes('synthetic probe reason') && !subsText.includes('Race probe') && !subsText.includes(draft.json.body_text.slice(0, 15)), `${subs.status} ${subsText.length} bytes`);
  const adminDeliveries = await call('GET', `/api/deliveries?revision_id=${revisionId}`, { cookie: admin.cookie });
  check("admin's own deliveries list does not show the owner's attempts", adminDeliveries.status === 200 && (adminDeliveries.json?.deliveries ?? []).length === 0, code(adminDeliveries));

  // ---- 12. Settings show_time_on_pdf
  const settingsNow = (await call('GET', '/api/settings/submission', { cookie: owner.cookie })).json.settings;
  const versionsBefore = (await call('GET', '/api/settings/submission/versions', { cookie: owner.cookie })).json.versions;
  const badSettings = await call('POST', '/api/settings/submission', { cookie: owner.cookie, body: { expected_seq: settingsNow.seq, to: settingsNow.recipients.to, auto_submit: settingsNow.auto_submit, show_time_on_pdf: 'no' } });
  check('settings: show_time_on_pdf not a boolean is 422 validation_error', code(badSettings) === '422:validation_error', code(badSettings));
  const saved = await call('POST', '/api/settings/submission', { cookie: owner.cookie, body: { expected_seq: settingsNow.seq, to: settingsNow.recipients.to, cc: settingsNow.recipients.cc, auto_submit: settingsNow.auto_submit, show_time_on_pdf: false } });
  const versionsAfter = (await call('GET', '/api/settings/submission/versions', { cookie: owner.cookie })).json.versions;
  const oldVersion = (await call('GET', `/api/settings/submission/versions/${settingsNow.id}`, { cookie: owner.cookie })).json?.settings;
  check('settings: a new version with show_time_on_pdf false; the earlier version is unchanged', saved.status === 201 && saved.json.settings.show_time_on_pdf === false && versionsAfter.length === versionsBefore.length + 1 && oldVersion?.show_time_on_pdf === settingsNow.show_time_on_pdf, `${code(saved)} ${versionsBefore.length}->${versionsAfter.length}`);
  const corrDraft2 = (await call('GET', `/api/revisions/${corrId}/send-draft`, { cookie: owner.cookie })).json;
  check('the Settings default reaches the send draft (time_rows false)', corrDraft2.time_rows === false);
  const corrJobs = monitor.prepare("SELECT payload_json FROM jobs WHERE revision_id = ? AND kind = 'render_pdf'").pluck().all(corrId);
  check("the correction's own sign-off PDF job is the sign-off default (time rows on), not the Settings value", corrJobs.length === 1 && JSON.parse(corrJobs[0]).variant?.startsWith('t1'), JSON.stringify(corrJobs));

  // ---- 13. round-4 recheck F-UX4B-1 and the password route
  const victim = meOther.id;
  const attackerTries = [];
  for (let attempt = 0; attempt < 6; attempt += 1) attackerTries.push((await call('POST', '/api/auth/login', { body: { email: `password:${victim}`, password: '<synthetic-password>' } })).status);
  const wrongCurrent = await call('POST', '/api/auth/password', { cookie: other.cookie, body: { current_password: '<synthetic-password>', new_password: '<synthetic-password>' } });
  const change = await call('POST', '/api/auth/password', { cookie: other.cookie, body: { current_password: passwords.employee, new_password: `Synthetic-${randomBytes(9).toString('base64url')}` } });
  check('F-UX4B-1: six failed sign-ins typed as password:<victim id> do not lock the victim out of the password change', change.status === 200 && wrongCurrent.status === 403 && wrongCurrent.json?.error?.code === 'current_password_incorrect', `sign-ins ${attackerTries.join(',')}; wrong current ${code(wrongCurrent)}; change ${code(change)}`);
  check('the sign-in limiter itself still answers 429 after five failures on one account', attackerTries.slice(5).every((status) => status === 429) && attackerTries.slice(0, 5).every((status) => status === 401), attackerTries.join(','));
  const sharedPassword = await call('POST', shared('/auth/password'), { cookie: owner.cookie, body: { current_password: 'x', new_password: 'y' } });
  check('the password route is not on the shared mount', sharedPassword.status === 404, code(sharedPassword));

  // ---- 14. round-4 shared clock guard still intact
  const clockIn = await call('POST', '/api/clock/in', { cookie: owner.cookie, body: { input_zone: 'America/Los_Angeles' } });
  const running = clockIn.json?.session;
  const sharedClockIn = await call('POST', shared('/clock/in'), { cookie: other.cookie, body: { input_zone: 'America/Los_Angeles' } });
  const runningId = running?.id;
  const sharedDelete = await call('DELETE', shared(`/sessions/${runningId}`), { cookie: other.cookie, body: { expected_version: running?.version ?? 1, reason: 'Synthetic probe' } });
  check('round 4: a grantee cannot clock in (404) or delete the running session (409 shared_clock_action)', sharedClockIn.status === 404 && code(sharedDelete) === '409:shared_clock_action', `${code(clockIn)} ${sharedClockIn.status} ${code(sharedDelete)}`);
} catch (error) {
  failures += 1;
  log(`FAIL unexpected error: ${error instanceof Error ? `${error.message}\n${error.stack}` : String(error)}`);
} finally {
  monitor?.close();
  for (const server of servers) server.child.kill();
  await new Promise((resolve) => setTimeout(resolve, 500));
  for (const [index, server] of servers.entries()) log(`server ${index} output (first 300 chars): ${server.output().replaceAll(work, '<work>').slice(0, 300).replaceAll('\n', ' | ')}`);
}
log(`failures: ${failures}`);
writeFileSync(outPath, `${lines.join('\n')}\n`);
process.exitCode = failures === 0 ? 0 : 1;
