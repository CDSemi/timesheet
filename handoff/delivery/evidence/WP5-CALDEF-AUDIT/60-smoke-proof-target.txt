function recentMonday(now = Date.now()) {
  // Step whole calendar dates from today's LA date; a day is not always 24 hours across DST.
  const [year, month, day] = laDateParts(new Date(now)).date.split('-').map(Number);
  for (let back = 1; back <= 7; back += 1) {
    const candidate = new Date(Date.UTC(year, month - 1, day - back));
    if (candidate.getUTCDay() === 1) return candidate.toISOString().slice(0, 10);
  }
  throw new Error('No Monday found');
}
