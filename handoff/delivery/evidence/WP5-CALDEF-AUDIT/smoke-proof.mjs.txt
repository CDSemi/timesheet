// WP5-CALDEF-AUDIT: independent check of recentMonday() in scripts/smoke-built-server.mjs (target export). The function text is read
// from the script itself (not retyped), evaluated in isolation, and compared at every 15 minutes over 2026-01-01..2028-01-01 (both DST
// changes of 2026 and 2027 included) with an oracle that walks back hour by hour and reads each LA calendar date and weekday from Intl.
import { readFileSync } from 'node:fs';

const path = process.argv[2];
const source = readFileSync(path, 'utf8');
const pick = (name) => {
  const start = source.indexOf(`function ${name}(`);
  if (start < 0) throw new Error(`no ${name}`);
  let depth = 0;
  for (let i = source.indexOf('{', start); i < source.length; i += 1) {
    if (source[i] === '{') depth += 1;
    if (source[i] === '}') {
      depth -= 1;
      if (depth === 0) return source.slice(start, i + 1);
    }
  }
  throw new Error(`unterminated ${name}`);
};
const text = `${pick('laDateParts')}\n${pick('recentMonday')}\nreturn recentMonday;`;
// A Date whose now() is the instant under test, so an older recentMonday() that reads Date.now() itself can be checked the same way.
let current = 0;
class FakeDate extends Date {
  static now() {
    return current;
  }
}
const inner = new Function('Date', text)(FakeDate);
const recentMonday = (ms) => {
  current = ms;
  return inner(ms);
};
console.log('function under test:');
console.log(pick('recentMonday'));

const fmt = new Intl.DateTimeFormat('en-CA', { timeZone: 'America/Los_Angeles', year: 'numeric', month: '2-digit', day: '2-digit', weekday: 'short' });
const laParts = (ms) => {
  const parts = fmt.formatToParts(new Date(ms));
  const get = (type) => parts.find((part) => part.type === type)?.value;
  return { date: `${get('year')}-${get('month')}-${get('day')}`, weekday: get('weekday') };
};
/** The latest LA date strictly before today's LA date whose weekday is Monday, found by walking back one hour at a time. */
function oracle(ms) {
  const today = laParts(ms).date;
  for (let back = 1; back <= 8 * 24 + 2; back += 1) {
    const seen = laParts(ms - back * 3_600_000);
    if (seen.date !== today && seen.weekday === 'Mon') return seen.date;
  }
  throw new Error('oracle found no Monday');
}

const from = Date.UTC(2026, 0, 1);
const to = Date.UTC(2028, 0, 1);
let checked = 0;
let mismatches = 0;
const samples = [];
for (let ms = from; ms < to; ms += 15 * 60_000) {
  checked += 1;
  let got;
  try {
    got = recentMonday(ms);
  } catch (error) {
    got = `THROW ${error.message}`;
  }
  const want = oracle(ms);
  if (got !== want) {
    mismatches += 1;
    if (samples.length < 10) samples.push(`${new Date(ms).toISOString()} got ${got} want ${want}`);
  }
}
for (const iso of ['2026-11-03T07:49:00Z', '2026-11-02T07:59:00Z', '2026-03-09T06:30:00Z', '2026-03-10T06:59:00Z', '2026-11-09T07:59:59Z', '2026-11-10T08:00:00Z']) {
  console.log(`${iso} (LA ${laParts(Date.parse(iso)).date} ${laParts(Date.parse(iso)).weekday}): recentMonday ${recentMonday(Date.parse(iso))}, oracle ${oracle(Date.parse(iso))}`);
}
console.log(`checked ${checked} instants (every 15 minutes, 2026-01-01..2028-01-01): mismatches ${mismatches}`);
for (const line of samples) console.log(`MISMATCH ${line}`);
process.exitCode = mismatches === 0 ? 0 : 1;
