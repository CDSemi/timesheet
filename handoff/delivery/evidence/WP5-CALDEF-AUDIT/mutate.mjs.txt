const [command, id] = process.argv.slice(2);
if (command === 'list') {
  for (const [key, value] of Object.entries(MUTANTS)) console.log(`${key}\t${value.file}\t${value.what}`);
  process.exit(0);
}
const mutant = MUTANTS[id];
if (mutant === undefined) throw new Error(`unknown mutant ${id}`);
const target = join(MUT, mutant.file);
const pristine = readFileSync(join(PRISTINE, mutant.file));
if (command === 'apply') {
  const current = readFileSync(target);
  if (!current.equals(pristine)) throw new Error(`${mutant.file} in export-mut is not pristine before ${id}`);
  let text = pristine.toString('utf8');
  for (const [find, replace] of mutant.edits) {
    const count = text.split(find).length - 1;
    if (count !== 1) throw new Error(`${id}: expected exactly one occurrence in ${mutant.file}, found ${count}`);
    text = text.replace(find, () => replace);
  }
  writeFileSync(target, text);
  console.log(`applied ${id} to ${mutant.file} (${mutant.edits.length} edit(s)): ${mutant.what}`);
} else if (command === 'restore') {
  writeFileSync(target, pristine);
  const back = readFileSync(target);
  console.log(`restored ${id} ${mutant.file}: byte-equal to the pristine export ${back.equals(pristine)}`);
  if (!back.equals(pristine)) process.exitCode = 1;
} else {
  throw new Error('usage: apply|restore|list <id>');
}
