// Longest test of a spec file in Playwright list-reporter logs, and the lines of a title fragment.
// Usage: node durations.mjs <specFragment> <titleFragment|-> <log> [<log> ...]
import { readFileSync } from 'node:fs';

const [spec, title, ...files] = process.argv.slice(2);
const toSeconds = (value, unit) => (unit === 'ms' ? Number(value) / 1000 : unit === 'm' ? Number(value) * 60 : Number(value));
for (const file of files) {
  const text = readFileSync(file, 'utf8').replace(/\r/g, '');
  let best = null;
  const matches = [];
  for (const line of text.split('\n')) {
    const m = /^\s+(ok|x)\s+\d+ \[(desktop|mobile)\] › (\S+) › (.*) \(([\d.]+)(ms|s|m)\)\s*$/.exec(line);
    if (!m || !m[3].includes(spec)) continue;
    const seconds = toSeconds(m[5], m[6]);
    if (best === null || seconds > best.seconds) best = { seconds, line: line.trim().slice(0, 200) };
    if (title !== '-' && m[4].includes(title)) matches.push(line.trim().slice(0, 220));
  }
  console.log(`${file.split('/').pop()}: longest ${spec} test ${best === null ? 'none' : `${best.seconds}s: ${best.line}`}`);
  for (const line of matches) console.log(`  title match: ${line}`);
}
