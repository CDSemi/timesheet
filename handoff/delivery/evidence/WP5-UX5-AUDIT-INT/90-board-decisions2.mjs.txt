// Read-only: all owner/coordinator decisions dated 2026-10-20 or later, plus pause_note, phase and recent runtime_observations.
import { readFileSync, writeFileSync } from 'node:fs';

const board = JSON.parse(readFileSync(process.argv[2], 'utf8'));
const out = [];
for (const key of ['status', 'phase', 'pause_note', 'active_package', 'current_source_digest', 'push_hold', 'checkpoint', 'git']) {
  out.push(`### ${key}\n${JSON.stringify(board[key], null, 2)}\n`);
}
const recent = (v) => typeof v?.date === 'string' && v.date >= '2026-10-20';
board.owner_decisions.forEach((v, i) => { if (recent(v)) out.push(`### owner_decisions[${i}] ${v.date}\n${JSON.stringify(v, null, 2)}\n`); });
board.coordinator_decisions.forEach((v, i) => { if (recent(v)) out.push(`### coordinator_decisions[${i}] ${v.date}\n${JSON.stringify(v, null, 2)}\n`); });
const ro = board.runtime_observations;
if (Array.isArray(ro)) ro.forEach((v, i) => { if (recent(v) || /UX5|UX4/.test(JSON.stringify(v))) out.push(`### runtime_observations[${i}]\n${JSON.stringify(v, null, 2)}\n`); });
else out.push(`### runtime_observations (${typeof ro})\n${JSON.stringify(ro, null, 2).slice(0, 20000)}\n`);
out.push(`### governance_backlog\n${JSON.stringify(board.governance_backlog, null, 2)}\n`);
writeFileSync(process.argv[3], out.join('\n') + '\n');
