// Writes masked LF copies of raw task files into the evidence folder: task folder -> <task>, repository -> <project>,
// profile segment -> <user>, setup tokens -> <setup-token>, non-example.invalid addresses -> <email>, ANSI codes removed.
// Usage: node mask.mjs <raw-root> <evidence-dir> <src>=<dst> [...]
import { readFileSync, writeFileSync } from 'node:fs';
import { join } from 'node:path';

const [rawRoot, outDir, ...pairs] = process.argv.slice(2);
const SEP = '(?:\\\\\\\\|\\\\|/)';
// The profile segment is read at run time (never written into this file).
const profile = (process.env.USERPROFILE ?? process.env.HOME ?? '').split(/[\\/]/).filter(Boolean).at(-1) ?? '';
if (profile === '') throw new Error('no profile segment to mask');
const rules = [
  [new RegExp(`[A-Za-z]:${SEP}\\.claude-tmp${SEP}timesheet${SEP}WP5-FINAL-AUDIT`, 'gi'), '<task>'],
  [/\/d\/\.claude-tmp\/timesheet\/WP5-FINAL-AUDIT/gi, '<task>'],
  [new RegExp(`[A-Za-z]:${SEP}Dropbox${SEP}Work\\.CDSemi${SEP}timesheet`, 'gi'), '<project>'],
  [/\/d\/Dropbox\/Work\.CDSemi\/timesheet/gi, '<project>'],
  [new RegExp(`(Users${SEP})${profile}`, 'gi'), '$1<user>'],
  [new RegExp(`\\b${profile}\\b`, 'gi'), '<user>'],
  [/[A-Z2-7]{5}(?:-[A-Z2-7]{5}){7}/g, '<setup-token>'],
  [/[A-Za-z0-9._%+-]+@(?![A-Za-z0-9.-]*\.invalid\b)[A-Za-z0-9.-]+\.[A-Za-z]{2,}/g, '<email>'],
  // eslint-disable-next-line no-control-regex
  [/\x1b\[[0-9;?]*[A-Za-z]/g, ''],
  [/\r\n?/g, '\n'],
];
let count = 0;
for (const pair of pairs) {
  const [src, dst] = pair.split('=');
  let text = readFileSync(join(rawRoot, src), 'utf8');
  for (const [pattern, replacement] of rules) text = text.replace(pattern, replacement);
  if (!text.endsWith('\n')) text += '\n';
  writeFileSync(join(outDir, dst), text);
  count += 1;
}
console.log(`masked ${count} files`);
