// Independent forbidden-file scan of an image file list (one absolute path per line, from `find / -xdev`).
// Usage: node image-scan.mjs <image-files.txt> <package.json of the release>
import { readFileSync } from 'node:fs';

const [listPath, packagePath] = process.argv.slice(2);
const paths = readFileSync(listPath, 'utf8').split(/\r?\n/).filter((line) => line !== '');
const pkg = JSON.parse(readFileSync(packagePath, 'utf8'));
const rules = [
  ['workbook (*.xlsx, *.xlsm, *.xls) anywhere', (p) => /\.(xlsx|xlsm|xls)$/i.test(p)],
  ['source map under /app/dist', (p) => p.startsWith('/app/dist/') && p.endsWith('.map')],
  ['.env file anywhere (except node internals)', (p) => /(^|\/)\.env(\.[^/]*)?$/.test(p)],
  ['handoff/ anywhere', (p) => /(^|\/)handoff(\/|$)/.test(p)],
  ['/app/tests or a tests/ tree of the project', (p) => p.startsWith('/app/tests')],
  ['/app/docs', (p) => p.startsWith('/app/docs')],
  ['/app/reference or reference/inputs', (p) => p.startsWith('/app/reference') || p.includes('reference/inputs')],
  ['.claude or .agents', (p) => /(^|\/)\.(claude|agents)(\/|$)/.test(p)],
  ['database files (*.db, *.sqlite*, -wal, -shm) outside node_modules', (p) => !p.includes('/node_modules/') && /\.(db|sqlite\d*)(-wal|-shm)?$/i.test(p)],
  ['TypeScript sources /app/src', (p) => p.startsWith('/app/src')],
  ['git metadata', (p) => /(^|\/)\.git(\/|$)/.test(p)],
  ['private data under /data in the image layers', (p) => p.startsWith('/data/') && p !== '/data'],
];
let total = 0;
for (const [label, test] of rules) {
  const hits = paths.filter(test);
  total += hits.length;
  console.log(`${hits.length === 0 ? 'OK  ' : 'HIT '} ${label}: ${hits.length}${hits.length ? ` e.g. ${hits.slice(0, 3).join(' | ')}` : ''}`);
}
const appTop = [...new Set(paths.filter((p) => p.startsWith('/app/')).map((p) => p.split('/')[2]))].sort();
console.log(`INFO /app top-level entries: ${JSON.stringify(appTop)}`);
const nm = new Set(paths.filter((p) => /^\/app\/node_modules\/(@[^/]+\/[^/]+|[^@/][^/]*)$/.test(p)).map((p) => p.replace('/app/node_modules/', '')));
const prod = Object.keys(pkg.dependencies ?? {});
const dev = Object.keys(pkg.devDependencies ?? {});
console.log(`INFO production dependencies present at top level: ${prod.filter((d) => nm.has(d)).length}/${prod.length}; missing: ${JSON.stringify(prod.filter((d) => !nm.has(d)))}`);
const devPresent = dev.filter((d) => nm.has(d));
console.log(`${devPresent.length === 0 ? 'OK  ' : 'HIT '} devDependencies present at top level: ${devPresent.length} ${JSON.stringify(devPresent)}`);
total += devPresent.length;
const distMaps = paths.filter((p) => p.endsWith('.map'));
console.log(`INFO paths: ${paths.length}; *.map anywhere: ${distMaps.length} (outside /app/dist: ${distMaps.filter((p) => !p.startsWith('/app/dist/')).length})`);
console.log(`FORBIDDEN TOTAL ${total}`);
process.exit(total === 0 ? 0 : 1);
