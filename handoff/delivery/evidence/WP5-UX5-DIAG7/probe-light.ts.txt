/* ---- DIAG7 light probe (scratch only): the controls.spec.ts:49 title gap test with a low-cost event recorder ----------------- */

// The same steps as the original test, in the same order, with the same browser round trips and the same early stop on the first
// wrong gap. The recorder takes no layout reads and runs no frame loop (the full frame probe changed the timing and hid the race):
// it logs epoch times of fetch start and end, resize and media-change events, DOM mutation batches (what was added or removed) and
// long tasks. The two box reads of each width also log which layout they saw (top bar height, clock in the bar or the title row).

import { appendFileSync } from 'node:fs';
import type { Locator } from '@playwright/test';
import { expect, test } from './fixtures.ts';

test.use({ locale: 'en-US', timezoneId: 'America/Los_Angeles' });

const OUT = process.env.DIAG_OUT ?? '';
const wall = () => Math.round((performance.timeOrigin + performance.now()) * 10) / 10;

function recorder() {
  const w = window as unknown as { __diag: unknown[]; __mark: (what: string, extra?: unknown) => void };
  const log: unknown[] = [];
  w.__diag = log;
  const at = () => Math.round((performance.timeOrigin + performance.now()) * 10) / 10;
  const push = (what: string, extra?: unknown) => log.push({ t: at(), what, ...(extra === undefined ? {} : { extra }) });
  w.__mark = push;
  const originalFetch = window.fetch.bind(window);
  window.fetch = (input: RequestInfo | URL, init?: RequestInit) => {
    const url = typeof input === 'string' ? input : input instanceof URL ? input.href : input.url;
    const method = init?.method ?? 'GET';
    push('fetch-start', `${method} ${url}`);
    return originalFetch(input, init).then(
      (response) => {
        push('fetch-end', `${response.status} ${method} ${url}`);
        return response;
      },
      (error: unknown) => {
        push('fetch-error', `${method} ${url} ${String(error)}`);
        throw error;
      },
    );
  };
  window.addEventListener('resize', () => push('resize-event', window.innerWidth));
  window.matchMedia('(width >= 768px)').addEventListener('change', (event) => push('shell-mq-change', event.matches));
  const describe = (node: Node) =>
    node instanceof Element ? `${node.tagName.toLowerCase()}${node.className && typeof node.className === 'string' ? `.${node.className.trim().split(/\s+/).join('.')}` : ''}` : node.nodeName;
  const observe = () => {
    new MutationObserver((mutations) => {
      const added: string[] = [];
      const removed: string[] = [];
      let text = 0;
      for (const mutation of mutations) {
        if (mutation.type === 'characterData') text += 1;
        mutation.addedNodes.forEach((node) => added.length < 6 && added.push(describe(node)));
        mutation.removedNodes.forEach((node) => removed.length < 6 && removed.push(describe(node)));
      }
      push('mutation', { n: mutations.length, added, removed, text });
    }).observe(document.documentElement, { childList: true, subtree: true, characterData: true });
  };
  if (document.documentElement !== null) observe();
  else document.addEventListener('DOMContentLoaded', observe);
  try {
    new PerformanceObserver((list) => {
      for (const entry of list.getEntries()) log.push({ t: Math.round((performance.timeOrigin + entry.startTime) * 10) / 10, what: 'longtask', extra: Math.round(entry.duration) });
    }).observe({ type: 'longtask', buffered: true });
  } catch {
    push('longtask-unsupported');
  }
}

async function readBox(locator: Locator, label: string) {
  return locator.evaluate((element, name) => {
    const box = element.getBoundingClientRect();
    const result = { top: box.top, bottom: box.bottom, left: box.left, right: box.right, width: box.width, height: box.height };
    const bar = document.querySelector('header.shell-bar')?.getBoundingClientRect().height ?? -1;
    (window as unknown as { __mark: (what: string, extra?: unknown) => void }).__mark(`read ${name}`, {
      top: box.top,
      bottom: box.bottom,
      iw: window.innerWidth,
      bar,
      barClock: document.querySelector('.bar-clock') !== null,
      clockGroup: document.querySelector('.clock-group') !== null,
    });
    return result;
  }, label);
}

async function px(locator: Locator, property: string): Promise<number> {
  return locator.evaluate((element, name) => parseFloat(element.ownerDocument.defaultView?.getComputedStyle(element).getPropertyValue(name) ?? '0'), property);
}

for (const theme of ['light', 'dark'] as const) {
  test(`diag7 light probe (${theme})`, async ({ page, signInThroughUi }, testInfo) => {
    await page.addInitScript(recorder);
    const node: { t: number; what: string; v?: unknown }[] = [];
    await page.emulateMedia({ colorScheme: theme, reducedMotion: 'reduce' });
    node.push({ t: wall(), what: 'sign-in start' });
    await signInThroughUi();
    await expect(page.locator('[data-day]')).toHaveCount(14);
    node.push({ t: wall(), what: 'days 14 seen' });
    const gaps: { width: number; gap: number }[] = [];
    let bad: { width: number; gap: number } | null = null;
    for (const width of [768, 1024, 1440, 1920]) {
      node.push({ t: wall(), what: `setViewportSize ${width} call` });
      await page.setViewportSize({ width, height: 900 });
      node.push({ t: wall(), what: `setViewportSize ${width} done` });
      const head = page.locator('header.page-head');
      const card = page.locator('.workbar');
      await expect(head).toBeVisible();
      node.push({ t: wall(), what: `head visible ${width}` });
      const gap = (await readBox(card, `card ${width}`)).top - (await readBox(head, `head ${width}`)).bottom;
      node.push({ t: wall(), what: `read done ${width}`, v: gap });
      gaps.push({ width, gap });
      if (Math.abs(gap - 32) >= 0.5) {
        bad = { width, gap };
        break;
      }
      expect(await px(head, 'margin-bottom'), `title row margin at ${width}px`).toBe(32);
    }
    // The settled layout after the reads (probe only; the original test stops at the failed assertion).
    await page.waitForTimeout(500);
    const settled = await page.evaluate(() => {
      const h = document.querySelector('header.page-head')?.getBoundingClientRect();
      const c = document.querySelector('.workbar')?.getBoundingClientRect();
      return { iw: window.innerWidth, gap: h === undefined || c === undefined ? null : c.top - h.bottom, headBottom: h?.bottom, cardTop: c?.top, bar: document.querySelector('header.shell-bar')?.getBoundingClientRect().height, barClock: document.querySelector('.bar-clock') !== null };
    });
    const diag = await page.evaluate(() => (window as unknown as { __diag: unknown[] }).__diag);
    if (OUT !== '') {
      appendFileSync(OUT, `${JSON.stringify({ title: testInfo.title, project: testInfo.project.name, repeat: testInfo.repeatEachIndex, workerIndex: testInfo.workerIndex, gaps, bad, settled, node, diag })}\n`);
    }
    expect(bad, 'every gap read was 32px').toBeNull();
  });
}
