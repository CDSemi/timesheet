/*
 * WP5-UX-AUDIT-B5 independent probes (auditor-written; not part of the reviewed source; kept out of git
 * through .git/info/exclude of the scratch clone). Synthetic data only. Each project starts its own built
 * server (dist/) on a fixed port of the task range (desktop PROBE_PORT_BASE, mobile PROBE_PORT_BASE + 1)
 * with DATA_DIR and DATABASE_PATH inside the task folder, and stops it through the child handle.
 *
 * Focus-ring method: real Tab / Shift+Tab presses. At each stop two viewport screenshots are taken
 * (focused, then the same element blurred; transitions fast-forwarded), decoded here, and compared along
 * lines perpendicular to each side of the focused element. The ring is the pixel run that changed most;
 * its rendered (composited) colour is compared with the rendered pixel just outside it (the adjacent
 * background), with the pixel just inside it, and with what was painted there before focus.
 * Hard expectations only guard the measurement itself; the findings are in the JSON records.
 */

function shot(name: string): string {

async function waitHealthy(origin: string, child: ChildProcess) {
  for (let attempt = 0; attempt < 150; attempt += 1) {
    if (child.exitCode !== null) throw new Error(`server exited early with ${child.exitCode}`);
    try {
      if ((await fetch(`${origin}/api/health`)).ok) return;
    } catch {
      // not listening yet
    }
    await new Promise((done) => setTimeout(done, 100));
  }
  throw new Error('server did not become healthy');
}

/* ---- PNG decoding and colour maths ------------------------------------------------------------- */

type RGB = [number, number, number];
interface Img {
  width: number;
  height: number;
  data: Buffer;
  bpp: number;
}

function decodePng(buf: Buffer): Img {
  let pos = 8;
  let width = 0;
  let height = 0;
  let bitDepth = 0;
  let colorType = 0;
  let interlace = 0;
  const idat: Buffer[] = [];
  while (pos < buf.length) {
    const length = buf.readUInt32BE(pos);
    const type = buf.toString('latin1', pos + 4, pos + 8);
    const chunk = buf.subarray(pos + 8, pos + 8 + length);
    if (type === 'IHDR') {
      width = chunk.readUInt32BE(0);
      height = chunk.readUInt32BE(4);
      bitDepth = chunk[8] ?? 0;
      colorType = chunk[9] ?? 0;
      interlace = chunk[12] ?? 0;
    } else if (type === 'IDAT') idat.push(chunk);
    else if (type === 'IEND') break;
    pos += 12 + length;
  }
  if (bitDepth !== 8 || interlace !== 0 || (colorType !== 2 && colorType !== 6)) throw new Error(`unsupported PNG ${bitDepth}/${colorType}/${interlace}`);
  const bpp = colorType === 6 ? 4 : 3;
  const raw = inflateSync(Buffer.concat(idat));
  const stride = width * bpp;
  const data = Buffer.alloc(height * stride);
  for (let y = 0; y < height; y += 1) {
    const filter = raw[y * (stride + 1)] ?? 0;
    const src = y * (stride + 1) + 1;
    const dst = y * stride;
    for (let i = 0; i < stride; i += 1) {
      const left = i >= bpp ? (data[dst + i - bpp] ?? 0) : 0;
      const up = y > 0 ? (data[dst - stride + i] ?? 0) : 0;
      const upLeft = i >= bpp && y > 0 ? (data[dst - stride + i - bpp] ?? 0) : 0;
      const x = raw[src + i] ?? 0;
      let value = x;
      if (filter === 1) value = x + left;
      else if (filter === 2) value = x + up;
      else if (filter === 3) value = x + ((left + up) >> 1);
      else if (filter === 4) {
        const p = left + up - upLeft;
        const pa = Math.abs(p - left);
        const pb = Math.abs(p - up);
        const pc = Math.abs(p - upLeft);
        value = x + (pa <= pb && pa <= pc ? left : pb <= pc ? up : upLeft);
      } else if (filter !== 0) throw new Error(`PNG filter ${filter}`);
      data[dst + i] = value & 0xff;
    }
  }
  return { width, height, data, bpp };
}

function pixel(img: Img, x: number, y: number): RGB | null {
  if (x < 0 || y < 0 || x >= img.width || y >= img.height) return null;
  const o = (y * img.width + x) * img.bpp;
  return [img.data[o] ?? 0, img.data[o + 1] ?? 0, img.data[o + 2] ?? 0];
}

function luminance([r, g, b]: RGB): number {
  const c = (v: number) => {
    const s = v / 255;
    return s <= 0.04045 ? s / 12.92 : ((s + 0.055) / 1.055) ** 2.4;
  };
  return 0.2126 * c(r) + 0.7152 * c(g) + 0.0722 * c(b);
}
function contrast(a: RGB, b: RGB): number {
  const [x, y] = [luminance(a), luminance(b)].sort((p, q) => q - p) as [number, number];
  return (x + 0.05) / (y + 0.05);
}
const hex = (c: RGB | null) => (c === null ? null : `#${c.map((v) => v.toString(16).padStart(2, '0')).join('')}`);
const r2 = (n: number) => Math.round(n * 100) / 100;

/* ---- Ring measurement ---------------------------------------------------------------------------- */

interface Rect {
  x: number;
  y: number;
  width: number;
  height: number;
}

interface SideResult {
  side: string;
  at: number;
  change: number;
  d: number | null;
  ring: string | null;
  under: string | null;
  outer: string | null;
  inner: string | null;
  outerRatio: number | null;
  innerRatio: number | null;
  bandCss: number;
}

function analyseLine(points: Array<{ d: number; f: RGB; u: RGB }>, dsf: number) {
  let best = -1;
  let bestCh = 1;
  points.forEach((point, index) => {
    const ch = contrast(point.f, point.u);
    if (ch > bestCh + 1e-9) {
      bestCh = ch;
      best = index;
    }
  });
  if (best < 0) return { change: 1, d: null, ring: null, under: null, outer: null, inner: null, outerRatio: null, innerRatio: null, bandCss: 0 };
  const ring = points[best]?.f as RGB;
  let start = best;
  let end = best;
  while (start - 1 >= 0 && contrast(points[start - 1]?.f as RGB, ring) < 1.15) start -= 1;
  while (end + 1 < points.length && contrast(points[end + 1]?.f as RGB, ring) < 1.15) end += 1;
  const outer = points[end + 1]?.f ?? null;
  const inner = points[start - 1]?.f ?? null;
  return {
    change: r2(bestCh),
    d: r2(points[best]?.d ?? 0),
    ring: hex(ring),
    under: hex(points[best]?.u ?? null),
    outer: hex(outer),
    inner: hex(inner),
    outerRatio: outer === null ? null : r2(contrast(ring, outer)),
    innerRatio: inner === null ? null : r2(contrast(ring, inner)),
    bandCss: r2((end - start + 1) / dsf),
  };
}

function ringOf(focused: Img, unfocused: Img, rect: Rect, dsf: number) {
  const sides: SideResult[] = [];
  for (const side of ['top', 'bottom', 'left', 'right'] as const) {
    let bestSide: SideResult | null = null;
    const vertical = side === 'top' || side === 'bottom';
    const inside = Math.max(0.5, Math.min(6, (vertical ? rect.height : rect.width) / 2 - 0.5));
    for (const at of [0.3, 0.5, 0.7]) {
      const points: Array<{ d: number; f: RGB; u: RGB }> = [];
      if (vertical) {
        const col = Math.floor((rect.x + at * rect.width) * dsf);
        const edge = side === 'top' ? rect.y : rect.y + rect.height;
        const sign = side === 'top' ? -1 : 1;
        const from = Math.floor((edge - sign * inside) * dsf);
        const to = Math.floor((edge + sign * 8) * dsf);
        for (let row = from; sign > 0 ? row <= to : row >= to; row += sign) {
          const f = pixel(focused, col, row);
          const u = pixel(unfocused, col, row);
          if (f === null || u === null) continue;
          points.push({ d: sign * ((row + 0.5) / dsf - edge), f, u });
        }
      } else {
        const row = Math.floor((rect.y + at * rect.height) * dsf);
        const edge = side === 'left' ? rect.x : rect.x + rect.width;
        const sign = side === 'left' ? -1 : 1;
        const from = Math.floor((edge - sign * inside) * dsf);
        const to = Math.floor((edge + sign * 8) * dsf);
        for (let col = from; sign > 0 ? col <= to : col >= to; col += sign) {
          const f = pixel(focused, col, row);
          const u = pixel(unfocused, col, row);
          if (f === null || u === null) continue;
          points.push({ d: sign * ((col + 0.5) / dsf - edge), f, u });
        }
      }
      const result = { side, at, ...analyseLine(points, dsf) };
      if (bestSide === null || result.change > bestSide.change) bestSide = result;
    }
    if (bestSide !== null) sides.push(bestSide);
  }
  const present = sides.filter((side) => side.change >= 3);
  // A side whose outer pixel lies beyond the viewport image (the ring touches the screen edge) has no
  // adjacent colour to measure; it counts as present but not in the minimum.
  const outerRatios = present.filter((side) => side.outerRatio !== null).map((side) => side.outerRatio ?? 0);
  const innerRatios = present.filter((side) => side.innerRatio !== null).map((side) => side.innerRatio ?? 0);
  return {
    sidesPresent: present.length,
    minOuter: outerRatios.length === 0 ? null : r2(Math.min(...outerRatios)),
    minInner: innerRatios.length === 0 ? null : r2(Math.min(...innerRatios)),
    minChange: present.length === 0 ? null : r2(Math.min(...present.map((side) => side.change))),
    maxChange: sides.length === 0 ? null : r2(Math.max(...sides.map((side) => side.change))),
    sides,
  };
}

interface StopInfo {
  body?: boolean;
  hasFocus?: boolean;
  id?: number;
  tag?: string;
  type?: string | null;
  role?: string | null;
  name?: string;
  focusVisible?: boolean;
  boxShadow?: string;
  outline?: string;
  rect?: Rect;
  inViewport?: boolean;
  pts?: number;
  vis?: number;
  covers?: string[];
  inDialog?: string | null;
  inPanel?: boolean;
  dataDay?: string | null;
  scrollY?: number;
}

async function stopInfo(page: Page): Promise<StopInfo> {
  return page.evaluate(() => {
    const w = window as unknown as { __ids?: WeakMap<Element, number>; __next?: number };
    w.__ids ??= new WeakMap();
    w.__next ??= 0;
    const a = document.activeElement as HTMLElement | null;
    if (a === null || a === document.body || a === document.documentElement) return { body: true, hasFocus: document.hasFocus() };
    let id = w.__ids.get(a);
    if (id === undefined) {
      w.__next += 1;
      id = w.__next;
      w.__ids.set(a, id);
    }
    const r = a.getBoundingClientRect();
    const cs = getComputedStyle(a);
    const labelled = a.getAttribute('aria-labelledby');
    const labels = (a as HTMLInputElement).labels;
    const name = (
      a.getAttribute('aria-label') ??
      (labelled !== null ? labelled.split(' ').map((part) => document.getElementById(part)?.textContent ?? '').join(' ') : null) ??
      (labels !== null && labels !== undefined && labels.length > 0 ? (labels[0]?.textContent ?? '') : null) ??
      a.innerText ??
      ''
    )
      .replace(/\s+/g, ' ')
      .trim()
      .slice(0, 60);
    let pts = 0;
    let vis = 0;
    const covers = new Set<string>();
    for (let i = 0; i < 5; i += 1) {
      for (let j = 0; j < 5; j += 1) {
        const x = r.left + 1 + ((r.width - 2) * (i + 0.5)) / 5;
        const y = r.top + 1 + ((r.height - 2) * (j + 0.5)) / 5;
        if (x < 0 || y < 0 || x >= innerWidth || y >= innerHeight) continue;
        pts += 1;
        const hit = document.elementFromPoint(x, y);
        const inLabel = labels !== null && labels !== undefined && [...labels].some((label) => hit !== null && label.contains(hit));
        if (hit !== null && (hit === a || a.contains(hit) || inLabel)) vis += 1;
        else if (hit !== null) covers.add(`${hit.tagName}.${typeof hit.className === 'string' ? hit.className.split(' ')[0] : ''}`);
      }
    }
    return {
      id,
      tag: a.tagName,
      type: a.getAttribute('type'),
      role: a.getAttribute('role'),
      name,
      focusVisible: a.matches(':focus-visible'),
      boxShadow: cs.boxShadow,
      outline: `${cs.outlineStyle} ${cs.outlineWidth} ${cs.outlineColor}`,
      rect: { x: r.left, y: r.top, width: r.width, height: r.height },
      inViewport: r.top >= 0 && r.left >= 0 && r.bottom <= innerHeight && r.right <= innerWidth,
      pts,
      vis,
      covers: [...covers],
      inDialog: a.closest('dialog')?.getAttribute('aria-labelledby') ?? null,
      inPanel: a.closest('.day-panel') !== null,
      dataDay: a.closest('[data-day]')?.getAttribute('data-day') ?? null,
      scrollY: Math.round(scrollY),
    };
  });
}

async function measureRing(page: Page, info: StopInfo, dsf: number) {
  if (info.rect === undefined) return null;
  const focused = decodePng(await page.screenshot({ animations: 'disabled' }));
  await page.evaluate(() => {
    const a = document.activeElement as HTMLElement;
    (window as unknown as { __probeEl: HTMLElement }).__probeEl = a;
    a.blur();
  });
  const unfocused = decodePng(await page.screenshot({ animations: 'disabled' }));
  const back = await page.evaluate(() => {
    const el = (window as unknown as { __probeEl: HTMLElement }).__probeEl;
    el.focus({ preventScroll: true });
    return document.activeElement === el;
  });
  return { refocused: back, ...ringOf(focused, unfocused, info.rect, dsf) };
}

interface WalkOptions {
  back?: boolean;
  max: number;
  measure?: boolean;
  leaveWhen?: (info: StopInfo) => boolean;
  /** Screenshot name prefix for stops that are entirely covered or show no ring. */
  label?: string;
}

/** Parks the pointer on a non-interactive spot of the sticky top bar, so no control is hovered while scrolling. */
async function parkPointer(page: Page) {
  const spot = await page.evaluate(() => {
    const bar = document.querySelector('.shell-bar')?.getBoundingClientRect();
    if (bar === undefined) return null;
    for (const fraction of [0.5, 0.45, 0.55, 0.4, 0.6, 0.35, 0.65]) {
      const x = bar.left + bar.width * fraction;
      const y = bar.top + 3;
      const hit = document.elementFromPoint(x, y);
      if (hit !== null && hit.closest('button, a, input, select, textarea, label, [tabindex]') === null) return { x, y, hit: `${hit.tagName}.${typeof hit.className === 'string' ? hit.className.split(' ')[0] : ''}` };
    }
    return null;
  });
  if (spot !== null) await page.mouse.move(spot.x, spot.y);
  return spot;
}

async function walk(page: Page, dsf: number, options: WalkOptions) {
  let shots = 0;
  const stops: Array<Record<string, unknown>> = [];
  const seen = new Set<number>();
  let lastId: number | undefined;
  let repeats = 0;
  for (let index = 0; index < options.max; index += 1) {
    await page.keyboard.press(options.back === true ? 'Shift+Tab' : 'Tab');
    // Let the shared 300 ms transition finish, so the recorded box-shadow is the settled ring.
    await page.waitForTimeout(380);
    const info = await stopInfo(page);
    if (info.body === true) {
      stops.push({ index, body: true, hasFocus: info.hasFocus });
      break;
    }
    // A date input keeps focus while Tab moves through its month, day and year segments.
    if (info.id !== undefined && info.id === lastId && repeats < 5) {
      repeats += 1;
      continue;
    }
    repeats = 0;
    lastId = info.id;
    if (info.id !== undefined && seen.has(info.id)) {
      stops.push({ index, cycleTo: info.id, name: info.name });
      break;
    }
    if (info.id !== undefined) seen.add(info.id);
    if (options.leaveWhen?.(info) === true) {
      stops.push({ index, left: true, name: info.name, tag: info.tag });
      break;
    }
    if (options.label !== undefined && shots < 6 && info.vis === 0) {
      shots += 1;
      await page.screenshot({ path: shot(`${options.label}-hidden-${index}-${String(info.name).replace(/[^A-Za-z0-9]+/g, '-').slice(0, 30)}-synthetic.png`) });
    }
    const ring = options.measure === false ? null : await measureRing(page, info, dsf);
    stops.push({ index, ...info, ring });
  }
  return stops;
}

function summarise(stops: Array<Record<string, unknown>>) {
  const measured = stops.filter((stop) => stop.ring !== undefined && stop.ring !== null) as Array<Record<string, unknown> & { ring: ReturnType<typeof ringOf> & { refocused: boolean }; focusVisible: boolean; inViewport: boolean; vis: number; pts: number; name: string; tag: string }>;
  const outer = measured.map((stop) => stop.ring.minOuter).filter((value): value is number => value !== null);
  const problems = measured
    .filter((stop) => !stop.focusVisible || stop.ring.sidesPresent < 4 || (stop.ring.minOuter ?? 0) < 3 || stop.vis === 0 || stop.vis < stop.pts || !stop.inViewport)
    .map((stop) => ({
      name: stop.name,
      tag: stop.tag,
      focusVisible: stop.focusVisible,
      sidesPresent: stop.ring.sidesPresent,
      minOuter: stop.ring.minOuter,
      minInner: stop.ring.minInner,
      vis: `${stop.vis}/${stop.pts}`,
      inViewport: stop.inViewport,
      covers: stop.covers,
      absentSides: stop.ring.sides.filter((side) => side.change < 3).map((side) => `${side.side}:${side.change}`),
    }));
  return {
    stops: stops.length,
    measured: measured.length,
    focusVisibleFalse: measured.filter((stop) => !stop.focusVisible).length,
    noRing: measured.filter((stop) => stop.ring.sidesPresent === 0).length,
    minOuter: outer.length === 0 ? null : Math.min(...outer),
    entirelyHidden: measured.filter((stop) => stop.vis === 0).map((stop) => stop.name),
    problems,
    end: stops.at(-1),
  };
}

function compact(stops: Array<Record<string, unknown>>) {
  return stops.map((stop) => {
    const ring = stop.ring as (ReturnType<typeof ringOf> & { refocused: boolean }) | null | undefined;
    return {
      index: stop.index,
      name: stop.name,
      tag: stop.tag,
      role: stop.role,
      fv: stop.focusVisible,
      vis: stop.vis === undefined ? undefined : `${String(stop.vis)}/${String(stop.pts)}`,
      inViewport: stop.inViewport,
      covers: stop.covers,
      boxShadow: stop.boxShadow,
      rect: stop.rect,
      day: stop.dataDay,
      dialog: stop.inDialog,
      panel: stop.inPanel,
      ring:
        ring === null || ring === undefined
          ? null
          : {
              refocused: ring.refocused,
              sidesPresent: ring.sidesPresent,
              minOuter: ring.minOuter,
              minInner: ring.minInner,
              minChange: ring.minChange,
              sides: ring.sides.map((side) => `${side.side} ch=${side.change} d=${side.d} ring=${side.ring} outer=${side.outer}(${side.outerRatio}) inner=${side.inner}(${side.innerRatio}) under=${side.under} band=${side.bandCss}`),
            },
      body: stop.body,
      hasFocus: stop.hasFocus,
      cycleTo: stop.cycleTo,
      left: stop.left,
    };
  });
}

async function freshStart(page: Page, hash: string) {
  await page.goto(`/${hash}`);
  await page.evaluate(() => {
    window.scrollTo(0, 0);
    (document.activeElement as HTMLElement | null)?.blur();
  });
}

async function seeded(person: (name?: string) => Promise<Person>) {
  const someone = await person();
  const days = await someone.api.displayedPeriodFreeWorkdays();
  const [completeDay = '', pendingDay = ''] = days;
  expect(completeDay, 'a past free workday').not.toBe('');
  await someone.api.seedCompleteDay(completeDay);
  if (pendingDay !== '') await someone.api.seedUnconfirmedBreaksDay(pendingDay);
  return { someone, completeDay, pendingDay };
}

/* ---- P6: zone note (B4-02), names, check words and shapes, text contrast ------------------------ */

test('P6 zone note, names, status words, text contrast', async ({ page, person, signInAs }, testInfo) => {
  const project = testInfo.project.name;
  const { someone, completeDay, pendingDay } = await seeded(person);
  await signInAs(someone.account);
  await freshStart(page, '#/timesheet');
  await expect(page.locator('[data-day]')).toHaveCount(14);
  const sheet = await someone.api.call<{ period: { due_at_utc: string; due_local_date: string; due_local_time: string }; reporting_zone: string }>('GET', `/api/timesheets/${someone.payrollDate}`);
  const zone = await page.evaluate(() => ({
    viewing: document.querySelector('[data-period-zone]')?.textContent,
    due: document.querySelector('[data-period-due]')?.textContent,
    noteTerms: [...document.querySelectorAll('[data-zone-note] dt')].map((element) => element.textContent),
    noteValues: [...document.querySelectorAll('[data-zone-note] dd')].map((element) => element.textContent),
    hint: document.querySelector('[data-zone-note] .hint')?.textContent,
    periodName: document.querySelector('.period-name')?.textContent,
    payroll: document.querySelector('.period-meta b')?.textContent,
  }));
  const parts = Object.fromEntries(
    new Intl.DateTimeFormat('en-US', { timeZone: 'Asia/Ho_Chi_Minh', weekday: 'short', year: 'numeric', month: '2-digit', day: '2-digit', hour: '2-digit', minute: '2-digit', hourCycle: 'h23' })
      .formatToParts(new Date(sheet.period.due_at_utc))
      .map((part) => [part.type, part.value]),
  );
  const expectedDue = `${parts.weekday} ${parts.month}/${parts.day}/${parts.year}, ${parts.hour}:${parts.minute}`;
  await page.locator('.period-bar').screenshot({ path: shot(`p6-${project}-zone-note-synthetic.png`) });
  // Names of the 14 day elements and their Edit buttons, the Check words and shapes.
  const names = await page.locator('[data-day]').evaluateAll((elements) => elements.map((element) => element.getAttribute('aria-label')));
  const checks = await page.locator('[data-check]').evaluateAll((elements) =>
    elements.map((element) => ({ key: element.getAttribute('data-check'), text: (element as HTMLElement).innerText.trim(), shape: element.querySelector('.shape') !== null })),
  );
  // Rendered text contrast (computed colours over the composited ancestor backgrounds), both themes.
  const textContrast: Record<string, unknown> = {};
  for (const scheme of SCHEMES) {
    await page.emulateMedia({ colorScheme: scheme });
    await page.waitForTimeout(400);
    textContrast[scheme] = await page.evaluate(() => {
      const parse = (text: string): [number, number, number, number] => {
        const n = (/rgba?\(([^)]*)\)/.exec(text)?.[1] ?? '0 0 0 0').split(/[\s,/]+/).filter((part) => part !== '').map(Number);
        return [n[0] ?? 0, n[1] ?? 0, n[2] ?? 0, n[3] ?? 1];
      };
      const over = (top: number[], bottom: number[]) => {
        const a = (top[3] ?? 1) + (bottom[3] ?? 1) * (1 - (top[3] ?? 1));
        const mix = (i: number) => ((top[i] ?? 0) * (top[3] ?? 1) + (bottom[i] ?? 0) * (bottom[3] ?? 1) * (1 - (top[3] ?? 1))) / (a === 0 ? 1 : a);
        return [mix(0), mix(1), mix(2), a];
      };
      const lum = (c: number[]) => {
        const f = (v: number) => {
          const s = v / 255;
          return s <= 0.04045 ? s / 12.92 : ((s + 0.055) / 1.055) ** 2.4;
        };
        return 0.2126 * f(c[0] ?? 0) + 0.7152 * f(c[1] ?? 0) + 0.0722 * f(c[2] ?? 0);
      };
      const ratio = (a: number[], b: number[]) => {
        const [x, y] = [lum(a), lum(b)].sort((p, q) => q - p);
        return ((x ?? 0) + 0.05) / ((y ?? 0) + 0.05);
      };
      const rows: Array<{ text: string; ratio: number; size: number; weight: number }> = [];
      for (const element of document.querySelectorAll<HTMLElement>('main *, header *, nav *')) {
        const own = [...element.childNodes].some((node) => node.nodeType === 3 && (node.textContent ?? '').trim() !== '');
        if (!own) continue;
        const r = element.getBoundingClientRect();
        const cs = getComputedStyle(element);
        if (r.width === 0 || r.height === 0 || cs.visibility === 'hidden' || cs.clipPath !== 'none') continue;
        const chain: string[] = [];
        for (let node: HTMLElement | null = element; node !== null; node = node.parentElement) chain.push(getComputedStyle(node).backgroundColor);
        const background = chain.reduceRight<number[]>((below, css) => over(parse(css), below), [255, 255, 255, 1]);
        const fg = over(parse(cs.color), background);
        rows.push({ text: (element.textContent ?? '').trim().slice(0, 30), ratio: Math.round(ratio(fg, background) * 100) / 100, size: parseFloat(cs.fontSize), weight: Number(cs.fontWeight) });
      }
      const under = rows.filter((row) => row.ratio < 4.5 && !(row.size >= 24 || (row.size >= 18.66 && row.weight >= 700)));
      return { nodes: rows.length, min: Math.min(...rows.map((row) => row.ratio)), under };
    });
  }
  record(`p6-${project}`, {
    completeDay,
    pendingDay,
    due: { utc: sheet.period.due_at_utc, local: `${sheet.period.due_local_date} ${sheet.period.due_local_time}`, reportingZone: sheet.reporting_zone, expectedDisplayZoneText: expectedDue },
    zone,
    dueMatches: zone.noteValues[2] === expectedDue,
    names,
    checks,
    textContrast,
  });
});

      if (project === 'desktop') await page.setViewportSize({ width, height: 800 });
      await freshStart(page, '#/timesheet');
      await expect(page.locator('[data-day]')).toHaveCount(14);
      await parkPointer(page);
      await page.getByRole('button', { name: `Edit ${pendingDay}`, exact: true }).focus();
      await page.keyboard.press('Enter');
      const editor = page.getByRole('dialog', { name: /Day editor/ });
      await expect(editor.getByRole('heading', { name: 'Figures (computed by the server)' })).toBeVisible();
      await page.waitForTimeout(300);
      await page.keyboard.press('Tab');
      await page.waitForTimeout(380);
      const first = await stopInfo(page);
      await page.keyboard.press('Shift+Tab');
      await page.waitForTimeout(380);
      const before = await stopInfo(page);
      const panel = await editor.evaluate((element) => ({
        tabIndexProperty: (element as HTMLElement).tabIndex,
        tabindexAttribute: element.getAttribute('tabindex'),
        scrollHeight: element.scrollHeight,
        clientHeight: element.clientHeight,
        isActive: document.activeElement === element,
        focusVisible: element.matches(':focus-visible'),
        boxShadow: getComputedStyle(element).boxShadow,
        outline: `${getComputedStyle(element).outlineStyle} ${getComputedStyle(element).outlineWidth} ${getComputedStyle(element).outlineColor}`,
      }));
      await page.screenshot({ path: shot(`p8-${project}-${width}-${scheme}-editor-dialog-focused-synthetic.png`) });
      const panelRing = panel.isActive ? await measureRing(page, before, dsf) : null;
      // Arrow keys scroll the focused panel (the purpose of a focusable scroller).
      const scrollBefore = await editor.evaluate((element) => element.scrollTop);
      if (panel.isActive) await page.keyboard.press('ArrowDown');
      await page.waitForTimeout(200);
      const scrollAfter = await editor.evaluate((element) => element.scrollTop);
      dialogStops[`w${width}`] = { firstAfterHeading: first.name, afterShiftTabFromFirst: { tag: before.tag, name: before.name, fv: before.focusVisible }, panel, panelRing: panelRing && { sidesPresent: panelRing.sidesPresent, maxChange: panelRing.maxChange, minOuter: panelRing.minOuter }, scrollBefore, scrollAfter };
      await page.keyboard.press('Escape');
      await expect(editor).toHaveCount(0);
    }
    result.dialogStop = dialogStops;
    if (project === 'desktop') await page.setViewportSize({ width: 1280, height: 800 });

    // C. Sticky top bar: from the end of the page, Shift+Tab back to the week-1 dates and the period buttons.
    await freshStart(page, '#/timesheet');
    await expect(page.locator('[data-day]')).toHaveCount(14);
    await parkPointer(page);
    await page.locator('[data-signature="employee"] a, [data-sheet-review-link]').first().focus();
    await page.waitForTimeout(300);
    const sticky: unknown[] = [];
    for (let step = 0; step < 45; step += 1) {
      await page.keyboard.press('Shift+Tab');
      await page.waitForTimeout(380);
      const info = await stopInfo(page);
      if (info.body === true) break;
      const bar = await page.evaluate(() => {
        const element = document.querySelector('.shell-bar');
        const r = element?.getBoundingClientRect();
        return r === undefined ? null : { top: r.top, bottom: r.bottom, position: getComputedStyle(element as Element).position };
      });
      const entry = { step, name: info.name, rectTop: info.rect?.y, rectBottom: info.rect === undefined ? undefined : info.rect.y + info.rect.height, vis: `${info.vis}/${info.pts}`, covers: info.covers, scrollY: info.scrollY, bar };
      sticky.push(entry);
      if (info.vis === 0 && sticky.filter((item) => (item as { vis: string }).vis.startsWith('0/')).length <= 3) {
        await page.screenshot({ path: shot(`p8-${project}-${scheme}-sticky-hidden-${step}-synthetic.png`) });
      }
    }
    result.stickyBackward = sticky;
    result.scrollPadding = await page.evaluate(() => ({ top: getComputedStyle(document.documentElement).scrollPaddingTop, bottom: getComputedStyle(document.documentElement).scrollPaddingBottom }));
    record(`p8-${project}-${scheme}`, result);
  });
}

test('P9 follow-ups', async ({ page, person, signInAs }, testInfo) => {
  const project = testInfo.project.name;

  }
  const segment = await page.evaluate(() => {

    return { type: input.type, value: input.value, fv: input.matches(':focus-visible'), focusWithin: input.matches(':focus-within'), boxShadow: getComputedStyle(input).boxShadow };
  });
  result.openADayShiftTab = { name: back.name, segment, ring: await measureRing(page, back, dsf) };

  // B. The Clock out dialog: Shift+Tab from its first control; is the <dialog> a stop, and is it painted?
  await page.getByRole('button', { name: 'Clock in' }).click();

  await expect(clockOut).toBeVisible();

  await clockOut.focus();
  await page.keyboard.press('Enter');
  const dialog = page.getByRole('dialog', { name: 'Clock out' });
  await expect(dialog).toBeVisible();
  await page.waitForTimeout(400);
  const opened = await stopInfo(page);
  await page.keyboard.press('Shift+Tab');
  await page.waitForTimeout(400);
  const before = await stopInfo(page);
  const dialogState = await dialog.evaluate((element) => ({ isActive: document.activeElement === element, scrollHeight: element.scrollHeight, clientHeight: element.clientHeight, overflowY: getComputedStyle(element).overflowY, boxShadow: getComputedStyle(element).boxShadow }));
  await page.screenshot({ path: shot(`p9-${project}-clock-out-shift-tab-synthetic.png`) });
  result.clockOutDialog = { opened: opened.name, afterShiftTab: { tag: before.tag, name: before.name, fv: before.focusVisible }, dialogState };
  await page.keyboard.press('Escape');
  await expect(dialog).toHaveCount(0);

  // C. Edge clipping: "Save day fields", the last control of the editor, reached by Tab.
  await freshStart(page, '#/timesheet');
  await expect(page.locator('[data-day]')).toHaveCount(14);
  await parkPointer(page);
  await page.getByRole('button', { name: `Edit ${completeDay}`, exact: true }).focus();

  const editor = page.getByRole('dialog', { name: /Day editor/ });
  await expect(editor.getByRole('heading', { name: 'Figures (computed by the server)' })).toBeVisible();
  await editor.getByLabel('Notes').focus();
  await page.keyboard.press('Tab');
  await page.waitForTimeout(500);
  const save = await stopInfo(page);
  const panelBox = await editor.boundingBox();
  if (save.rect !== undefined) {
    await page.screenshot({ path: shot(`p9-${project}-save-day-fields-edge-synthetic.png`), clip: { x: Math.max(0, save.rect.x - 16), y: Math.max(0, save.rect.y - 16), width: save.rect.width + 32, height: Math.min(save.rect.height + 32, (page.viewportSize()?.height ?? 800) - Math.max(0, save.rect.y - 16)) } });
  }
  result.saveEdge = { name: save.name, rect: save.rect, panelBottom: panelBox === null ? null : panelBox.y + panelBox.height, ring: await measureRing(page, save, dsf) };
  await page.keyboard.press('Escape');

  // D. The tap-target loops of the specs filter visible controls: how many matched controls are not visible?
  if (project === 'mobile') {
    // Batch mode on the phone: the day button of 2026-09-30-like rows next to the selection box (crop for a visual check).
    await freshStart(page, '#/timesheet');
    await expect(page.locator('[data-day]')).toHaveCount(14);
    await page.getByRole('button', { name: 'Change several days' }).click();
    const days = await page.locator('[data-day]').evaluateAll((elements) => elements.map((element) => element.getAttribute('data-day')));

      const ring = await measureRing(page, info, dsf);
      crops.push({ day, name: info.name, minOuter: ring?.minOuter, sides: ring?.sides.map((side) => `${side.side} ch=${side.change} ring=${side.ring} outer=${side.outer}(${side.outerRatio}) under=${side.under}`) });

        const r = element.getBoundingClientRect();
        return r.width === 0 || r.height === 0;
      });
      return { all: all.length, hidden: hidden.length, hiddenNames: hidden.map((element) => `${element.tagName} ${(element.getAttribute('aria-label') ?? element.textContent ?? '').trim().slice(0, 30)}`) };
    });
    result.tapTargetFilter = counts;
  }
  record(`p9-${project}`, result);
