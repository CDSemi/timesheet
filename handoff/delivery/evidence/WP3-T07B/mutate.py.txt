saved SHA-256. Usage (repository root, NODE_EXE pointing at the Node 24 binary, TEMP/TMP outside
the repository):
python mutate.py.txt <out-file>

def sha(data: bytes) -> str:
    return hashlib.sha256(data).hexdigest()


def run_tests() -> tuple[int, str]:
    node = os.environ.get('NODE_EXE', 'node')
    result = subprocess.run([node, 'node_modules/vitest/vitest.mjs', 'run', *TESTS], capture_output=True, text=True, encoding='utf-8', errors='replace', cwd=ROOT)
    out = result.stdout + result.stderr
    summary = [line.strip() for line in out.splitlines() if line.strip().startswith(('Tests ', 'Test Files'))]
    failing = sorted({line.strip().split(' > ')[-1][:110] for line in out.splitlines() if line.strip().startswith('FAIL ') and ' > ' in line})
    detail = ' | '.join(summary)
    if failing:
        detail += '\n    failing: ' + '\n    failing: '.join(failing[:5])
    return result.returncode, detail


def main(out_path: str) -> int:
    originals: dict[str, bytes] = {}
    for _, file, _ in MUTATIONS:
        if file not in originals:
            originals[file] = (ROOT / file).read_bytes()
    lines = []
    for file, data in originals.items():
        lines.append(f'baseline sha256 {file} {sha(data)}')
    code, detail = run_tests()
    lines.append(f'baseline exit={code} {detail}')
    # Compute every mutation before any file is opened for writing.
    planned = []
    for label, file, pairs in MUTATIONS:
        text = originals[file].decode('utf-8')
        for old, new in pairs:
            if text.count(old) != 1:
                raise SystemExit(f'pattern for {label!r} matches {text.count(old)} times: {old[:60]!r}')
            text = text.replace(old, new)
        mutated = text.encode('utf-8')
        if mutated == originals[file]:
            raise SystemExit(f'mutation {label!r} changes nothing')
        planned.append((label, file, mutated))
    caught = 0
    for label, file, mutated in planned:
        target = ROOT / file
        try:
            with open(target, 'wb') as handle:
                handle.write(mutated)
            code, detail = run_tests()
        finally:
            with open(target, 'wb') as handle:
                handle.write(originals[file])
        restored = 'byte-identical' if sha((ROOT / file).read_bytes()) == sha(originals[file]) else 'NOT IDENTICAL'
        verdict = 'CAUGHT' if code != 0 else 'SURVIVED'
        caught += 1 if code != 0 else 0
        lines.append(f'{label}: exit={code} {verdict} restored={restored}\n    {detail}')
    lines.append(f'summary: {caught} of {len(planned)} mutations caught')
    Path(out_path).write_text('\n'.join(lines) + '\n', encoding='utf-8', newline='\n')
    return 0 if caught == len(planned) else 1


if __name__ == '__main__':
    sys.exit(main(sys.argv[1]))
