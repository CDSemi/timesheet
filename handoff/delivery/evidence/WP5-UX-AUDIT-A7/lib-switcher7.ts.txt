// WP5-UX-AUDIT-A7 delta probe (scope 1: the switcher, AC-16 and AC-01 after FIX8). Built on the A6 switcher probe
// (lib-switcher.ts): a grantee G receives a view-only share from owner A and an edit share from owner B; a third
// person C shares nothing. For the fixed sequence of choices (A, B, my timesheets, B, A, my timesheets) the probe
// records the address, the share bar, the heading, the owner-only controls, which owner's data is on the page and
// every API request; then the isolation. New in A7: the sequence runs three times, once per input method -
// 'keyboard' (focus the select, arrow keys to the choice, Tab to Open, Enter), 'touch' (the choice, then a tap on
// Open) and 'pointer' (the choice, then a click on Open, the A6 method) - and every landing records the switcher's
// own geometry (the shell bar, the select and Open: in the viewport, not covered at the centre). Width, height and
// device zone come from the Playwright project (390x844 and 320x640, zone note shown or not). The same steps run
// on the 255ba17 build and on the bf954c0 build (before FIX8) for comparison. Synthetic data only.
import { writeResult } from './common.ts';

export interface Switcher7Options {
  build: string;
}

const DAY_BUTTONS = 'button.sheet-date, button.sheet-day-button';
type Mode = 'keyboard' | 'touch' | 'pointer';

export function registerSwitcher7Probe(fx: any, options: Switcher7Options): void {
  const { test, expect, newPerson, statusInPage } = fx;
  test.use({ locale: 'en-US' });

  test(`switcher routes by keyboard, touch and pointer; read-only shared view, owner-only controls and isolation (${options.build})`, async ({ page, browser, adminSeed, builtServer }: any, testInfo: any) => {
    test.setTimeout(900_000);
    page.setDefaultTimeout(15_000);
    const ownerA = await newPerson(adminSeed, builtServer, { displayName: 'Synthetic Owner A' });
    const ownerB = await newPerson(adminSeed, builtServer, { displayName: 'Synthetic Owner B' });
    const grantee = await newPerson(adminSeed, builtServer, { displayName: 'Synthetic Grantee G', signature: 'none' });
    const other = await newPerson(adminSeed, builtServer, { displayName: 'Synthetic Other C', signature: 'none' });
    const notes: Record<string, string> = {};
    const sessionIds: Record<string, string> = {};
    for (const [key, person] of [['A', ownerA], ['B', ownerB], ['G', grantee], ['C', other]] as const) {
      const [day = ''] = await person.api.displayedPeriodFreeWorkdays();
      await person.api.seedCompleteDay(day);
      notes[key] = `note-${key}-${Date.now().toString(36)}-synthetic`;
      await person.api.putDay(day, { notes: notes[key] });
      sessionIds[key] = (await person.api.dayView(day)).sessions[0]?.id ?? '';
    }
    await ownerA.api.call('POST', '/api/shares', { grantee_email: grantee.account.email, items: { timesheets: 'view', ot_read: false, pdf_download: false } }, 201);
    await ownerB.api.call('POST', '/api/shares', { grantee_email: grantee.account.email, items: { timesheets: 'edit', ot_read: false, pdf_download: false } }, 201);
    const ids: Record<string, string> = { A: ownerA.account.id, B: ownerB.account.id, G: grantee.account.id, C: other.account.id };
    const tokenise = (text: string) => Object.entries(ids).reduce((acc, [key, id]) => acc.split(id).join(`<${key}>`), text);

    const requests: string[] = [];
    page.on('request', (request: any) => {
      const { pathname, search } = new URL(request.url());
      if (pathname.startsWith('/api/')) requests.push(tokenise(`${request.method()} ${pathname}${search}`));
    });
    await page.emulateMedia({ reducedMotion: 'reduce' });
    await page.goto('/#/timesheet');
    await page.getByLabel('Email').fill(grantee.account.email);
    await page.getByLabel('Password').fill(grantee.account.password);
    await page.getByRole('button', { name: 'Sign in' }).click();
    await expect(page.locator('[data-day]')).toHaveCount(14);
    const select = page.locator('.shell-switcher select');
    const open = page.locator('.shell-switcher button[type="submit"]');
    await expect(select).toHaveCount(1);
    const switcherOptions = await select.locator('option').evaluateAll((list: any[]) => list.map((o) => ({ value: o.value, label: o.textContent })));
    const optionValues: string[] = switcherOptions.map((o: any) => o.value);
    const viewport = { width: await page.evaluate('window.innerWidth'), height: await page.evaluate('window.innerHeight') };
    const zone = await page.evaluate('Intl.DateTimeFormat().resolvedOptions().timeZone');

    const snapshot = async () =>
      page.evaluate(`(() => {
        const count = (sel) => document.querySelectorAll(sel).length;
        const names = [...document.querySelectorAll(${JSON.stringify(DAY_BUTTONS)})].map((b) => (b.getAttribute('aria-label') || '').split(' ')[0]);
        const bar = document.querySelector('[data-share-bar]');
        const body = document.body.innerText;
        return {
          hash: window.location.hash,
          shareBar: bar === null ? null : bar.getAttribute('data-share-bar'),
          heading: (document.querySelector('main h1, h1') || { textContent: '' }).textContent.replace(/\\s+/g, ' ').trim(),
          days: count('[data-day]'),
          clockButtons: [...document.querySelectorAll('button')].filter((b) => /^Clock (in|out)$/.test(b.textContent.trim())).length,
          openADay: count('.open-day'),
          changeSeveralDays: [...document.querySelectorAll('button')].filter((b) => b.textContent.trim() === 'Change several days').length,
          labelPickers: count('[data-label-picker]'),
          checkboxes: count('input[type=checkbox]'),
          editDayButtons: names.filter((n) => n === 'Edit').length,
          viewDayButtons: names.filter((n) => n === 'View').length,
          signatureLines: count('[data-signature], [data-sheet-review-link]'),
          zoneNote: count('[data-zone-note]'),
          switcherValue: (document.querySelector('.shell-switcher select') || { value: null }).value,
          notesOnPage: ${JSON.stringify(Object.keys(notes))}.filter((key) => body.includes(${JSON.stringify(notes)}[key])),
          sessionsOnPage: ${JSON.stringify(Object.keys(sessionIds))}.filter((key) => document.querySelector('[data-range="' + ${JSON.stringify(sessionIds)}[key] + '"]') !== null),
        };
      })()`);

    /** The switcher's geometry at the top of the page: bar height, the select and Open inside the viewport and hit at their centre. */
    const geometry = async () =>
      page.evaluate(`(() => {
        window.scrollTo(0, 0);
        const box = (el) => { if (el === null) return null; const r = el.getBoundingClientRect(); return { x: Math.round(r.x * 10) / 10, y: Math.round(r.y * 10) / 10, w: Math.round(r.width * 10) / 10, h: Math.round(r.height * 10) / 10 }; };
        const hitOk = (el) => { if (el === null) return false; const r = el.getBoundingClientRect(); const h = document.elementFromPoint(r.left + r.width / 2, r.top + r.height / 2); return h !== null && (h === el || el.contains(h)); };
        const inView = (el) => { if (el === null) return false; const r = el.getBoundingClientRect(); return r.left >= -0.5 && r.top >= -0.5 && r.right <= window.innerWidth + 0.5 && r.bottom <= window.innerHeight + 0.5; };
        const sel = document.querySelector('.shell-switcher select');
        const open = document.querySelector('.shell-switcher button[type="submit"]');
        const label = document.querySelector('.shell-switcher label .muted');
        return {
          shellBar: box(document.querySelector('.shell-bar')),
          select: box(sel), selectInView: inView(sel), selectHit: hitOk(sel),
          open: box(open), openInView: inView(open), openHit: hitOk(open),
          labelText: label === null ? null : label.textContent, labelVisible: label !== null && label.getBoundingClientRect().height > 0,
          scrollWidth: document.documentElement.scrollWidth, innerWidth: window.innerWidth,
        };
      })()`);

    async function choose(mode: Mode, value: string) {
      if (mode === 'keyboard') {
        await select.focus();
        const current = await select.inputValue();
        const diff = optionValues.indexOf(value) - optionValues.indexOf(current);
        for (let i = 0; i < Math.abs(diff); i += 1) await page.keyboard.press(diff > 0 ? 'ArrowDown' : 'ArrowUp');
      } else {
        await select.selectOption(value);
      }
    }

    async function submit(mode: Mode): Promise<Record<string, unknown>> {
      if (mode === 'keyboard') {
        await page.keyboard.press('Tab');
        const onOpen = await page.evaluate(`document.activeElement !== null && document.activeElement.matches('.shell-switcher button[type="submit"]')`);
        await page.keyboard.press('Enter');
        return { tabReachedOpen: onOpen };
      }
      if (mode === 'touch') await open.tap();
      else await open.click();
      return {};
    }

    const runs: any[] = [];
    for (const mode of ['keyboard', 'touch', 'pointer'] as const) {
      // Every run starts on the grantee's own timesheet.
      await page.evaluate("window.location.hash = '#/timesheet'");
      await expect(page.locator('[data-share-bar]')).toHaveCount(0);
      await expect(page.locator('[data-day]')).toHaveCount(14);
      await page.waitForTimeout(400);
      const steps: any[] = [];
      const targets: Array<'A' | 'B' | ''> = ['A', 'B', '', 'B', 'A', ''];
      for (const target of targets) {
        const value = target === '' ? '' : ids[target];
        const before = await snapshot();
        const geometryBefore = await geometry();
        requests.length = 0;
        await choose(mode, value);
        await page.waitForTimeout(600);
        const chosen = await select.inputValue();
        const afterChoice = { hash: tokenise((await snapshot()).hash), requests: [...requests], chosen: tokenise(chosen) };
        const how = await submit(mode);
        if (target === '') {
          await expect(page.locator('[data-share-bar]')).toHaveCount(0);
          await expect(page.getByRole('heading', { name: 'Timesheet', exact: true })).toBeVisible();
        } else {
          await expect(page.locator(`[data-share-bar="${value}"]`)).toBeVisible();
        }
        await expect(page.locator('[data-day]')).toHaveCount(14);
        await page.waitForTimeout(800);
        const after = await snapshot();
        steps.push({
          from: tokenise(before.hash),
          target: target === '' ? 'my timesheets' : target,
          geometryBefore,
          afterChoice,
          how,
          landed: { ...after, hash: tokenise(after.hash), shareBar: after.shareBar === null ? null : tokenise(after.shareBar), switcherValue: tokenise(after.switcherValue ?? '') },
          requests: [...requests].sort(),
        });
      }
      runs.push({ mode, steps, geometryEnd: await geometry() });
    }

    // Isolation: G opens C's (never shared) address; C sees no switcher and none of A's or B's data.
    requests.length = 0;
    await page.evaluate(`window.location.hash = '#/shared/${ids.C}'`);
    await page.waitForTimeout(1_500);
    const gOnC = await snapshot();
    const isolation: Record<string, unknown> = {
      grantee_on_C_address: { ...gOnC, hash: tokenise(gOnC.hash) },
      grantee_on_C_requests: [...requests].sort(),
      grantee_C_timesheet_status: await statusInPage(page, 'GET', `/api/shared/${ids.C}/timesheets/${other.payrollDate}`),
      grantee_A_put_status: await statusInPage(page, 'PUT', `/api/shared/${ids.A}/days/${(await ownerA.api.displayedPeriodFreeWorkdays())[0]}`, { category: 'Off', leave_minutes: 0, leave_kind: null, wfh: false, notes: '', expected_version: null }),
    };
    const use = testInfo.project.use as any;
    const context = await browser.newContext({ baseURL: builtServer.origin, timezoneId: use.timezoneId, locale: 'en-US', viewport: use.viewport, isMobile: use.isMobile ?? false, hasTouch: use.hasTouch ?? false });
    const third = await context.newPage();
    third.setDefaultTimeout(15_000);
    await third.goto('/#/timesheet');
    await third.getByLabel('Email').fill(other.account.email);
    await third.getByLabel('Password').fill(other.account.password);
    await third.getByRole('button', { name: 'Sign in' }).click();
    await expect(third.locator('[data-day]')).toHaveCount(14);
    isolation.C_switchers = await third.locator('.shell-switcher').count();
    await third.evaluate(`window.location.hash = '#/shared/${ids.A}'`);
    await third.waitForTimeout(1_500);
    const body = await third.locator('body').innerText();
    isolation.C_on_A_address_notes = Object.keys(notes).filter((key) => body.includes(notes[key] ?? '\u0000'));
    isolation.C_on_A_address_sessions = [];
    for (const key of Object.keys(sessionIds)) if ((await third.locator(`[data-range="${sessionIds[key]}"]`).count()) > 0) (isolation.C_on_A_address_sessions as string[]).push(key);
    isolation.C_A_timesheet_status = await statusInPage(third, 'GET', `/api/shared/${ids.A}/timesheets/${ownerA.payrollDate}`);
    isolation.C_B_timesheet_status = await statusInPage(third, 'GET', `/api/shared/${ids.B}/timesheets/${ownerB.payrollDate}`);
    await context.close();

    writeResult(`switcher7-${options.build}-${testInfo.project.name}.json`, {
      build: options.build,
      project: testInfo.project.name,
      viewport,
      zone,
      switcher_options: switcherOptions.map((o: any) => ({ value: tokenise(o.value), label: o.label })),
      runs,
      isolation,
    });

    // The facts that must hold on every build, every input method and every width.
    for (const run of runs) {
      for (const step of run.steps) {
        const landed = step.landed;
        expect(step.geometryBefore.selectInView && step.geometryBefore.selectHit && step.geometryBefore.openInView && step.geometryBefore.openHit, `${run.mode} ${step.from}: the switcher is in view and not covered`).toBe(true);
        expect(step.afterChoice.hash, `${run.mode}: the choice alone does not navigate`).toBe(step.from);
        expect(step.afterChoice.requests, `${run.mode}: the choice alone sends nothing`).toEqual([]);
        expect(step.afterChoice.chosen, `${run.mode}: the choice took the target`).toBe(step.target === 'my timesheets' ? '' : `<${step.target}>`);
        if (run.mode === 'keyboard') expect(step.how.tabReachedOpen, 'Tab from the select reaches Open').toBe(true);
        if (step.target === 'my timesheets') {
          expect(landed.hash).toBe('#/timesheet');
          expect(landed.shareBar).toBeNull();
          expect(landed.sessionsOnPage).toEqual(['G']);
          expect(landed.notesOnPage.filter((key: string) => key !== 'G')).toEqual([]);
          expect(landed.clockButtons).toBe(1);
        } else {
          expect(landed.hash).toMatch(new RegExp(`^#/shared/<${step.target}>(/timesheet)?$`));
          expect(landed.shareBar).toBe(`<${step.target}>`);
          expect(landed.sessionsOnPage).toEqual([step.target]);
          expect(landed.notesOnPage.filter((key: string) => key !== step.target)).toEqual([]);
          expect(landed.clockButtons).toBe(0);
          expect(landed.signatureLines).toBe(0);
          if (step.target === 'A') expect([landed.openADay, landed.changeSeveralDays, landed.labelPickers, landed.checkboxes, landed.editDayButtons, landed.viewDayButtons]).toEqual([0, 0, 0, 0, 0, 14]);
          if (step.target === 'B') expect([landed.openADay, landed.changeSeveralDays, landed.labelPickers, landed.editDayButtons, landed.viewDayButtons]).toEqual([1, 1, 14, 14, 0]);
          for (const request of step.requests) expect(request.startsWith(`GET /api/shared/<${step.target}>/`) || ['GET /api/shares', 'GET /api/auth/me'].includes(request), request).toBe(true);
        }
      }
    }
    expect(isolation.C_switchers).toBe(0);
    expect((isolation.C_on_A_address_notes as string[]).filter((key) => key !== 'C')).toEqual([]);
    expect((isolation.C_on_A_address_sessions as string[]).filter((key) => key !== 'C')).toEqual([]);
    expect([403, 404]).toContain(isolation.grantee_C_timesheet_status);
    expect([403, 404]).toContain(isolation.C_A_timesheet_status);
    expect([403, 404]).toContain(isolation.C_B_timesheet_status);
    expect([403, 404]).toContain(isolation.grantee_A_put_status);
    expect((isolation.grantee_on_C_address as any).notesOnPage.filter((key: string) => key === 'C')).toEqual([]);
    expect((isolation.grantee_on_C_address as any).sessionsOnPage.filter((key: string) => key === 'C')).toEqual([]);
  });
}
