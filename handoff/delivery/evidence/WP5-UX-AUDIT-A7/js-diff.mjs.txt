// Compares two built client bundles: reports their sizes, the first differing offsets with context, and whether
// they are equal once the CSS asset file names (index-<hash>.css) are replaced by a placeholder.
// Usage: node js-diff.mjs <a.js> <b.js>
import { readFileSync } from 'node:fs';

const [a, b] = process.argv.slice(2).map((p) => readFileSync(p, 'utf8'));
console.log(`sizes: ${a.length} vs ${b.length}`);
const diffs = [];
const n = Math.min(a.length, b.length);
for (let i = 0; i < n && diffs.length < 6; i += 1) {
  if (a[i] !== b[i]) {
    diffs.push(i);
    i += 200;
  }
}
for (const i of diffs) console.log(`offset ${i}:\n  a: ${JSON.stringify(a.slice(Math.max(0, i - 80), i + 80))}\n  b: ${JSON.stringify(b.slice(Math.max(0, i - 80), i + 80))}`);
const norm = (s) => s.replace(/index-[A-Za-z0-9_-]{8}\.css/g, 'index-<css>.css').replace(/index-[A-Za-z0-9_-]{8}\.js/g, 'index-<js>.js');
console.log(`equal after replacing asset names: ${norm(a) === norm(b)}`);
