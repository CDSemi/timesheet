for c in $(git rev-list $BASE..$RC); do

  git diff-tree --no-commit-id --name-status -r $c -- . ':(exclude)handoff'

echo "== git diff --stat bf954c0..255ba17 outside handoff/"
git diff --stat=200 $BASE $RC -- . ':(exclude)handoff'

git diff --stat $BASE $RC -- src/server src/domain src/client ':(exclude)src/client/styles.css' tests/e2e/fixtures.ts tests/e2e/dayButton.ts tests/client tests/integration tests/domain tests/server package.json package-lock.json vite.config.ts playwright.config.ts tsconfig.json tsconfig.server.json tsconfig.client.json tsconfig.test.json eslint.config.js scripts docs
echo "(end)"
echo "== removed lines in the delta (any file outside handoff/), expected 0"
git diff -U0 $BASE $RC -- . ':(exclude)handoff' > "$T/raw/delta-U0.txt"
echo "removed lines: $(grep -cE '^-[^-]' "$T/raw/delta-U0.txt")   added lines: $(grep -cE '^\+[^+]' "$T/raw/delta-U0.txt")"
echo "== tests/e2e/sharing.spec.ts: test( declarations and expect( counts"

echo "(end)"
echo "== src/server and src/server/pdf since 014bd47 (expected empty)"
git diff --stat 014bd47 $RC -- src/server
echo "(end)"
echo "== src/domain diff since 014bd47"
