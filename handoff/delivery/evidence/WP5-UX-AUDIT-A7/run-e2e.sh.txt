#!/usr/bin/env bash
# The area-A e2e specs on both projects (own output folder, JSON report), then AC-13 once, in the clean 255ba17 export.
source <task>/env.sh
cd "$T/export" || exit 2
mkdir -p "$T/tmp/e2e-output"
SPECS="tests/e2e/sharing.spec.ts tests/e2e/isolation.spec.ts tests/e2e/day-editor.spec.ts tests/e2e/timesheet.spec.ts tests/e2e/review.spec.ts tests/e2e/submission.spec.ts tests/e2e/pdf-visual.spec.ts"
{
  echo '$ playwright test tests/e2e/{sharing,isolation,day-editor,timesheet,review,submission,pdf-visual}.spec.ts --output <task>/tmp/e2e-output --reporter=list,json (projects desktop and mobile)'
  PLAYWRIGHT_JSON_OUTPUT_NAME="$(cygpath -w "$T/raw/04-e2e-report.json")" node node_modules/@playwright/test/cli.js test $SPECS --output "$(cygpath -w "$T/tmp/e2e-output")" --reporter=list,json
  echo "exit $?"
} > "$T/raw/04-e2e.txt" 2>&1
{
  echo '$ vitest run tests/integration/ac13-two-week.test.ts'
  node node_modules/vitest/vitest.mjs run tests/integration/ac13-two-week.test.ts
  echo "exit $?"
} > "$T/raw/05-ac13.txt" 2>&1
echo done > "$T/raw/e2e-done.txt"
