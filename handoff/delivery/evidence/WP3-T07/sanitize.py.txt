import os, re, sys
repo = sys.argv[1]
lad = os.environ['LOCALAPPDATA']
user = os.environ.get('USERNAME', '')
def clean(text):
    text = re.sub(r'\x1b\[[0-9;]*[A-Za-z]', '', text).replace('\r\n', '\n').replace('\r', '\n')
    text = ''.join(ch for ch in text if ch == '\n' or ch == '\t' or ord(ch) >= 32)
    bs = chr(92)
    for old, new in [(repo.replace('/', bs), '<project>'), (repo, '<project>'), (lad.replace(bs, '/'), '<local-appdata>'), (lad, '<local-appdata>')]:
        text = text.replace(old, new)
    if user:
        text = text.replace(user, '<user>')
    lines = [l.rstrip() for l in text.split('\n')]
    while lines and lines[-1] == '':
        lines.pop()
    return '\n'.join(lines) + '\n'
jobs = {
  '00-commands.txt': (['# WP3-T07 dependency and lockfile evidence (Node 24.21.0 by full path; npm run through it).', '', 'registry.txt', 'pkgjson.diff', 'lock.txt', 'install2.txt']),
  '01-red.txt': (['# Red-first (red.py.txt): the new test file against missing modules, then a stub renderer.', 'red.txt']),
  '02-green.txt': (['# The new test file, verbose reporter (after the final change).', 'green.txt']),
  '03-mutations.txt': (['# Mutation checks (mutate.py.txt): each mutated file is computed in memory, written, tested, restored.', 'mut.txt']),
  '04-verify.txt': (["# NODE_OPTIONS='--trace-deprecation --pending-deprecation' npm run verify (Node 24.21.0 by full path; TEMP on the local drive).", 'verify.txt']),
  '05-digest.txt': (['digest.txt']),
  '06-visual.txt': (['visual.txt']),
}
raw = sys.argv[2]; out = sys.argv[3]
for name, parts in jobs.items():
    body = []
    for part in parts:
        p = os.path.join(raw, part)
        if part and os.path.isfile(p):
            body.append(open(p, encoding='utf-8', errors='replace').read().rstrip('\n'))
        else:
            body.append(part)
    with open(os.path.join(out, name), 'w', encoding='utf-8', newline='\n') as handle:
        handle.write(clean('\n'.join(body)))
print('written', list(jobs))
