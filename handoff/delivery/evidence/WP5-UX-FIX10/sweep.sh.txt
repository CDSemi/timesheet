# Usage: bash sweep.sh <set-name> then label/instant pairs are fixed below. Full e2e (both projects), sequential.
# Each run: bash run-shift.sh <label> <instant|real> [extra playwright args]; log in logs/sweep-<label>.txt.
cd /d/.claude-tmp/timesheet/WP5-UX-FIX10
run() {
  label=$1; instant=$2; shift 2
  bash run-shift.sh "$label" "$instant" "$@" > "logs/sweep-$label.txt" 2>&1
  echo "$label finished $(date -u +%Y-%m-%dT%H:%M:%SZ)" >> logs/sweep-progress.txt
}
