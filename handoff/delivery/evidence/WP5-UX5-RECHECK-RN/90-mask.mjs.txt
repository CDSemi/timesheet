// Copy task-folder files into the evidence folder as masked LF text with exactly one final newline.
// usage: node mask.mjs <evidence-dir> <dst-name>=<src-path>[+<src-path>...] ...
// Several sources for one destination are concatenated under "==== <source name>" headers.
import { readFileSync, writeFileSync } from 'node:fs';
import { basename, join } from 'node:path';

const [evidenceDir, ...specs] = process.argv.slice(2);
const report = [];
for (const spec of specs) {
  const eq = spec.indexOf('=');
  const dst = spec.slice(0, eq);
  const srcs = spec.slice(eq + 1).split('+');
  let text = '';
  for (const src of srcs) {
    const body = readFileSync(src, 'utf8');
    text += srcs.length > 1 ? `==== ${basename(src)}\n${body}${body.endsWith('\n') ? '' : '\n'}` : body;
  }
  let masked = text
    .replace(/\u001b\[[0-9;?]*[A-Za-z]/g, '')
    .replace(/\r\n?/g, '\n')
    .replace(/[A-Za-z]:[\\/]+Users[\\/]+[^\\/\s'"]+/gi, '<user>')
    .replace(/\/[a-z]\/Users\/[^/\s'"]+/gi, '<user>')
    .replace(/<user>/gi, '<user>');
  masked = masked.replace(/\n+$/, '') + '\n';
  if (!dst.endsWith('.txt')) throw new Error(`evidence name must end in .txt: ${dst}`);
  writeFileSync(join(evidenceDir, dst), masked);
  report.push(`${dst} <- ${srcs.map((s) => basename(s)).join(' + ')} (${masked.length} chars)`);
}
console.log(report.join('\n'));
