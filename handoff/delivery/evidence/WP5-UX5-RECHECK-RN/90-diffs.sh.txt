# Scope and text diffs, read-only git queries in the repository.
. /d/.claude-tmp/timesheet/WP5-UX5-RECHECK-RN/env.sh
cd "$R" || exit 2
{
  printf 'time_utc %s\n' "$(date -u +%Y-%m-%dT%H:%M:%SZ)"
  printf '$ git log --oneline -3 %s\n' "$TARGET"
  git log --format='%H %P %s' -3 "$TARGET"
  printf '$ git diff --name-status %s..%s -- . %s\n' "$PREV" "$TARGET" "':!handoff'"
  git diff --name-status "$PREV".."$TARGET" -- . ':!handoff'
  printf 'exit %s\n' "$?"
  printf '$ git diff --stat %s..%s -- . %s\n' "$PREV" "$TARGET" "':!handoff'"
  git diff --stat "$PREV".."$TARGET" -- . ':!handoff'
  printf '$ git diff --check %s..%s -- . %s\n' "$PREV" "$TARGET" "':!handoff'"
  git diff --check "$PREV".."$TARGET" -- . ':!handoff'
  printf 'exit %s\n' "$?"
  printf '$ git diff --name-only %s..%s -- package.json package-lock.json .npmrc Dockerfile vite.config.ts src tests scripts migrations\n' "$PREV" "$TARGET"
  git diff --name-only "$PREV".."$TARGET" -- package.json package-lock.json .npmrc Dockerfile vite.config.ts src tests scripts migrations
  printf 'exit %s (no line above = no change)\n' "$?"
  printf '$ git diff --stat 1fe5ad3 %s -- Dockerfile vite.config.ts compose.yaml docker-compose.yml package.json package-lock.json\n' "$TARGET"
  git diff --stat 1fe5ad3 "$TARGET" -- Dockerfile vite.config.ts compose.yaml docker-compose.yml package.json package-lock.json
  printf 'exit %s (no line above = no change)\n' "$?"
  printf '$ git log --oneline 1fe5ad3..%s -- Dockerfile vite.config.ts\n' "$TARGET"
  git log --oneline 1fe5ad3.."$TARGET" -- Dockerfile vite.config.ts
  printf 'exit %s (no line above = no commit)\n' "$?"
  printf '$ git rev-parse 1fe5ad3\n'; git rev-parse 1fe5ad3
} > "$T/scope.raw" 2>&1
git diff "$PREV".."$TARGET" -- . ':!handoff' > "$T/doc-diff.raw" 2>&1
git diff --word-diff=plain "$PREV".."$TARGET" -- . ':!handoff' > "$T/doc-worddiff.raw" 2>&1
for f in DEVELOPMENT.md DEVELOPMENT.vi.md README.md README.vi.md docs/12_RELEASE_NOTES.md docs/12_RELEASE_NOTES.vi.md; do
  n=$(echo "$f" | tr '/' '_')
  git show "$TARGET:$f" > "$T/target-$n"
  git show "$PREV:$f" > "$T/prev-$n"
done
echo "diffs done"
