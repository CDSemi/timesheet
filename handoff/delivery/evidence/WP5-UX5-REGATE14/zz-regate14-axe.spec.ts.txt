import { appendFileSync, readFileSync } from 'node:fs';
import type { Locator, Page } from '@playwright/test';
import { expect, newPerson, test } from './fixtures.ts';

/*
 * WP5-UX5-REGATE14 probe (the verifier's own, never part of the repository): axe-core 4.14.0 (loaded from the task folder; no dependency added)
 * on the Timesheet page with the Day editor open on a past workday that has a session, light and dark, at 1440, 1280 (desktop project) and 390px
 * (mobile project): (1) as opened, before a session save; (2) after a session save ("Session saved."); (3) with the stale notice shown (the day
 * changed on another device, then "Save day fields" is refused). Tags WCAG 2.0 to 2.2 A and AA. Synthetic data only, a person of its own per test.
 */
const AXE_PATH = process.env.AXE_PATH ?? '';
const OUT = process.env.AXE_OUT ?? '';
const TAGS = ['wcag2a', 'wcag2aa', 'wcag21a', 'wcag21aa', 'wcag22aa'];
const SAVED = 'Synthetic note saved before the editor opened';
const ELSEWHERE = 'Synthetic note saved on another device';
const TYPED = 'Synthetic note typed here and not saved';
let axeSource = '';
function note(line: string): void {
  if (OUT !== '') appendFileSync(OUT, `${line}\n`);
}
const frames = (page: Page) => page.evaluate('new Promise((done) => requestAnimationFrame(() => requestAnimationFrame(done)))');

interface Result {
  version: string;
  violations: Array<{ id: string; impact: string | null; nodes: string[] }>;
  incomplete: string[];
}

async function scan(page: Page, label: string): Promise<number> {
  if (axeSource === '') axeSource = readFileSync(AXE_PATH, 'utf8');
  await frames(page);
  if (!(await page.evaluate<boolean>(`typeof window.axe !== 'undefined'`))) await page.evaluate(axeSource);
  const result = await page.evaluate<Result, string[]>(async (tags) => {
    const axe = (window as unknown as { axe: { run: (c: unknown, o: unknown) => Promise<Record<string, unknown>>; version: string } }).axe;
    const r = (await axe.run(document, { runOnly: { type: 'tag', values: tags }, resultTypes: ['violations', 'incomplete'] })) as {
      violations: Array<{ id: string; impact: string | null; nodes: Array<{ target: unknown }> }>;
      incomplete: Array<{ id: string }>;
    };
    return { version: axe.version, violations: r.violations.map((v) => ({ id: v.id, impact: v.impact, nodes: v.nodes.map((n) => JSON.stringify(n.target)) })), incomplete: [...new Set(r.incomplete.map((i) => i.id))] };
  }, TAGS);
  note(`${label}: axe ${result.version} violations=${result.violations.length} [${result.violations.map((v) => `${v.id}(${v.impact}) ${v.nodes.join(' ')}`).join('; ')}] incomplete=[${result.incomplete.join(',')}]`);
  return result.violations.length;
}

async function openDay(page: Page, date: string): Promise<Locator> {
  await page.getByLabel('Open a day').fill(date);
  await page.getByRole('button', { name: 'Open day', exact: true }).click();
  const editor = page.getByRole('dialog', { name: /Day editor/ });
  await expect(editor).toBeVisible();
  await expect(editor.getByRole('heading', { name: 'Times', exact: true })).toBeVisible();
  return editor;
}

for (const scheme of ['light', 'dark'] as const) {
  test(`axe Timesheet with the Day editor open on a past workday with a session: before and after a session save, and with the stale notice - ${scheme}`, async ({
    page,
    adminSeed,
    builtServer,
    signInPageAs,
    isMobile,
  }, testInfo) => {
    test.setTimeout(300_000);
    const project = testInfo.project.name;
    await page.emulateMedia({ colorScheme: scheme, reducedMotion: 'reduce' });
    const person = await newPerson(adminSeed, builtServer, { displayName: `Synthetic Axe ${project} ${scheme}`, signature: 'none' });
    const [date = ''] = await person.api.displayedPeriodFreeWorkdays();
    expect(date, 'a past free workday').not.toBe('');
    await person.api.seedCompleteDay(date);
    await person.api.putDay(date, { notes: SAVED });
    const widths = isMobile ? [390] : [1440, 1280];
    const ends = ['17:45', '17:50'];
    let total = 0;
    let scans = 0;
    let pass = 0;
    for (const width of widths) {
      await page.setViewportSize({ width, height: 800 });
      if (pass === 0) await signInPageAs(person.account, '#/timesheet');
      else await page.reload();
      await expect(page.locator('[data-day]')).toHaveCount(14);
      await expect.poll(() => page.evaluate<number>('window.innerWidth')).toBe(width);
      const tag = `[${project} ${scheme} ${width}]`;
      const editor = await openDay(page, date);
      await expect(editor.locator('.session-card')).toHaveCount(1);
      total += await scan(page, `${tag} Day editor open on past workday ${date === '' ? '' : 'with a session'}, as opened (before a session save)`);
      scans += 1;
      // A session save: Edit session, End time, Save, Save in "Save this session?".
      await editor.getByRole('button', { name: /^Edit session/ }).click();
      const edit = page.getByRole('dialog', { name: 'Edit session', exact: true });
      await edit.getByLabel('End time', { exact: true }).fill(ends[pass] ?? '17:55');
      await edit.getByRole('button', { name: 'Save', exact: true }).click();
      const confirm = page.getByRole('alertdialog', { name: 'Save this session?' });
      await expect(confirm).toBeVisible();
      await confirm.getByRole('button', { name: 'Save', exact: true }).click();
      await expect(confirm).toHaveCount(0);
      await expect(editor.getByRole('status').filter({ hasText: 'Session saved.' })).toBeVisible();
      total += await scan(page, `${tag} Day editor after a session save ("Session saved." shown)`);
      scans += 1;
      // The day fields change on another device, then Save day fields is refused: the stale notice.
      await person.api.putDay(date, { notes: `${ELSEWHERE} ${pass}`, wfh: pass === 0 });
      await editor.getByRole('form', { name: 'Day fields' }).getByLabel('Notes').fill(`${TYPED} ${pass}`);
      await editor.getByRole('button', { name: 'Save day fields' }).click();
      const stale = editor.getByRole('alert').filter({ hasText: 'This day changed since it was loaded' });
      await expect(stale).toBeVisible();
      total += await scan(page, `${tag} Day editor with the stale notice shown`);
      scans += 1;
      pass += 1;
    }
    note(`[${project} ${scheme}] Day editor TOTAL violations ${total} in ${scans} scans`);
    expect(total).toBe(0);
  });
}
