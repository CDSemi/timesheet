# Usage: bash digest.sh <tag>   (the export form, the ls-tree form, and npm run digest in the repository)
. <task>/env.sh
TAG=$1
D=$T/tmp-digest-$TAG
mkdir -p $D
cd $T/export
find . -type f -not -path './handoff/*' -not -path './node_modules/*' -not -path './dist/*' -not -path './.git/*' -not -path './test-results/*' -not -path './playwright-report/*' | sed 's|^\./||' | LC_ALL=C sort > $D/ex-files.txt
rm -f $D/ex-hashes.txt
while IFS= read -r f; do
  h=$(git hash-object --no-filters -- "$f")
  printf '%s %s\n' "$f" "$h" >> $D/ex-hashes.txt
done < $D/ex-files.txt
LC_ALL=C sort $D/ex-hashes.txt > $D/ex-sorted.txt
echo "export form: $(sha256sum $D/ex-sorted.txt | cut -d' ' -f1) lines $(wc -l < $D/ex-sorted.txt)" > $T/out/01-digest-export-$TAG.txt
cd $REPO
git ls-tree -r --format="%(path) %(objectname)" $TARGET | grep -v '^handoff/' | LC_ALL=C sort > $D/lstree-sorted.txt
echo "ls-tree form: $(sha256sum $D/lstree-sorted.txt | cut -d' ' -f1) lines $(wc -l < $D/lstree-sorted.txt)" > $T/out/01-digest-lstree-$TAG.txt
cmp $D/ex-sorted.txt $D/lstree-sorted.txt && echo "export list byte-identical to ls-tree list" >> $T/out/01-digest-lstree-$TAG.txt
cd $REPO
npm run digest > $T/out/01-digest-npm-$TAG.txt 2>&1; echo "npm run digest (repository) exit $?" >> $T/out/01-digest-npm-$TAG.txt
cat $T/out/01-digest-export-$TAG.txt $T/out/01-digest-lstree-$TAG.txt $T/out/01-digest-npm-$TAG.txt
