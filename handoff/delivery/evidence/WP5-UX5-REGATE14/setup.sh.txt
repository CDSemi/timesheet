. <task>/env.sh
cd $REPO
git rev-parse HEAD > $T/out/00-head.txt 2>&1
git rev-parse origin/main >> $T/out/00-head.txt 2>&1
git branch --show-current >> $T/out/00-head.txt 2>&1
git log -1 --format='%H %s' >> $T/out/00-head.txt 2>&1
git status --short -- . ':!handoff' > $T/out/00-nonhandoff-status-before.txt 2>&1
git diff --name-status $PREV..$TARGET -- . ':!handoff' > $T/out/20-namestatus.txt 2>&1
git diff --stat $PREV..$TARGET -- . ':!handoff' > $T/out/20-diffstat-nonhandoff.txt 2>&1
git diff --stat $PREV..$TARGET -- package.json package-lock.json > $T/out/25-lock-and-scope.txt 2>&1
echo "lock/package diff lines: $(wc -l < $T/out/25-lock-and-scope.txt)" >> $T/out/25-lock-and-scope.txt
git diff $PREV..$TARGET -- src > $T/out/22-src-diff.txt 2>&1
git diff $PREV..$TARGET -- tests > $T/out/22-tests-diff.txt 2>&1
git diff $PREV..$TARGET -- docs > $T/out/22-docs-diff.txt 2>&1
git diff $PREV..$TARGET -- scripts > $T/out/22-scripts-diff.txt 2>&1
git diff --check $PREV..$TARGET > $T/out/21-diff-check.txt 2>&1; echo "diff --check exit $?" >> $T/out/21-diff-check.txt
git diff --name-only $PREV..$TARGET -- handoff > $T/out/20-handoff-paths.txt 2>&1
git archive $TARGET | tar -x -C $T/export
git archive $PREV | tar -x -C $T/export-prev
ls $T/export
cat $T/out/00-head.txt $T/out/20-namestatus.txt $T/out/00-nonhandoff-status-before.txt $T/out/25-lock-and-scope.txt $T/out/21-diff-check.txt
