/*
 * WP5-UX3-AUDIT-A items 3 and 10 (task-local probe, never in the repository): the identity block and account menu by

test.use({ locale: 'en-US', timezoneId: 'America/Los_Angeles' });

const SHARE_ITEMS = { timesheets: 'view', ot_read: false, pdf_download: false };

async function focusInfo(page: Page): Promise<string> {
  return page.evaluate(() => {
    const a = document.activeElement as HTMLElement | null;
    if (a === null) return 'null';
    const name = a.tagName.toLowerCase();
    const text = (a.getAttribute('aria-label') ?? a.textContent ?? '').trim().replace(/\s+/g, ' ').slice(0, 50);
    const fv = a.matches(':focus-visible');
    const shadow = getComputedStyle(a).boxShadow;
    return `${name}${a.id ? `#${a.id}` : ''} "${text}" focus-visible=${fv} ring=${shadow === 'none' ? 'none' : 'painted'}`;
  });
}

test('desktop: identity button, account menu by keyboard and pointer, focus after each entry', async ({ page, builtServer, signInPageAs, isMobile }) => {
  test.skip(isMobile, 'desktop');
  test.setTimeout(180_000);
  const admin = builtServer.credentials.admin;
  await page.setViewportSize({ width: 1440, height: 900 });
  await signInPageAs(admin);
  await expect(page.locator('[data-day]')).toHaveCount(14);
  const identity = page.getByRole('banner').getByRole('button', { name: /, account menu$/ });
  note('shell.txt', `identity accessible name: ${await identity.evaluate((e) => e.textContent?.replace(/\s+/g, ' ').trim())} | aria-expanded ${await identity.getAttribute('aria-expanded')} aria-controls ${await identity.getAttribute('aria-controls')}`);

  // Tab from the top of the page to the identity button: the bar order.
  await page.evaluate(() => (document.activeElement as HTMLElement | null)?.blur());
  await page.keyboard.press('Tab');
  const order: string[] = [];
  for (let i = 0; i < 12; i += 1) {
    order.push(await focusInfo(page));
    if (await identity.evaluate((e) => e === document.activeElement)) break;
    await page.keyboard.press('Tab');
  }
  note('shell.txt', `bar tab order to the identity button: ${order.join(' -> ')}`);

  // Enter opens and focuses the first entry; Tab order inside; Tab past Sign out closes.
  await identity.focus();
  await page.keyboard.press('Enter');
  const menu = page.locator('#account-menu');
  await expect(menu).toBeVisible();
  const inside: string[] = [await focusInfo(page)];
  for (let i = 0; i < 8; i += 1) {
    await page.keyboard.press('Tab');
    inside.push(await focusInfo(page));
    if (!(await menu.isVisible())) break;
  }
  note('shell.txt', `Enter opens; focus then Tab: ${inside.join(' -> ')}; menu visible after tabbing past: ${await menu.isVisible()} aria-expanded ${await identity.getAttribute('aria-expanded')}`);

  // Escape returns focus to the button.
  await identity.focus();
  await page.keyboard.press('Enter');
  await expect(menu).toBeVisible();
  await page.keyboard.press('Tab');
  await page.keyboard.press('Escape');
  note('shell.txt', `Escape from the second entry: menu visible ${await menu.isVisible()}, focus ${await focusInfo(page)}`);

  // Outside click closes; hash change closes.
  await identity.click();
  await expect(menu).toBeVisible();
  note('shell.txt', `pointer opening keeps focus on: ${await focusInfo(page)}`);
  await page.getByRole('heading', { level: 1 }).click();
  note('shell.txt', `click outside: menu visible ${await menu.isVisible()}`);
  await identity.click();
  await expect(menu).toBeVisible();
  await page.evaluate(() => {
    window.location.hash = '#/ot';
  });
  await expect(page.getByRole('heading', { level: 1 })).toHaveText(/Overtime/);
  note('shell.txt', `hash change to #/ot: menu visible ${await menu.isVisible()}`);

  // Entries by keyboard: the new page's h1, and whether it keeps focus after the data loads.
  for (const [entry, heading] of [
    ['Profile', 'Profile'],
    ['Settings', 'Settings'],
    ['Administration', 'Administration'],
  ] as const) {
    await identity.focus();
    await page.keyboard.press('Enter');
    await expect(menu).toBeVisible();
    await menu.getByRole('link', { name: entry, exact: true }).focus();
    await page.keyboard.press('Enter');
    await expect(page.getByRole('heading', { level: 1, name: heading })).toBeVisible();
    await expect.poll(async () => page.evaluate(() => document.activeElement?.tagName ?? '')).toBe('H1');
    const right = await focusInfo(page);
    await page.waitForTimeout(2500);
    const later = await focusInfo(page);
    note('shell.txt', `keyboard entry ${entry}: at once ${right}; after 2.5 s ${later}; hash ${await page.evaluate(() => location.hash)}`);
  }
  // Same page: Administration again returns focus to the identity button.
  await identity.focus();
  await page.keyboard.press('Enter');
  await menu.getByRole('link', { name: 'Administration', exact: true }).focus();
  await page.keyboard.press('Enter');
  await page.waitForTimeout(300);
  note('shell.txt', `same-page entry Administration: focus ${await focusInfo(page)} menu visible ${await menu.isVisible()}`);

  // Pointer entry: the heading takes focus without a ring.
  await page.evaluate(() => {
    window.location.hash = '#/timesheet';
  });
  await expect(page.locator('[data-day]')).toHaveCount(14);
  await identity.click();
  await menu.getByRole('link', { name: 'Profile', exact: true }).click();
  await expect(page.getByRole('heading', { level: 1, name: 'Profile' })).toBeVisible();
  await page.waitForTimeout(400);
  note('shell.txt', `pointer entry Profile: focus ${await focusInfo(page)}`);

  // A slow Administration chunk (3 s): the heading still takes focus; a 7 s chunk: the shell gives up after 5 s.
  for (const delay of [3000, 7000]) {
    await page.evaluate(() => {
      window.location.hash = '#/timesheet';
    });
    await page.reload();
    await expect(page.locator('[data-day]')).toHaveCount(14);
    await page.route('**/assets/AdminScreen-*.js', async (route) => {
      await new Promise((resolve) => setTimeout(resolve, delay));
      await route.continue();
    });
    await identity.focus();
    await page.keyboard.press('Enter');
    await menu.getByRole('link', { name: 'Administration', exact: true }).focus();
    await page.keyboard.press('Enter');
    await expect(page.getByRole('heading', { level: 1, name: 'Administration' })).toBeVisible({ timeout: 15_000 });
    await page.waitForTimeout(1000);
    note('shell.txt', `Administration chunk delayed ${delay} ms: focus after load ${await focusInfo(page)}`);
    await page.unroute('**/assets/AdminScreen-*.js');
  }
  // A person who moves focus while the chunk loads keeps it.
  await page.evaluate(() => {
    window.location.hash = '#/timesheet';
  });
  await page.reload();
  await expect(page.locator('[data-day]')).toHaveCount(14);
  await page.route('**/assets/AdminScreen-*.js', async (route) => {
    await new Promise((resolve) => setTimeout(resolve, 2500));
    await route.continue();
  });
  await identity.focus();
  await page.keyboard.press('Enter');
  await menu.getByRole('link', { name: 'Administration', exact: true }).focus();
  await page.keyboard.press('Enter');
  await page.waitForTimeout(300);
  await page.keyboard.press('Tab');
  const moved = await focusInfo(page);
  await expect(page.getByRole('heading', { level: 1, name: 'Administration' })).toBeVisible({ timeout: 15_000 });
  await page.waitForTimeout(800);
  note('shell.txt', `moved focus to ${moved} while loading; after load focus ${await focusInfo(page)}`);
  await page.unroute('**/assets/AdminScreen-*.js');
});

test('phone: More panel by keyboard, Escape, entries and focus', async ({ page, builtServer, signInPageAs, isMobile }) => {
  test.skip(!isMobile, 'phone');
  test.setTimeout(120_000);
  await signInPageAs(builtServer.credentials.admin);
  await expect(page.locator('[data-day]')).toHaveCount(14);
  const more = page.getByRole('navigation', { name: 'Main' }).getByRole('button', { name: 'More' });
  await more.focus();
  await page.keyboard.press('Enter');
  const panel = page.locator('#more-panel');
  await expect(panel).toBeVisible();
  note('shell.txt', `phone More by Enter: focus ${await focusInfo(page)}`);
  await page.keyboard.press('Tab');
  note('shell.txt', `phone More, Tab: focus ${await focusInfo(page)}`);
  await page.keyboard.press('Escape');
  note('shell.txt', `phone More, Escape: panel visible ${await panel.isVisible()} focus ${await focusInfo(page)}`);
  for (const [entry, heading] of [
    ['Profile', 'Profile'],
    ['Settings', 'Settings'],
    ['Import', 'Import and opening balance'],
    ['Administration', 'Administration'],
  ] as const) {
    await more.focus();
    await page.keyboard.press('Enter');
    await panel.getByRole('link', { name: entry, exact: true }).focus();
    await page.keyboard.press('Enter');
    await expect(page.getByRole('heading', { level: 1, name: heading })).toBeVisible();
    await page.waitForTimeout(1500);
    note('shell.txt', `phone keyboard entry ${entry}: focus after 1.5 s ${await focusInfo(page)}`);
  }
  await more.focus();
  await page.keyboard.press('Enter');
  await panel.getByRole('link', { name: 'Administration', exact: true }).focus();
  await page.keyboard.press('Enter');
  await page.waitForTimeout(300);
  note('shell.txt', `phone same-page entry: focus ${await focusInfo(page)} panel visible ${await panel.isVisible()}`);
  // Outside press closes.
  await more.click();
  await expect(panel).toBeVisible();
  await page.getByRole('heading', { level: 1 }).click({ force: true });
  note('shell.txt', `phone outside press: panel visible ${await panel.isVisible()}`);
});

test('wordmark at its edges, with and without a share; Settings sub-pages and redirects; the employee and /api/admin', async ({
  page,
  adminSeed,
  builtServer,
  signInPageAs,
  isMobile,
}) => {
  test.skip(isMobile, 'desktop');
  test.setTimeout(180_000);
  const me = await newPerson(adminSeed, builtServer, { displayName: 'Synthetic Wordmark Reader', signature: 'none' });
  const owner = await newPerson(adminSeed, builtServer, { displayName: 'Owner', signature: 'none' });
  const adminRequests: string[] = [];
  page.on('request', (request) => {
    if (request.url().includes('/api/admin/')) adminRequests.push(request.url().replace(/^https?:\/\/[^/]+/, ''));
  });
  await page.setViewportSize({ width: 1440, height: 900 });
  await signInPageAs(me.account);
  await expect(page.locator('[data-day]')).toHaveCount(14);
  const wordmark = page.locator('header.shell-bar .wordmark');
  const report = async (label: string) => {
    const bar = await page.locator('header.shell-bar').boundingBox();
    const shown = await wordmark.isVisible();
    const overflow = await page.evaluate('document.documentElement.scrollWidth > document.documentElement.clientWidth');
    return `${label}: wordmark ${shown ? 'shown' : 'hidden'}, bar ${bar?.height}px, page overflow ${overflow}`;
  };
  const lines: string[] = [];
  for (const width of [767, 768, 1023, 1024, 1279, 1280, 1920]) {
    await page.setViewportSize({ width, height: 900 });
    await page.waitForTimeout(150);
    lines.push(await report(`no share ${width}`));
  }
  await owner.api.call('POST', '/api/shares', { grantee_email: me.account.email, items: SHARE_ITEMS }, 201);
  await page.reload();
  await expect(page.locator('header.shell-bar .shell-switcher')).toBeVisible();
  for (const width of [1023, 1024, 1279, 1280, 1281, 1920]) {
    await page.setViewportSize({ width, height: 900 });
    await page.waitForTimeout(150);
    lines.push(await report(`share ${width}`));
  }
  note('shell.txt', `wordmark: ${lines.join(' | ')}`);

  // Settings sub-pages and redirects as an employee.
  await page.setViewportSize({ width: 1280, height: 900 });
  const visits: string[] = [];
  for (const hash of ['#/settings', '#/settings/profile', '#/settings/admin', '#/admin', '#/import']) {
    await page.goto(`/${hash}`);
    await page.waitForTimeout(1200);
    const h1 = await page.getByRole('heading', { level: 1 }).first().textContent();
    const nav = await page.getByRole('navigation', { name: 'Settings sections' }).getByRole('link').allTextContents().catch(() => []);
    visits.push(`${hash} -> ${await page.evaluate(() => location.hash)} h1 "${h1}" sections [${nav.join(', ')}]`);
  }
  note('shell.txt', `employee visits: ${visits.join(' | ')}; /api/admin/ requests: ${adminRequests.length} ${adminRequests.join(' ')}`);
  // The read-only Profile: no input, no button inside the profile card.
  await page.goto('/#/settings/profile');
  await expect(page.getByRole('heading', { level: 1, name: 'Profile' })).toBeVisible();
  const editable = await page.locator('.profile-screen input, .profile-screen textarea, .profile-screen select, .profile-screen button').count();
  note('shell.txt', `profile editable controls: ${editable}; facts: ${(await page.locator('.profile-facts').innerText()).replace(/\s+/g, ' ')}`);
  // Back button: #/admin for an administrator replaces the address (checked by settings-routes.spec.ts); the employee goes to the timesheet.
});
