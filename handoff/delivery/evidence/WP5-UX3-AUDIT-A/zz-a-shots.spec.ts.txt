import { expect, newPerson, openAccount, test } from './fixtures.ts';
import { namedDayButton } from './dayButton.ts';
import { note, shot } from './zz-alib.ts';

/*
 * WP5-UX3-AUDIT-A item 1 (task-local probe, never in the repository): synthetic screenshots of the owner's items for a visual
 * judgement, plus a few measurements (title row, meta row, toolbar gap, pills).
 */
test.use({ locale: 'en-US', timezoneId: 'America/Los_Angeles' });

test('desktop shots: sheet with the editor beside, account menu, floating panel, light and dark', async ({ page, adminSeed, builtServer, signInPageAs, isMobile }) => {
  test.skip(isMobile, 'desktop');
  test.setTimeout(240_000);
  const me = await newPerson(adminSeed, builtServer, { displayName: 'Synthetic Owner View', signature: 'none' });
  const free = await me.api.displayedPeriodFreeWorkdays();
  if (free[0] !== undefined) await me.api.seedCompleteDay(free[0]);
  if (free[1] !== undefined) await me.api.seedUnconfirmedBreaksDay(free[1]);
  for (const theme of ['light', 'dark'] as const) {
    await page.emulateMedia({ colorScheme: theme, reducedMotion: 'reduce' });
    await page.setViewportSize({ width: 1440, height: 1000 });
    if (theme === 'light') await signInPageAs(me.account);
    else await page.reload();
    await expect(page.locator('[data-day]')).toHaveCount(14);
    // Measurements of the title row, meta row and toolbar (U-07, U-08, U-10, U-11, U-12).
    const facts = await page.evaluate(() => {
      const h1 = document.querySelector('main h1')?.getBoundingClientRect();
      const clock = document.querySelector('.page-head .clock-group, .page-head [role="region"]')?.getBoundingClientRect();
      const nameLine = document.querySelector('.page-head p');
      const meta = [...document.querySelectorAll('.period-meta > *, .period-bar p > span')].map((e) => Math.round(e.getBoundingClientRect().top));
      const sw = document.querySelector('.tools .switch-label, .tools label:has(input[role="switch"])')?.getBoundingClientRect();
      const btn = [...document.querySelectorAll('.tools button')].find((b) => b.textContent?.includes('Change several days'))?.getBoundingClientRect();
      const od = document.querySelector('.tools .open-day');
      const odText = od?.querySelector('.open-day-text')?.getBoundingClientRect();
      const odField = od?.querySelector('input')?.getBoundingClientRect();
      return {
        titleRow: h1 && clock ? `h1 centre ${(h1.top + h1.height / 2).toFixed(1)} clock centre ${(clock.top + clock.height / 2).toFixed(1)} clock right ${clock.right.toFixed(0)} of ${document.documentElement.clientWidth}` : `h1 ${!!h1} clock ${!!clock}`,
        nameLine: nameLine?.textContent ?? 'none',
        metaRows: [...new Set(meta)].length,
        gap: sw && btn ? (btn.left - sw.right).toFixed(1) : 'n/a',
        openDay: odText && odField ? `text centre ${(odText.top + odText.height / 2).toFixed(1)} field centre ${(odField.top + odField.height / 2).toFixed(1)}` : 'n/a',
      };
    });
    note('shots.txt', `${theme} 1440: ${JSON.stringify(facts)}`);
    await namedDayButton(page, free[1] ?? '', 'Edit').click();
    const editor = page.getByRole('dialog', { name: /Day editor/ });
    await expect(editor).toBeVisible();
    await page.mouse.move(0, 0);
    await page.waitForTimeout(400);
    await page.screenshot({ path: shot(`a-sheet-editor-1440-${theme}-synthetic.png`), animations: 'disabled' });
    await page.keyboard.press('Escape');
    await openAccount(page);
    await page.waitForTimeout(300);
    await page.screenshot({ path: shot(`a-account-menu-1440-${theme}-synthetic.png`), clip: { x: 640, y: 0, width: 800, height: 520 }, animations: 'disabled' });
    await page.keyboard.press('Escape');
    // Batch run of three plus the editor column.
    await page.getByRole('button', { name: 'Change several days' }).click();
    const dates = await page.locator('.week-grid').first().locator('.sheet-day').evaluateAll((nodes) => nodes.map((n) => n.getAttribute('data-day') ?? ''));
    for (const index of [1, 2, 3]) await page.getByLabel(`Select ${dates[index]}`).check();
    await page.mouse.move(0, 0);
    await page.waitForTimeout(400);
    const grid = await page.locator('.sheet-week').first().boundingBox();
    if (grid !== null) await page.screenshot({ path: shot(`a-lifted-run-1440-${theme}-synthetic.png`), clip: { x: grid.x - 10, y: grid.y - 10, width: grid.width + 20, height: grid.height + 30 }, animations: 'disabled' });
    await page.getByRole('button', { name: 'Done' }).click();
    // Floating panel at 1280.
    await page.setViewportSize({ width: 1280, height: 860 });
    await namedDayButton(page, free[0] ?? '', 'Edit').click();
    await expect(editor).toBeVisible();
    await page.mouse.move(0, 0);
    await page.waitForTimeout(400);
    if (theme === 'light') await page.screenshot({ path: shot(`a-floating-1280-${theme}-synthetic.png`), animations: 'disabled' });
    await page.keyboard.press('Escape');
  }
});

test('phone shots: the first screen, the away row at 320, the editor sheet', async ({ page, adminSeed, builtServer, signInPageAs, isMobile }) => {
  test.skip(!isMobile, 'phone');
  test.setTimeout(180_000);
  const me = await newPerson(adminSeed, builtServer, { displayName: 'Synthetic Phone View', signature: 'none' });
  const free = await me.api.displayedPeriodFreeWorkdays();
  if (free[0] !== undefined) await me.api.seedUnconfirmedBreaksDay(free[0]);
  await page.emulateMedia({ colorScheme: 'light', reducedMotion: 'reduce' });
  await signInPageAs(me.account);
  await expect(page.locator('[data-day]')).toHaveCount(14);
  await page.screenshot({ path: shot('a-phone-390-first-light-synthetic.png'), animations: 'disabled' });
  // Away state (Today on a window day shows Current period), at 320.
  const today = page.getByRole('button', { name: 'Today', exact: true });
  const periods = await me.api.periodsCurrent();
  const window = periods.in_progress.payroll_date !== periods.current.payroll_date;
  if (window) await today.click();
  await page.setViewportSize({ width: 320, height: 700 });
  await page.waitForTimeout(400);
  const row = await page.evaluate(() =>
    [...document.querySelectorAll('.period-bar a, .period-bar button')]
      .map((e) => {
        const r = e.getBoundingClientRect();
        return `${(e.textContent ?? '').trim()}[${e.getAttribute('aria-label') ?? ''}] y${Math.round(r.top)} h${Math.round(r.height)}`;
      })
      .join(', '),
  );
  note('shots.txt', `phone 320 ${window ? 'away (window day)' : 'not a window day'}: ${row}`);
  const bar = await page.locator('.period-bar').first().boundingBox();
  if (bar !== null) await page.screenshot({ path: shot('a-phone-320-period-row-light-synthetic.png'), clip: { x: 0, y: Math.max(0, bar.y - 4), width: 320, height: Math.min(700 - Math.max(0, bar.y - 4), bar.height + 120) }, animations: 'disabled' });
  await page.setViewportSize({ width: 390, height: 844 });
  if (window) await page.getByRole('button', { name: 'Current period' }).click();
  await page.evaluate('window.scrollTo(0, 0)');
  await namedDayButton(page, free[0] ?? '', 'Edit').click();
  const editor = page.getByRole('dialog', { name: /Day editor/ });
  await expect(editor).toBeVisible();
  await page.waitForTimeout(400);
  await page.emulateMedia({ colorScheme: 'dark', reducedMotion: 'reduce' });
  await page.waitForTimeout(300);
  await page.screenshot({ path: shot('a-phone-390-editor-dark-synthetic.png'), animations: 'disabled' });
});
