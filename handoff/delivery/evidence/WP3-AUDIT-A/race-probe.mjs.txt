// WP3-AUDIT-A multi-process deadline/manual race probe (synthetic data, capture mode only).
// Each round: a fresh SQLite WAL file; one OS process signs off manually through the production
// service, another runs the production `cli.js run-jobs --once --now` deadline pass; both are
// released at the same wall-clock instant (the manual racer is delayed by a cycling offset).
// Then two more run-jobs passes drain the PDF and send jobs. Checks: one winner, one revision,
// one ledger set, one send, no invented signed_at, integrity ok.
import { spawn } from 'node:child_process';
import { existsSync, mkdirSync, readdirSync, readFileSync } from 'node:fs';
import { join } from 'node:path';
import { count, fixedClock, freshDb, m, png, policy, REPO, session, sessionUser } from './common.mjs';

const ROUNDS = Number(process.env.ROUNDS ?? 20);
const BASE = process.env.RUN_DIR ?? '<scratch>/work/runs/race';
const EARLY = '2026-09-20T12:00:00Z';
const NOW = '2026-09-30T00:30:00Z';
const DELAYS = [0, 2, 5, 10, 20, 40, 80, 120, 160, 200];
const HERE = import.meta.dirname;

const run = (args, env = {}) =>
  new Promise((resolve) => {
    const child = spawn(process.execPath, args, { env: { ...process.env, ...env }, cwd: REPO });
    let out = '';
    let err = '';
    child.stdout.on('data', (d) => {
      out += d;
    });
    child.stderr.on('data', (d) => {
      err += d;
    });
    child.on('close', (code) => resolve({ code, out: out.trim(), err: err.trim() }));
  });

const totals = { manual: 0, deadline: 0, failed: 0 };
for (let round = 1; round <= ROUNDS; round += 1) {
  const dir = join(BASE, `r${String(round).padStart(2, '0')}`);
  mkdirSync(dir, { recursive: true });
  const dbPath = join(dir, 't.db');
  const dataDir = join(dir, 'private-data');
  const s = await freshDb(dbPath, EARLY);
  const { db } = s;
  const owner = sessionUser(db, s.employee.id);
  const early = fixedClock(EARLY);
  policy(db, EARLY, owner.id, s.calendarId, 'auto_deduct');
  m.ledger.postCredit(
    { db, clock: early },
    { userId: owner.id, sourceKey: 'audit-opening', minutes: 300, workDate: '2026-09-01', actorUserId: null, origin: 'system', reason: 'Synthetic opening balance' },
  );
  m.cmds.createSession({ db, clock: early, user: owner }, '2026-09-15', session('2026-09-15', '09:00', '18:00'));
  m.cmds.createSession({ db, clock: early, user: owner }, '2026-09-16', session('2026-09-16', '09:00', '13:00'));
  m.sigs.saveSignature(db, early, new m.files.FileStore(dataDir), owner.id, png(), 'image/png');
  m.settings.saveSubmissionSettings(db, early, owner.id, { expectedSeq: 0, to: ['<email>'], autoSubmit: true });
  m.auto.setAutomationActivation(db, early, { actorUserId: s.admin.id, activeFrom: '2026-09-21T00:00:00Z', reason: 'Synthetic audit activation' });
  db.prepare("UPDATE users SET status = 'deactivated' WHERE id = ?").run(s.admin.id);
  const review = m.review.buildReviewPayload(db, fixedClock(NOW), owner, '2026-10-02');
  db.close();

  const startAt = Date.now() + 1500;
  const delay = DELAYS[(round - 1) % DELAYS.length];
  const [manual, deadline] = await Promise.all([
    run([join(HERE, 'race-child.mjs'), 'manual', dbPath, dataDir, String(startAt), String(delay), review.payloadHash, String(review.expectedVersion)]),
    run([join(HERE, 'race-child.mjs'), 'deadline', dbPath, dataDir, String(startAt), '0']),
  ]);
  const env = { DATABASE_PATH: dbPath, DATA_DIR: dataDir, MAIL_FROM: '<email>' };
  const drain1 = await run([join(REPO, 'dist/server/cli.js'), 'run-jobs', '--once', '--now', '2026-09-30T00:40:00Z'], env);
  const drain2 = await run([join(REPO, 'dist/server/cli.js'), 'run-jobs', '--once', '--now', '2026-09-30T00:50:00Z'], env);

  const check = m.db.openDatabase(dbPath);
  const revs = check.prepare('SELECT id, origin, review_state, reviewed_sha256, created_at FROM timesheet_revisions WHERE user_id = ?').all(owner.id);
  const rev = revs[0];
  const signoffs = check.prepare('SELECT signed_at FROM signoffs WHERE user_id = ?').all(owner.id);
  const finKeys = check
    .prepare("SELECT source_key, delta_minutes, origin FROM ot_ledger WHERE user_id = ? AND source_key LIKE 'finalization:%' ORDER BY source_key")
    .all(owner.id);
  const lines = count(check, 'SELECT count(*) FROM revision_ledger_lines WHERE user_id = ?', owner.id);
  const jobs = check.prepare("SELECT kind, state FROM jobs WHERE user_id = ? AND kind IN ('render_pdf','send_email') ORDER BY kind").all(owner.id);
  const attempts = check.prepare('SELECT state FROM delivery_attempts WHERE user_id = ?').all(owner.id);
  const audits = check
    .prepare("SELECT operation FROM audit_events WHERE owner_user_id = ? AND operation IN ('timesheet.signoff','timesheet.auto_finalize')")
    .pluck()
    .all(owner.id);
  const integrity = check.prepare('PRAGMA integrity_check').pluck().get();
  const fk = check.prepare('PRAGMA foreign_key_check').all().length;
  const balance = m.ledger.getBalance(check, owner.id);
  check.close();
  const captureDir = join(dataDir, 'mail-capture');
  const folders = existsSync(captureDir) ? readdirSync(captureDir).filter((name) => !name.startsWith('.tmp')) : [];
  const metas = folders.map((name) => JSON.parse(readFileSync(join(captureDir, name, 'metadata.json'), 'utf8')));
  const captures = metas.filter((meta) => meta.pdf_sha256 !== undefined).length;
  const notices = metas.filter((meta) => meta.pdf_sha256 === undefined).map((meta) => (meta.envelope.to ?? []).join('+')).join('|') || 'none';

  const manualResult = JSON.parse(manual.out.split('\n').pop() || '{}');
  const winner = rev === undefined ? 'none' : rev.origin === 'employee' ? 'manual' : 'deadline';
  const problems = [];
  if (revs.length !== 1) problems.push(`revisions=${revs.length}`);
  if (winner === 'manual') {
    if (manualResult.outcome !== 'created') problems.push('manual winner not created');
    if (signoffs.length !== 1 || signoffs[0].signed_at !== '2026-09-30T00:30:05Z') problems.push(`signoffs=${JSON.stringify(signoffs)}`);
    if (rev.review_state !== 'signed') problems.push('review_state');
  } else if (winner === 'deadline') {
    if (!(manualResult.outcome === 'error' && manualResult.status === 409)) problems.push(`manual loser outcome ${JSON.stringify(manualResult)}`);
    if (signoffs.length !== 0) problems.push('automatic revision has a signoff');
    if (rev.review_state !== 'pending' || rev.reviewed_sha256 !== null) problems.push('automatic review_state/reviewed hash');
  }
  const ledgerOk =
    finKeys.length === 2 &&
    finKeys.some((e) => e.source_key === 'finalization:day:2026-09-15:credit' && e.delta_minutes === 60) &&
    finKeys.some((e) => e.source_key === 'finalization:day:2026-09-16:deficit_debit' && e.delta_minutes === -240);
  if (!ledgerOk) problems.push(`ledger=${JSON.stringify(finKeys)}`);
  if (lines !== 2) problems.push(`lines=${lines}`);
  if (jobs.length !== 2 || jobs.some((j) => j.state !== 'succeeded')) problems.push(`jobs=${JSON.stringify(jobs)}`);
  if (attempts.length !== 1 || attempts[0].state !== 'accepted') problems.push(`attempts=${JSON.stringify(attempts)}`);
  if (captures !== 1) problems.push(`captures=${captures}`);
  if (audits.length !== 1) problems.push(`audits=${audits.join(',')}`);
  if (integrity !== 'ok' || fk !== 0) problems.push(`integrity=${integrity} fk=${fk}`);
  if (balance.postedMinutes !== 120) problems.push(`posted=${balance.postedMinutes}`);
  if (manual.code !== 0 || deadline.code !== 0 || drain1.code !== 0 || drain2.code !== 0) {
    problems.push(`exit codes ${manual.code}/${deadline.code}/${drain1.code}/${drain2.code} ${manual.err.slice(0, 200)} ${deadline.err.slice(0, 200)}`);
  }
  if (problems.length > 0) totals.failed += 1;
  else totals[winner] += 1;
  const manualText = `${manualResult.outcome}${manualResult.code ? `/${manualResult.code}` : ''}`;
  console.log(
    `round ${String(round).padStart(2, '0')} delay=${String(delay).padStart(3)}ms winner=${winner} manual=${manualText} deadline_pass=${deadline.out.split('\n').pop()} ` +
      `revisions=${revs.length} signoffs=${signoffs.length} signed_at=${signoffs[0]?.signed_at ?? 'null'} ledger=${finKeys.map((e) => e.delta_minutes).join(',')} ` +
      `lines=${lines} jobs=${jobs.map((j) => `${j.kind}:${j.state}`).join(',')} attempts=${attempts.map((a) => a.state).join(',')} captures=${captures} notices=${notices} ` +
      `posted=${balance.postedMinutes} integrity=${integrity} fk=${fk} ${problems.length ? `FAIL ${problems.join('; ')}` : 'OK'}`,
  );
}
console.log(`TOTAL rounds=${ROUNDS} manual_won=${totals.manual} deadline_won=${totals.deadline} failed=${totals.failed}`);
process.exitCode = totals.failed === 0 ? 0 : 1;
