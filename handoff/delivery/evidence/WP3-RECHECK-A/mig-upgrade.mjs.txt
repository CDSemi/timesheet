// WP3-RECHECK-A migration upgrade probe (reviewed build 2f2520e). For a COPY of each source database
//   wp2      schema v3 made by the accepted WP2 source 5fafeae (mig-seed-wp2.mjs),
//   a1cd566  schema v6 made by the first WP3 freeze a1cd566 (mig-seed-a1cd566.mjs),
// it runs the production `cli.js migrate`, then checks: applied versions, every pre-existing row kept
// column for column, old checksums unchanged, triggers kept, append-only refusals, integrity and
// foreign keys, a second migrate applies nothing, and that WP3 and the fix-round behaviour work on the
// upgraded file (sign-off, PDF, capture, share, review hint, history attribution, B-01 bound).
import { cpSync, existsSync, readdirSync, readFileSync } from 'node:fs';
import { createRequire } from 'node:module';
import { join } from 'node:path';
import { fixedClock, m, png, REPO, run, SCRATCH, session, sessionUser } from './common.mjs';

const Database = createRequire(join(REPO, 'package.json'))('better-sqlite3');
const DIR = `${SCRATCH}/work/mig`;
let failures = 0;
const check = (label, condition, detail = '') => {
  console.log(`${condition ? 'PASS' : 'FAIL'}  ${label}${detail ? `  [${detail}]` : ''}`);
  if (!condition) failures += 1;
};

function snapshot(path) {
  const db = new Database(path, { readonly: true });
  const tables = db.prepare("SELECT name FROM sqlite_master WHERE type = 'table' AND name NOT LIKE 'sqlite_%' ORDER BY name").pluck().all();
  const data = {};
  for (const table of tables) {
    const columns = db.prepare(`PRAGMA table_info("${table}")`).all().map((c) => c.name);
    data[table] = { columns, rows: db.prepare(`SELECT * FROM "${table}" ORDER BY rowid`).all() };
  }
  const triggers = db.prepare("SELECT name FROM sqlite_master WHERE type = 'trigger' ORDER BY name").pluck().all();
  const migrations = db.prepare('SELECT version, name, checksum FROM schema_migrations ORDER BY version').all();
  db.close();
  return { tables, data, triggers, migrations };
}

const APPEND_ONLY = [
  ['audit_events', "UPDATE audit_events SET reason = 'x' WHERE rowid = (SELECT min(rowid) FROM audit_events)", 'DELETE FROM audit_events WHERE rowid = (SELECT min(rowid) FROM audit_events)'],
  ['ot_ledger', 'UPDATE ot_ledger SET delta_minutes = delta_minutes + 1 WHERE rowid = (SELECT min(rowid) FROM ot_ledger)', 'DELETE FROM ot_ledger WHERE rowid = (SELECT min(rowid) FROM ot_ledger)'],
  ['calendar_versions', "UPDATE calendar_versions SET note = 'x' WHERE rowid = (SELECT min(rowid) FROM calendar_versions)", 'DELETE FROM calendar_versions WHERE rowid = (SELECT min(rowid) FROM calendar_versions)'],
  ['work_policies', "UPDATE work_policies SET note = 'x' WHERE rowid = (SELECT min(rowid) FROM work_policies)", 'DELETE FROM work_policies WHERE rowid = (SELECT min(rowid) FROM work_policies)'],
  ['timesheet_revisions', "UPDATE timesheet_revisions SET payload_json = '{}' WHERE rowid = (SELECT min(rowid) FROM timesheet_revisions)", 'DELETE FROM timesheet_revisions WHERE rowid = (SELECT min(rowid) FROM timesheet_revisions)'],
  ['signoffs', "UPDATE signoffs SET signer_name = 'x' WHERE rowid = (SELECT min(rowid) FROM signoffs)", 'DELETE FROM signoffs WHERE rowid = (SELECT min(rowid) FROM signoffs)'],
  ['revision_ledger_lines', "UPDATE revision_ledger_lines SET outcome = 'unchanged' WHERE rowid = (SELECT min(rowid) FROM revision_ledger_lines)", 'DELETE FROM revision_ledger_lines WHERE rowid = (SELECT min(rowid) FROM revision_ledger_lines)'],
  ['submission_settings', 'UPDATE submission_settings SET auto_submit = 1 - auto_submit WHERE rowid = (SELECT min(rowid) FROM submission_settings)', 'DELETE FROM submission_settings WHERE rowid = (SELECT min(rowid) FROM submission_settings)'],
  ['attachments', "UPDATE attachments SET sha256 = 'x' WHERE rowid = (SELECT min(rowid) FROM attachments)", 'DELETE FROM attachments WHERE rowid = (SELECT min(rowid) FROM attachments)'],
  ['timesheet_shares', "UPDATE timesheet_shares SET created_at = '2000-01-01T00:00:00Z' WHERE rowid = (SELECT min(rowid) FROM timesheet_shares)", 'DELETE FROM timesheet_shares WHERE rowid = (SELECT min(rowid) FROM timesheet_shares)'],
];

function appendOnly(path) {
  const db = new Database(path);
  const results = [];
  for (const [table, update, remove] of APPEND_ONLY) {
    const count = db.prepare(`SELECT count(*) FROM "${table}"`).pluck().get();
    for (const [kind, sql] of [['UPDATE', update], ['DELETE', remove]]) {
      if (count === 0) {
        results.push({ table, kind, outcome: 'no-row' });
        continue;
      }
      db.exec('SAVEPOINT probe');
      let outcome;
      try {
        const changes = db.prepare(sql).run().changes;
        outcome = changes === 0 ? 'no-change' : 'ALLOWED';
      } catch (error) {
        outcome = `refused (${String(error.message).slice(0, 60)})`;
      }
      db.exec('ROLLBACK TO probe; RELEASE probe');
      results.push({ table, kind, outcome });
    }
  }
  db.close();
  return results;
}

async function upgrade(name, source, expectApplied) {
  const target = join(DIR, `${name}-upgraded.db`);
  if (existsSync(target)) throw new Error(`Refusing to reuse ${target}`);
  cpSync(source, target);
  const before = snapshot(target);
  const env = { DATABASE_PATH: target, DATA_DIR: join(DIR, `${name}-upgraded-private-data`) };
  const first = await run([join(REPO, 'dist/server/cli.js'), 'migrate'], env);
  const after = snapshot(target);
  console.log(`INFO  ${name}: cli migrate exit=${first.code} out=${JSON.stringify(first.out)} err=${JSON.stringify(first.err.slice(0, 200))}`);
  const applied = after.migrations.filter((row) => !before.migrations.some((old) => old.version === row.version)).map((row) => row.version);
  check(`${name}: production cli.js migrate exits 0 and applies exactly [${expectApplied}]`, first.code === 0 && JSON.stringify(applied) === JSON.stringify(expectApplied), JSON.stringify(applied));
  check(`${name}: the checksums of the already applied migrations are unchanged`, before.migrations.every((old) => after.migrations.find((row) => row.version === old.version)?.checksum === old.checksum));
  let kept = 0;
  const problems = [];
  const added = [];
  for (const table of before.tables) {
    if (table === 'schema_migrations') continue;
    const old = before.data[table];
    const now = after.data[table];
    if (now === undefined) {
      problems.push(`${table} dropped`);
      continue;
    }
    const newColumns = now.columns.filter((c) => !old.columns.includes(c));
    if (newColumns.length) added.push(`${table}+${newColumns.join('+')}`);
    if (now.rows.length !== old.rows.length) problems.push(`${table} rows ${old.rows.length}->${now.rows.length}`);
    old.rows.forEach((row, index) => {
      for (const column of old.columns) if (now.rows[index]?.[column] !== row[column]) problems.push(`${table}[${index}].${column}`);
    });
    kept += old.rows.length;
  }
  const newTables = after.tables.filter((t) => !before.tables.includes(t));
  check(`${name}: every pre-existing row kept column for column (${kept} rows in ${before.tables.length - 1} tables)`, problems.length === 0, problems.slice(0, 5).join(','));
  console.log(`INFO  ${name}: new tables [${newTables.join(', ')}]; added columns [${added.join(', ')}]`);
  check(`${name}: every pre-existing trigger is kept (${before.triggers.length} before, ${after.triggers.length} after)`, before.triggers.every((t) => after.triggers.includes(t)));
  const db = new Database(target, { readonly: true });
  const integrity = db.prepare('PRAGMA integrity_check').pluck().get();
  const fk = db.prepare('PRAGMA foreign_key_check').all();
  const userVersion = db.pragma('user_version', { simple: true });
  db.close();
  check(`${name}: integrity_check ok, foreign_key_check empty, user_version 6`, integrity === 'ok' && fk.length === 0 && userVersion === 6, `${integrity} fk=${fk.length} uv=${userVersion}`);
  const refusals = appendOnly(target);
  const allowed = refusals.filter((r) => r.outcome === 'ALLOWED');
  const refused = refusals.filter((r) => r.outcome.startsWith('refused'));
  console.log(`INFO  ${name}: append-only probe ${refusals.map((r) => `${r.table}.${r.kind}=${r.outcome.split(' ')[0]}`).join(' ')}`);
  check(`${name}: UPDATE/DELETE on append-only tables refused by triggers (${refused.length} refused, ${allowed.length} allowed, ${refusals.length - refused.length - allowed.length} without a row)`, allowed.length === 0, allowed.map((r) => `${r.table}.${r.kind}`).join(','));
  const second = await run([join(REPO, 'dist/server/cli.js'), 'migrate'], env);
  const again = snapshot(target);
  check(`${name}: a second migrate applies nothing`, second.code === 0 && again.migrations.length === after.migrations.length, JSON.stringify(second.out));
  return target;
}
