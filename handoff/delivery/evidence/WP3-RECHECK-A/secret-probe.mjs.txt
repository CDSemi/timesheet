// WP3-RECHECK-A SMTP-secret probe: the built server and the run-jobs CLI are started with
// OUTBOUND_MODE=smtp and a random SMTP password but WITHOUT the owner-only PRODUCTION_SENDING_ENABLED
// flag (never set by this audit). Both must refuse to start, and neither output may contain the secret.
// A capture-mode CLI run with the same SMTP variables present must not print the secret either.
import { randomBytes } from 'node:crypto';
import { join } from 'node:path';
import { createServer } from 'node:net';
import { REPO, run, SCRATCH } from './common.mjs';

const port = await new Promise((resolve, reject) => {
  const probe = createServer();
  probe.once('error', reject);
  probe.listen(0, '127.0.0.1', () => {
    const address = probe.address();
    probe.close(() => resolve(address.port));
  });
});

const secret = `smtp-secret-${randomBytes(12).toString('hex')}`;
const base = { DATABASE_PATH: join(SCRATCH, 'work', 'secret', 't.db'), DATA_DIR: join(SCRATCH, 'work', 'secret', 'private-data'), MAIL_FROM: '<email>', NODE_ENV: 'development', PORT: String(port), HOST: '127.0.0.1', JOB_RUNNER: 'off', SMTP_HOST: '127.0.0.1', SMTP_PORT: '2525', SMTP_USER: 'synthetic-user', SMTP_PASS: secret };
delete process.env.PRODUCTION_SENDING_ENABLED;
let failures = 0;
const check = (label, condition, detail = '') => {
  console.log(`${condition ? 'PASS' : 'FAIL'}  ${label}${detail ? `  [${detail}]` : ''}`);
  if (!condition) failures += 1;
};
const migrate = await run([join(REPO, 'dist/server/cli.js'), 'migrate'], { ...base, OUTBOUND_MODE: 'capture' });
check('migrate a scratch database (capture mode)', migrate.code === 0);
const server = await run([join(REPO, 'dist/server/index.js')], { ...base, OUTBOUND_MODE: 'smtp' });
check('server with OUTBOUND_MODE=smtp and no PRODUCTION_SENDING_ENABLED refuses to start, without printing the secret',
  server.code !== 0 && /PRODUCTION_SENDING_ENABLED/.test(server.err + server.out) && !(server.err + server.out).includes(secret), `exit ${server.code}: ${(server.err || server.out).split('\n').find((l) => l.includes('PRODUCTION_SENDING_ENABLED'))?.slice(0, 160)}`);
const cli = await run([join(REPO, 'dist/server/cli.js'), 'run-jobs', '--once', '--now', '2026-10-05T12:00:00Z'], { ...base, OUTBOUND_MODE: 'smtp' });
check('run-jobs with OUTBOUND_MODE=smtp and no flag refuses, without printing the secret', cli.code !== 0 && !(cli.err + cli.out).includes(secret), `exit ${cli.code}`);
const capture = await run([join(REPO, 'dist/server/cli.js'), 'run-jobs', '--once', '--now', '2026-10-05T12:00:00Z'], { ...base, OUTBOUND_MODE: 'capture' });
check('run-jobs in capture mode with SMTP variables present: exit 0, the secret never printed', capture.code === 0 && !(capture.err + capture.out).includes(secret), capture.out.split('\n').pop());
console.log(failures === 0 ? 'SECRET PROBE PASSED' : `SECRET PROBE FAILED (${failures})`);
process.exitCode = failures === 0 ? 0 : 1;
