import { appendFileSync } from 'node:fs';
import { join } from 'node:path';
import type { Page } from '@playwright/test';
import { expect, screenshotPath, test } from './fixtures.ts';

/*
 * WP5-UX-REGATE2 probe (scratch only, never committed): direct observation of B2-01 at 390, 360 and
 * 320px and of B-01 (first day row against the tab bar) with and without the zone note.
 */
const OUT = process.env.UXREGATE_OUT ?? '';

function note(line: string): void {
  if (OUT !== '') appendFileSync(join(OUT, 'obs-b2.txt'), `${line}\n`);
}

async function firstRowVsTabBar(page: Page, label: string, enforce = true): Promise<void> {
  const tabs = await page.locator('.shell-tabs').boundingBox();
  const row = await page.locator('[data-day]').first().boundingBox();
  const bottom = (row?.y ?? 0) + (row?.height ?? 0);
  const zoneNote = await page.locator('[data-zone-note]').count();
  note(
    `B-01 ${label}: viewport ${JSON.stringify(page.viewportSize())} scrollY=${await page.evaluate('window.scrollY')} zone-note count=${zoneNote} first [data-day] top=${row?.y} bottom=${bottom} ; tab bar top=${tabs?.y} ; bottom<=top: ${bottom <= (tabs?.y ?? 0) + 0.5}`,
  );
  if (enforce) expect(bottom).toBeLessThanOrEqual((tabs?.y ?? 0) + 0.5);
}

test.use({ locale: 'en-US', timezoneId: 'Asia/Ho_Chi_Minh' });

test.describe('phone only', () => {
  test.skip(({ isMobile }) => !isMobile, 'phone viewport probes');

  test('B-01 with the zone note shown (browser zone Asia/Ho_Chi_Minh differs from the reporting zone)', async ({ page, signInThroughUi }) => {
    await signInThroughUi();
    await expect(page.locator('[data-day]')).toHaveCount(14);
    await expect(page.locator('[data-zone-note]')).toBeVisible();
    await firstRowVsTabBar(page, 'zone note shown');
    await page.screenshot({ path: screenshotPath('phone-first-screen-zone-note-synthetic.png'), animations: 'disabled' });
  });

  test.describe('zone equal', () => {
    test.use({ locale: 'en-US', timezoneId: 'America/Los_Angeles' });
    test('B-01 with the viewing zone equal to the reporting zone', async ({ page, signInThroughUi }) => {
      await signInThroughUi();
      await expect(page.locator('[data-day]')).toHaveCount(14);
      await expect(page.locator('[data-zone-note]')).toHaveCount(0);
      await firstRowVsTabBar(page, 'zone equal, no note');
    });
  });

  for (const width of [390, 360, 320]) {
    test(`B2-01 Open a day at ${width}px`, async ({ page, signInThroughUi }) => {
      await page.setViewportSize({ width, height: 844 });
      await signInThroughUi();
      await expect(page.locator('[data-day]')).toHaveCount(14);
      const field = page.getByLabel('Open a day');
      await field.fill('2026-10-09');
      await expect(field).toHaveValue('2026-10-09');
      const box = await field.boundingBox();
      const minimum = await page.evaluate<number>(
        "(() => { const probe = document.createElement('div'); probe.style.width = 'var(--date-field-min)'; document.body.append(probe); const px = probe.getBoundingClientRect().width; probe.remove(); return px; })()",
      );
      const clip = await field.evaluate((input) => ({ scrollWidth: input.scrollWidth, clientWidth: input.clientWidth }));
      const page$ = await page.evaluate('({ scroll: document.documentElement.scrollWidth, client: document.documentElement.clientWidth, inner: window.innerWidth })');
      const row = await page.locator('.tools .open-day').boundingBox();
      const button = await page.getByRole('button', { name: 'Open day', exact: true }).boundingBox();
      note(
        `B2-01 width=${width}: field value="2026-10-09" field width=${box?.width}px (height ${box?.height}) ; --date-field-min=${minimum}px ; input scrollWidth=${clip.scrollWidth} clientWidth=${clip.clientWidth} (clipped=${clip.scrollWidth > clip.clientWidth}) ; page ${JSON.stringify(page$)} ; row box=${JSON.stringify(row)} ; Open day button box=${JSON.stringify(button)}`,
      );
      expect(box?.width ?? 0).toBeGreaterThanOrEqual(minimum - 0.5);
      expect(clip.scrollWidth).toBeLessThanOrEqual(clip.clientWidth);
      await page.locator('.tools .open-day').screenshot({ path: screenshotPath(`open-day-w${width}-synthetic.png`) });
      await firstRowVsTabBar(page, `after open-day at ${width}`, width === 390);
    });
  }
});
