async function signIn(page: any, account: { email: string; password: string }, hash: string) {
  await page.goto(`/${hash}`);
  await page.getByLabel('Email').fill(account.email);
  await page.getByLabel('Password').fill(account.password);
  await page.getByRole('button', { name: 'Sign in' }).click();

function recordApi(page: any): string[] {
  const list: string[] = [];
  page.on('request', (request: any) => {
    const { pathname } = new URL(request.url());
    if (pathname.startsWith('/api/')) list.push(`${request.method()} ${pathname}`);
  });
  return list;
}

/** Local date `YYYY-MM-DD` and time `HH:MM` of an instant in a zone, by Intl (an oracle independent of the app). */
function local(instant: string, zone: string): { date: string; time: string } {
  const parts = new Intl.DateTimeFormat('en-CA', {
    timeZone: zone,
    year: 'numeric',
    month: '2-digit',
    day: '2-digit',
    hour: '2-digit',
    minute: '2-digit',
    hourCycle: 'h23',
  }).formatToParts(new Date(instant));
  const get = (type: string) => parts.find((part) => part.type === type)?.value ?? '';
  return { date: `${get('year')}-${get('month')}-${get('day')}`, time: `${get('hour')}:${get('minute')}` };
}

function dayDiff(later: string, earlier: string): number {
  return Math.round((Date.parse(`${later}T00:00:00Z`) - Date.parse(`${earlier}T00:00:00Z`)) / 86_400_000);
}

/** The text the sheet should print for a session in a zone, and the start-date marker. */
function expectedRange(session: { start_utc: string; end_utc: string | null }, workDate: string, zone: string) {
  const start = local(session.start_utc, zone);
  const startDate = start.date === workDate ? null : `${start.date.slice(5, 7)}/${start.date.slice(8, 10)}`;
  if (session.end_utc === null) return { text: start.time, startDate };
  const end = local(session.end_utc, zone);
  const later = dayDiff(end.date, start.date);
  return { text: `${start.time}-${end.time}${later > 0 ? `+${later}` : ''}`, startDate };
}

const COLLECT_DAYS = `[...document.querySelectorAll('[data-day]')].map((el) => ({
  date: el.getAttribute('data-day'),
  ranges: [...el.querySelectorAll('[data-range]')].map((r) => {
    const sub = r.querySelector('.cell-sub');
    const marker = sub === null ? '' : sub.textContent;
    return { key: r.getAttribute('data-range'), startDate: sub === null ? null : marker, text: r.textContent.slice(marker.length) };
  }),
  ot: el.querySelector('[data-ot]') === null ? '' : el.querySelector('[data-ot]').textContent,
}))`;

const COLLECT_REVIEW = `[...document.querySelectorAll('[data-review-day]')].map((el) => ({
  date: el.getAttribute('data-review-day'),
  ranges: [...el.querySelectorAll('[data-range]')].map((r) => {
    const sub = r.querySelector('.cell-sub');
    const marker = sub === null ? '' : sub.textContent;
    return { key: r.getAttribute('data-range'), startDate: sub === null ? null : marker, text: r.textContent.slice(marker.length) };
  }),
  ot: el.querySelector('[data-ot]') === null ? '' : el.querySelector('[data-ot]').textContent,
}))`;

test('R-07: a device-zone change shows other clock times but never regroups a day', async ({ browser, adminSeed, builtServer }, testInfo) => {
  test.setTimeout(300_000);
  const person = await newPerson(adminSeed, builtServer, { displayName: 'Synthetic Zone Probe', signature: 'none' });
  const free = await person.api.displayedPeriodFreeWorkdays();
  expect(free.length, 'three past free workdays').toBeGreaterThanOrEqual(3);
  const late = free[0] ?? '';
  const early = free[1] ?? '';
  const overnight = free.at(-1) ?? '';
  expect(addDays(overnight, 1) <= person.todayLocal, 'the overnight end is not in the future').toBe(true);
  await person.api.createSession(late, '23:00', '23:50', []);
  await person.api.createSession(early, '00:20', '01:10', []);
  await person.api.call(
    'POST',
    `/api/days/${overnight}/sessions`,
    {
      start: { local: `${overnight}T22:00`, zone: LA },
      end: { local: `${addDays(overnight, 1)}T02:00`, zone: LA },
      input_zone: LA,
      breaks_confirmed: true,
      breaks: [],
    },
    201,
  );
  const sheet = await person.api.call<any>('GET', `/api/timesheets/${person.payrollDate}`);
  const owner = new Map<string, string>();
  const sessions: any[] = [];
  for (const day of sheet.days) for (const session of day.sessions) {
    owner.set(session.id, day.work_date);
    sessions.push({ ...session, work_date: day.work_date });
  }
  expect(sessions.length).toBe(3);

  const zones = ['America/Los_Angeles', 'Asia/Tokyo', 'Pacific/Kiritimati', 'Pacific/Pago_Pago', 'Asia/Ho_Chi_Minh'];
  const runs: any[] = [];
  const use = testInfo.project.use as any;
  for (const zone of zones) {
    const context = await browser.newContext({
      baseURL: builtServer.origin,
      timezoneId: zone,
      locale: 'en-US',
      viewport: use.viewport,
      isMobile: use.isMobile ?? false,
      hasTouch: use.hasTouch ?? false,
      deviceScaleFactor: use.deviceScaleFactor ?? 1,
    });
    const page = await context.newPage();
    await signIn(page, person.account, '#/timesheet');
    await expect(page.locator('[data-day]')).toHaveCount(14);
    await expect(page.locator(`[data-range="${sessions[0].id}"]`)).toHaveCount(1);
    const resolved = await page.evaluate('Intl.DateTimeFormat().resolvedOptions().timeZone');
    const days = await page.evaluate(COLLECT_DAYS);
    const periodZone = await page.locator('.period-bar [data-period-zone]').innerText();
    const zoneNote = await page.locator('[data-zone-note]').count();
    const total = await page.locator('[data-ot-total]').innerText();
    await page.evaluate(`window.location.hash = '#/review/${person.payrollDate}'`);
    await expect(page.locator('[data-review-day]')).toHaveCount(14);
    const review = await page.evaluate(COLLECT_REVIEW);
    runs.push({ zone, resolved, periodZone, zoneNote, total, days, review });
    await context.close();
  }

  const reference = runs[0];
  const checks: string[] = [];
  for (const run of runs) {
    expect(run.days.map((day: any) => day.date), `${run.zone}: 14 accounting dates`).toEqual(sheet.days.map((day: any) => day.work_date));
    for (const day of run.days) {
      for (const range of day.ranges) {
        expect(owner.get(range.key), `${run.zone}: session ${range.key} stays on its accounting date`).toBe(day.date);
        const session = sessions.find((item) => item.id === range.key);
        const want = expectedRange(session, day.date, run.resolved);
        expect(range.text, `${run.zone}: ${range.key} text`).toBe(want.text);
        expect(range.startDate, `${run.zone}: ${range.key} start-date marker`).toBe(want.startDate);
      }
    }
    expect(run.days.map((day: any) => day.ot), `${run.zone}: OT cells`).toEqual(reference.days.map((day: any) => day.ot));
    expect(run.total, `${run.zone}: Overtime Total`).toBe(reference.total);
    expect(run.periodZone, `${run.zone}: period bar names the viewing zone`).toBe(`Times in ${run.resolved}`);
    expect(run.zoneNote, `${run.zone}: zone note only when the zones differ`).toBe(run.resolved === sheet.reporting_zone ? 0 : 1);
    expect(run.review, `${run.zone}: the review is the same in every device zone`).toEqual(reference.review);
    checks.push(`${run.zone} (${run.resolved}): ok`);
  }
  for (const day of reference.review) {
    for (const range of day.ranges) {
      const session = sessions.find((item) => item.id === range.key);
      const want = expectedRange(session, day.date, sheet.reporting_zone);
      expect(range.text, `review ${range.key} in the reporting zone`).toBe(want.text);
    }
  }
  writeResult(`r07-${testInfo.project.name}.json`, {
    reporting_zone: sheet.reporting_zone,
    sessions: sessions.map((item) => ({ id: item.id, work_date: item.work_date, start_utc: item.start_utc, end_utc: item.end_utc })),
    runs: runs.map((run) => ({
      zone: run.zone,
      resolved: run.resolved,
      periodZone: run.periodZone,
      zoneNote: run.zoneNote,
      total: run.total,
      sessionCells: run.days.filter((day: any) => day.ranges.length > 0),
      reviewSessionCells: run.review.filter((day: any) => day.ranges.length > 0),
    })),
    checks,
  });
});

  const attributed = ownerHistory.audit_events.filter((event: any) => event.via_share === true);
  result.owner_history_via_share = attributed.map((event: any) => ({ operation: event.operation, actor_display_name: event.actor_display_name }));
  expect(attributed.some((event: any) => event.actor_display_name === 'Synthetic Grantee B')).toBe(true);

  // A third user: nothing of the owner, by screen or by id swap.
  const context = await browser.newContext({ baseURL: builtServer.origin, timezoneId: 'Asia/Ho_Chi_Minh', locale: 'en-US', viewport: (testInfo.project.use as any).viewport });
  const third = await context.newPage();
  await signIn(third, other.account, '#/timesheet');
  await expect(third.locator('[data-day]')).toHaveCount(14);
  const isolation = {
    note_on_page: (await third.locator('body').innerText()).includes(note),
    owner_session_in_dom: await third.locator(`[data-range="${ownerSession}"]`).count(),
    shared_timesheet_status: await statusInPage(third, 'GET', `/api/shared/${ownerId}/timesheets/${ownerP.payrollDate}`),
    shared_day_status: await statusInPage(third, 'GET', `/api/shared/${ownerId}/days/${ownerDay}`),
    own_day_sessions: (await other.api.dayView(ownerDay)).sessions.length,
    owner_session_put_status: await statusInPage(third, 'PUT', `/api/sessions/${ownerSession}`, {
      start: { local: `${ownerDay}T09:00`, zone: LA },
      end: { local: `${ownerDay}T10:00`, zone: LA },
      input_zone: LA,
      breaks: [],
      breaks_confirmed: true,
      expected_version: 1,
    }),
  };
  await third.evaluate(`window.location.hash = '#/shared/${ownerId}/timesheet'`);
  await third.waitForTimeout(1_500);
  (isolation as any).shared_route_shows_owner_note = (await third.locator('body').innerText()).includes(note);
  (isolation as any).shared_route_owner_session = await third.locator(`[data-range="${ownerSession}"]`).count();
  result.isolation = isolation;
  expect(isolation).toMatchObject({ note_on_page: false, owner_session_in_dom: 0, own_day_sessions: 0 });
  expect([403, 404]).toContain(isolation.shared_timesheet_status);
  expect([403, 404]).toContain(isolation.shared_day_status);
  expect([403, 404]).toContain(isolation.owner_session_put_status);
  expect((isolation as any).shared_route_shows_owner_note).toBe(false);
  expect((isolation as any).shared_route_owner_session).toBe(0);
  expect((await ownerP.api.dayView(ownerDay)).sessions[0]?.id).toBe(ownerSession);
  await context.close();
  writeResult(`sharing-${testInfo.project.name}.json`, result);
});

test('privacy: the signature image is on the Review only, never on the Timesheet page, before or after sign-off', async ({ page, adminSeed, builtServer }, testInfo) => {
  test.setTimeout(300_000);
  const signer = await newPerson(adminSeed, builtServer, { displayName: 'Synthetic Signer' });
  const [day = ''] = await signer.api.displayedPeriodFreeWorkdays();
  await signer.api.seedCompleteDay(day);
  const requests = recordApi(page);
  const result: Record<string, unknown> = {};

  await signIn(page, signer.account, '#/timesheet');
  await expect(page.locator('[data-day]')).toHaveCount(14);
  await expect(page.locator('[data-signature="employee"]')).toHaveText('Not signed yet');
  await page.waitForTimeout(1_000);
  result.timesheet_before = {
    images: await page.locator('img').count(),
    signature_requests: requests.filter((entry) => entry.includes('/api/signatures')),
    strip: await page.locator('[data-signature="employee"]').innerText(),
  };

  requests.length = 0;
  await page.evaluate(`window.location.hash = '#/review/${signer.payrollDate}'`);
  await expect(page.getByRole('heading', { name: 'Review and sign off', level: 1 })).toBeVisible();
  const image = page.getByRole('img', { name: 'Your saved signature image' });
  await expect(image).toBeVisible();
  result.review = {
    images: await page.locator('img').count(),
    image_src: await image.getAttribute('src'),
    signature_requests: requests.filter((entry) => entry.includes('/api/signatures')),
    image_inside_sheet: await page.locator('[data-review-day] img, article.sheet img').count(),
  };
  await page.getByLabel('Your name').fill('Synthetic Signer');
  const acknowledgement = page.getByRole('checkbox', { name: /I acknowledge the incomplete evidence/ });
  if ((await acknowledgement.count()) > 0) await acknowledgement.check();
  await page.getByRole('button', { name: 'Sign off & Submit' }).click();
  await expect(page.getByRole('heading', { name: 'Submitted' })).toBeVisible();
  const finalization = await signer.api.call<any>('GET', `/api/timesheets/${signer.payrollDate}/finalization`);
  const signedLocal = local(finalization.signoff.signed_at, (await signer.api.today()).reportingZone).date;

  requests.length = 0;
  await page.evaluate(`window.location.hash = '#/timesheet'`);
  await expect(page.locator('[data-day]')).toHaveCount(14);
  await expect(page.locator('[data-signature="employee"]')).toHaveText('Signed by Synthetic Signer');
  await page.waitForTimeout(1_000);
  result.timesheet_after = {
    images: await page.locator('img').count(),
    signature_requests: requests.filter((entry) => entry.includes('/api/signatures')),
    strip: await page.locator('[data-signature="employee"]').innerText(),
    strip_date: await page.locator('[data-signature="employee-date"]').innerText(),
    expected_date: `${signedLocal.slice(5, 7)}/${signedLocal.slice(8, 10)}/${signedLocal.slice(0, 4)}`,
  };
  expect(result.timesheet_before).toMatchObject({ images: 0, signature_requests: [] });
  expect(result.review).toMatchObject({ images: 1, image_inside_sheet: 0 });
  expect((result.review as any).image_src).toMatch(/^\/api\/signatures\//);
  expect(result.timesheet_after).toMatchObject({ images: 0, signature_requests: [] });
  expect((result.timesheet_after as any).strip_date).toBe((result.timesheet_after as any).expected_date);
  writeResult(`privacy-${testInfo.project.name}.json`, result);
});
