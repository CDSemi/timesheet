function instance(name, ports) {
  const dir = join(work, name);
  mkdirSync(dir, { recursive: true });
  return { dir, databasePath: join(dir, 'timesheet.db'), dataDir: join(dir, 'private-data'), ports };

const envFor = (inst, port) => ({
  ...process.env,
  NODE_ENV: 'development',
  PORT: String(port),
  HOST: '127.0.0.1',
  DATABASE_PATH: inst.databasePath,
  DATA_DIR: inst.dataDir,

const cli = (inst, args) => {
  const result = spawnSync(process.execPath, [join(exportDir, 'dist', 'server', 'cli.js'), ...args], { env: envFor(inst, inst.ports[0]), encoding: 'utf8', cwd: exportDir });
  return { status: result.status, output: `${result.stdout}${result.stderr}`.replaceAll(work, '<work>').trim() };
};
const servers = [];
function startServer(inst, port) {
  const child = spawn(process.execPath, [join(exportDir, 'dist', 'server', 'index.js')], { env: envFor(inst, port), cwd: exportDir, stdio: ['ignore', 'pipe', 'pipe'] });
  let output = '';
  child.stdout.on('data', (chunk) => (output += String(chunk)));
  child.stderr.on('data', (chunk) => (output += String(chunk)));
  const server = { child, port, output: () => output, exited: false };
  child.on('exit', () => (server.exited = true));
  servers.push(server);
  return server;
}
async function stopServer(server) {
  if (server.exited) return;
  server.child.kill();
  for (let wait = 0; wait < 40 && !server.exited; wait += 1) await new Promise((resolve) => setTimeout(resolve, 100));
}
async function waitHealthy(port) {
  for (let attempt = 0; attempt < 80; attempt += 1) {
    try {
      const response = await fetch(`http://127.0.0.1:${port}/api/health`, { headers: { connection: 'close' }, signal: AbortSignal.timeout(2000) });
      if (response.ok) return true;
    } catch {
      // not listening yet
    }
    await new Promise((resolve) => setTimeout(resolve, 250));
  }
  return false;
}
async function call(method, path, { cookie, body, port } = {}) {
  const headers = { connection: 'close', origin: `http://localhost:${port}` };
  if (cookie) headers.cookie = cookie;
  if (body !== undefined) headers['content-type'] = 'application/json';
  const response = await fetch(`http://127.0.0.1:${port}${path}`, { method, headers, body: body === undefined ? undefined : JSON.stringify(body) });
  const text = await response.text();
  let json = null;
  try {
    json = JSON.parse(text);
  } catch {
    json = null;
  }
  return { status: response.status, json, text };
}
const code = (res) => `${res.status}${res.json?.error?.code ? `:${res.json.error.code}` : ''}`;
async function login(email, password, port) {
  const res = await fetch(`http://127.0.0.1:${port}/api/auth/login`, {
    method: 'POST',
    headers: { connection: 'close', origin: `http://localhost:${port}`, 'content-type': 'application/json' },
    body: JSON.stringify({ email, password }),
  });
  await res.text();
  return { status: res.status, cookie: (res.headers.get('set-cookie') ?? '').split(';')[0] };
}
const addDaysIso = (date, days) => new Date(Date.parse(`${date}T00:00:00Z`) + days * 86400000).toISOString().slice(0, 10);
let clockOffset = 0;
const runJobs = (inst, passes = 3) => {
  const out = [];
  for (let pass = 0; pass < passes; pass += 1) {
    clockOffset += 150;
    out.push(cli(inst, ['run-jobs', '--once', '--now', new Date(Date.now() + clockOffset * 1000).toISOString().replace(/\.\d{3}Z$/, 'Z')]));
  }
  return out.map((item) => `${item.status}:${item.output}`).join(' ; ');
};
async function signOff(user, payroll, port) {
  const review = await call('GET', `/api/timesheets/${payroll}/review`, { cookie: user.cookie, port });
  if (review.status !== 200) return { status: review.status, code: code(review), id: null };
  const res = await call('POST', `/api/timesheets/${payroll}/signoff`, {
    cookie: user.cookie,
    port,
    body: { expected_version: review.json.expected_version, reviewed_hash: review.json.payload_hash, incomplete_evidence_acknowledged: true },
  });
  return { status: res.status, code: code(res), id: res.json?.revision?.id ?? null };
}
async function editDay(user, payroll, port, note) {
  const review = (await call('GET', `/api/timesheets/${payroll}/review`, { cookie: user.cookie, port })).json;
  const day = review.payload.days[2].work_date;
  const edited = await call('PUT', `/api/days/${day}`, { cookie: user.cookie, port, body: { category: 'Worked', leave_minutes: 0, wfh: false, notes: note, reason: 'Synthetic correction probe' } });
  const after = (await call('GET', `/api/timesheets/${payroll}/review`, { cookie: user.cookie, port })).json;
  return { edited: code(edited), expected_version: after.expected_version, payload_hash: after.payload_hash };
}
async function correct(user, payroll, port, prepared) {
  const res = await call('POST', `/api/timesheets/${payroll}/revisions`, {
    cookie: user.cookie,
    port,
    body: { expected_version: prepared.expected_version, reviewed_hash: prepared.payload_hash, incomplete_evidence_acknowledged: true, reason: 'Synthetic correction probe', send_email: false },
  });
  return { status: res.status, code: code(res), id: res.json?.revision?.id ?? null };
}
async function sendBody(user, revisionId, port, extra = {}) {
  const draft = (await call('GET', `/api/revisions/${revisionId}/send-draft`, { cookie: user.cookie, port })).json;
  return { request_key: randomUUID(), to: draft.to, cc: draft.cc, subject: draft.subject, body_text: draft.body_text, time_rows: draft.time_rows, ot_rows: draft.ot_rows, ...extra };
}
const send = (user, revisionId, port, body) => call('POST', `/api/revisions/${revisionId}/send`, { cookie: user.cookie, port, body });
/** The seed gives one employee no recipients; save synthetic ones (example.invalid) when none are set. */
async function ensureRecipients(user, port) {
  const current = await call('GET', '/api/settings/submission', { cookie: user.cookie, port });
  const settings = current.json?.settings ?? null;
  if ((settings?.recipients?.to ?? []).length > 0) return 'kept';
  const saved = await call('POST', '/api/settings/submission', { cookie: user.cookie, port, body: { expected_seq: settings?.seq ?? 0, to: ['payroll@example.invalid'], cc: [], auto_submit: false } });
  return code(saved);
}

let monitor;
let monitor3;
try {
  log(`node ${process.version}`);
  /* ---------------------------------------------------------------- 1. option A on two processes */
  const one = instance('one', [portA, portB]);
  check('1 cli migrate', cli(one, ['migrate']).status === 0);
  check('1 cli seed', cli(one, ['seed']).status === 0);
  const serverA = startServer(one, portA);
  const serverB = startServer(one, portB);

  monitor = new Database(one.databasePath, { readonly: true });

  const rev1 = await signOff(owner, payroll, portA);
  check('1 owner signs revision 1', rev1.status === 201, rev1.code);
  const s1 = await send(owner, rev1.id, portA, await sendBody(owner, rev1.id, portA));

  const keyed = await sendBody(owner, rev2.id, portB, { reason: 'Synthetic corrected copy (retry key)' });
  const rounds = [];
  for (let round = 0; round < 4; round += 1) {
    const before = dbState();
    const bodyA = round === 1 ? keyed : await sendBody(owner, rev2.id, portA, round === 0 ? {} : { reason: `Synthetic reason A${round}` });
    const bodyB = await sendBody(owner, rev2.id, portB, round === 2 ? {} : { reason: `Synthetic reason B${round}` });
    const first = round === 3 ? call('POST', `/api/revisions/${rev2.id}/resend`, { cookie: owner.cookie, port: portA, body: { reason: 'Synthetic resend reason' } }) : send(owner, rev2.id, portA, bodyA);
    const results = await Promise.all([first, send(owner, rev2.id, portB, bodyB)]);

  }
  check('1 two processes, four rounds: every send or resend of revision 2 is 409 delivery_in_progress while revision 1 waits; nothing written', rounds.every((item) => item === '409:delivery_in_progress+409:delivery_in_progress unchanged true'), rounds.join(' | '));
  const message = await send(owner, rev2.id, portA, await sendBody(owner, rev2.id, portA, { reason: 'Synthetic message probe' }));
  log(`1 refusal message: ${JSON.stringify(message.json?.error?.message)}`);

  const otherSend = otherRev.id === null ? { status: 0 } : await send(other, otherRev.id, portB, await sendBody(other, otherRev.id, portB));
  check("1 another owner's send of the same payroll date is not blocked", otherSend.status === 201, `${otherRev.code} ${code(otherSend)}`);
  const prevRev = await signOff(owner, previous, portA);
  const prevSend = prevRev.id === null ? { status: 0 } : await send(owner, prevRev.id, portB, await sendBody(owner, prevRev.id, portB));

  log(`1 run-jobs: ${runJobs(one)}`);

  for (let round = 1; round <= 6; round += 1) {

      })(),
    ]);
    const sendsAfter = count("SELECT count(*) FROM revision_sends WHERE user_id = (SELECT user_id FROM timesheet_revisions WHERE id = ?)", current);
    raceResults.push(`r${round}: waiting-before ${waiting}; old ${code(oldSend)}; correction ${newPath.corrected.code}; new ${newPath.sent === null ? '-' : code(newPath.sent)}; sends +${sendsAfter - sendsBefore}`);
