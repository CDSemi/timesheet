// WP4-RECHECK-B probe R2: the BUILT server (dist/) over real HTTP, synthetic example.invalid users, capture mode,
// job runner off. It uploads packages already built by r1-cost.mjs under <task>/work/pkgs (never the repository) and
// records, per upload: the status and reason, the time, the latency of GET /api/health sent 300 ms after the upload
// started (own connection), the response size and, for an accepted preview, the stored imports.report_json size.
// DATA_DIR and DATABASE_PATH are explicit under the work folder. The server child is stopped through its handle.
// Usage (from the export): node ../probes/r2-http.mjs <work folder>
import { spawn, spawnSync } from 'node:child_process';
import { randomBytes } from 'node:crypto';
import { mkdirSync, readFileSync } from 'node:fs';
import { request as httpRequest } from 'node:http';
import { createRequire } from 'node:module';
import { createServer } from 'node:net';
import { join } from 'node:path';
import { performance } from 'node:perf_hooks';
import { fileURLToPath } from 'node:url';

const require = createRequire(new URL('../export/package.json', import.meta.url));
const Database = require('better-sqlite3');
const work = process.argv[2];
if (!work) throw new Error('work folder required');
mkdirSync(work, { recursive: true });
const PKG_DIR = fileURLToPath(new URL('../work/pkgs/', import.meta.url));
const port = await new Promise((resolve) => {
  const s = createServer();
  s.listen(0, '127.0.0.1', () => {
    const p = s.address().port;
    s.close(() => resolve(p));
  });
});
const origin = `http://localhost:${port}`;
const pw = randomBytes(18).toString('base64url');
const DATABASE_PATH = join(work, 'timesheet.db');
const env = {
  ...process.env,
  NODE_ENV: 'development',
  PORT: String(port),
  HOST: '127.0.0.1',
  DATABASE_PATH,
  DATA_DIR: join(work, 'data'),
  APP_ORIGINS: origin,
  MAIL_FROM: 'probe-sender@example.invalid',
  OUTBOUND_MODE: 'capture',
  JOB_RUNNER: 'off',
  SEED_ADMIN_PASSWORD: randomBytes(18).toString('base64url'),
  SEED_EMPLOYEE_PASSWORD: pw,
  SEED_EMPLOYEE2_PASSWORD: randomBytes(18).toString('base64url'),
};
for (const args of [['migrate'], ['seed']]) {
  const r = spawnSync(process.execPath, ['dist/server/cli.js', ...args], { env, encoding: 'utf8' });
  console.log(`cli ${args[0]}: exit ${r.status}`);
}
const server = spawn(process.execPath, ['dist/server/index.js'], { env, stdio: ['ignore', 'ignore', 'pipe'] });
let serverErr = '';
server.stderr.on('data', (c) => { serverErr += c.toString('utf8'); });
const fail = (e) => {
  console.log('FATAL', e?.message ?? e);
  server.kill('SIGTERM');
  process.exitCode = 2;
};
process.on('uncaughtException', fail);
process.on('unhandledRejection', fail);

function send(method, path, { body, headers = {} } = {}) {
  return new Promise((resolve) => {
    const t0 = performance.now();
    const req = httpRequest({ host: '127.0.0.1', port, method, path, agent: false, headers: { origin, ...headers, ...(body ? { 'content-length': body.length } : {}) } }, (res) => {
      let size = 0;
      const head = [];
      res.on('data', (c) => { size += c.length; if (head.length < 4) head.push(c); });
      res.on('end', () => resolve({ status: res.statusCode, ms: Math.round(performance.now() - t0), headers: res.headers, size, text: Buffer.concat(head).toString('utf8').slice(0, 4000) }));
    });
    req.on('error', (e) => resolve({ status: 0, ms: Math.round(performance.now() - t0), error: e.code ?? e.message, size: 0, text: '' }));
    if (body) req.write(body);
    req.end();
  });
}
for (let i = 0; i < 100; i += 1) {
  const r = await send('GET', '/api/ready');
  if (r.status === 200) break;
  await new Promise((r2) => setTimeout(r2, 100));
}
const login = await send('POST', '/api/auth/login', { body: Buffer.from(JSON.stringify({ email: 'employee@example.invalid', password: pw })), headers: { 'content-type': 'application/json' } });
const cookie = String(login.headers['set-cookie']?.[0] ?? '').split(';')[0];
console.log(`login: ${login.status}`);

const CASES = [
  'ACCEPT_tracked_template',
  'ACCEPT_60_dated_sheets',
  'P1b_three_15_8_MiB_sheets_of_c_65_KB_',
  'B_02_Holiday_Dates_150_000_cells_in_column_C',
  'B_02_sparse_far_row_A9999999',
  'P7_3_payroll_sheet_s_x_199_000_f_8_5_f_',
  'H2a_4_parts_of_1_digit_tag_names_',
  'H4b_4_parts_one_c_each_with_4_MiB_of_attributes',
  'H5d_one_1_MiB_shared_string_x_100_holiday_names',
  'H5b_one_4_MiB_shared_string_x_2000_holiday_names',
];
const db = new Database(DATABASE_PATH, { readonly: true, fileMustExist: true });
for (const name of CASES) {
  const bytes = readFileSync(join(PKG_DIR, `${name}.xlsx`));
  const upload = send('POST', '/api/imports', { body: bytes, headers: { cookie, 'content-type': 'application/vnd.openxmlformats-officedocument.spreadsheetml.sheet' } });
  await new Promise((r) => setTimeout(r, 300));
  const health = await send('GET', '/api/health');
  const done = await upload;
  let reason = '';
  try {
    const parsed = JSON.parse(done.text);
    reason = parsed?.error ? ` ${parsed.error.code}${parsed.error.details?.reason ? '/' + parsed.error.details.reason : ''}` : '';
  } catch {
    reason = done.status === 201 || done.status === 200 ? ' (body larger than the captured head)' : '';
  }
  const stored = db.prepare('SELECT count(*) AS n, coalesce(max(length(report_json)), 0) AS largest, coalesce(sum(length(report_json)), 0) AS total FROM imports').get();
  console.log(`${name}: upload ${bytes.length} B -> ${done.status}${reason}${done.error ? ' ' + done.error : ''} in ${done.ms} ms, response ${(done.size / 1024 / 1024).toFixed(2)} MiB; GET /api/health 300 ms later -> ${health.status} after ${health.ms} ms; imports rows ${stored.n}, largest report_json ${(stored.largest / 1024 / 1024).toFixed(2)} MiB`);
}
const idle = await send('GET', '/api/health');
console.log(`idle GET /api/health afterwards -> ${idle.status} in ${idle.ms} ms`);
db.close();
server.kill('SIGTERM');
await new Promise((r) => server.once('exit', r));
const errLines = serverErr.split('\n').filter((l) => /RangeError|Error|internal/i.test(l)).slice(0, 4).map((l) => l.slice(0, 200));
console.log(`server stopped through its handle: ${server.signalCode ?? server.exitCode}; server stderr error lines: ${errLines.length ? errLines.join(' / ') : 'none'}`);
