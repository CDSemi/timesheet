// WP4-RECHECK-B probe R1: cost of one synchronous preview (reader + mapping + report-sized JSON) for the first
// audit's cost packages (P1b, B-02, P7 rebuilt) and for new hostile packages that try to get around the WP4-FIXB
// limits. Each case runs in a fresh child process (process.execPath, stopped through its handle on timeout), so the
// maxRSS printed belongs to that case only. In memory only: no workbook is written to disk.
// Usage: node r1-cost.mjs [filter]      (parent: runs every case whose name contains filter)
//        node r1-cost.mjs --case <name> (child: one case)

const MiB = 1024 * 1024;
const NS = 'http://schemas.openxmlformats.org/spreadsheetml/2006/main';
const RNS = 'http://schemas.openxmlformats.org/officeDocument/2006/relationships';
const WS_TYPE = `${RNS}/worksheet`;
const HEAD = `<?xml version="1.0"?><worksheet xmlns="${NS}"><sheetData><row r="1">`;
const TAIL = '</row></sheetData></worksheet>';

/** A worksheet whose single row is `unit` repeated to about `bytes` bytes. */
const fill = (unit, bytes, head = HEAD, tail = TAIL) => head + unit.repeat(Math.floor((bytes - head.length - tail.length) / unit.length)) + tail;

/** Template entries plus `extra` worksheet parts registered as sheets named by `names` (all distinct parts). */
function withSheets(entries, parts) {
  let workbook = strFromU8(entries['xl/workbook.xml']);
  let rels = strFromU8(entries['xl/_rels/workbook.xml.rels']);
  parts.forEach(({ name, part, xml, relId }, i) => {
    const id = relId ?? `rIdR${i}`;
    if (xml !== undefined) entries[part] = strToU8(xml);
    workbook = workbook.replace('</sheets>', `<sheet name="${name}" sheetId="${500 + i}" r:id="${id}"/></sheets>`);
    if (relId === undefined) {
      rels = rels.replace('</Relationships>', `<Relationship Id="${id}" Type="${WS_TYPE}" Target="${part.replace('xl/', '')}"/></Relationships>`);
    }
  });
  entries['xl/workbook.xml'] = strToU8(workbook);
  entries['xl/_rels/workbook.xml.rels'] = strToU8(rels);
  return entries;
}

/** Four worksheet parts (sheet1..3 replaced, one added) filled to `each` bytes with `unit`. */
function fourParts(unit, each) {
  const e = unpack(readTemplateBytes());
  for (const part of ['xl/worksheets/sheet1.xml', 'xl/worksheets/sheet2.xml', 'xl/worksheets/sheet3.xml']) e[part] = strToU8(fill(unit, each));
  return withSheets(e, [{ name: 'Extra', part: 'xl/worksheets/sheet90.xml', xml: fill(unit, each) }]);
}
const EACH = 4 * MiB - 6 * 1024; // four parts plus the template's small parts stay under the 16 MiB total

/** Template whose shared strings gain one big string (index 66) of about `bytes` bytes. */
function withBigSharedString(e, bytes, first = 'a') {
  const sst = strFromU8(e['xl/sharedStrings.xml']);
  const big = first + 'a'.repeat(bytes - sst.length - 40);
  e['xl/sharedStrings.xml'] = strToU8(sst.replace('</sst>', `<si><t>${big}</t></si></sst>`));
  return e;
}

/** Holiday Dates sheet with `rows` rows: A = a valid serial date, B = shared string `index`. */
function holidaySheet(rows, index) {
  const out = [];
  for (let r = 2; r < 2 + rows; r += 1) out.push(`<row r="${r}"><c r="A${r}"><v>${46023 + r}</v></c><c r="B${r}" t="s"><v>${index}</v></c></row>`);
  return `<?xml version="1.0"?><worksheet xmlns="${NS}"><sheetData>${out.join('')}</sheetData></worksheet>`;
}

/** One synthetic dated sheet whose labels reference shared string `index`, registered under `count` dated names. */
function labelAmplifier(count, index, bigBytes, first = 'a') {
  const e = withBigSharedString(unpack(buildSyntheticWorkbook({ periods: [{ payrollDate: '2026-01-09' }] })), bigBytes, first);
  const part = 'xl/worksheets/sheet100.xml';
  e[part] = strToU8(strFromU8(e[part]).replace(/t="inlineStr"><is><t>Worked<\/t><\/is>/g, `t="s"><v>${index}</v>`));
  const names = [];
  for (let i = 1; i < count; i += 1) {
    const d = new Date(Date.UTC(2026, 0, 9) + i * 14 * 86400000).toISOString().slice(0, 10).replace(/-/g, '.');
    names.push({ name: d, part, relId: 'rIdSyn0' });
  }
  return withSheets(e, names);
}

/** One dated sheet (payroll 2026-01-09) whose label B14 is an inline string of `bytes` characters `ch`. */
function inlineLabel(sheets, bytes, ch) {
  const periods = [];
  for (let i = 0; i < sheets; i += 1) periods.push({ payrollDate: new Date(Date.UTC(2026, 0, 9) + i * 14 * 86400000).toISOString().slice(0, 10) });
  const e = unpack(buildSyntheticWorkbook({ periods }));
  for (let i = 0; i < sheets; i += 1) {
    const part = `xl/worksheets/sheet${100 + i}.xml`;
    e[part] = strToU8(strFromU8(e[part]).replace(/(<c r="B14"[^>]*t="inlineStr"><is><t>)Worked(<\/t>)/, `$1${ch.repeat(bytes)}$2`));
  }
  return e;
}

// Two children per case: one builds the package and writes it under the task folder (never the repository), a fresh
// one reads it and previews it, so the maxRSS baseline is not inflated by building the hostile strings.
import { mkdirSync, readFileSync, writeFileSync } from 'node:fs';
import { join } from 'node:path';
const PKG_DIR = fileURLToPath(new URL('../work/pkgs/', import.meta.url));
const fileOf = (name) => join(PKG_DIR, `${name.replace(/[^A-Za-z0-9]+/g, '_').slice(0, 80)}.xlsx`);
const args = process.argv.slice(2);
if (args[0] === '--build') {
  mkdirSync(PKG_DIR, { recursive: true });
  writeFileSync(fileOf(args[1]), pack(CASES[args[1]]()));
} else if (args[0] === '--run') {
  const name = args[1];
  const bytes = new Uint8Array(readFileSync(fileOf(name)));
  const baseline = process.resourceUsage().maxRSS;
  const t0 = performance.now();
  let outcome;
  let jsonMs = 0;
  try {
    const out = previewWorkbook(bytes);
    const t1 = performance.now();
    let size;
    try {
      size = `${(JSON.stringify(out).length / MiB).toFixed(2)} MiB preview JSON, ${(JSON.stringify(out.findings).length / 1024).toFixed(1)} KiB findings JSON, sources listed ${out.findings.reduce((n, f) => n + f.sources.length, 0)} of ${out.findings.reduce((n, f) => n + f.sourceCount, 0)}`;
    } catch (error) {
      size = `JSON.stringify THROWN ${error.name}: ${error.message}`;
    }
    jsonMs = Math.round(performance.now() - t1);
    outcome = `accepted (sheets=${out.sheets.length}, periods=${out.periods.length}, holidays=${out.holidays.length}, errors=${out.summary.errors}, findings=${out.findings.length}); ${size}`;
  } catch (error) {
    outcome = error.name === 'WorkbookRejectedError' ? `rejected:${error.code}` : `THROWN ${error.name}: ${String(error.message).slice(0, 120)}`;
  }
  const ms = Math.round(performance.now() - t0);
  console.log(`${name} | upload=${bytes.length} B | preview ms=${ms - jsonMs} (+${jsonMs} ms JSON) | maxRSS ${Math.round(baseline / 1024)} -> ${Math.round(process.resourceUsage().maxRSS / 1024)} MiB | ${outcome}`);
} else {
  const filter = args[0] ?? '';
  const opts = { encoding: 'utf8', env: process.env, timeout: 300_000, maxBuffer: 64 * MiB };
  const clean = (s) => (s ?? '').trim().split('\n').filter((l) => !l.includes('ExperimentalWarning') && !l.includes('--trace-warnings')).slice(0, 2).join(' / ').slice(0, 300);
  for (const name of Object.keys(CASES).filter((n) => n.includes(filter))) {
    const build = spawnSync(process.execPath, [fileURLToPath(import.meta.url), '--build', name], opts);
    if (build.status !== 0) { process.stdout.write(`${name} | BUILD FAILED exit=${build.status} ${clean(build.stderr)}\n`); continue; }
    const child = spawnSync(process.execPath, [fileURLToPath(import.meta.url), '--run', name], opts);
    const err = clean(child.stderr);
    process.stdout.write(`${(child.stdout ?? '').trim() || name} | exit=${child.status}${child.signal ? ' signal=' + child.signal : ''}${err ? ' | stderr=' + err : ''}\n`);
  }
}
