import { createServer } from 'node:http';
const srv = createServer((req, res) => { res.writeHead(200, { 'content-type': 'application/json' }); res.end('{"status":"ok"}'); });
srv.listen(Number(process.argv[2]), '127.0.0.1', () => console.log('occupied', process.argv[2]));
setTimeout(() => srv.close(), Number(process.argv[3] ?? 60000));
