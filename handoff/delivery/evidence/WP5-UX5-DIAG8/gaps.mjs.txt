// Usage: node gaps.mjs <diag jsonl>  -> quantiles (ms) of: click end -> read end, read end -> next click start, click end -> the page's change of period.
import { readFileSync } from 'node:fs';

const lines = readFileSync(process.argv[2], 'utf8').split('\n').filter((l) => l.trim() !== '');
const a = [];
const b = [];
const c = [];
for (const line of lines) {
  const run = JSON.parse(line);
  const changes = run.pageLog.filter((e) => e.due !== null).map((e) => e.t);
  const steps = run.steps;
  for (let i = 1; i < steps.length; i += 1) {
    const s = steps[i];
    a.push(s.readEnd - s.clickEnd);
    if (i + 1 < steps.length) b.push(steps[i + 1].clickStart - s.readEnd);
    const next = changes.find((t) => t > s.clickStart);
    if (next !== undefined) c.push(next - s.clickEnd);
  }
}
const q = (xs, p) => [...xs].sort((x, y) => x - y)[Math.min(xs.length - 1, Math.floor(xs.length * p))];
const show = (name, xs) => console.log(`${name}: n=${xs.length} min=${q(xs, 0)} p25=${q(xs, 0.25)} median=${q(xs, 0.5)} p75=${q(xs, 0.75)} p90=${q(xs, 0.9)} max=${q(xs, 0.999)}`);
show('click end -> read end', a);
show('read end -> next click start (includes setViewportSize, and a check() of a new kind)', b);
show('click end -> the period changes on the page', c);
