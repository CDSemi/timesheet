# Usage: [SHIFT_I=<UTC ISO instant>] [PW_EXPORT=<dir>] bash run-pw.sh <label> [playwright args...]
# One Playwright run in the scratch export with its own output folder. SHIFT_I simulates the wall clock of the Node processes
# (shiftclock.mjs of FIX10/DIAG7, unchanged); without it the clock is the real one. DIAG_OUT is where probe specs write jsonl.
. <task>/env.sh
L=$1; shift
mkdir -p $T/pw-$L $T/shots-$L
export E2E_SCREENSHOT_DIR="$TW\\shots-$L"
export DIAG_OUT="$TW\\out\\diag-$L.jsonl"
if [ -z "$SHIFT_I" ]; then
  export NODE_OPTIONS="--trace-deprecation"
  export AB8_SHIFT_MS=0
else
  TARGET_MS=$(date -u -d "$SHIFT_I" +%s%3N)
  NOW_MS=$(date -u +%s%3N)
  export AB8_SHIFT_MS=$((TARGET_MS - NOW_MS))
  export NODE_OPTIONS="--trace-deprecation --import file:///$(cygpath -m $T)/shiftclock.mjs"
fi
echo "== $L: real start $(date -u +%Y-%m-%dT%H:%M:%SZ), simulated start ${SHIFT_I:-real}, AB8_SHIFT_MS=$AB8_SHIFT_MS"
cd "${PW_EXPORT:-$T/export}" || exit 2
echo "\$ node node_modules/@playwright/test/cli.js test $* --output=<task>\\pw-$L"
node.exe node_modules/@playwright/test/cli.js test "$@" --output="$TW\\pw-$L"
echo "exit=$?"
echo "== $L real end $(date -u +%Y-%m-%dT%H:%M:%SZ)"
