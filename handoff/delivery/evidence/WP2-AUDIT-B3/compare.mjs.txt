// WP2-AUDIT-B3 comparison (auditor-owned): node compare.mjs <style dir>
// Diffs the base (f79413b) and reviewed (a3d1b65) real-screen captures of zz-audit-b3-style:
// target selector values, element count, DOM paths, bounding boxes and computed properties.
import { readdirSync, readFileSync } from 'node:fs';
import { join } from 'node:path';

const dir = process.argv[2];
const files = readdirSync(dir).filter((name) => name.startsWith('reviewed-') && name.endsWith('.json')).sort();
let totalElements = 0;
let totalValues = 0;
let differences = 0;
const targetSummary = new Map();
for (const name of files) {
  const key = name.slice('reviewed-'.length, -'.json'.length);
  const reviewed = JSON.parse(readFileSync(join(dir, name), 'utf8'));
  const base = JSON.parse(readFileSync(join(dir, `base-${key}.json`), 'utf8'));
  const lines = [];
  // Targets.
  for (const [index, target] of reviewed.targets.entries()) {
    const other = base.targets[index];
    const same = JSON.stringify(target) === JSON.stringify(other);
    if (!same) lines.push(`  TARGET DIFF ${target.selector}: base ${JSON.stringify(other)} reviewed ${JSON.stringify(target)}`);
    if (target.count > 0) {
      const distinct = [...new Set(target.values.map((value) => JSON.stringify(value)))];
      const entry = targetSummary.get(target.selector) ?? new Set();
      for (const value of distinct) entry.add(`${key.split('-')[0]} ${value}`);
      targetSummary.set(target.selector, entry);
    }
  }
  // Elements.
  if (reviewed.elements.length !== base.elements.length) lines.push(`  ELEMENT COUNT base ${base.elements.length} reviewed ${reviewed.elements.length}`);
  const count = Math.min(reviewed.elements.length, base.elements.length);
  for (let i = 0; i < count; i += 1) {
    const a = base.elements[i];
    const b = reviewed.elements[i];
    totalValues += b.values.split(';').length + 1;
    if (a.path !== b.path) { lines.push(`  PATH ${i}: base ${a.path} reviewed ${b.path}`); continue; }
    if (a.box !== b.box) lines.push(`  BOX ${b.path}: base ${a.box} reviewed ${b.box}`);
    if (a.values !== b.values) {
      const av = a.values.split(';');
      const bv = b.values.split(';');
      const changed = bv.filter((value, j) => value !== av[j]).map((value, j) => `${av[bv.indexOf(value)]} -> ${value}`);
      lines.push(`  VALUES ${b.path}: ${changed.join(' | ')}`);
    }
  }
  totalElements += count;
  differences += lines.length;
  console.log(`${key}: ${reviewed.elements.length} visible elements; targets ${reviewed.targets.map((t) => `${t.selector}=${t.count}`).join(', ')}; differences ${lines.length}`);
  for (const line of lines) console.log(line);
}
console.log(`screens ${files.length}; elements compared ${totalElements}; values compared ${totalValues}; differences ${differences}`);
console.log('distinct computed values of the tokenized targets (project, values):');
for (const [selector, values] of targetSummary) console.log(`  ${selector}: ${[...values].join(' ; ')}`);
