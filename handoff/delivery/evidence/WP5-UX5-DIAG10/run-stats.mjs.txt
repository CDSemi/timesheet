// Statistics of one Playwright list-reporter run log: the import copies, and every test above one third of its own timeout.
// Usage: node run-stats.mjs <log> <outFile> <sourceDir>
import { readFileSync, writeFileSync } from 'node:fs';
import { join } from 'node:path';

const [log, outFile, sourceDir] = process.argv.slice(2);
const LINE = /^\s+(ok|x|!)\s+\d+\s+\[(desktop|mobile)\]\s+›\s+tests\\e2e\\([^:]+):(\d+):(\d+)\s+›\s+(.*?)\s+\((\d+(?:\.\d+)?)(ms|s|m)\)\s*$/;
const text = readFileSync(log, 'utf8');
const rows = [];
for (const line of text.split('\n')) {
  const m = LINE.exec(line);
  if (m === null) continue;
  const s = m[8] === 'ms' ? Number(m[7]) / 1000 : m[8] === 'm' ? Number(m[7]) * 60 : Number(m[7]);
  rows.push({ status: m[1], project: m[2], spec: m[3], line: Number(m[4]), title: m[6], seconds: s });
}
const cache = new Map();
function timeoutOf(row) {
  if (!cache.has(row.spec)) cache.set(row.spec, readFileSync(join(sourceDir, row.spec), 'utf8').split('\n'));
  const lines = cache.get(row.spec);
  for (let i = row.line - 1; i < Math.min(lines.length, row.line + 8); i += 1) {
    const m = /test\.setTimeout\((\d[\d_]*)\)/.exec(lines[i]);
    if (m !== null) return Number(m[1].replace(/_/g, '')) / 1000;
  }
  return 30;
}
const pct = (sorted, p) => (sorted.length === 0 ? NaN : sorted[Math.min(sorted.length, Math.max(1, Math.ceil((p / 100) * sorted.length))) - 1]);
const out = [];
out.push(`log ${log.split(/[\\/]/).pop()}: timed rows ${rows.length}; failed rows ${rows.filter((r) => r.status === 'x').length}`);
for (const l of text.split('\n')) if (/^\s+\d+ (passed|failed|skipped|flaky)/.test(l)) out.push(l.trim());
for (const project of ['desktop', 'mobile']) {
  for (const which of ['upload, preview with source cells', 'the opening balance: validation']) {
    const sel = rows.filter((r) => r.project === project && /^importo?f?-r\d+\.spec\.ts$|^import\.spec\.ts$|^importfix\.spec\.ts$/.test(r.spec) && r.title.startsWith(which));
    const sorted = sel.map((r) => r.seconds).sort((a, b) => a - b);
    out.push(`${project} ${which}: n=${sorted.length} median=${pct(sorted, 50)} p95=${pct(sorted, 95)} max=${sorted[sorted.length - 1]} failed=${sel.filter((r) => r.status === 'x').length}`);
  }
}
out.push('');
out.push('tests above one third of their own timeout, or failed (project spec timeout seconds title):');
const flagged = rows.map((r) => ({ r, t: timeoutOf(r) })).filter(({ r, t }) => r.seconds > t / 3 || r.status === 'x').sort((a, b) => b.r.seconds / b.t - a.r.seconds / a.t);
for (const { r, t } of flagged) out.push(`${r.project} ${r.spec}:${r.line} ${r.status === 'x' ? 'FAILED ' : ''}${r.seconds} s of ${t} s | ${r.title.slice(0, 80)}`);
out.push(`flagged: ${flagged.length}`);
writeFileSync(outFile, out.join('\n') + '\n');
