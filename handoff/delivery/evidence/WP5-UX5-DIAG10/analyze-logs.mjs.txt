// Aggregates per-test durations from the Playwright list-reporter logs of the gates REGATE5 to REGATE11.
// Usage: node analyze-logs.mjs <evidenceRoot> <outFile>
import { readdirSync, readFileSync, writeFileSync, existsSync } from 'node:fs';
import { join } from 'node:path';

const [evidenceRoot, outFile] = process.argv.slice(2);
// argv[5] (optional): the gate numbers to read, comma separated; default 5 to 11.
const gates = (process.argv[5] ?? '5,6,7,8,9,10,11').split(',').map((n) => `WP5-UX5-REGATE${n}`);
const wanted = /^(04-e2e-(desktop|mobile)|log-sweep-(full|seven)-[0-9]+(-rerun)?)\.txt$/;
const LINE = /^\s+(ok|x|!|-)\s+(\d+)\s+\[(desktop|mobile)\]\s+›\s+tests\\e2e\\([^:]+):(\d+):(\d+)\s+›\s+(.*?)\s+\((\d+(?:\.\d+)?)(ms|s|m)\)\s*$/;

function seconds(value, unit) {
  const n = Number(value);
  return unit === 'ms' ? n / 1000 : unit === 'm' ? n * 60 : n;
}

function percentile(sorted, p) {
  if (sorted.length === 0) return NaN;
  const rank = Math.ceil((p / 100) * sorted.length);
  return sorted[Math.min(sorted.length, Math.max(1, rank)) - 1];
}

const runs = [];
for (const gate of gates) {
  const dir = join(evidenceRoot, gate);
  if (!existsSync(dir)) continue;
  for (const name of readdirSync(dir).sort()) {
    if (!wanted.test(name)) continue;
    const text = readFileSync(join(dir, name), 'utf8');
    const rows = [];
    for (const line of text.split('\n')) {
      const m = LINE.exec(line);
      if (m === null) continue;
      rows.push({ status: m[1], project: m[3], spec: m[4], line: Number(m[5]), title: m[7], seconds: seconds(m[8], m[9]) });
    }
    runs.push({ gate, name, rows });
  }
}

const out = [];
out.push(`runs read: ${runs.length}`);
for (const run of runs) out.push(`  ${run.gate}/${run.name}: ${run.rows.length} timed tests`);

// 1. import.spec.ts, every test, per project.
const byTest = new Map();
for (const run of runs) {
  for (const row of run.rows) {
    const key = `${row.project}\t${row.spec}\t${row.title}`;
    let entry = byTest.get(key);
    if (entry === undefined) {
      entry = { project: row.project, spec: row.spec, title: row.title, values: [], where: [] };
      byTest.set(key, entry);
    }
    entry.values.push(row.seconds);
    entry.where.push(`${run.gate.replace('WP5-UX5-', '')}/${run.name.replace('.txt', '')}:${row.seconds}${row.status === 'x' ? ':FAILED' : ''}`);
  }
}

function stats(values) {
  const sorted = [...values].sort((a, b) => a - b);
  return { n: sorted.length, median: percentile(sorted, 50), p95: percentile(sorted, 95), max: sorted[sorted.length - 1] };
}

out.push('');
out.push('== import.spec.ts, every test, both projects (seconds; nearest-rank percentiles; limit 30 s unless stated)');
for (const entry of [...byTest.values()].filter((e) => e.spec === 'import.spec.ts').sort((a, b) => (a.project + a.title).localeCompare(b.project + b.title))) {
  const s = stats(entry.values);
  out.push(`${entry.project} n=${s.n} median=${s.median} p95=${s.p95} max=${s.max} | ${entry.title.slice(0, 90)}`);
}

out.push('');
out.push('== import.spec.ts: the first test, every sample');
for (const entry of byTest.values()) {
  if (entry.spec === 'import.spec.ts' && entry.title.startsWith('upload, preview with source cells')) {
    out.push(`${entry.project}: ${entry.where.join(' ')}`);
  }
}

// 2. every test with p95 above 10 s (about one third of 30 s), with n >= 3.
out.push('');
out.push('== tests with p95 > 10 s or any failed sample (n, median, p95, max); a test with its own setTimeout is marked by the reader later');
const rows = [];
for (const entry of byTest.values()) {
  const s = stats(entry.values);
  const failed = entry.where.some((w) => w.endsWith(':FAILED'));
  if (s.p95 > 10 || failed) rows.push({ entry, s, failed });
}
rows.sort((a, b) => b.s.p95 - a.s.p95);
for (const { entry, s, failed } of rows) {
  out.push(`${entry.project} ${entry.spec} n=${s.n} median=${s.median} p95=${s.p95} max=${s.max}${failed ? ' FAILED' : ''} | ${entry.title.slice(0, 110)}`);
}

// 2b. Each test's own timeout: the line of the test( call in the latest gate log (REGATE11, the HEAD source) and a test.setTimeout within the next 8 lines.
const sourceDir = process.argv[4];
const timeoutOf = new Map();
if (sourceDir !== undefined) {
  const latest = runs.filter((r) => r.gate === 'WP5-UX5-REGATE11');
  const cache = new Map();
  for (const run of latest) {
    for (const row of run.rows) {
      const key = `${row.project}\t${row.spec}\t${row.title}`;
      if (timeoutOf.has(key)) continue;
      if (!cache.has(row.spec)) cache.set(row.spec, readFileSync(join(sourceDir, row.spec), 'utf8').split('\n'));
      const lines = cache.get(row.spec);
      let found = 30;
      for (let i = row.line - 1; i < Math.min(lines.length, row.line + 8); i += 1) {
        const m = /test\.setTimeout\((\d[\d_]*)\)/.exec(lines[i]);
        if (m !== null) { found = Number(m[1].replace(/_/g, '')) / 1000; break; }
      }
      timeoutOf.set(key, found);
    }
  }
  out.push('');
  out.push('== tests whose p95 is above one third of their own timeout (timeout from test.setTimeout within 8 lines of the test call in the HEAD source; 30 s otherwise); also the top by p95/timeout');
  const ranked = [];
  for (const [key, entry] of byTest) {
    const timeout = timeoutOf.get(key);
    if (timeout === undefined) continue;
    const s = stats(entry.values);
    ranked.push({ entry, s, timeout, ratio: s.p95 / timeout, maxRatio: s.max / timeout });
  }
  ranked.sort((a, b) => b.ratio - a.ratio);
  for (const r of ranked.slice(0, 25)) {
    out.push(`${r.entry.project} ${r.entry.spec} timeout=${r.timeout} n=${r.s.n} median=${r.s.median} p95=${r.s.p95} max=${r.s.max} p95/timeout=${r.ratio.toFixed(2)} max/timeout=${r.maxRatio.toFixed(2)}${r.ratio > 1 / 3 ? ' ABOVE-ONE-THIRD' : ''} | ${r.entry.title.slice(0, 90)}`);
  }
  out.push(`tests with a known timeout: ${ranked.length}; above one third at p95: ${ranked.filter((r) => r.ratio > 1 / 3).length}; with any sample above one third of the timeout: ${ranked.filter((r) => r.maxRatio > 1 / 3).length}`);
  out.push('');
  out.push('== tests with any sample above one third of their timeout (max/timeout > 0.34), sorted by max/timeout');
  for (const r of [...ranked].filter((x) => x.maxRatio > 1 / 3).sort((a, b) => b.maxRatio - a.maxRatio)) {
    out.push(`${r.entry.project} ${r.entry.spec} timeout=${r.timeout} n=${r.s.n} median=${r.s.median} p95=${r.s.p95} max=${r.s.max} max/timeout=${r.maxRatio.toFixed(2)} | ${r.entry.title.slice(0, 90)}`);
  }
}

// 2c. The samples above 10 s of every test with a 30 s timeout, so the reader sees which runs they come from.
out.push('');
out.push('== samples above 10 s of the tests that have the default 30 s timeout (gate/run:seconds)');
for (const [key, entry] of byTest) {
  if (timeoutOf.get(key) !== 30) continue;
  const slow = entry.where.filter((w) => Number(w.split(':')[1]) > 10);
  if (slow.length === 0) continue;
  out.push(`${entry.project} ${entry.spec} | ${entry.title.slice(0, 70)} | ${slow.join(' ')}`);
}

// 3. distribution of the test-level durations of all runs: how many samples are above 20 s and 25 s.
let above20 = 0;
let above25 = 0;
let all = 0;
for (const run of runs) for (const row of run.rows) { all += 1; if (row.seconds > 20) above20 += 1; if (row.seconds > 25) above25 += 1; }
out.push('');
out.push(`samples: ${all}; above 20 s: ${above20}; above 25 s: ${above25}`);

writeFileSync(outFile, out.join('\n') + '\n');
