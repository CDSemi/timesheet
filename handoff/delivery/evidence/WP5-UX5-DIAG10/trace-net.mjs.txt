// Lists the network entries of a Playwright trace (relative to the first one), compact.
// Usage: node trace-net.mjs <trace.network.jsonl> <outFile> <contextWallTimeMs> <shiftMs>
// The context wall time comes from the shifted (simulated) clock of the test runner; the browser stamps requests with
// the real clock, so the shift is taken off the wall time first.
import { readFileSync, writeFileSync } from 'node:fs';

const [file, outFile, wallArg, shiftArg] = process.argv.slice(2);
const wall0 = Number(wallArg) - Number(shiftArg ?? 0);
const rows = [];
for (const line of readFileSync(file, 'utf8').split('\n')) {
  if (line.length === 0) continue;
  const e = JSON.parse(line);
  if (e.type !== 'resource-snapshot') continue;
  const s = e.snapshot;
  const started = Date.parse(s.startedDateTime);
  const t = s.timings ?? {};
  rows.push({
    start: started - wall0,
    url: s.request.url.replace(/^https?:\/\/[^/]+/, ''),
    method: s.request.method,
    status: s.response.status,
    time: s.time,
    wait: t.wait,
    receive: t.receive,
    send: t.send,
    page: s._pageref,
  });
}
rows.sort((a, b) => a.start - b.start);
const out = [`entries ${rows.length}`];
for (const r of rows) {
  out.push(`${String(r.start).padStart(7)} +${String(Math.round(r.time)).padStart(6)} wait=${Math.round(r.wait ?? -1)} recv=${Math.round(r.receive ?? -1)} ${r.method} ${r.status} ${r.url.slice(0, 90)}`);
}
writeFileSync(outFile, out.join('\n') + '\n');
