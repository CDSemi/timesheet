# Usage: bash make-copies.sh <source spec> <prefix> <count>  - copies a spec file N times into the scratch export (new names, same directory).
source /d/.claude-tmp/timesheet/WP5-UX5-DIAG10/env.sh
SRC=$1; PREFIX=$2; N=$3
i=1
while [ $i -le $N ]; do
  nn=$(printf "%02d" $i)
  cp "$T/export/tests/e2e/$SRC" "$T/export/tests/e2e/$PREFIX$nn.spec.ts"
  i=$((i + 1))
done
ls $T/export/tests/e2e | grep "^$PREFIX"
