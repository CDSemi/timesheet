import { appendFileSync, readFileSync } from 'node:fs';
import type { Page } from '@playwright/test';
import { expect, newPerson, test } from './fixtures.ts';

/*
 * WP5-UX5-REGATE8 probe (adapted from the FIX11 probe) (the worker's own, never part of the repository): axe-core 4.14.0 (loaded from the task folder; no dependency
 * added) on the Review unsigned, signed, and signed and reloaded at 1280, 900, 768, 480 and 320px, light and dark, and with the week
 * card focused at 900 and 768px (where it scrolls and shows its ring). Tags WCAG 2.0 to 2.2 A and AA. Synthetic data only.
 */
test.use({ locale: 'en-US', timezoneId: 'America/Los_Angeles' });

const AXE_PATH = process.env.AXE_PATH ?? '';
const OUT = process.env.AXE_OUT ?? '';
const TAGS = ['wcag2a', 'wcag2aa', 'wcag21a', 'wcag21aa', 'wcag22aa'];
const WIDTHS = [1280, 900, 768, 480, 390, 320];
let axeSource = '';
function note(line: string): void {
  if (OUT !== '') appendFileSync(OUT, `${line}\n`);
}
const frames = (page: Page) => page.evaluate('new Promise((done) => requestAnimationFrame(() => requestAnimationFrame(done)))');

interface Result {
  version: string;
  violations: Array<{ id: string; impact: string | null; nodes: string[] }>;
  incomplete: string[];
}

async function scan(page: Page, label: string): Promise<number> {
  if (axeSource === '') axeSource = readFileSync(AXE_PATH, 'utf8');
  await frames(page);
  if (!(await page.evaluate<boolean>(`typeof window.axe !== 'undefined'`))) await page.evaluate(axeSource);
  const result = await page.evaluate<Result, string[]>(async (tags) => {
    const axe = (window as unknown as { axe: { run: (c: unknown, o: unknown) => Promise<Record<string, unknown>>; version: string } }).axe;
    const r = (await axe.run(document, { runOnly: { type: 'tag', values: tags }, resultTypes: ['violations', 'incomplete'] })) as {
      violations: Array<{ id: string; impact: string | null; nodes: Array<{ target: unknown }> }>;
      incomplete: Array<{ id: string }>;
    };
    return { version: axe.version, violations: r.violations.map((v) => ({ id: v.id, impact: v.impact, nodes: v.nodes.map((n) => JSON.stringify(n.target)) })), incomplete: [...new Set(r.incomplete.map((i) => i.id))] };
  }, TAGS);
  note(`${label}: axe ${result.version} violations=${result.violations.length} [${result.violations.map((v) => `${v.id}(${v.impact}) ${v.nodes.join(' ')}`).join('; ')}] incomplete=[${result.incomplete.join(',')}]`);
  return result.violations.length;
}

for (const scheme of ['light', 'dark'] as const) {
  test(`axe Review unsigned, signed, signed and reloaded - ${scheme}`, async ({ page, adminSeed, builtServer, isMobile }, testInfo) => {
    test.setTimeout(420_000);
    const project = testInfo.project.name;
    const widths = WIDTHS;
    await page.emulateMedia({ colorScheme: scheme, reducedMotion: 'reduce' });
    const person = await newPerson(adminSeed, builtServer, { displayName: `Synthetic Axe ${scheme}`, signature: 'saved' });
    const free = await person.api.displayedPeriodFreeWorkdays();
    await person.api.seedCompleteDay(free[0] ?? '');
    await page.goto(`/#/review/${person.payrollDate}`);
    await page.getByLabel('Email').fill(person.account.email);
    await page.getByLabel('Password').fill(person.account.password);
    await page.getByRole('button', { name: 'Sign in' }).click();
    await expect(page.locator('[data-review-day]')).toHaveCount(14);
    const tag = `${project} ${scheme}`;
    let scans = 0;
    let total = 0;
    const run = async (label: string) => {
      scans += 1;
      total += await scan(page, `[${tag}] ${label}`);
    };
    for (const width of widths) {
      await page.setViewportSize({ width, height: 900 });
      await run(`Review unsigned ${width}`);
    }
    if (!isMobile) {
      for (const width of [900, 768]) {
        await page.setViewportSize({ width, height: 900 });
        const card = page.locator('[data-paper="review"] .sheet-week').first();
        await page.keyboard.press('Shift');
        await card.focus();
        await expect(card).toBeFocused();
        await run(`Review unsigned ${width} week card focused`);
        await card.evaluate((element) => element.blur());
      }
    }
    await page.locator('[data-sign-off]').click();
    const dialog = page.getByRole('dialog', { name: 'Sign off confirmation' });
    await expect(dialog).toBeVisible();
    await dialog.getByRole('checkbox', { name: /I acknowledge the incomplete evidence/ }).check();
    await dialog.getByRole('button', { name: 'Sign off', exact: true }).click();
    await expect(page.locator('[data-result="created"]')).toContainText('Signed.');
    for (const width of widths) {
      await page.setViewportSize({ width, height: 900 });
      await run(`Review signed ${width}`);
    }
    await page.reload();
    await expect(page.locator('[data-sign-off]')).toHaveText('Sign correction');
    for (const width of widths) {
      await page.setViewportSize({ width, height: 900 });
      await expect.poll(() => page.evaluate<boolean>(`(() => { const i = document.querySelector('img.signed-image'); return !!i && i.complete && i.naturalWidth > 0; })()`)).toBe(true);
      await run(`Review signed and reloaded ${width}`);
    }
    note(`[${tag}] TOTAL violations ${total} in ${scans} scans`);
    expect(total).toBe(0);
  });

  test(`axe Timesheet page - ${scheme}`, async ({ page, adminSeed, builtServer }, testInfo) => {
    test.setTimeout(240_000);
    const project = testInfo.project.name;
    await page.emulateMedia({ colorScheme: scheme, reducedMotion: 'reduce' });
    const person = await newPerson(adminSeed, builtServer, { displayName: `Synthetic Axe Sheet ${scheme}`, signature: 'saved' });
    const free = await person.api.displayedPeriodFreeWorkdays();
    await person.api.seedCompleteDay(free[0] ?? '');
    await page.goto('/');
    await page.getByLabel('Email').fill(person.account.email);
    await page.getByLabel('Password').fill(person.account.password);
    await page.getByRole('button', { name: 'Sign in' }).click();
    await expect(page.locator('[data-day]')).toHaveCount(14);
    const tag = `${project} ${scheme}`;
    let total = 0;
    for (const width of [1280, 320]) {
      await page.setViewportSize({ width, height: 900 });
      total += await scan(page, `[${tag}] Timesheet page ${width}`);
    }
    note(`[${tag}] Timesheet TOTAL violations ${total} in 2 scans`);
    expect(total).toBe(0);
  });
}

import { namedDayButton } from './dayButton.ts';

for (const scheme of ['light', 'dark'] as const) {
  test(`axe Timesheet page with the Day editor open and typed Notes - ${scheme}`, async ({ page, adminSeed, builtServer }, testInfo) => {
    test.setTimeout(240_000);
    const project = testInfo.project.name;
    await page.emulateMedia({ colorScheme: scheme, reducedMotion: 'reduce' });
    const person = await newPerson(adminSeed, builtServer, { displayName: `Synthetic Axe Editor ${scheme}`, signature: 'none' });
    const free = await person.api.displayedPeriodFreeWorkdays();
    const day = free[0] ?? '';
    await page.goto('/');
    await page.getByLabel('Email').fill(person.account.email);
    await page.getByLabel('Password').fill(person.account.password);
    await page.getByRole('button', { name: 'Sign in' }).click();
    await expect(page.locator('[data-day]')).toHaveCount(14);
    const tag = `${project} ${scheme}`;
    let total = 0;
    for (const width of [1280, 390]) {
      await page.setViewportSize({ width, height: 900 });
      await expect.poll(() => page.evaluate<number>('window.innerWidth')).toBe(width);
      await namedDayButton(page, day, 'Edit').click();
      const editor = page.getByRole('dialog', { name: /Day editor/ });
      await expect(editor).toHaveAttribute('data-day-editor', day);
      const notes = editor.getByLabel('Notes');
      await expect(notes).toBeEditable();
      await notes.fill('Synthetic unsaved note');
      await expect(notes).toHaveValue('Synthetic unsaved note');
      total += await scan(page, `[${tag}] Timesheet Day editor open, typed Notes ${width}`);
      await page.keyboard.press('Escape');
      const keep = page.getByRole('button', { name: /Keep editing/i });
      if (await keep.isVisible()) {
        await page.getByRole('button', { name: /Discard/i }).first().click();
      }
      await expect(editor).toHaveCount(0);
    }
    note(`[${tag}] Day editor TOTAL violations ${total} in 2 scans`);
    expect(total).toBe(0);
  });
}
