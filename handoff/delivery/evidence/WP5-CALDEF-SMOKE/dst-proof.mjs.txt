import { readFileSync } from 'node:fs';
const load = (file) => {
  const src = readFileSync(file, 'utf8');
  const a = src.indexOf('function laDateParts');
  const b = src.indexOf('const local = ');
  return new Function(`${src.slice(a, b)}; return recentMonday;`);
};
const mk = (file) => {
  const body = load(file)();
  return (ms) => {
    const RealNow = Date.now;
    Date.now = () => ms;
    try { return body(ms); } catch (e) { return `ERR:${e.message}`; } finally { Date.now = RealNow; }
  };
};
const oldFn = mk(process.argv[2]);
const newFn = mk(process.argv[3]);
const fmt = new Intl.DateTimeFormat('en-CA', { timeZone: 'America/Los_Angeles', year: 'numeric', month: '2-digit', day: '2-digit' });
// independent oracle: LA date -> weekday via UTC -> days back to previous Monday (1..7)
const oracle = (ms) => {
  const [y, m, d] = fmt.format(new Date(ms)).split('-').map(Number);
  const dow = new Date(Date.UTC(y, m - 1, d)).getUTCDay();
  const back = ((dow + 6) % 7) || 7;
  return new Date(Date.UTC(y, m - 1, d - back)).toISOString().slice(0, 10);
};
const laMs = (s) => { // LA wall time -> instant, search offset
  for (const off of [7, 8]) { const t = Date.parse(`${s}:00Z`) + off * 3600000; const p = new Intl.DateTimeFormat('en-CA',{timeZone:'America/Los_Angeles',hour12:false,year:'numeric',month:'2-digit',day:'2-digit',hour:'2-digit',minute:'2-digit'}).format(new Date(t)).replace(',', ''); if (p.replace(' ','T').replace(/^(.{10}T)24/,'$100') === s.replace(' ','T')) return t; }
  throw new Error('no instant ' + s);
};
