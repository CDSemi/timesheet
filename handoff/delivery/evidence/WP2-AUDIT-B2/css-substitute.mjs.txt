// WP2-AUDIT-B2 static check (auditor-owned): substitute the tokens added by WP2-FIXB with their
// :root values and compare the result with styles.css at 8fae685.
// Usage: node css-substitute.mjs <old.css> <new.css>
import { readFileSync } from 'node:fs';

const escape = (text) => text.replace(/[.*+?^${}()|[\]\\]/g, (ch) => `\\${ch}`);
const [oldPath, newPath] = process.argv.slice(2);
const oldCss = readFileSync(oldPath, 'utf8').replace(/\r\n/g, '\n');
const newCss = readFileSync(newPath, 'utf8').replace(/\r\n/g, '\n');
const decl = /^[ \t]*(--[a-z0-9-]+):[ \t]*([^;\n]+);[ \t]*$/gm;
const oldTokens = new Set([...oldCss.matchAll(decl)].map((m) => m[1]));
const added = new Map([...newCss.matchAll(decl)].filter((m) => !oldTokens.has(m[1])).map((m) => [m[1], m[2].trim()]));
console.log(`tokens added in new: ${added.size}`);
for (const [k, v] of added) console.log(`  ${k}: ${v}`);
let substituted = newCss;
let removed = 0;
for (const k of added.keys()) {
  const before = substituted;
  substituted = substituted.replace(new RegExp(`^[ \\t]*${escape(k)}:[^;\\n]+;\\n`, 'm'), '');
  if (before !== substituted) removed += 1;
}
console.log(`declarations removed: ${removed}`);
let uses = 0;
for (const [k, v] of added) {
  substituted = substituted.replace(new RegExp(`var\\(${escape(k)}\\)`, 'g'), () => {
    uses += 1;
    return v;
  });
}
console.log(`var() uses substituted: ${uses}`);
const remaining = [...substituted.matchAll(/var\((--[a-z0-9-]+)\)/g)].map((m) => m[1]).filter((k) => !oldTokens.has(k));
console.log(`var() uses of unknown tokens left: ${remaining.length}`);
const a = oldCss.split('\n');
const b = substituted.split('\n');
let diffs = 0;
const max = Math.max(a.length, b.length);
for (let i = 0; i < max; i += 1) {
  if (a[i] !== b[i]) {
    diffs += 1;
    if (diffs <= 20) console.log(`line ${i + 1}: old=${JSON.stringify(a[i])} new=${JSON.stringify(b[i])}`);
  }
}
console.log(`old lines ${a.length}, substituted new lines ${b.length}, differing lines ${diffs}`);
console.log(diffs === 0 ? 'RESULT: IDENTICAL after substitution' : 'RESULT: DIFFERENT');
process.exitCode = diffs === 0 ? 0 : 1;
