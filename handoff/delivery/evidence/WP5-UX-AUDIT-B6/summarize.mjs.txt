// WP5-UX-AUDIT-B6: summarise probe JSON files of one tag. Usage: node summarize.mjs <probe-out dir> <tag> <out.txt>
import { readdirSync, readFileSync, writeFileSync } from 'node:fs';
import { join } from 'node:path';

const [dir, tag, outFile] = process.argv.slice(2);
const lines = [];
const files = readdirSync(dir).filter((name) => name.startsWith(`${tag}-`) && name.endsWith('.json')).sort();
const totals = { walks: 0, stops: 0, hidden: 0, noRing: 0, weakRing: 0, unmeasured: 0, refBlur: 0, refPrevious: 0, refNone: 0, notFv: 0 };
for (const file of files) {
  const data = JSON.parse(readFileSync(join(dir, file), 'utf8'));
  if (file.includes('-p1-walks-')) {
    lines.push(`== ${file}: geometry ${JSON.stringify(data.geometry)} errors ${JSON.stringify(data.errors)}`);
    for (const walk of data.walks) {
      totals.walks += 1;
      const stops = walk.stops ?? [];
      totals.stops += stops.length;
      const hidden = stops.filter((stop) => stop.visible === 0);
      const measured = stops.filter((stop) => stop.ring && stop.ring.sidesPresent !== null);
      const unmeasured = stops.filter((stop) => !stop.ring || stop.ring.sidesPresent === null);
      const noRing = measured.filter((stop) => stop.ring.sidesPresent === 0);
      const weak = measured.filter((stop) => stop.ring.sidesPresent > 0 && (stop.ring.minAdjacentBest ?? 0) < 3);
      const notFv = stops.filter((stop) => !stop.fv && !stop.fw);
      for (const stop of stops) {
        if (!stop.ring) continue;
        if (stop.ring.ref === 'blur') totals.refBlur += 1;
        else if (stop.ring.ref === 'previous') totals.refPrevious += 1;
        else if (stop.ring.ref === 'next') totals.refNext = (totals.refNext ?? 0) + 1;
        else totals.refNone += 1;
      }
      const partial = stops.filter((stop) => stop.visible > 0 && stop.visible < 25);
      totals.partial = (totals.partial ?? 0) + partial.length;
      for (const stop of partial) lines.push(`    PARTLY COVERED ${stop.tag} "${stop.name}" visible ${stop.visible}/25 covers ${stop.covers.join('|')}`);
      totals.hidden += hidden.length;
      totals.noRing += noRing.length;
      totals.weakRing += weak.length;
      totals.unmeasured += unmeasured.length;
      totals.notFv += notFv.length;
      const minVisible = stops.length === 0 ? null : Math.min(...stops.map((stop) => stop.visible));
      const minAdj = measured.length === 0 ? null : Math.min(...measured.filter((s) => s.ring.sidesPresent > 0).map((stop) => stop.ring.minAdjacentBest ?? 99));
      lines.push(`  ${walk.label} ${walk.direction}: stops ${stops.length}, hidden ${hidden.length}, min visible ${minVisible}/25, measured ${measured.length}, no ring ${noRing.length}, ring<3 ${weak.length}, unmeasured ${unmeasured.length}, min ring-adjacent ${minAdj}${walk.note ? `, note ${walk.note}` : ''}`);
      const daySeq = stops.filter((stop) => stop.dayButton).map((stop) => stop.day);
      if (daySeq.length > 0) lines.push(`    day buttons in order: ${daySeq.length} ${JSON.stringify(daySeq) === JSON.stringify([...daySeq].sort((a, b) => (walk.direction === 'forward' ? a.localeCompare(b) : b.localeCompare(a)))) ? 'ascending in walk direction' : 'NOT IN ORDER ' + daySeq.join(',')}`);
      for (const stop of hidden) lines.push(`    HIDDEN ${stop.tag} "${stop.name}" covers ${stop.covers.join('|')} rect ${JSON.stringify(stop.rect)}`);
      for (const stop of noRing) lines.push(`    NO RING ${stop.tag}${stop.type ? `[${stop.type}]` : ''} "${stop.name}" fv=${stop.fv} fw=${stop.fw} ref=${stop.ring.ref} maxChange=${stop.ring.maxChange} shadow=${stop.boxShadow.slice(0, 80)}`);
      for (const stop of weak) lines.push(`    RING<3 ${stop.tag} "${stop.name}" sides=${stop.ring.sidesPresent} adj=${stop.ring.minAdjacentBest} outer=${stop.ring.minOuter} ref=${stop.ring.ref}`);
      for (const stop of unmeasured) lines.push(`    UNMEASURED ${stop.tag}${stop.type ? `[${stop.type}]` : ''} "${stop.name}" ref=${stop.ring?.ref} blurTook=${stop.ring?.blurTook} fv=${stop.fv} fw=${stop.fw} computed shadow=${stop.boxShadow.slice(0, 90)}`);
      for (const stop of notFv) lines.push(`    NOT :focus-visible ${stop.tag}${stop.type ? `[${stop.type}]` : ''} "${stop.name}" fw=${stop.fw}`);
    }
  }
}
lines.unshift(`TOTALS ${JSON.stringify(totals)}`);
writeFileSync(outFile, `${lines.join('\n')}\n`);
