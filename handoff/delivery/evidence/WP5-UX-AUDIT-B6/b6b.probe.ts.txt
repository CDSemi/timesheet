/*
 * WP5-UX-AUDIT-B6 follow-up probes (auditor-written; outside git tracking): WCAG 2.2 SC 2.5.8 target size
 * (minimum 24x24 CSS px or the spacing exception) on the desktop widths, and the two AX-07 notices the gate
 * could not reach (administrator new-user password notice, Settings policy notice). Synthetic data only;
 * own built server on the task port range, stopped through its handle.
 */

const REPO = resolve(import.meta.dirname, '..');
const OUT = process.env.PROBE_OUT ?? join(REPO, '..', 'probe-out');
const WORK = process.env.PROBE_WORK ?? join(REPO, '..', 'probe-work');
const PORT_BASE = Number(process.env.PROBE_PORT_BASE ?? '48310');
const TAG = process.env.PROBE_TAG ?? 'b6b';
mkdirSync(OUT, { recursive: true });
mkdirSync(WORK, { recursive: true });

function record(name: string, data: unknown) {
  writeFileSync(join(OUT, `${TAG}-${name}.json`), `${JSON.stringify(data, null, 1)}\n`);
}

interface Credentials {
  email: string;
  password: string;
}

async function waitHealthy(origin: string, child: ChildProcess) {
  for (let attempt = 0; attempt < 150; attempt += 1) {
    if (child.exitCode !== null) throw new Error(`server exited early with ${child.exitCode}`);
    try {
      if ((await fetch(`${origin}/api/health`)).ok) return;
    } catch {
      // not listening yet
    }
    await new Promise((done) => setTimeout(done, 100));
  }
  throw new Error('server did not become healthy');
}

const test = base.extend<{ adminSeed: SeedClient }, { server: { origin: string; admin: Credentials } }>({
  server: [
    async ({}, use, workerInfo) => {
      const port = workerInfo.project.name === 'desktop' ? PORT_BASE : PORT_BASE + 1;
      const work = mkdtempSync(join(WORK, `${TAG}-${workerInfo.project.name}-`));
      const origin = `http://127.0.0.1:${port}`;
      const admin = { email: 'admin@example.invalid', password: randomBytes(18).toString('base64url') };
      const env: NodeJS.ProcessEnv = {
        ...process.env,
        NODE_ENV: 'development',
        HOST: '127.0.0.1',
        PORT: String(port),
        DATABASE_PATH: join(work, 'timesheet.db'),
        DATA_DIR: join(work, 'private-data'),
        APP_ORIGINS: origin,
        OUTBOUND_MODE: 'capture',
        MAIL_FROM: 'timesheet-capture@example.invalid',
        SEED_ADMIN_PASSWORD: admin.password,
        SEED_EMPLOYEE_PASSWORD: randomBytes(18).toString('base64url'),
      };
      for (const command of ['migrate', 'seed']) {
        const result = spawnSync(process.execPath, ['dist/server/cli.js', command], { cwd: REPO, env, encoding: 'utf8' });
        if (result.status !== 0) throw new Error(`cli ${command} failed: ${result.status} ${result.stderr}`);
      }
      const errFd = openSync(join(OUT, `${TAG}-server-${workerInfo.project.name}-stderr.txt`), 'w');
      const child = spawn(process.execPath, ['dist/server/index.js'], { cwd: REPO, env, stdio: ['ignore', 'ignore', errFd] });
      try {
        await waitHealthy(origin, child);
        await use({ origin, admin });
      } finally {
        if (child.exitCode === null) {
          const exited = new Promise((done) => child.once('exit', done));
          child.kill();
          await exited;
        }
        closeSync(errFd);
        writeFileSync(join(OUT, `${TAG}-server-${workerInfo.project.name}-stopped.json`), `${JSON.stringify({ port, exitCode: child.exitCode, signal: child.signalCode })}\n`);
      }
    },
    { scope: 'worker', timeout: 90_000 },
  ],
  baseURL: async ({ server }, use) => {
    await use(server.origin);
  },
  adminSeed: async ({ server }, use) => {
    await use(await new SeedClient(server.origin).signIn(server.admin.email, server.admin.password));
  },
});

test.use({ timezoneId: 'Asia/Ho_Chi_Minh', locale: 'en-US' });

async function signIn(page: Page, who: Credentials, hash = '#/timesheet') {
  await page.goto(`/${hash}`);
  await page.getByLabel('Email').fill(who.email);
  await page.getByLabel('Password').fill(who.password);
  await page.getByRole('button', { name: 'Sign in' }).click();
  await expect(page.getByRole('navigation', { name: 'Main' })).toBeVisible();
}

/** SC 2.5.8: every visible target under 24x24 CSS px, with the spacing test (a 24px circle on its centre must not touch another target). */
async function targets(page: Page, state: string) {
  return page.evaluate((label) => {
    const selector = 'button, a[href], input:not([type="hidden"]), select, textarea, [role="option"], summary, label.pick';
    const all: Array<{ element: HTMLElement; rect: DOMRect }> = [];
    for (const element of document.querySelectorAll<HTMLElement>(selector)) {
      if (element.getClientRects().length === 0 || element.closest('[inert]') !== null) continue;
      if (getComputedStyle(element).visibility === 'hidden') continue;
      const rect = element.getBoundingClientRect();
      if (rect.width === 0 || rect.height === 0) continue;
      // An inline link inside a sentence is exempt (inline exception).
      if (element.tagName === 'A' && getComputedStyle(element).display === 'inline' && (element.parentElement?.innerText.length ?? 0) > (element.innerText.length + 20)) continue;
      all.push({ element, rect });
    }
    const small: Array<{ tag: string; text: string; w: number; h: number; labelArea: string | null; spacingOk: boolean; nearest: string }> = [];
    for (const { element, rect } of all) {
      // A control whose own label activates it: the label box counts as part of the target.
      let union = rect;
      const labels = (element as HTMLInputElement).labels ? [...((element as HTMLInputElement).labels ?? [])] : [];
      for (const lab of labels) {
        const r = lab.getBoundingClientRect();
        const left = Math.min(union.left, r.left);
        const top = Math.min(union.top, r.top);
        const right = Math.max(union.right, r.right);
        const bottom = Math.max(union.bottom, r.bottom);
        union = new DOMRect(left, top, right - left, bottom - top);
      }
      if (union.width >= 24 && union.height >= 24) continue;
      if (rect.width >= 24 && rect.height >= 24) continue;
      const cx = rect.left + rect.width / 2;
      const cy = rect.top + rect.height / 2;
      let nearest = Number.POSITIVE_INFINITY;
      let nearestText = '';
      for (const other of all) {
        if (other.element === element || other.element.contains(element) || element.contains(other.element)) continue;
        if (labels.some((lab) => lab.contains(other.element) || other.element.contains(lab))) continue;
        const r = other.rect;
        const dx = Math.max(r.left - cx, 0, cx - r.right);
        const dy = Math.max(r.top - cy, 0, cy - r.bottom);
        const distance = Math.hypot(dx, dy);
        if (distance < nearest) {
          nearest = distance;
          nearestText = `${other.element.tagName.toLowerCase()} ${(other.element.innerText || other.element.getAttribute('aria-label') || '').trim().slice(0, 30)}`;
        }
      }
      small.push({
        tag: `${element.tagName.toLowerCase()}${element.getAttribute('type') ? `[${element.getAttribute('type')}]` : ''}`,
        text: (element.innerText || element.getAttribute('aria-label') || '').trim().slice(0, 40),
        w: Math.round(rect.width * 10) / 10,
        h: Math.round(rect.height * 10) / 10,
        labelArea: labels.length === 0 ? null : `${Math.round(union.width)}x${Math.round(union.height)}`,
        spacingOk: nearest >= 12,
        nearest: `${Math.round(nearest * 10) / 10}px to ${nearestText}`,
      });
    }
    return { state: label, count: all.length, small };
  }, state);
}

test('P9 desktop target size and AX-07 notices', async ({ page, adminSeed, server }, testInfo) => {
  test.skip(testInfo.project.name !== 'desktop', 'desktop widths');
  const out: Record<string, unknown> = {};
  const me = await newPerson(adminSeed, { origin: server.origin } as never, { displayName: 'Synthetic Target Person', signature: 'none' });
  const owner = await newPerson(adminSeed, { origin: server.origin } as never, { displayName: 'Synthetic Target Owner', signature: 'none' });
  const [day = ''] = await me.api.displayedPeriodFreeWorkdays();
  await me.api.seedCompleteDay(day);
  await owner.api.call('POST', '/api/shares', { grantee_email: me.account.email, items: { timesheets: 'view', ot_read: false, pdf_download: false } }, 201);
  await me.api.call('POST', '/api/shares', { grantee_email: owner.account.email, items: { timesheets: 'view', ot_read: false, pdf_download: false } }, 201);
  await signIn(page, me.account);
  const results: unknown[] = [];
  for (const width of [1280, 768]) {
    await page.setViewportSize({ width, height: 900 });
    await page.goto('/#/timesheet');
    await page.reload();
    await expect(page.locator('[data-day]')).toHaveCount(14);
    results.push({ width, ...(await targets(page, 'timesheet')) });
    await page.getByRole('button', { name: 'Change several days', exact: true }).click();
    results.push({ width, ...(await targets(page, 'batch')) });
    await page.getByRole('button', { name: 'Done' }).click();
    await page.locator(`[data-day="${day}"] [data-day-button]`).click();
    await expect(page.locator(`dialog[data-day-editor="${day}"]`).getByRole('heading', { name: /^Times/ })).toBeVisible();
    await page.locator(`dialog[data-day-editor="${day}"]`).getByRole('button', { name: 'Add session' }).click();
    results.push({ width, ...(await targets(page, 'editor-with-form')) });
    await page.keyboard.press('Escape');
    for (const hash of ['#/settings', '#/ot', '#/history', '#/import', `#/shared/${owner.account.id}`]) {
      await page.goto(`/${hash}`);
      await page.reload();
      await page.waitForTimeout(600);
      results.push({ width, ...(await targets(page, hash.startsWith('#/shared') ? '#/shared' : hash)) });
    }
  }
  out.targets = results;

  // The switcher choice without Open, then another own screen through the navigation: what does it show?
  try {
    await page.setViewportSize({ width: 1280, height: 900 });
    await page.goto('/#/timesheet');
    await page.reload();
    const select = page.getByRole('combobox', { name: 'Shared with me' });
    await select.selectOption(owner.account.id);
    await page.getByRole('navigation', { name: 'Main' }).getByRole('link', { name: 'Settings' }).click();
    await expect(page.getByRole('heading', { name: 'Settings', level: 1 })).toBeVisible();
    out.switcherAfterNavigation = { hash: await page.evaluate(() => window.location.hash), value: await select.inputValue(), owner: owner.account.id, shareBar: await page.locator('[data-share-bar]').count() };
  } catch (error) {
    out.switcherAfterNavigation = { error: String(error).slice(0, 300) };
  }

  // AX-07: the Settings policy notice (an invalid draft) is a status message.
  try {
    await page.setViewportSize({ width: 1280, height: 900 });
    await page.goto('/#/settings');
    await page.reload();
    const form = page.getByRole('form', { name: 'New policy version' });
    await expect(form).toBeVisible();
    const fields = await form.locator('input').evaluateAll((inputs) => inputs.map((input) => `${(input as HTMLInputElement).type}:${(input as HTMLInputElement).labels?.[0]?.innerText.trim().slice(0, 30) ?? ''}`));
    // Clear the first number field to make the draft invalid.
    const number = form.locator('input[type="number"], input[inputmode="numeric"], input[type="text"]').first();
    await number.fill('');
    await page.waitForTimeout(300);
    const statuses = await form.locator('[role="status"]').allInnerTexts();
    const notices = await form.locator('.notice').evaluateAll((elements) => elements.map((element) => ({ role: element.getAttribute('role'), text: (element as HTMLElement).innerText.slice(0, 80) })));
    out.policyNotice = { fields, statuses, notices };
  } catch (error) {
    out.policyNotice = { error: String(error).slice(0, 300) };
  }

  // AX-07: the administrator new-user password notice.
  try {
    await page.goto('/#/timesheet');
    await page.getByRole('button', { name: 'Sign out' }).first().click();
    await expect(page.getByRole('button', { name: 'Sign in' })).toBeVisible();
    await signIn(page, server.admin, '#/admin');
    await page.goto('/#/admin');
    await page.waitForTimeout(600);
    const forms = await page.locator('form').evaluateAll((elements) => elements.map((element) => element.getAttribute('aria-label') ?? element.querySelector('h2, h3')?.textContent ?? ''));
    const password = page.getByLabel(/temporary password/i).first();
    await password.fill('short');
    await page.waitForTimeout(300);
    const status = await password.evaluate((input) => {
      const form = input.closest('form');
      return [...(form?.querySelectorAll('.notice') ?? [])].map((element) => ({ role: element.getAttribute('role'), text: (element as HTMLElement).innerText.slice(0, 120) }));
    });
    out.adminNotice = { forms, status };
  } catch (error) {
    out.adminNotice = { error: String(error).slice(0, 300) };
  }
  record('p9-targets-notices-desktop', out);
});

/* ---- P10 (phone): small checkboxes and radios and their label targets; first screen of a grantee ------ */

test('P10 phone label targets and grantee first screen', async ({ page, adminSeed, server }, testInfo) => {
  test.skip(testInfo.project.name !== 'mobile', 'phone widths');
  const out: Record<string, unknown> = {};
  const me = await newPerson(adminSeed, { origin: server.origin } as never, { displayName: 'Synthetic Phone Person', signature: 'saved' });
  const owner = await newPerson(adminSeed, { origin: server.origin } as never, { displayName: 'Synthetic Phone Owner', signature: 'none' });
  const [day = ''] = await me.api.displayedPeriodFreeWorkdays();
  await me.api.seedCompleteDay(day);
  await me.api.call('POST', '/api/shares', { grantee_email: owner.account.email, items: { timesheets: 'view', ot_read: false, pdf_download: false } }, 201);
  await signIn(page, me.account);
  const smallInputs = async (state: string) =>
    page.evaluate((label) => {
      const rows: Array<{ type: string; input: string; label: string | null; labelClass: string; ok: boolean }> = [];
      for (const input of document.querySelectorAll<HTMLInputElement>('input[type="checkbox"], input[type="radio"]')) {
        if (input.getClientRects().length === 0 || input.closest('[inert]') !== null) continue;
        const r = input.getBoundingClientRect();
        const lab = input.closest('label') ?? input.labels?.[0] ?? null;
        const l = lab?.getBoundingClientRect();
        rows.push({
          type: input.type,
          input: `${Math.round(r.width * 10) / 10}x${Math.round(r.height * 10) / 10}`,
          label: l === undefined ? null : `${Math.round(l.width * 10) / 10}x${Math.round(l.height * 10) / 10}`,
          labelClass: lab?.className ?? '',
          ok: l !== undefined && l.width >= 44 && l.height >= 44,
        });
      }
      return { state: label, inputs: rows.length, notOk: rows.filter((row) => !row.ok), sample: rows.slice(0, 3) };
    }, state);
  const rows: unknown[] = [];
  for (const width of [390, 320]) {
    await page.setViewportSize({ width, height: width === 320 ? 640 : 844 });
    await page.goto('/#/settings');
    await page.reload();
    await page.waitForTimeout(600);
    rows.push({ width, ...(await smallInputs('settings')) });
    await page.goto('/#/import');
    await page.reload();
    await page.waitForTimeout(600);
    rows.push({ width, ...(await smallInputs('import')) });
    await page.goto('/#/timesheet');
    await page.reload();
    await expect(page.locator('[data-day]')).toHaveCount(14);
    await page.locator(`[data-day="${day}"] [data-day-button]`).click();
    await expect(page.locator(`dialog[data-day-editor="${day}"]`).getByRole('heading', { name: /^Times/ })).toBeVisible();
    rows.push({ width, ...(await smallInputs('editor')) });
    await page.keyboard.press('Escape');
    await page.locator('[data-sheet-review-link], a[href*="review"]').first().click();
    await page.waitForTimeout(800);
    rows.push({ width, ...(await smallInputs('review')) });
  }
  out.labelTargets = rows;
  // The first screen of a person who has a received share (the switcher widens the shell bar) at 390x844.
  await page.getByRole('navigation', { name: 'Main' }).getByRole('button', { name: 'More' }).click().catch(() => undefined);
  await page.goto('/#/timesheet');
  await page.getByRole('navigation', { name: 'Main' }).getByRole('button', { name: 'More' }).click();
  await page.getByRole('button', { name: 'Sign out' }).click();
  await expect(page.getByRole('button', { name: 'Sign in' })).toBeVisible();
  const firstScreen: unknown[] = [];
  for (const who of [me.account, owner.account]) {
    await page.setViewportSize({ width: 390, height: 844 });
    await signIn(page, who);
    await page.goto('/#/timesheet');
    await page.reload();
    await expect(page.locator('[data-day]')).toHaveCount(14);
    await page.evaluate(() => window.scrollTo(0, 0));
    firstScreen.push(
      await page.evaluate((name) => {
        const first = document.querySelector('tr.sheet-row[data-day]')?.getBoundingClientRect();
        const tabs = document.querySelector('.shell-tabs')?.getBoundingClientRect();
        const bar = document.querySelector('.shell-bar')?.getBoundingClientRect();
        return { who: name, switcher: document.querySelector('form.shell-switcher') !== null, shellBar: bar?.height ?? null, firstRowBottom: first?.bottom ?? null, tabBarTop: tabs?.top ?? null, visible: first !== undefined && tabs !== undefined && first.bottom <= tabs.top };
      }, who === me.account ? 'grantor (no received share)' : 'grantee (received share)'),
    );
    await page.getByRole('navigation', { name: 'Main' }).getByRole('button', { name: 'More' }).click();
    await page.getByRole('button', { name: 'Sign out' }).click();
    await expect(page.getByRole('button', { name: 'Sign in' })).toBeVisible();
  }
  out.firstScreen = firstScreen;
  record('p10-phone-labels-first-screen', out);
});
