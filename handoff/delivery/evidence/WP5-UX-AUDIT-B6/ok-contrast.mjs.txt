// WP5-UX-AUDIT-B6: --ok text contrast over every light and dark surface token, plain and under the selection
// tint (and the tint twice, as a phone selected row over a head cell renders it), from the colours the
// browser computed in P5. Usage: node ok-contrast.mjs <p5.json> <out.txt>
import { readFileSync, writeFileSync } from 'node:fs';

const [input, output] = process.argv.slice(2);
const data = JSON.parse(readFileSync(input, 'utf8'));
const parse = (text) => {
  const [r = 0, g = 0, b = 0, a = 1] = (/rgba?\(([^)]*)\)/.exec(text)?.[1] ?? '').split(/[\s,/]+/).filter(Boolean).map(Number);
  return [r, g, b, a];
};
const over = (top, bottom) => {
  const a = top[3] + bottom[3] * (1 - top[3]);
  const mix = (i) => (top[i] * top[3] + bottom[i] * bottom[3] * (1 - top[3])) / (a || 1);
  return [mix(0), mix(1), mix(2), a];
};
const lum = ([r, g, b]) => {
  const f = (v) => {
    const s = v / 255;
    return s <= 0.04045 ? s / 12.92 : ((s + 0.055) / 1.055) ** 2.4;
  };
  return 0.2126 * f(r) + 0.7152 * f(g) + 0.0722 * f(b);
};
const ratio = (a, b) => {
  const [x, y] = [lum(a), lum(b)].sort((p, q) => q - p);
  return Math.round(((x + 0.05) / (y + 0.05)) * 100) / 100;
};
const lines = [];
for (const theme of ['light', 'dark']) {
  const tokens = data[theme].tokens;
  const ok = parse(tokens.ok);
  const tint = parse(tokens.selected);
  let lowest = 99;
  for (const { token, colour } of tokens.surfaces) {
    const base = parse(colour);
    const plain = ratio(over(ok, base), base);
    const once = over(tint, base);
    const tinted = ratio(over(ok, once), once);
    const twice = over(tint, once);
    const tinted2 = ratio(over(ok, twice), twice);
    lowest = Math.min(lowest, plain, tinted, tinted2);
    lines.push(`${theme} --ok ${tokens.ok} on ${token} ${colour}: plain ${plain}, with the selection tint ${tinted}, tint twice ${tinted2}`);
  }
  lines.push(`${theme} lowest ${lowest}`);
}
writeFileSync(output, `${lines.join('\n')}\n`);
