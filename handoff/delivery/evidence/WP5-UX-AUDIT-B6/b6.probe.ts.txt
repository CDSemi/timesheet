/*
 * WP5-UX-AUDIT-B6 independent probes (auditor-written for this audit; not part of the reviewed source;
 * kept out of git through .git/info/exclude of the scratch clone). Synthetic data only. Each project
 * starts its own built server (dist/) on a fixed port of the task range (desktop PROBE_PORT_BASE, mobile
 * PROBE_PORT_BASE + 1) with DATA_DIR and DATABASE_PATH inside the task folder, stopped through its handle.
 *
 * Keyboard: every walk uses real Tab / Shift+Tab presses; the pickers, toggles, dialogs and the switcher
 * are driven with real Enter / Escape / arrow presses. Focus ring: at each stop the viewport is captured
 * with the control focused and again unfocused (blur(); when blur() does not take, the screenshot taken
 * before the key press that brought focus, if nothing scrolled), and the ring is read on lines across each
 * side (the pixel run that changed most; its colour against the rendered pixel just outside and inside it).
 * Occlusion: a 5x5 elementFromPoint grid per stop. Hard expectations only guard the probe itself; the
 * results are written as JSON and judged in the review.
 */

async function waitHealthy(origin: string, child: ChildProcess) {
  for (let attempt = 0; attempt < 150; attempt += 1) {
    if (child.exitCode !== null) throw new Error(`server exited early with ${child.exitCode}`);
    try {
      if ((await fetch(`${origin}/api/health`)).ok) return;
    } catch {
      // not listening yet
    }
    await new Promise((done) => setTimeout(done, 100));
  }
  throw new Error('server did not become healthy');
}

    if (day !== '') await person.api.seedCompleteDay(day);

/* ---- PNG decoding and colour maths ------------------------------------------------------------------ */

type RGB = [number, number, number];
interface Img {
  width: number;
  height: number;
  data: Buffer;
  bpp: number;
}

function decodePng(buf: Buffer): Img {
  let pos = 8;
  let width = 0;
  let height = 0;
  let bitDepth = 0;
  let colorType = 0;
  let interlace = 0;
  const idat: Buffer[] = [];
  while (pos < buf.length) {
    const length = buf.readUInt32BE(pos);
    const type = buf.toString('latin1', pos + 4, pos + 8);
    const chunk = buf.subarray(pos + 8, pos + 8 + length);
    if (type === 'IHDR') {
      width = chunk.readUInt32BE(0);
      height = chunk.readUInt32BE(4);
      bitDepth = chunk[8] ?? 0;
      colorType = chunk[9] ?? 0;
      interlace = chunk[12] ?? 0;
    } else if (type === 'IDAT') idat.push(chunk);
    else if (type === 'IEND') break;
    pos += 12 + length;
  }
  if (bitDepth !== 8 || interlace !== 0 || (colorType !== 2 && colorType !== 6)) throw new Error(`unsupported PNG ${bitDepth}/${colorType}/${interlace}`);
  const bpp = colorType === 6 ? 4 : 3;
  const raw = inflateSync(Buffer.concat(idat));
  const stride = width * bpp;
  const data = Buffer.alloc(height * stride);
  for (let y = 0; y < height; y += 1) {
    const filter = raw[y * (stride + 1)] ?? 0;
    const src = y * (stride + 1) + 1;
    const dst = y * stride;
    for (let i = 0; i < stride; i += 1) {
      const left = i >= bpp ? (data[dst + i - bpp] ?? 0) : 0;
      const up = y > 0 ? (data[dst - stride + i] ?? 0) : 0;
      const upLeft = i >= bpp && y > 0 ? (data[dst - stride + i - bpp] ?? 0) : 0;
      const x = raw[src + i] ?? 0;
      let value = x;
      if (filter === 1) value = x + left;
      else if (filter === 2) value = x + up;
      else if (filter === 3) value = x + ((left + up) >> 1);
      else if (filter === 4) {
        const p = left + up - upLeft;
        const pa = Math.abs(p - left);
        const pb = Math.abs(p - up);
        const pc = Math.abs(p - upLeft);
        value = x + (pa <= pb && pa <= pc ? left : pb <= pc ? up : upLeft);
      } else if (filter !== 0) throw new Error(`PNG filter ${filter}`);
      data[dst + i] = value & 0xff;
    }
  }
  return { width, height, data, bpp };
}

function pixel(img: Img, x: number, y: number): RGB | null {
  if (x < 0 || y < 0 || x >= img.width || y >= img.height) return null;
  const o = (y * img.width + x) * img.bpp;
  return [img.data[o] ?? 0, img.data[o + 1] ?? 0, img.data[o + 2] ?? 0];
}

function luminance([r, g, b]: RGB): number {
  const c = (v: number) => {
    const s = v / 255;
    return s <= 0.04045 ? s / 12.92 : ((s + 0.055) / 1.055) ** 2.4;
  };
  return 0.2126 * c(r) + 0.7152 * c(g) + 0.0722 * c(b);
}
function contrast(a: RGB, b: RGB): number {
  const [x, y] = [luminance(a), luminance(b)].sort((p, q) => q - p) as [number, number];
  return (x + 0.05) / (y + 0.05);
}
const hex = (c: RGB | null) => (c === null ? null : `#${c.map((v) => v.toString(16).padStart(2, '0')).join('')}`);
const r2 = (n: number) => Math.round(n * 100) / 100;

interface Rect {
  x: number;
  y: number;
  width: number;
  height: number;
}

function analyseLine(points: Array<{ d: number; f: RGB; u: RGB }>, dsf: number) {
  let best = -1;
  let bestCh = 1;
  points.forEach((point, index) => {
    const ch = contrast(point.f, point.u);
    if (ch > bestCh + 1e-9) {
      bestCh = ch;
      best = index;
    }
  });
  if (best < 0) return { change: 1, d: null as number | null, ring: null as string | null, outer: null as string | null, inner: null as string | null, outerRatio: null as number | null, innerRatio: null as number | null, bandCss: 0 };
  const ring = points[best]?.f as RGB;
  let start = best;
  let end = best;
  while (start - 1 >= 0 && contrast(points[start - 1]?.f as RGB, ring) < 1.15) start -= 1;
  while (end + 1 < points.length && contrast(points[end + 1]?.f as RGB, ring) < 1.15) end += 1;
  const outer = points[end + 1]?.f ?? null;
  const inner = points[start - 1]?.f ?? null;
  return {
    change: r2(bestCh),
    d: r2(points[best]?.d ?? 0),
    ring: hex(ring),
    outer: hex(outer),
    inner: hex(inner),
    outerRatio: outer === null ? null : r2(contrast(ring, outer)),
    innerRatio: inner === null ? null : r2(contrast(ring, inner)),
    bandCss: r2((end - start + 1) / dsf),
  };
}

/** Differential ring reading: focused image against an unfocused image of the same scroll state. */
function ringOf(focused: Img, unfocused: Img, rect: Rect, dsf: number) {
  const sides: Array<{ side: string; at: number } & ReturnType<typeof analyseLine>> = [];
  for (const side of ['top', 'bottom', 'left', 'right'] as const) {
    let bestSide: ({ side: string; at: number } & ReturnType<typeof analyseLine>) | null = null;
    const vertical = side === 'top' || side === 'bottom';
    const inside = Math.max(0.5, Math.min(6, (vertical ? rect.height : rect.width) / 2 - 0.5));
    for (const at of [0.3, 0.5, 0.7]) {
      const points: Array<{ d: number; f: RGB; u: RGB }> = [];
      if (vertical) {
        const col = Math.floor((rect.x + at * rect.width) * dsf);
        const edge = side === 'top' ? rect.y : rect.y + rect.height;
        const sign = side === 'top' ? -1 : 1;
        const from = Math.floor((edge - sign * inside) * dsf);
        const to = Math.floor((edge + sign * 8) * dsf);
        for (let row = from; sign > 0 ? row <= to : row >= to; row += sign) {
          const f = pixel(focused, col, row);
          const u = pixel(unfocused, col, row);
          if (f === null || u === null) continue;
          points.push({ d: sign * ((row + 0.5) / dsf - edge), f, u });
        }
      } else {
        const row = Math.floor((rect.y + at * rect.height) * dsf);
        const edge = side === 'left' ? rect.x : rect.x + rect.width;
        const sign = side === 'left' ? -1 : 1;
        const from = Math.floor((edge - sign * inside) * dsf);
        const to = Math.floor((edge + sign * 8) * dsf);
        for (let col = from; sign > 0 ? col <= to : col >= to; col += sign) {
          const f = pixel(focused, col, row);
          const u = pixel(unfocused, col, row);
          if (f === null || u === null) continue;
          points.push({ d: sign * ((col + 0.5) / dsf - edge), f, u });
        }
      }
      const result = { side, at, ...analyseLine(points, dsf) };
      if (bestSide === null || result.change > bestSide.change) bestSide = result;
    }
    if (bestSide !== null) sides.push(bestSide);
  }
  const present = sides.filter((side) => side.change >= 3);
  const adjacent = present.map((side) => Math.max(side.outerRatio ?? 0, side.innerRatio ?? 0)).filter((value) => value > 0);
  const outerRatios = present.filter((side) => side.outerRatio !== null).map((side) => side.outerRatio ?? 0);
  return {
    sidesPresent: present.length,
    minOuter: outerRatios.length === 0 ? null : r2(Math.min(...outerRatios)),
    minAdjacentBest: adjacent.length === 0 ? null : r2(Math.min(...adjacent)),
    minChange: present.length === 0 ? null : r2(Math.min(...present.map((side) => side.change))),
    maxChange: sides.length === 0 ? null : r2(Math.max(...sides.map((side) => side.change))),
    sides,
  };
}

/** Intrinsic outset-ring reading (no reference image): the strongest pixel within 0-5px outside each side against the pixel 7-9px out. */
function outsetRingOf(img: Img, rect: Rect, dsf: number) {
  const sides: Array<{ side: string; ring: string | null; far: string | null; ratio: number; d: number }> = [];
  for (const side of ['top', 'bottom', 'left', 'right'] as const) {
    let best: { side: string; ring: string | null; far: string | null; ratio: number; d: number } | null = null;
    for (const at of [0.3, 0.5, 0.7]) {
      const vertical = side === 'top' || side === 'bottom';
      const sign = side === 'top' || side === 'left' ? -1 : 1;
      const edge = vertical ? (side === 'top' ? rect.y : rect.y + rect.height) : side === 'left' ? rect.x : rect.x + rect.width;
      const across = vertical ? rect.x + at * rect.width : rect.y + at * rect.height;
      const read = (d: number) => {
        const along = Math.floor((edge + sign * d) * dsf);
        const fixed = Math.floor(across * dsf);
        return vertical ? pixel(img, fixed, along) : pixel(img, along, fixed);
      };
      const far = read(9);
      if (far === null) continue;
      for (let d = 0.25; d <= 5; d += 0.5 / dsf) {
        const value = read(d);
        if (value === null) continue;
        const ratio = contrast(value, far);
        if (best === null || ratio > best.ratio) best = { side, ring: hex(value), far: hex(far), ratio: r2(ratio), d: r2(d) };
      }
    }
    if (best !== null) sides.push(best);
  }
  return { min: sides.length === 0 ? null : r2(Math.min(...sides.map((side) => side.ratio))), sides };
}

/* ---- Stop inspection -------------------------------------------------------------------------------- */

interface StopInfo {
  key: number;
  tag: string;
  type: string | null;
  role: string | null;
  name: string;
  rect: Rect;
  visible: number;
  covers: string[];
  inside: boolean;
  fv: boolean;
  fw: boolean;
  boxShadow: string;
  scrollY: number;
  scrollers: number[];
  day: string | null;
  dayButton: boolean;
  picker: string | null;
}

async function stopInfo(page: Page, scope: string): Promise<StopInfo | null> {
  return page.evaluate((scopeSelector) => {
    const w = window as unknown as { __b6keys?: WeakMap<Element, number>; __b6next?: number; __b6last?: Element };
    const active = document.activeElement as HTMLElement | null;
    if (active === null || active === document.body || active === document.documentElement) return null;
    w.__b6keys = w.__b6keys ?? new WeakMap();
    let key = w.__b6keys.get(active);
    if (key === undefined) {
      w.__b6next = (w.__b6next ?? 0) + 1;
      key = w.__b6next;
      w.__b6keys.set(active, key);
    }
    w.__b6last = active;
    const labels = (active as HTMLInputElement).labels ? [...((active as HTMLInputElement).labels ?? [])] : [];
    const rect = active.getBoundingClientRect();
    let visible = 0;
    const covers = new Set<string>();
    for (let i = 0; i < 5; i += 1) {
      for (let j = 0; j < 5; j += 1) {
        const x = rect.left + 1 + ((rect.width - 2) * (i + 0.5)) / 5;
        const y = rect.top + 1 + ((rect.height - 2) * (j + 0.5)) / 5;
        if (x < 0 || y < 0 || x >= window.innerWidth || y >= window.innerHeight) {
          covers.add('outside the viewport');
          continue;
        }
        const hit = document.elementFromPoint(x, y);
        if (hit !== null && (active.contains(hit) || labels.some((label) => label.contains(hit)))) visible += 1;
        else if (hit !== null) {
          const cover = hit.closest('.shell-bar, .shell-tabs, .share-bar, .editor-head, dialog, .more-panel') ?? hit;
          covers.add(`${cover.tagName.toLowerCase()}.${String(cover.className).split(' ')[0]}`);
        }
      }
    }
    const scrollers: number[] = [];
    for (let node: Element | null = active.parentElement; node !== null; node = node.parentElement) {
      if (node.scrollHeight > node.clientHeight + 1 && getComputedStyle(node).overflowY !== 'visible') scrollers.push(Math.round(node.scrollTop));
    }
    const name = (active.getAttribute('aria-label') ?? active.textContent ?? '').replace(/\s+/g, ' ').trim().slice(0, 70);
    return {
      key,
      tag: active.tagName.toLowerCase(),
      type: active.getAttribute('type'),
      role: active.getAttribute('role'),
      name,
      rect: { x: rect.x, y: rect.y, width: rect.width, height: rect.height },
      visible,
      covers: [...covers],
      inside: active.closest(scopeSelector) !== null,
      fv: active.matches(':focus-visible'),
      fw: active.matches(':focus-within'),
      boxShadow: getComputedStyle(active).boxShadow,
      scrollY: Math.round(window.scrollY),
      scrollers,
      day: active.closest('[data-day]')?.getAttribute('data-day') ?? null,
      dayButton: active.hasAttribute('data-day-button'),
      picker: active.closest('[data-label-picker]')?.getAttribute('data-label-picker') ?? null,
    };
  }, scope);
}

async function capture(page: Page): Promise<Buffer> {
  return page.screenshot({ animations: 'disabled', caret: 'hide' });
}

interface RingResult {
  ref: 'blur' | 'previous' | 'none';
  blurTook: boolean;
  sidesPresent: number | null;
  minOuter: number | null;
  minAdjacentBest: number | null;
  minChange: number | null;
  maxChange: number | null;
}

async function measureRing(page: Page, info: StopInfo, previous: { png: Buffer; info: StopInfo } | null, dsf: number, focusedPng: Buffer): Promise<RingResult> {
  const blurTook = await page.evaluate(() => {
    const w = window as unknown as { __b6last?: HTMLElement };
    const element = w.__b6last;
    if (element === undefined) return false;
    element.blur();
    return document.activeElement !== element && !element.matches(':focus-within');
  });
  let refPng: Buffer | null = null;
  let ref: RingResult['ref'] = 'none';
  if (blurTook) {
    refPng = await capture(page);
    ref = 'blur';
    await page.evaluate(() => {
      const w = window as unknown as { __b6last?: HTMLElement };
      w.__b6last?.focus({ preventScroll: true });
    });
  } else if (previous !== null && previous.info.scrollY === info.scrollY && JSON.stringify(previous.info.scrollers) === JSON.stringify(info.scrollers)) {
    refPng = previous.png;
    ref = 'previous';
  }
  if (refPng === null) return { ref, blurTook, sidesPresent: null, minOuter: null, minAdjacentBest: null, minChange: null, maxChange: null };
  const result = ringOf(decodePng(focusedPng), decodePng(refPng), info.rect, dsf);
  return { ref, blurTook, sidesPresent: result.sidesPresent, minOuter: result.minOuter, minAdjacentBest: result.minAdjacentBest, minChange: result.minChange, maxChange: result.maxChange };
}

interface WalkStop extends StopInfo {
  ring?: RingResult;
}

/** Focuses the first or last visible focusable control of `scope` (the starting point of a walk). */
async function focusEdge(page: Page, scope: string, edge: 'first' | 'last'): Promise<boolean> {
  return page.evaluate(
    ([scopeSelector, which]) => {
      const root = document.querySelector(scopeSelector);
      const selector = 'a[href], button:not([disabled]), input:not([disabled]), select:not([disabled]), textarea:not([disabled]), [tabindex]:not([tabindex="-1"])';
      const candidates = root === null ? [] : [...root.querySelectorAll<HTMLElement>(selector)].filter((element) => element.getClientRects().length > 0 && element.closest('[inert]') === null);
      const target = which === 'first' ? candidates[0] : candidates.at(-1);
      if (target === undefined) return false;
      target.focus();
      return document.activeElement === target || target.contains(document.activeElement);
    },
    [scope, edge] as const,
  );
}

async function walk(page: Page, scope: string, direction: 'forward' | 'backward', label: string, dsf: number, rings: boolean) {
  const stops: WalkStop[] = [];
  const ok = await focusEdge(page, scope, direction === 'forward' ? 'first' : 'last');
  if (!ok) return { label, direction, stops, note: 'no focusable control' };
  await page.keyboard.press('Shift');
  // Re-establish keyboard modality on the starting control with a real key press pair.
  await page.keyboard.press(direction === 'forward' ? 'Shift+Tab' : 'Tab');
  await page.keyboard.press(direction === 'forward' ? 'Tab' : 'Shift+Tab');
  const seen = new Set<number>();
  let last: number | undefined;
  let repeats = 0;
  let previous: { png: Buffer; info: StopInfo } | null = null;
  // A stop whose ring could not be measured yet (blur() did not take and the page scrolled since the
  // previous stop): the screenshot of the next stop serves as its unfocused image if nothing scrolled.
  let pending: { stop: WalkStop; png: Buffer } | null = null;
  for (let index = 0; index < 220; index += 1) {
    await settle(page);
    const info = await stopInfo(page, scope);
    if (info === null || !info.inside) break;
    if (info.key === last && repeats < 8) {
      // A date or time field keeps focus while the key moves through its segments.
      repeats += 1;
    } else {
      if (info.key !== last && seen.has(info.key)) break;
      repeats = 0;
      last = info.key;
      seen.add(info.key);
      const stop: WalkStop = { ...info };
      if (rings) {
        const focusedPng = await capture(page);
        if (pending !== null) {
          const before = pending.stop;
          if (before.scrollY === info.scrollY && JSON.stringify(before.scrollers) === JSON.stringify(info.scrollers)) {
            const result = ringOf(decodePng(pending.png), decodePng(focusedPng), before.rect, dsf);
            before.ring = { ref: 'next' as never, blurTook: false, sidesPresent: result.sidesPresent, minOuter: result.minOuter, minAdjacentBest: result.minAdjacentBest, minChange: result.minChange, maxChange: result.maxChange };
          }
          pending = null;
        }
        stop.ring = await measureRing(page, info, previous, dsf, focusedPng);
        if (stop.ring.sidesPresent === null) pending = { stop, png: focusedPng };
        previous = { png: focusedPng, info };
      }
      stops.push(stop);
    }
    await page.keyboard.press(direction === 'forward' ? 'Tab' : 'Shift+Tab');
  }
  return { label, direction, stops };
}

/* ---- P3: names (label in name) on every screen ------------------------------------------------------ */

async function nameScan(page: Page, screen: string) {
  const count = await page.evaluate(() => {
    const selector = 'button, a[href], [role="button"], [role="link"], [role="tab"], [role="option"], [role="menuitem"], summary';
    let n = 0;
    for (const element of document.querySelectorAll<HTMLElement>(selector)) {
      if (element.getClientRects().length === 0 || element.closest('[inert], [aria-hidden="true"]') !== null) continue;
      element.setAttribute('data-b6-i', String(n));
      n += 1;
    }
    return n;
  });
  const rows: Array<{ screen: string; i: number; role: string; name: string; visible: string; contains: boolean; symbolic: boolean; ariaLabel: string | null; raw?: string }> = [];
  for (let i = 0; i < count; i += 1) {
    const locator = page.locator(`[data-b6-i="${i}"]`);
    try {
      const [snapshot, visible, ariaLabel] = await Promise.all([
        locator.ariaSnapshot({ timeout: 3_000 }),
        locator.evaluate((element) => (element as HTMLElement).innerText.replace(/\s+/g, ' ').trim()),
        locator.getAttribute('aria-label'),
      ]);
      let first = (snapshot.split('\n')[0] ?? '').replace(/:$/, '');
      // YAML quotes a whole entry that contains ": " as - '...' (with '' for a quote), maybe followed by ": text".
      const quoted = /^- '((?:[^']|'')*)'/.exec(first);
      if (quoted !== null) first = `- ${(quoted[1] ?? '').replace(/''/g, "'")}`;

    } catch (error) {
      rows.push({ screen, i, role: 'error', name: String(error).slice(0, 120), visible: '', contains: true, symbolic: false, ariaLabel: null });

  }

    for (const element of document.querySelectorAll('[data-b6-i]')) element.removeAttribute('data-b6-i');
  });

}

/* ---- P5: contrast of text (rendered colours), AX-09 tokens, status words and shapes --------------------- */

test('P5 contrast', async ({ page, adminSeed, server }, testInfo) => {
  const project = testInfo.project.name;
  const out: Record<string, unknown> = {};
  const world = await newWorld(adminSeed, server.origin, { mutual: true });
  await signIn(page, world.me.account);
  for (const theme of ['light', 'dark'] as const) {
    await page.emulateMedia({ colorScheme: theme, reducedMotion: 'reduce' });
    const scan = async (state: string) =>
      page.evaluate((label) => {
        const parse = (text: string): [number, number, number, number] => {
          const numbers = (/rgba?\(([^)]*)\)/.exec(text)?.[1] ?? '').split(/[\s,/]+/).filter((part) => part !== '').map(Number);
          const [r = 0, g = 0, b = 0, a = 1] = numbers;
          return [r, g, b, a];
        };
        const over = (top: number[], bottom: number[]) => {
          const a = (top[3] ?? 1) + (bottom[3] ?? 1) * (1 - (top[3] ?? 1));
          const mix = (i: number) => ((top[i] ?? 0) * (top[3] ?? 1) + (bottom[i] ?? 0) * (bottom[3] ?? 1) * (1 - (top[3] ?? 1))) / (a || 1);
          return [mix(0), mix(1), mix(2), a];
        };
        const lum = (c: number[]) => {
          const f = (v: number) => {
            const s = v / 255;
            return s <= 0.04045 ? s / 12.92 : ((s + 0.055) / 1.055) ** 2.4;
          };
          return 0.2126 * f(c[0] ?? 0) + 0.7152 * f(c[1] ?? 0) + 0.0722 * f(c[2] ?? 0);
        };
        const ratio = (a: number[], b: number[]) => {
          const [x, y] = [lum(a), lum(b)].sort((p, q) => q - p) as [number, number];
          return (x + 0.05) / (y + 0.05);
        };
        const rows: Array<{ state: string; text: string; color: string; bg: string; ratio: number; size: string; weight: string; cls: string }> = [];
        const walker = document.createTreeWalker(document.body, NodeFilter.SHOW_TEXT);
        const seen = new Set<Element>();
        for (let node = walker.nextNode(); node !== null; node = walker.nextNode()) {
          const element = node.parentElement;
          if (element === null || seen.has(element) || (node.textContent ?? '').trim() === '') continue;
          seen.add(element);
          if (element.getClientRects().length === 0 || element.closest('[inert], [aria-hidden="true"], .visually-hidden, .sr-only') !== null) continue;
          const style = getComputedStyle(element);
          if (style.visibility === 'hidden' || Number(style.opacity) === 0 || style.clipPath === 'inset(50%)') continue;
          const chain: number[][] = [];
          for (let e: Element | null = element; e !== null; e = e.parentElement) chain.push(parse(getComputedStyle(e).backgroundColor));
          const bg = chain.reduceRight((below, c) => over(c, below), [255, 255, 255, 1] as number[]);
          const page = getComputedStyle(document.body).backgroundColor;
          void page;
          const fg = over(parse(style.color), bg);
          rows.push({ state: label, text: (node.textContent ?? '').trim().slice(0, 40), color: style.color, bg: `rgb(${bg.slice(0, 3).map((v) => Math.round(v)).join(',')})`, ratio: Math.round(ratio(fg, bg) * 100) / 100, size: style.fontSize, weight: style.fontWeight, cls: element.className.toString().slice(0, 40) });
        }
        return rows;
      }, state);
    const all: unknown[] = [];
    await gotoTimesheet(page);
    all.push(...(await scan(`${theme} timesheet`)));
    await page.getByRole('button', { name: 'Change several days', exact: true }).click();
    for (const date of [world.complete, world.wfh, world.unconfirmed].filter((d) => d !== '')) await page.getByRole('checkbox', { name: `Select ${date}` }).check();
    all.push(...(await scan(`${theme} batch-selected`)));
    await page.getByRole('button', { name: 'Done' }).click();
    await page.getByRole('button', { name: 'Show details', exact: true }).click();
    all.push(...(await scan(`${theme} details`)));
    await page.getByRole('button', { name: 'Show details', exact: true }).click();
    const button = page.locator(`[data-day="${world.complete}"] [data-day-button]`);
    await button.click();
    await expect(page.locator(`dialog[data-day-editor="${world.complete}"]`).getByRole('heading', { name: /^Times/ })).toBeVisible();
    all.push(...(await scan(`${theme} editor`)));
    await page.keyboard.press('Escape');
    for (const hash of ['#/settings', '#/ot', '#/history', '#/import', `#/shared/${world.owner.account.id}`]) {
      await page.goto(`/${hash}`);
      await page.waitForTimeout(500);
      all.push(...(await scan(`${theme} ${hash.startsWith('#/shared') ? '#/shared' : hash}`)));
    }
    await gotoTimesheet(page);
    await page.locator('[data-sheet-review-link], a[href*="review"]').first().click();
    await page.waitForTimeout(700);
    all.push(...(await scan(`${theme} review`)));
    // Status words and shapes of the Check cells.
    await gotoTimesheet(page);
    const checks = await page.locator('[data-check]').evaluateAll((elements) =>
      elements.map((element) => ({ key: element.getAttribute('data-check'), text: (element as HTMLElement).innerText.trim(), shape: element.querySelector('.shape, .live-dot, svg') !== null })),
    );
    // AX-09: --ok over every light/dark surface token, with and without the selection tint.
    const tokens = await page.evaluate(() => {
      const probe = document.createElement('div');
      document.body.append(probe);
      const read = (value: string) => {
        probe.style.backgroundColor = value;
        const computed = getComputedStyle(probe).backgroundColor;
        probe.style.backgroundColor = '';
        return computed;
      };
      const result = { ok: read('var(--ok)'), selected: read('var(--day-selected)'), surfaces: ['--card', '--bg', '--off', '--sheet-head', '--sheet-head-off', '--day-nonworking', '--attention-bg', '--line-soft'].map((token) => ({ token, colour: read(`var(${token})`) })) };
      probe.remove();

    });

      ([label, w]) => {
        const selector = 'button, a[href], input, select, textarea, label.pick, [role="option"], summary';
        const small: Array<{ tag: string; text: string; w: number; h: number }> = [];
        let controls = 0;
        for (const element of document.querySelectorAll<HTMLElement>(selector)) {
          if (element.getClientRects().length === 0 || element.closest('[inert]') !== null) continue;
          const style = getComputedStyle(element);
          if (style.visibility === 'hidden') continue;
          if (element.closest('label.pick') !== null && element.tagName === 'INPUT') continue;
          const r = element.getBoundingClientRect();

          if (r.width > 0 && (r.right > window.innerWidth + 0.5 || r.left < -0.5) && element.closest('[inert]') === null && getComputedStyle(element).position !== 'fixed') {
            let clipped = false;
            for (let p = element.parentElement; p !== null; p = p.parentElement) {
              const o = getComputedStyle(p).overflowX;
              if (o === 'hidden' || o === 'auto' || o === 'scroll' || o === 'clip') {
                clipped = true;
                break;
              }
            }
            if (!clipped) overflow.push(`${element.tagName.toLowerCase()}.${String(element.className).split(' ')[0]} ${Math.round(r.left)}-${Math.round(r.right)}`);
          }
        }
        return { state: label, width: w, controls, small, scrollWidth: document.documentElement.scrollWidth, innerWidth: window.innerWidth, overflow: overflow.slice(0, 20) };
      },
      [state, width] as const,
    );

/* ---- P7: modal behaviour, keyboard order and hover over focus (R-10) ------------------------------------ */

test('P7 modal and order', async ({ page, adminSeed, server }, testInfo) => {
  const project = testInfo.project.name;
  const world = await newWorld(adminSeed, server.origin);
  await page.emulateMedia({ reducedMotion: 'reduce' });
  await signIn(page, world.me.account);
  const out: Record<string, unknown> = {};
  const sizes = project === 'desktop' ? [1280, 1024, 768] : [390, 320];
  for (const width of sizes) {
    await page.setViewportSize({ width, height: width >= 768 ? 800 : width === 320 ? 640 : 844 });
    await gotoTimesheet(page);
    // Keyboard order: day buttons and label pickers in Tab order.
    await page.evaluate(() => (document.activeElement as HTMLElement | null)?.blur());
    await page.evaluate(() => window.scrollTo(0, 0));
    const order: string[] = [];
    const trail: string[] = [];
    const seenStops = new Set<number>();
    for (let i = 0; i < 160; i += 1) {
      await page.keyboard.press('Tab');
      const item = await page.evaluate(() => {
        const w = window as unknown as { __b6ord?: WeakMap<Element, number>; __b6ordNext?: number };
        const a = document.activeElement as HTMLElement | null;
        if (a === null || a === document.body) return { key: -1, text: '' };
        w.__b6ord = w.__b6ord ?? new WeakMap();
        let key = w.__b6ord.get(a);
        if (key === undefined) {
          w.__b6ordNext = (w.__b6ordNext ?? 0) + 1;
          key = w.__b6ordNext;
          w.__b6ord.set(a, key);
        }
        let text = '';
        if (a.hasAttribute('data-day-button')) text = `day ${a.closest('[data-day]')?.getAttribute('data-day') ?? ''}`;
        else {
          const picker = a.closest('[data-label-picker]');
          if (picker !== null) text = `label ${picker.getAttribute('data-label-picker') ?? ''}`;
        }
        return { key, text };
      });
      trail.push(`${item.key}:${item.text}`);
      if (item.key > 0 && seenStops.has(item.key)) break;
      if (item.key > 0) seenStops.add(item.key);
      if (item.text !== '') order.push(item.text);
    }
    const dayOrder = order.filter((item) => item.startsWith('day ')).map((item) => item.slice(4));
    const sorted = [...dayOrder].sort();
    // Modal behaviour of the editor.
    const button = page.locator(`[data-day="${world.complete}"] [data-day-button]`);
    await button.focus();
    await page.keyboard.press('Enter');
    const editor = page.locator(`dialog[data-day-editor="${world.complete}"]`);
    await expect(editor.getByRole('heading', { name: /^Times/ })).toBeVisible();
    await settle(page);
    const onOpen = await page.evaluate(() => `${document.activeElement?.tagName ?? ''} ${(document.activeElement?.textContent ?? '').trim().slice(0, 40)}`);
    const modal = await editor.evaluate((element) => element.matches(':modal'));
    let outside = 0;
    for (let i = 0; i < 60; i += 1) {
      await page.keyboard.press(i % 3 === 2 ? 'Shift+Tab' : 'Tab');
      const inEditor = await page.evaluate((date) => {
        const a = document.activeElement;
        return a === null || a === document.body || a.closest(`dialog[data-day-editor="${date}"]`) !== null;
      }, world.complete);
      if (!inEditor) outside += 1;
    }
    await editor.evaluate((element) => (element.querySelector('h2') as HTMLElement | null)?.focus());
    await page.keyboard.press('Escape');
    await settle(page);
    const closed = (await editor.count()) === 0;
    const returned = await button.evaluate((element) => element === document.activeElement);
    // R-10: hover over a keyboard-focused control.
    const clockIn = page.getByRole('button', { name: 'Clock in', exact: true });
    await clockIn.focus();
    await page.keyboard.press('Shift+Tab');
    await page.keyboard.press('Tab');
    await page.waitForTimeout(100);
    await settle(page);
    const focusedShadow = await clockIn.evaluate((element) => `${getComputedStyle(element).boxShadow} fv=${element.matches(':focus-visible')}`);
    await clockIn.hover();
    await settle(page);
    const hoveredShadow = await clockIn.evaluate((element) => ({ shadow: getComputedStyle(element).boxShadow, fv: element.matches(':focus-visible') }));
    await page.mouse.move(0, 0);
    out[String(width)] = { trail: trail.slice(0, 12), stops: seenStops.size, dayOrder, ascending: JSON.stringify(dayOrder) === JSON.stringify(sorted) && dayOrder.length === 14, labelCount: order.filter((item) => item.startsWith('label ')).length, onOpen, modal, tabStopsOutside: outside, closed, returned, hover: { focusedShadow, hoveredShadow } };
  }
  record(`p7-modal-order-${project}`, out);
});
