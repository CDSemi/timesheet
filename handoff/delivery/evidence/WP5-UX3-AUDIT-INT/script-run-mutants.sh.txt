# Mutant phase (after the probes). Usage: bash run-mutants.sh [id ...]
. <task>/env.sh
export NODE_OPTIONS="--trace-deprecation"
mkdir -p $T/logs
echo "mutants start $(date -u +%Y-%m-%dT%H:%M:%SZ) $*" >> $T/logs/phase.txt
node $T/run-mutants.mjs "$@"
echo "mutants end $(date -u +%Y-%m-%dT%H:%M:%SZ)" >> $T/logs/phase.txt
echo done > $T/out/mutants.done
