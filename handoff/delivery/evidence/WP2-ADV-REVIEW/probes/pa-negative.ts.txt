let pass = 0;
let fail = 0;
const observations: string[] = [];
function check(id: string, ok: boolean, detail: unknown) {
  if (ok) pass += 1; else fail += 1;
  console.log(`${ok ? 'PASS' : 'FAIL'}  ${id}  ${typeof detail === 'string' ? detail : JSON.stringify(detail)}`);
}
function observe(id: string, detail: unknown) {
  const line = `OBSERVED  ${id}  ${typeof detail === 'string' ? detail : JSON.stringify(detail)}`;
  observations.push(line);
  console.log(line);
}
function errOf(fn: () => unknown): { status: number | null; code: string } | null {
  try {
    fn();
    return null;
  } catch (error) {
    const e = error as { status?: number; code?: string; message?: string };
    return { status: e.status ?? null, code: e.code ?? String(e.message) };
  }
}
