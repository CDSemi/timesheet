function owner(opening: number): string {
  const id = randomUUID();
  db.prepare(`INSERT INTO users (id, email, display_name, role, status, password_hash, calendar_id, created_at, updated_at)
    VALUES (?, ?, 'Synthetic Racer', 'employee', 'active', 'login-disabled-synthetic', ?, ?, ?)`).run(id, `race-${id}@example.invalid`, calendarId, NOW, NOW);
  postCredit(ctx, { userId: id, sourceKey: 'opening', minutes: opening, workDate: '2026-09-21', actorUserId: null, origin: 'system' });

}

const children: ChildProcess[] = [];
function nextMsg(child: ChildProcess): Promise<any> {
  return new Promise((resolve, reject) => {
    const onError = (error: Error) => { child.off('message', onMessage); reject(error); };
    const onMessage = (message: unknown) => { child.off('error', onError); resolve(message); };
    child.once('message', onMessage);
    child.once('error', onError);
  });
}
for (let i = 0; i < 4; i += 1) {
  const child = fork(join(import.meta.dirname, 'pb-child.ts'), [root, dbPath], { stdio: ['ignore', 'inherit', 'inherit', 'ipc'] });
  children.push(child);
}
const ready = await Promise.all(children.map(nextMsg));
const pids = new Set(ready.map((m) => m.pid));
console.log(`racer processes: ${pids.size} distinct pids (parent pid ${process.pid})`);

async function race(ops: Array<{ kind: string; input: unknown }>, opts: { blockerMs?: number; unsafeDelayMs?: number } = {}) {
  const startAt = Date.now() + 120;
  const racers = children.slice(0, ops.length);
  const done = racers.map(nextMsg);
  racers.forEach((child, index) => child.send({ op: ops[index], startAt, nowIso: NOW, unsafeDelayMs: opts.unsafeDelayMs }));
  if (opts.blockerMs) {
    // A third writer holds the write lock across the release instant, so every racer queues on
    // the busy handler and they all contend when it commits.
    while (Date.now() < startAt - 10) { await new Promise((r) => setTimeout(r, 1)); }
    db.exec('BEGIN IMMEDIATE');
    const until = Date.now() + opts.blockerMs;
    while (Date.now() < until) { /* hold the lock */ }
    db.exec('COMMIT');
  }
  const outcomes = await Promise.all(done);
  const latestStart = Math.max(...outcomes.map((o) => o.startedAtMs));
  const earliestFinish = Math.min(...outcomes.map((o) => o.finishedAtMs));
  return { outcomes, overlap: latestStart < earliestFinish, distinctPids: new Set(outcomes.map((o) => o.pid)).size };
}
const label = (o: any) => (o.ok ? o.status : `${o.http ?? '-'} ${o.code}`);
const newDeltas = (id: string) => listLedgerEntries(db, id).filter((e: any) => e.sourceKey !== 'opening').map((e: any) => e.deltaMinutes);

let failures = 0;
let cancelUnchangedAfterUse = 0;
async function scenario(name: string, rounds: number, setup: () => { ops: any[]; verify: (outcomes: any[]) => string | null }, opts = {}) {
  const dist = new Map<string, number>();
  let overlap = 0; let bad = 0; let pidsOk = 0;
  for (let r = 0; r < rounds; r += 1) {
    const { ops, verify } = setup();
    const res = await race(ops, opts);
    if (res.overlap) overlap += 1;
    if (res.distinctPids === ops.length) pidsOk += 1;
    const key = res.outcomes.map(label).sort().join(' + ');
    dist.set(key, (dist.get(key) ?? 0) + 1);
    const problem = verify(res.outcomes);
    if (problem !== null) { bad += 1; console.log(`  round ${r}: ${problem}`); }
  }
  if (bad > 0) failures += 1;
  console.log(`${bad === 0 ? 'PASS' : 'FAIL'}  ${name}: ${rounds} rounds, separate processes ${pidsOk}/${rounds}, overlapping call windows ${overlap}/${rounds}, invariant violations ${bad}; outcomes: ${[...dist].map(([k, v]) => `${k} x${v}`).join('; ')}`);
}
