// WP4-FIXB4 summary of run-cost.mjs / cost.mjs output: per case, the runs, the median and the largest preview time and
// maxRSS delta, and the outcome. With a second file, puts the same case of both side by side (after | before).
// Usage: node stat.mjs <after.txt> [before.txt]
import { readFileSync } from 'node:fs';

function parse(file) {
  const cases = new Map();
  for (const line of readFileSync(file, 'utf8').split('\n')) {
    const ms = /(?:preview|full previewImport) ms=(\d+)/.exec(line);
    const rss = /\(\+(\d+)\)(?: OVER-[A-Z]+)? \|/.exec(line);
    if (!ms || !rss) continue;
    const name = line.split(' | ')[0];
    const outcome = (/\| (accepted|rejected:[a-z_]+|THROWN[^|]*|\d{3} [^|]*)/.exec(line)?.[1] ?? '?').replace(/ \(.*$/, '').trim();
    const upload = /upload=(\d+) B/.exec(line)?.[1];
    const entry = cases.get(name) ?? { ms: [], rss: [], outcome, upload };
    entry.ms.push(Number(ms[1]));
    entry.rss.push(Number(rss[1]));
    cases.set(name, entry);
  }
  return cases;
}
const median = (values) => [...values].sort((a, b) => a - b)[Math.floor(values.length / 2)];
const show = (e) => (e ? `${e.ms.length}x median ${median(e.ms)} ms max ${Math.max(...e.ms)} ms, +${median(e.rss)} MiB max +${Math.max(...e.rss)} MiB, ${e.outcome}` : 'not run');
const after = parse(process.argv[2]);
const before = process.argv[3] ? parse(process.argv[3]) : null;
let worstMs = 0;
let worstRss = 0;
for (const [name, entry] of after) {
  worstMs = Math.max(worstMs, ...entry.ms);
  worstRss = Math.max(worstRss, ...entry.rss);
  const short = name.length > 110 ? `${name.slice(0, 107)}...` : name;
  console.log(before ? `${short} | upload ${entry.upload} B | after: ${show(entry)} | before: ${show(before.get(name))}` : `${short} | upload ${entry.upload} B | ${show(entry)}`);
}
console.log(`== ${after.size} cases; largest single run: ${worstMs} ms (${((100 * worstMs) / 500).toFixed(0)} % of 500 ms), +${worstRss} MiB (${((100 * worstRss) / 150).toFixed(0)} % of 150 MiB)`);
