export const MiB = 1024 * 1024;
const NS = 'http://schemas.openxmlformats.org/spreadsheetml/2006/main';
const RNS = 'http://schemas.openxmlformats.org/officeDocument/2006/relationships';
const WS_TYPE = `${RNS}/worksheet`;
const HEAD = `<?xml version="1.0"?><worksheet xmlns="${NS}"><sheetData><row r="1">`;
const TAIL = '</row></sheetData></worksheet>';
const addDays = (d, n) => new Date(Date.parse(`${d}T00:00:00Z`) + n * 86400000).toISOString().slice(0, 10);

/** A worksheet whose single row is `unit` repeated to about `bytes` bytes. */
const fill = (unit, bytes, head = HEAD, tail = TAIL) => head + unit.repeat(Math.floor((bytes - head.length - tail.length) / unit.length)) + tail;

/** Template entries plus worksheet parts registered as sheets named by `names`. */
function withSheets(entries, parts) {
  let workbook = strFromU8(entries['xl/workbook.xml']);
  let rels = strFromU8(entries['xl/_rels/workbook.xml.rels']);
  parts.forEach(({ name, part, xml, relId }, i) => {
    const id = relId ?? `rIdR${i}`;
    if (xml !== undefined) entries[part] = strToU8(xml);
    workbook = workbook.replace('</sheets>', `<sheet name="${name}" sheetId="${500 + i}" r:id="${id}"/></sheets>`);
    if (relId === undefined) {
      rels = rels.replace('</Relationships>', `<Relationship Id="${id}" Type="${WS_TYPE}" Target="${part.replace('xl/', '')}"/></Relationships>`);
    }
  });
  entries['xl/workbook.xml'] = strToU8(workbook);
  entries['xl/_rels/workbook.xml.rels'] = strToU8(rels);
  return entries;
}

/** Four worksheet parts (sheet1..3 replaced, one added) filled to `each` bytes with `unit`. */
function fourParts(unit, each) {
  const e = unpack(readTemplateBytes());
  for (const part of ['xl/worksheets/sheet1.xml', 'xl/worksheets/sheet2.xml', 'xl/worksheets/sheet3.xml']) e[part] = strToU8(fill(unit, each));
  return withSheets(e, [{ name: 'Extra', part: 'xl/worksheets/sheet90.xml', xml: fill(unit, each) }]);
}
/** `count` worksheet parts of `xml` each, as sheets P0.. */
function nParts(count, xml) {
  return withSheets(unpack(readTemplateBytes()), Array.from({ length: count }, (_, i) => ({ name: `P${i}`, part: `xl/worksheets/p${i}.xml`, xml })));
}
const EACH = 4 * MiB - 6 * 1024; // four parts plus the template's small parts stay under the 16 MiB total

/** Template whose shared strings gain one big string (index 66) of about `bytes` bytes. */
function withBigSharedString(e, bytes, first = 'a', body = 'a') {
  const sst = strFromU8(e['xl/sharedStrings.xml']);
  const big = first + body.repeat(Math.floor((bytes - sst.length - 40) / body.length));
  e['xl/sharedStrings.xml'] = strToU8(sst.replace('</sst>', `<si><t>${big}</t></si></sst>`));
  return e;
}

/** Holiday Dates sheet with `rows` rows: A = a valid serial date (or shared string `dateIndex`), B = shared string `index`. */
function holidaySheet(rows, index, dateIndex) {
  const out = [];
  for (let r = 2; r < 2 + rows; r += 1) {
    const a = dateIndex === undefined ? `<c r="A${r}"><v>${46023 + r}</v></c>` : `<c r="A${r}" t="s"><v>${dateIndex}</v></c>`;
    out.push(`<row r="${r}">${a}<c r="B${r}" t="s"><v>${index}</v></c></row>`);
  }
  return `<?xml version="1.0"?><worksheet xmlns="${NS}"><sheetData>${out.join('')}</sheetData></worksheet>`;
}

/** One synthetic dated sheet whose labels reference shared string `index`, registered under `count` dated names. */
function labelAmplifier(count, index, bigBytes, first = 'a', body = 'a') {
  const e = withBigSharedString(unpack(buildSyntheticWorkbook({ periods: [{ payrollDate: '2026-01-09' }] })), bigBytes, first, body);
  const part = 'xl/worksheets/sheet100.xml';
  e[part] = strToU8(strFromU8(e[part]).replace(/t="inlineStr"><is><t>Worked<\/t><\/is>/g, `t="s"><v>${index}</v>`));
  const names = [];
  for (let i = 1; i < count; i += 1) names.push({ name: addDays('2026-01-09', 14 * i).replace(/-/g, '.'), part, relId: 'rIdSyn0' });
  return withSheets(e, names);
}

/** Dated sheets (payroll 2026-01-09 + 14 n) whose label B14 is an inline string of `bytes` characters `ch`. */
function inlineLabel(sheets, bytes, ch) {
  const periods = Array.from({ length: sheets }, (_, i) => ({ payrollDate: addDays('2026-01-09', 14 * i) }));
  const e = unpack(buildSyntheticWorkbook({ periods }));
  for (let i = 0; i < sheets; i += 1) {
    const part = `xl/worksheets/sheet${100 + i}.xml`;
    e[part] = strToU8(strFromU8(e[part]).replace(/(<c r="B14"[^>]*t="inlineStr"><is><t>)Worked(<\/t>)/, `$1${ch.repeat(bytes)}$2`));
  }
  return e;
}

/** A realistic workbook: `n` dated sheets with clocks on weekdays, a vacation day and a holiday label per period. */
function realistic(n) {
  const periods = Array.from({ length: n }, (_, i) => {
    const days = {};
    for (let d = 0; d < 14; d += 1) if (d % 7 < 5) days[d] = { label: 'Worked', start: 8 * 60, end: 17 * 60 + 30 };
    days[2] = { label: 'Vacation' };
    days[9] = { label: 'Work from home', start: 9 * 60, end: 18 * 60 };
    return { payrollDate: addDays('2026-01-09', 14 * i), days };
  });
  return unpack(buildSyntheticWorkbook({ periods }));
}

function attrs(count, from = 0) {
  let out = '';
  for (let i = from; i < from + count; i += 1) out += ` a${i.toString(36)}=""`;
  return out;
}

export function buildCase(name) {
  const builder = CASES[name];
  if (!builder) throw new Error(`unknown case ${name}`);
  return pack(builder());
}
