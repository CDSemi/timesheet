const esc = (s) => s.replace(/&/g, '&amp;').replace(/</g, '&lt;').replace(/"/g, '&quot;');

/** `unit` repeated so that the whole part stays under `bytes` (UTF-8 bytes counted). */
function fillPart(unit, bytes = PART, wrap = row1) {
  const overhead = Buffer.byteLength(wrap(''));
  const n = Math.floor((bytes - overhead) / Buffer.byteLength(unit));
  return wrap(unit.repeat(n));
}
const addDays = (d, n) => new Date(Date.parse(`${d}T00:00:00Z`) + n * 86_400_000).toISOString().slice(0, 10);
const dated = (i) => addDays('2026-01-09', 14 * i).replace(/-/g, '.');

/** A minimal package: content types, workbook, relationships, the given worksheets and optional shared strings. */
function pkg(sheets, { sst, rels, extra = {}, workbookTail = '' } = {}) {
  const entries = {};
  const overrides = sheets.map((_, i) => `<Override PartName="/xl/worksheets/sheet${i + 1}.xml" ContentType="application/vnd.openxmlformats-officedocument.spreadsheetml.worksheet+xml"/>`).join('');
  entries['[Content_Types].xml'] = `<?xml version="1.0" encoding="UTF-8"?><Types xmlns="http://schemas.openxmlformats.org/package/2006/content-types"><Default Extension="xml" ContentType="application/xml"/><Override PartName="/xl/workbook.xml" ContentType="application/vnd.openxmlformats-officedocument.spreadsheetml.sheet.main+xml"/>${overrides}</Types>`;
  entries['xl/workbook.xml'] = `<?xml version="1.0" encoding="UTF-8"?><workbook xmlns="${NS}" xmlns:r="${RNS}"><sheets>${sheets.map((s, i) => `<sheet name="${esc(s.name)}" sheetId="${i + 1}"${s.hidden ? ' state="hidden"' : ''} r:id="rId${i + 1}"/>`).join('')}</sheets>${workbookTail}</workbook>`;
  entries['xl/_rels/workbook.xml.rels'] = rels ?? `<?xml version="1.0" encoding="UTF-8"?><Relationships xmlns="http://schemas.openxmlformats.org/package/2006/relationships">${sheets.map((_, i) => `<Relationship Id="rId${i + 1}" Type="${RNS}/worksheet" Target="worksheets/sheet${i + 1}.xml"/>`).join('')}</Relationships>`;
  sheets.forEach((s, i) => { entries[`xl/worksheets/sheet${i + 1}.xml`] = s.xml; });
  if (sst !== undefined) entries['xl/sharedStrings.xml'] = sst;
  Object.assign(entries, extra);
  const zippable = {};
  for (const [name, data] of Object.entries(entries)) zippable[name] = [typeof data === 'string' ? strToU8(data) : data, { level: 6, mtime: 1_767_225_600_000 }];
  return zipSync(zippable);
}

/** A dated payroll sheet cloned from the template's form, its B14 label replaced by `labelXml` (raw `<t>` content). */
function datedWithLabel(count, labelXml, { formula } = {}) {
  const periods = Array.from({ length: count }, (_, i) => ({ payrollDate: addDays('2026-01-09', 14 * i) }));
  const e = unpack(buildSyntheticWorkbook({ periods }));
  for (let i = 0; i < count; i += 1) {
    const part = `xl/worksheets/sheet${100 + i}.xml`;
    let xml = strFromU8(e[part]);
    if (labelXml !== undefined) xml = xml.replace(/(<c r="B14"[^>]*t="inlineStr"><is><t>)Worked(<\/t>)/, `$1${labelXml}$2`);
    if (formula !== undefined) xml = xml.replace('</sheetData>', `<row r="400"><c r="AA400"><f>${formula}</f><v>0</v></c></row></sheetData>`);
    e[part] = strToU8(xml);
  }
  const zippable = {};
  for (const [name, data] of Object.entries(e)) zippable[name] = [data, { level: 6, mtime: 1_767_225_600_000 }];
  return zipSync(zippable);
}

/** 61 dated sheets (the 64-sheet limit with the 3 support sheets) whose 14 labels are `label(i, d)`. */
function sixtyOneLabels(label, extra = {}) {
  const periods = Array.from({ length: 61 }, (_, i) => {
    const days = {};
    for (let d = 0; d < 14; d += 1) days[d] = { label: label(i, d) };
    return { payrollDate: addDays('2024-01-12', 14 * i), days, ...extra };
  });
  return buildSyntheticWorkbook({ periods });
}

/** Holiday Dates sheet of the template replaced by `rows` holiday rows named `name(r)` (inline strings). */
function withHolidays(bytes, rows, name) {
  const e = unpack(bytes);
  const out = [];
  for (let r = 2; r < 2 + rows; r += 1) out.push(`<row r="${r}"><c r="A${r}"><v>${44000 + r}</v></c>${inlineCell(`B${r}`, name(r))}</row>`);
  e['xl/worksheets/sheet2.xml'] = strToU8(ws(out.join('')));
  const zippable = {};
  for (const [n, data] of Object.entries(e)) zippable[n] = [data, { level: 6, mtime: 1_767_225_600_000 }];
  return zipSync(zippable);
}

const farCells = (count, tail = '/>') => {
  let out = '';
  for (let i = 0; i < count; i += 1) out += `<c r="${String.fromCharCode(65 + (i % 26))}${String.fromCharCode(65 + (Math.floor(i / 26) % 26))}${String.fromCharCode(65 + (Math.floor(i / 676) % 26))}${1000000 + i}"${tail}`;
  return out;
};
