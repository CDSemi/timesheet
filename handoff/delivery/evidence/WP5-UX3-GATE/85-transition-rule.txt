}

*,
*::before,
*::after {
  box-sizing: border-box;
  /*
   * The one shared transition: only the properties that actually change, never `all` (which also animates layout).
   * It is declared here, on every element and pseudo-element, so that an element which needs the fast or the slow
   * duration sets --transition-duration on itself (for example `--transition-duration: var(--duration-slow)`) and
   * still uses `transition: var(--transition)`. The durations become 0.01ms under prefers-reduced-motion above.
   */
  --transition:
    color var(--transition-duration) var(--ease-standard), background-color var(--transition-duration) var(--ease-standard),
    background-position var(--transition-duration) var(--ease-standard), border-color var(--transition-duration) var(--ease-standard),
    box-shadow var(--transition-duration) var(--ease-standard), transform var(--transition-duration) var(--ease-standard);
}
