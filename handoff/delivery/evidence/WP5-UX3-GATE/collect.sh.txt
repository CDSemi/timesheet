# Copies the masked evidence into handoff/delivery/evidence/WP5-UX3-GATE/ (masked LF .txt, scripts as .txt, 14 synthetic PNG).
. <task>/env.sh
mkdir -p $E
m() { node.exe "$TW\\mask.mjs" "$1" "$2" >> <task>/tmp/mask-last.txt 2>&1; }
cd $T/out
for f in *.txt; do
  case "$f" in ex-files.txt|ex-hashes.txt|ex-sorted.txt|lstree-sorted.txt|keys-list.txt|lsfiles.txt|51-*-files.txt|51-*-hashes.txt) continue;; esac
  m "$T/out/$f" "$E/$f"
done
cd $T/logs
for f in control-*.txt sweep-*.txt rep-*.txt probe-*.txt cmp-dist.txt cfg.txt keys.txt digest-after.txt; do
  [ -f "$f" ] && m "$T/logs/$f" "$E/log-$f"
done
cd $T/obs
for f in *.txt; do m "$T/obs/$f" "$E/obs-$f"; done
for s in env.sh digest.sh run-main.sh run-rest.sh run-drill.sh run-shift.sh run-probe.sh sweep.sh cmp-dist.sh cfg.sh keys.sh collect.sh chain1.sh; do m "$T/$s" "$E/$s.txt"; done
for s in mask.mjs zone.mjs run-tz.mjs tzprobe.mjs tzchild.mjs contrast5.mjs shiftclock.mjs probe-now.mjs zoom.mjs seam.mjs zz-lib.ts zz-png.ts zz-a11y.spec.ts zz-layout.spec.ts zz-lift.spec.ts zz-phone.spec.ts zz-headers.spec.ts zz-shots.spec.ts zz-week.spec.ts zz-weekdbg.spec.ts zz-dom.spec.ts zz-measure.spec.ts zz-focus.spec.ts zz-control.spec.ts; do m "$T/$s" "$E/$s.txt"; done
cp $T/shots-shots2/*-synthetic.png $E/
ls $E | wc -l
