#!/bin/bash
export MSYS_NO_PATHCONV=1
T="D:/.claude-tmp/timesheet/INGAT-T0T8-GATE"
REG=registry:3@sha256:01cd312e67026f06f5d89c3e133a4a9adc7cb9eeb9b1194e0b2954dfd770efd2
{
echo "== pull registry"
docker pull "$REG"; echo "exit $?"
echo "== start registry"
docker run -d --rm --name ingat-gate-registry -p 127.0.0.1:49820:5000 "$REG"; echo "exit $?"
echo "== create builder"
docker buildx create --name ingat-gate-builder --driver docker-container --driver-opt network=host; echo "exit $?"
docker buildx inspect ingat-gate-builder --bootstrap; echo "exit $?"
} > "$T/g9-setup.txt" 2>&1
