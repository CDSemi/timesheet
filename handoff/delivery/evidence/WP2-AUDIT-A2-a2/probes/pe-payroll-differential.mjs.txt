// WP2-AUDIT-A2 attempt 2: payroll-exception differential probe (WP2-A2-02 recheck). Synthetic data only.
// Question: can the administrator learn, from anything the payroll-exception flow answers or from any
// read afterwards, whether employees have timesheets (stored pay_periods rows) in a pay period?
// Method: fresh SQLite WAL files with the same seed and the same fixed clock.
//   W0: no employee activity.  W1: two employees edit days in four pay periods (past, current, next,
//   one after), so stored pay_periods rows exist there; the administrator does nothing personal.
// The administrator then sends the same sequence of payroll-exception requests in both worlds and
// reads every admin-reachable route afterwards. Bodies are compared byte for byte after masking UUIDs
// (calendar and user ids are random per database). The stored rows must really be refreshed in W1 and
// the audit event must keep the refreshed flag (not readable by the administrator).
// W3: finalized periods (direct UPDATE, as WP3 will set the column): 409 period_finalized is kept,
// writes nothing, carries period dates only, and is identical for one versus two finalized timesheets.
// Passwords are generated at run time and never printed.
// Usage: node pe-payroll-differential.mjs <tree-root> <work-dir>

const UUID = /[0-9a-f]{8}-[0-9a-f]{4}-[0-9a-f]{4}-[0-9a-f]{4}-[0-9a-f]{12}/gi;
const mask = (text) => String(text).replace(UUID, '<uuid>');
const sha = (text) => createHash('sha256').update(text).digest('hex').slice(0, 12);
let pass = 0;
let fail = 0;
const check = (name, ok, detail = '') => {
  if (ok) pass += 1;
  else fail += 1;
  out(`${ok ? 'PASS' : 'FAIL'} ${name}${detail === '' ? '' : `  ${mask(detail)}`}`);
};

async function world(name) {
  const db = openDatabase(join(workDir, `${name}.db`));
  migrate(db);
  const pw = { admin: randomBytes(18).toString('base64url'), employee: randomBytes(18).toString('base64url') };
  const seed = await seedSynthetic(db, clock, { passwords: pw });
  const app = createApp({ db, clock, config, loginLimiter: new LoginRateLimiter(), staticDir: null });
  async function req(method, path, cookie, body) {
    const headers = { origin: ORIGIN };
    if (cookie !== undefined) headers.cookie = cookie;
    const init = { method, headers };
    if (body !== undefined) {
      headers['content-type'] = 'application/json';
      init.body = JSON.stringify(body);
    }
    const res = await app.request(path, init);
    const text = await res.text();
    let json = null;
    try { json = JSON.parse(text); } catch { json = null; }
    return { status: res.status, text, json, setCookie: res.headers.get('set-cookie') };
  }
  const login = async (email, password) => (await req('POST', '/api/auth/login', undefined, { email, password })).setCookie.split(';')[0];
  const admin = await login('admin@example.invalid', pw.admin);
  const emp = await login('employee@example.invalid', pw.employee);
  const pw2 = randomBytes(18).toString('base64url');
  const created = await req('POST', '/api/admin/users', admin, { email: 'employee2@example.invalid', display_name: 'Example Employee Two', role: 'employee', password: pw2, calendar_id: seed.calendarId });
  const emp2 = await login('employee2@example.invalid', pw2);
  const counts = () => Object.fromEntries(['audit_events', 'payroll_exceptions', 'pay_periods', 'timesheets', 'day_entries', 'work_sessions'].map((table) => [table, db.prepare(`SELECT count(*) FROM ${table}`).pluck().get()]));
  return { name, db, req, admin, emp, emp2, calendarId: seed.calendarId, createdStatus: created.status, ids: { employee: seed.users.find((u) => u.role !== 'admin').id, employee2: created.json.user.id }, counts };
}

async function employeeActivity(w) {
  const steps = [];
  const step = async (label, promise) => { const r = await promise; steps.push(`${label} -> ${r.status}${r.status >= 300 ? ` ${r.json?.error?.code ?? ''}` : ''}`); return r; };
  await step('employee session 09-21 (period of payroll 10-02, past) with reason', w.req('POST', '/api/days/2026-09-21/sessions', w.emp, { start: at('2026-09-21', '09:00'), end: at('2026-09-21', '17:00'), input_zone: LA, breaks_confirmed: true, breaks: [], reason: 'synthetic late entry' }));
  await step('employee session 10-07 (payroll 10-16, current)', w.req('POST', '/api/days/2026-10-07/sessions', w.emp, { start: at('2026-10-07', '08:00'), end: at('2026-10-07', '18:00'), input_zone: LA, breaks_confirmed: true, breaks: [] }));
  await step('employee2 session 10-08 (payroll 10-16, current)', w.req('POST', '/api/days/2026-10-08/sessions', w.emp2, { start: at('2026-10-08', '07:00'), end: at('2026-10-08', '19:00'), input_zone: LA, breaks_confirmed: true, breaks: [] }));
  await step('employee explicit Vacation 10-20 (payroll 10-30, next)', w.req('PUT', '/api/days/2026-10-20', w.emp, { category: 'Vacation', leave_minutes: 480, leave_kind: 'vacation', wfh: false, notes: '' }));
  await step('employee2 explicit Off 11-02 (payroll 11-13)', w.req('PUT', '/api/days/2026-11-02', w.emp2, { category: 'Off', leave_minutes: 0, wfh: false, notes: '' }));
  for (const line of steps) out(`  setup ${line}`);
  return steps.every((line) => / -> 20[01]$/.test(line));
}

async function adminRun(w) {
  const view = new Map();
  for (const [label, body] of SEQUENCE) view.set(label, await w.req('POST', '/api/admin/payroll-exceptions', w.admin, body(w)));
  for (const path of READS) view.set(`GET ${path}`, await w.req('GET', path, w.admin));
  return view;
}
function diffPaths(a, b, path = '$', acc = []) {
  if (acc.length >= 12) return acc;
  if (typeof a !== typeof b || a === null || b === null || typeof a !== 'object') {
    if (a !== b) acc.push(`${path}: ${JSON.stringify(a)} -> ${JSON.stringify(b)}`);
    return acc;
  }
  for (const key of new Set([...Object.keys(a), ...Object.keys(b)])) diffPaths(a[key], b[key], `${path}.${key}`, acc);
  return acc;
}
function compare(label, before, after) {
  const differing = [];
  for (const [name, x] of before) {
    const y = after.get(name);
    const a = mask(x.text);
    const b = mask(y.text);
    const same = x.status === y.status && a === b;
    if (!same) differing.push(name);
    let detail = '';
    if (!same) {
      try { detail = ` | ${x.status}->${y.status} ${diffPaths(JSON.parse(a), JSON.parse(b)).join('; ')}`; } catch { detail = ` | ${x.status}->${y.status} non-JSON body differs`; }
    }
    out(`  ${same ? 'same' : 'DIFF'} ${name} [${x.status} ${sha(a)}]${detail}`);
  }
  out(`  ${label}: ${before.size - differing.length}/${before.size} identical`);
  return differing;
}
