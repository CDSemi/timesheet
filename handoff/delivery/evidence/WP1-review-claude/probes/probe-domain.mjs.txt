let checks = 0;
const failures = [];
function check(label, ok, detail) {
  checks += 1;
  if (!ok && failures.length < 40) failures.push(`${label}: ${detail ?? ''}`);
  else if (!ok) failures.push('…');
}
const section = (name) => console.log(`\n## ${name}`);
const report = (name, before) => console.log(`${name}: ${checks - before} checks, failures so far ${failures.length}`);

// Deterministic PRNG (mulberry32) so the run is reproducible.
let seed = 0x5eed2026;
function rnd() {
  seed |= 0;
  seed = (seed + 0x6d2b79f5) | 0;
  let t = Math.imul(seed ^ (seed >>> 15), 1 | seed);
  t = (t + Math.imul(t ^ (t >>> 7), 61 | t)) ^ t;
  return ((t ^ (t >>> 14)) >>> 0) / 4294967296;
}
const ri = (a, b) => a + Math.floor(rnd() * (b - a + 1));
const pick = (xs) => xs[ri(0, xs.length - 1)];

function engineVersions(oracleVersions) {
  return oracleVersions.map((v, i) => ({
    id: `cal-${i + 1}`,
    seq: v.seq,
    effectiveFrom: v.effectiveFrom,
    weekdays: [...v.weekdays],
    dates: [...v.holidays].map((date) => ({ date, kind: 'holiday', name: 'Synthetic' })),
  }));
}

function randomDay(z) {
  const workDate = pick(interestingDates);
  // approximate local midnight of workDate via the oracle's own date function
  let t = O.parseDate(workDate) * O.DAY - 14 * 3600;
  while (z.localDateOf(t) < workDate) t += 60;
  const midnight = t;
  const sessions = [];
  let cursor = midnight + ri(-4 * 3600, 20 * 3600) + (rnd() < 0.5 ? ri(0, 59) : 0);
  const count = ri(1, 3);
  for (let i = 0; i < count; i += 1) {
    const len = rnd() < 0.15 ? ri(1, 600) : ri(10 * 60, 16 * 3600) + (rnd() < 0.5 ? ri(0, 59) : 0);
    const start = cursor;
    const end = start + len;
    const breaks = [];
    let bc = start + ri(0, Math.max(0, Math.floor(len / 3)));
    for (let k = ri(0, 3); k > 0 && bc < end - 1; k -= 1) {
      const bl = Math.min(end - bc, ri(1, 3600));
      breaks.push({ start: bc, end: bc + bl, countsAsWork: rnd() < 0.2 });
      bc = bc + bl + ri(0, 7200);
    }
    sessions.push({ start, end, breaks, breaksConfirmed: rnd() < 0.95 });
    cursor = end + ri(0, 4 * 3600);
  }
  if (rnd() < 0.05) sessions[sessions.length - 1].end = null;
  return { workDate, sessions };
}

const coverage = { multiDate: 0, laDstCrossing: 0, subMinuteSeconds: 0, secondVersionUsed: 0, multiSession: 0, breakAcrossMidnight: 0, bothClasses: 0, perSessionFloorWouldDiffer: 0, mathRoundWouldDiffer: 0 };
const laTransitions = ['2026-03-08T10:00:00Z', '2026-11-01T09:00:00Z', '2027-03-14T10:00:00Z', '2027-11-07T09:00:00Z'].map(O.parseIsoUtc);
{
  const before = checks;
  let compared = 0;
  const statusCounts = {};
  for (let trial = 0; trial < 12_000; trial += 1) {
    const z = pick(zones);
    const { workDate, sessions } = randomDay(z);
    const B = pick([480, 480, 480, 450, 600, 1]);
    const N = pick([30, 30, 0, 45, 60]);
    const M = pick([30, 30, 15, 60, 1, 7]);
    const weekdays = rnd() < 0.85 ? new Set([1, 2, 3, 4, 5]) : new Set([1, 2, 3, 4, 5, 6].filter(() => rnd() < 0.7).concat([3]));
    const v1 = { effectiveFrom: '2025-01-01', seq: 1, weekdays, holidays: new Set(holidays2026) };
    const versions = [v1];
    if (rnd() < 0.4) {
      // a later version effective mid-shift: different holidays/weekdays from its own date on
      const eff = O.addDays(workDate, pick([0, 1]));
      versions.push({ effectiveFrom: eff, seq: 2, weekdays: new Set([1, 2, 3, 4, 5, 6, 7].filter(() => rnd() < 0.6)), holidays: new Set(rnd() < 0.5 ? [O.addDays(workDate, 1)] : []) });
      if (versions[1].weekdays.size === 0) versions[1].weekdays.add(1);
    }
    const expected = O.workDayOracle({ sessions, localDateOf: z.localDateOf, versions, B, N, M });
    let actual;
    try {
      actual = eng.computeWorkDay({
        workDate,
        reportingZone: z.zone,
        policy: { id: 'p', requiredMinutes: B, thresholdMinutes: N, roundingStepMinutes: M },
        calendarVersions: engineVersions(versions),
        sessions: sessions.map((s) => ({
          startUtc: s.start,
          endUtc: s.end,
          breaksConfirmed: s.breaksConfirmed,
          breaks: s.breaks.map((b) => ({ startUtc: b.start, endUtc: b.end, countsAsWork: b.countsAsWork, confirmed: true })),
        })),
      });
    } catch (error) {
      actual = { status: `throw:${error.code ?? error.message}` };
    }
    statusCounts[expected.status] = (statusCounts[expected.status] ?? 0) + 1;
    compared += 1;
    if (expected.status !== 'complete') {
      check(`trial ${trial} status`, actual.status === expected.status, `${z.zone} ${workDate} engine ${actual.status} oracle ${expected.status}`);
      if (actual.status === expected.status && expected.status !== 'calendar_missing') {
        check(`trial ${trial} no invented minutes`, actual.creditedMinutes === null && actual.regularMinutes === null, JSON.stringify(actual));
      }
      continue;
    }
    const same =
      actual.status === 'complete' &&
      actual.regularSeconds === expected.rs &&
      actual.nonworkingSeconds === expected.os &&
      actual.regularMinutes === expected.R &&
      actual.nonworkingMinutes === expected.O &&
      actual.normalExcessMinutes === expected.E &&
      actual.eligibleMinutes === expected.T &&
      actual.creditedMinutes === expected.credit;
    check(`trial ${trial}`, same, `${z.zone}/${z.oracle} ${workDate} B${B} N${N} M${M} engine ${JSON.stringify({ s: actual.status, rs: actual.regularSeconds, os: actual.nonworkingSeconds, c: actual.creditedMinutes })} oracle ${JSON.stringify(expected)} sessions ${JSON.stringify(sessions)}`);
    // Coverage of the behaviours the review prompt names (and probe sensitivity to two classic mistakes).
    if (actual.segments.length > 1) coverage.multiDate += 1;
    if (sessions.length > 1) coverage.multiSession += 1;
    if (z.zone === 'America/Los_Angeles' && sessions.some((s) => laTransitions.some((x) => s.start < x && x < s.end))) coverage.laDstCrossing += 1;
    if (expected.rs % 60 !== 0 || expected.os % 60 !== 0) coverage.subMinuteSeconds += 1;
    if (versions.length > 1 && actual.segments.some((seg) => seg.calendarVersionId === 'cal-2')) coverage.secondVersionUsed += 1;
    if (expected.rs > 0 && expected.os > 0) coverage.bothClasses += 1;
    if (sessions.some((s) => s.breaks.some((b) => !b.countsAsWork && z.localDateOf(b.start) !== z.localDateOf(b.end - 1)))) coverage.breakAcrossMidnight += 1;
    const perSession = sessions.reduce(
      (acc, s) => {
        const m = O.secondsByLocalDate(O.netIntervals(s), z.localDateOf);
        let r = 0;
        let o = 0;
        for (const [d, sec] of m) (O.classify(versions, d) === 'normal' ? (r += sec) : (o += sec));
        return { R: acc.R + Math.floor(r / 60), O: acc.O + Math.floor(o / 60) };
      },
      { R: 0, O: 0 },
    );
    if (O.otOracle(perSession.R, perSession.O, B, N, M).credit !== expected.credit || perSession.R !== expected.R) coverage.perSessionFloorWouldDiffer += 1;
    if (Math.round(expected.T / M) * M !== expected.credit) coverage.mathRoundWouldDiffer += 1;
    // The per-segment seconds must add up to net work, and every segment date must be classified with its own version.
    const net = sessions.flatMap(O.netIntervals).reduce((sum, [a, b]) => sum + b - a, 0);
    check(`trial ${trial} segments sum`, actual.segments.reduce((s, x) => s + x.seconds, 0) === net, `${net}`);
    for (const seg of actual.segments) {
      const cls = O.classify(versions, seg.localDate);
      check(`trial ${trial} segment ${seg.localDate}`, seg.dayClass === cls, `${seg.dayClass} vs ${cls}`);
    }
  }
  console.log('oracle status distribution', statusCounts, 'trials', compared);
  console.log('coverage of complete trials', coverage);
  report('randomized interval engine', before);
}

section('3. Local time resolution (R-07) against the statutory LA rule');
{
  const before = checks;
  const days = ['2026-03-08', '2026-11-01', '2027-03-14', '2027-11-07', '2026-07-15', '2026-01-15', '2026-03-07', '2026-10-31'];
  for (const day of days) {
    for (let minute = 0; minute < 1440; minute += 1) {
      const hh = String(Math.floor(minute / 60)).padStart(2, '0');
      const mm = String(minute % 60).padStart(2, '0');
      const local = `${day}T${hh}:${mm}`;
      const wall = O.parseDate(day) * O.DAY + minute * 60;
      const cands = O.laCandidates(wall);
      const attempt = (opts) => {
        try {
          const r = eng.resolveLocalDateTime(local, 'America/Los_Angeles', opts);
          return { utc: r.utc, offset: r.offsetSeconds };
        } catch (error) {
          return { error: error.code };
        }
      };
      const plain = attempt({});
      if (cands.length === 0) {
        check(`${local} gap`, plain.error === 'nonexistent_local_time', JSON.stringify(plain));
        check(`${local} gap fold`, attempt({ fold: 0 }).error === 'nonexistent_local_time' && attempt({ offset: '-08:00' }).error === 'nonexistent_local_time', '');
      } else if (cands.length === 2) {
        check(`${local} ambiguous`, plain.error === 'ambiguous_local_time', JSON.stringify(plain));
        check(`${local} fold0`, attempt({ fold: 0 }).utc === cands[0], '');
        check(`${local} fold1`, attempt({ fold: 1 }).utc === cands[1], '');
        check(`${local} offset -07`, attempt({ offset: '-07:00' }).utc === cands[0], '');
        check(`${local} offset -08`, attempt({ offset: '-08:00' }).utc === cands[1], '');
        check(`${local} bad offset`, attempt({ offset: '-05:00' }).error === 'offset_mismatch', '');
      } else {
        check(`${local} unique`, plain.utc === cands[0], `${JSON.stringify(plain)} vs ${cands[0]}`);
        check(`${local} unique fold ignored`, attempt({ fold: 1 }).utc === cands[0], '');
      }
    }
  }
  report('local resolution, every minute of 8 days', before);
}
