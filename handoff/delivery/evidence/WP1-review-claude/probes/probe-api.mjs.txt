let checks = 0;
const failures = [];
const notes = [];
function check(label, ok, detail = '') {
  checks += 1;
  console.log(`${ok ? 'PASS' : 'FAIL'}  ${label}${detail ? `  — ${detail}` : ''}`);
  if (!ok) failures.push(label);
}
const note = (text) => {
  notes.push(text);
  console.log(`NOTE  ${text}`);
};

const SEED_CAL = [{ effectiveFrom: '2026-01-01', seq: 1, weekdays: new Set([1, 2, 3, 4, 5]), holidays: new Set(['2026-01-01', '2026-01-02', '2026-02-16', '2026-05-25', '2026-07-03', '2026-09-07', '2026-11-26', '2026-11-27', '2026-12-25']) }];
const LA = 'America/Los_Angeles';
const utcOfLa = (local) => {
  const [d, hm] = local.split('T');
  const [h, m] = hm.split(':').map(Number);
  const c = O.laCandidates(O.parseDate(d) * O.DAY + h * 3600 + m * 60);
  if (c.length !== 1) throw new Error(`oracle: ${local} is not unique in LA`);
  return c[0];
};
const body = (start, end, breaks = [], extra = {}) => ({ start, end, input_zone: LA, breaks, breaks_confirmed: true, ...extra });

/** Oracle for the API sessions of one work date, seed calendar and seed policy 480/30/30. */
function expectFor(sessionsUtc) {
  return O.workDayOracle({
    sessions: sessionsUtc.map((s) => ({ ...s, breaksConfirmed: true })),
    localDateOf: O.laLocalDate,
    versions: SEED_CAL,
    B: 480,
    N: 30,
    M: 30,
  });
}
function compareCalc(label, calc, want) {
  const ok =
    calc?.status === 'complete' &&
    calc.regular_seconds === want.rs &&
    calc.nonworking_seconds === want.os &&
    calc.regular_minutes === want.R &&
    calc.nonworking_minutes === want.O &&
    calc.eligible_minutes === want.T &&
    calc.credited_minutes === want.credit;
  check(label, ok, `API R ${calc?.regular_minutes} O ${calc?.nonworking_minutes} T ${calc?.eligible_minutes} credit ${calc?.credited_minutes} | oracle R ${want.R} O ${want.O} T ${want.T} credit ${want.credit}`);
}

console.log('\n## B. Two-user isolation (AC-01) on every implemented endpoint');
{
  const t = await createTestContext('2026-09-29T20:00:00Z');
  const emp = await t.login('employee');
  const adm = await t.login('admin');
  const created = await t.request('POST', '/api/days/2026-09-21/sessions', { cookie: emp, body: body(la('2026-09-21T09:00'), la('2026-09-21T19:00'), [], { breaks_confirmed: true }) });
  const empSession = created.body.session.id;
  const ghost = randomUUID();
  const update = { ...body(la('2026-09-21T08:00'), la('2026-09-21T09:00')), expected_version: 1, reason: 'probe' };
  for (const [method, payload] of [['GET'], ['PUT', update], ['DELETE', { expected_version: 1, reason: 'probe' }]]) {
    const other = await t.request(method, `/api/sessions/${empSession}`, { cookie: adm, body: payload });
    const missing = await t.request(method, `/api/sessions/${ghost}`, { cookie: adm, body: payload });
    check(`B1 ${method} another user's session is indistinguishable from a missing one`, other.status === 404 && JSON.stringify(other.body) === JSON.stringify(missing.body), `${other.status} ${JSON.stringify(other.body)}`);
  }
  const row = t.db.prepare('SELECT version, start_utc, end_utc FROM work_sessions WHERE id = ?').get(empSession);
  check('B1 employee session untouched after admin attempts', row.version === 1 && row.start_utc === '2026-09-21T16:00:00Z', JSON.stringify(row));
  const admSheet = await t.request('GET', '/api/timesheets/2026-10-02', { cookie: adm });
  check('B2 admin timesheet totals and days exclude the employee data', admSheet.body.totals.provisional_credited_minutes === 0 && admSheet.body.days.every((d) => d.sessions.length === 0 && d.entry === null), JSON.stringify(admSheet.body.totals));
  const admDay = await t.request('GET', '/api/days/2026-09-21', { cookie: adm });
  check('B3 admin day view of the same date shows nothing of the employee', admDay.body.sessions.length === 0 && admDay.body.entry === null && admDay.body.calculation.status === 'no_records');
  const admClock = await t.request('POST', '/api/clock/in', { cookie: adm, body: { input_zone: LA } });
  const empOut = await t.request('POST', '/api/clock/out', { cookie: emp, body: { breaks: [], breaks_confirmed: true } });
  check('B4 employee cannot clock out the admin running session', admClock.status === 201 && empOut.status === 409 && empOut.body.error.code === 'no_open_session', `${admClock.status}/${empOut.status}`);
  // Owner/identity injection into every write body is rejected by the strict schemas.
  const injections = [
    ['POST', '/api/days/2026-09-22/sessions', { ...body(la('2026-09-22T09:00'), la('2026-09-22T10:00')), user_id: t.userIds.admin }],
    ['PUT', `/api/sessions/${empSession}`, { ...update, owner_user_id: t.userIds.admin }],
    ['PUT', `/api/sessions/${empSession}`, { ...update, work_date: '2026-09-22' }],
    ['DELETE', `/api/sessions/${empSession}`, { expected_version: 1, user_id: t.userIds.admin }],
    ['PUT', '/api/days/2026-09-22', { category: 'Worked', leave_minutes: 0, wfh: false, notes: '', user_id: t.userIds.admin }],
    ['POST', '/api/clock/in', { input_zone: LA, user_id: t.userIds.admin }],
    ['POST', '/api/clock/out', { breaks: [], breaks_confirmed: true, session_id: empSession }],
    ['POST', '/api/policies', { effective_from: '2026-10-05', required_minutes: 480, threshold_minutes: 30, rounding_step_minutes: 30, reference_start: '08:00', reference_end: '17:00', breaks: [{ start_offset_minutes: 120, duration_minutes: 60, counts_as_work: false }], deficit_mode: 'ignore', user_id: t.userIds.admin }],
  ];
  for (const [method, path, payload] of injections) {
    const r = await t.request(method, path, { cookie: emp, body: payload });
    check(`B5 ${method} ${path.replace(empSession, '{own}')} rejects identity/extra fields`, r.status === 422 && r.body.error.code === 'validation_error', `${r.status} ${r.body?.error?.code}`);
  }
  const overlap = await t.request('POST', '/api/days/2026-09-21/sessions', { cookie: emp, body: body(la('2026-09-21T18:00'), la('2026-09-21T20:00')) });
  check('B6 overlap error only names the caller own session', overlap.status === 422 && overlap.body.error.details?.conflicting_session_id === empSession, JSON.stringify(overlap.body.error));
  const policies = await t.request('GET', '/api/policies', { cookie: adm });
  check('B7 admin sees only its own policy history', policies.body.policies.length === 1 && t.db.prepare('SELECT user_id FROM work_policies WHERE id = ?').pluck().get(policies.body.policies[0].id) === t.userIds.admin);
  const audit = t.db.prepare('SELECT DISTINCT actor_user_id, owner_user_id FROM audit_events WHERE owner_user_id = ? AND operation LIKE ?').all(t.userIds.employee, 'work_session.%');
  check('B8 every employee session audit event was performed by the employee', audit.length === 1 && audit[0].actor_user_id === t.userIds.employee, JSON.stringify(audit));
  t.close();
}

  check('D1 update audit keeps exact before/after including breaks', updated.status === 200 && b.breaks[0].end_utc === '2026-09-24T19:30:00Z' && a.breaks[0].end_utc === '2026-09-24T20:00:00Z' && b.version === 1 && a.version === 2 && ev.occurred_at === '2026-09-29T20:00:00Z' && ev.actor_user_id === t.userIds.employee, JSON.stringify({ b: b.breaks, a: a.breaks }));
  const afterDelete = await t.request('DELETE', `/api/sessions/${id}`, { cookie, body: { expected_version: 2 } });
  const again = await t.request('PUT', `/api/sessions/${id}`, { cookie, body: { ...body(la('2026-09-24T09:00'), la('2026-09-24T17:30')), expected_version: 3 } });
  check('D2 deleted session cannot be updated (404)', afterDelete.status === 200 && again.status === 404, `${afterDelete.status}/${again.status}`);
  const breaksLeft = t.db.prepare('SELECT count(*) FROM session_breaks WHERE session_id = ?').pluck().get(id);
  check('D3 breaks of a deleted session are removed with it', breaksLeft === 0, String(breaksLeft));
  const v = t.db.prepare('SELECT version FROM timesheets WHERE user_id = ?').pluck().get(t.userIds.employee);
  check('D4 timesheet version bumped per change (create, update, delete → 3 or more incl. day-entry creation)', v >= 3, String(v));
  t.close();
}

{
  const t = await createTestContext('2026-09-29T20:00:00Z');
  const cookie = await t.login('employee');
  const created = await t.request('POST', '/api/days/2026-09-24/sessions', { cookie, body: body(la('2026-09-24T09:00'), la('2026-09-24T17:00'), [
    { start: la('2026-09-24T10:00'), end: la('2026-09-24T10:15'), counts_as_work: false },
    { start: la('2026-09-24T12:00'), end: la('2026-09-24T12:30'), counts_as_work: false }]) });
  const id = created.body.session.id;
  const breaks = t.db.prepare('SELECT id FROM session_breaks WHERE session_id = ? ORDER BY start_utc').pluck().all(id);
  const attempt = (sql, ...args) => {
    try {
      t.db.prepare(sql).run(...args);
      return 'accepted';
    } catch (error) {
      return `rejected: ${error.message}`;
    }
  };
  const outside = attempt("UPDATE session_breaks SET end_utc = '2026-09-25T02:00:00Z' WHERE id = ?", breaks[0]);
  const overlapping = attempt("UPDATE session_breaks SET start_utc = '2026-09-24T19:10:00Z', end_utc = '2026-09-24T19:40:00Z' WHERE id = ?", breaks[1]);
  console.log(`INFO  raw UPDATE moving a break outside its session: ${outside}`);
  console.log(`INFO  raw UPDATE making two breaks overlap: ${overlapping}`);
  if (outside === 'accepted' || overlapping === 'accepted') note('E: session_breaks has INSERT-only triggers; raw UPDATEs that move a break outside its session or onto another break are accepted. The API never updates break rows (delete + insert), so this is optional hardening only (OPT-1).');
  const ownerMove = attempt('UPDATE work_sessions SET user_id = ? WHERE id = ?', t.userIds.admin, id);
  check('E1 session owner cannot be changed in SQL', ownerMove.startsWith('rejected'), ownerMove);
  const tsMove = attempt('UPDATE timesheets SET user_id = ? WHERE user_id = ?', t.userIds.admin, t.userIds.employee);
  check('E2 a timesheet with day entries cannot be moved to another user', tsMove.startsWith('rejected'), tsMove);
  const policyEdit = attempt('UPDATE work_policies SET threshold_minutes = 0');
  check('E3 policy versions immutable', policyEdit.includes('immutable_policy_version'), policyEdit);
  t.close();
}
