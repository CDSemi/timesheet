const sha = (text) => createHash('sha256').update(text).digest('hex').slice(0, 16);
const run = (args, log) => {
  const result = spawnSync(process.execPath, args, { cwd: REPO, encoding: 'utf8', stdio: ['ignore', 'pipe', 'pipe'], maxBuffer: 64 * 1024 * 1024 });
  writeFileSync(log, `${result.stdout ?? ''}${result.stderr ?? ''}\nexit=${result.status}\n`);
  return result;
};

for (const id of process.argv.slice(2)) {
  const mutant = M[id];
  if (mutant === undefined) throw new Error(`unknown mutant ${id}`);
  const paths = mutant.files ?? [...new Set(mutant.edits.map(([file]) => file))];
  const original = Object.fromEntries(paths.map((file) => [file, readFileSync(`${REPO}/${file}`, 'utf8')]));
  const before = Object.fromEntries(paths.map((file) => [file, sha(original[file])]));
  const files = { ...original };
  if (mutant.transform !== undefined) mutant.transform(files);
  else {
    for (const [file, find, replace] of mutant.edits) {
      if (files[file].split(find).length !== 2) throw new Error(`${id}: the edit target is not unique or missing in ${file}: ${find.slice(0, 60)}`);
      files[file] = files[file].replace(find, () => replace);
    }
  }
  let summary;
  try {
    for (const file of paths) writeFileSync(`${REPO}/${file}`, files[file]);
    const log = `${OUT}/${id}.log`;
    const [kind, ...rest] = mutant.run;
    let result;
    if (kind === 'vitest') {
      result = run(['node_modules/vitest/vitest.mjs', 'run', ...rest], log);
    } else {
      const build = run([NPM_CLI, 'run', 'build'], `${OUT}/${id}.build.log`);
      if (build.status !== 0) throw new Error(`${id}: the mutant does not build`);
      const [grep, project] = rest;
      result = run(['node_modules/@playwright/test/cli.js', 'test', 'tests/e2e/calendar-default.spec.ts', `--project=${project}`, '--workers=2', '-g', grep], log);
    }
    const text = readFileSync(log, 'utf8');
    const failed = [...text.matchAll(/^\s+(?:×|x)\s+(?:\d+ \[\w+\] › )?(.*?)(?: \(?\d[\d.]*m?s\)?)?$/gm)].map((match) => match[1].replace(/^tests\\e2e\\calendar-default\.spec\.ts:\d+:\d+ › /, '').slice(0, 110));
    const counts = /(\d+) failed/.exec(text)?.[0] ?? (/Tests\s+(\d+) failed/.exec(text)?.[0] ?? '0 failed');
    summary = { exit: result.status, counts, failed: [...new Set(failed)].slice(0, 6) };
  } finally {
    for (const file of paths) writeFileSync(`${REPO}/${file}`, original[file]);
  }
  const after = Object.fromEntries(paths.map((file) => [file, sha(readFileSync(`${REPO}/${file}`, 'utf8'))]));
  const restored = paths.every((file) => before[file] === after[file]);
  const line = `${id}\t${summary.exit === 0 ? 'SURVIVED' : 'killed'}\texit=${summary.exit}\t${summary.counts}\trestore=${restored ? 'ok' : 'FAILED'}\t${mutant.what}\t=> ${summary.failed.join(' | ')}\n`;
  appendFileSync(`${OUT}/00-summary.tsv`, line);
  process.stdout.write(line);
}
