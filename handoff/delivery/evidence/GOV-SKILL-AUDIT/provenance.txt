== Part 3: probe script (lockhash.mjs, kept in the audit temp folder)
// Read-only probe: compare skills-lock.json computedHash values with candidate hash methods.
// Usage: node lockhash.mjs <lockfile> <skillsRoot> [name ...]
import { readFileSync, readdirSync, statSync } from 'node:fs';
import { createHash } from 'node:crypto';
import { join, relative, sep } from 'node:path';

const [lockPath, root, ...names] = process.argv.slice(2);
const lock = JSON.parse(readFileSync(lockPath, 'utf8')).skills;

function walk(dir) {
  const out = [];
  for (const entry of readdirSync(dir)) {
    const full = join(dir, entry);
    if (statSync(full).isDirectory()) out.push(...walk(full));
    else out.push(full);
  }
  return out;
}

function folderHash(dir, withPath) {
  const files = walk(dir)
    .map((full) => ({ full, rel: relative(dir, full).split(sep).join('/') }))
    .sort((a, b) => (a.rel < b.rel ? -1 : a.rel > b.rel ? 1 : 0));
  const hash = createHash('sha256');
  for (const f of files) {
    if (withPath) hash.update(f.rel);
    hash.update(readFileSync(f.full));
  }
  return hash.digest('hex');
}

for (const name of names.length ? names : Object.keys(lock)) {
  const dir = join(root, name);
  let skillMd = '-';
  let fh = '-';
  let fhNoPath = '-';
  try {
    skillMd = createHash('sha256').update(readFileSync(join(dir, 'SKILL.md'))).digest('hex');
    fh = folderHash(dir, true);
    fhNoPath = folderHash(dir, false);
  } catch {
    // missing folder
  }
  const want = lock[name]?.computedHash ?? '(not in lock)';
  const match = want === skillMd ? 'skillmd' : want === fh ? 'folder+path' : want === fhNoPath ? 'folder' : 'none';
  console.log(`${name} lock=${want.slice(0, 12)} skillmd=${skillMd.slice(0, 12)} folder+path=${fh.slice(0, 12)} folder=${fhNoPath.slice(0, 12)} match=${match}`);
}
