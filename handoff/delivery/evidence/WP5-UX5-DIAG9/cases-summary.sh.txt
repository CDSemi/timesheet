# Usage: bash cases-summary.sh <variant> <project> <out file>
# One block per midnight case: the simulated instants of the stages (midnight = 0 s) and the test result.
. <task>/env.sh
cd $T
V=$1; P=$2; OUT=$3
: > $OUT
for n in C1-all-before C2-between-today-and-click C3-between-click-and-poll C4-poll-starts-at-midnight C5-after-today-before-signin C6-all-after; do
  f=logs/$V-$P-$n.txt
  echo "== $V $P $n" >> $OUT
  if [ ! -f $f ]; then echo "(not run)" >> $OUT; continue; fi
  grep -E "^DIAG9 (start|after-today|after-signin|before-click|after-click): go|^DIAG9 in-poll: (arrive|go)|^DIAG9 (start|after-today|before-click|in-poll): arrive" $f >> $OUT
  grep -E "^  (ok|x)  *1 |^  [0-9]+ (passed|failed)|Expected: |Received: " $f >> $OUT
  grep -E "^exit=" $f >> $OUT
done
