# Usage: bash dense.sh <variant tag> <delta ms> <list file> <project>...
# For every listed file:line and project: one single-worker run launched 7 s before local midnight (2026-10-27T07:00:00Z) whose test body
# starts <delta> ms before the midnight is replaced... no: the body starts at midnight minus <delta> ms, so the midnight falls <delta> ms into the body.
# Results go to out/dense-<variant>-<delta>.txt, one line per run.
. <task>/env.sh
V=$1; DELTA=$2; LIST=$3; shift 3
cd $T
RES=out/dense-$V-$DELTA.txt
: > $RES
while IFS= read -r target; do
  [ -z "$target" ] && continue
  for P in "$@"; do
    tag=$(echo "$target" | tr '/:.' '___')
    L="dense-$V-$DELTA-$tag-$P"
    MID_ISO=2026-10-27T07:00:00Z
    MID_MS=$(date -u -d "$MID_ISO" +%s%3N)
    export DIAG9_MID_MS=$MID_MS
    export DIAG9_AT_TEST_START=-$DELTA
    unset DIAG9_LOG
    START_ISO=$(date -u -d "@$(( (MID_MS / 1000) - 7 ))" +%Y-%m-%dT%H:%M:%SZ)
    bash $T/run-shift.sh "$L" "$START_ISO" "$target" --project=$P --workers=1 > "logs/$L.txt" 2>&1
    summary=$(grep -E '^\s+[0-9]+ (passed|failed|skipped)' "logs/$L.txt" | tr -s ' ' | tr '\n' ';')
    ex=$(grep '^exit=' "logs/$L.txt")
    arrive=$(grep -m1 'DIAG9 test-start: arrive' "logs/$L.txt" | sed 's/.*midnight //')
    echo "$target $P delta=$DELTA $ex $summary | $arrive" >> $RES
  done
done < "$LIST"
echo dense-done >> $RES
