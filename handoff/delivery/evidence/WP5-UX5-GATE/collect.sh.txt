# Copies the masked evidence into handoff/delivery/evidence/WP5-UX5-GATE/ (masked LF .txt, scripts as .txt, 15 synthetic PNG).
. <task>/env.sh
mkdir -p $E
m() { node "$TW\\mask.mjs" "$1" "$2" >> $T/tmp/mask-last.txt 2>&1; }
cd $T/out
for f in *.txt; do
  case "$f" in ex-files.txt|ex-hashes.txt|ex-sorted.txt|lstree-sorted.txt|keys-list.txt|51-*-files.txt|51-*-hashes.txt|25-removed-*.txt) continue;; esac
  m "$T/out/$f" "$E/$f"
done
cd $T/logs
for f in control-*.txt sweep-*.txt rerun-*.txt probe-*.txt vprobe-*.txt cmp-dist.txt csp.txt hist.txt; do
  [ -f "$f" ] && m "$T/logs/$f" "$E/log-$f"
done
cd $T/obs
for f in *.txt; do m "$T/obs/$f" "$E/obs-$f"; done
cd $T
for s in env.sh setup.sh digest.sh run-main.sh run-rest.sh run-integ.sh run-drill.sh run-shift.sh run-probe.sh run-csp.sh run-hist.sh run-vprobe.sh sweep.sh cmp-dist.sh cfg.sh keys.sh scope.sh chain1.sh collect.sh; do [ -f "$s" ] && m "$T/$s" "$E/$s.txt"; done
for s in mask.mjs zone.mjs run-tz.mjs tzprobe.mjs tzchild.mjs contrast5.mjs shiftclock.mjs probe-now.mjs mapdist.mjs wp3check.mjs zz-lib.ts zz-png.ts zz-control.spec.ts zz-bp.spec.ts zz-bar.spec.ts zz-editor5.spec.ts zz-ctl.spec.ts zz-phone.spec.ts zz-review.spec.ts zz-axe5.spec.ts zz-tok.spec.ts zz-shots.spec.ts zz-csp.spec.ts zz-hist.spec.ts zz-session.spec.ts zz-modal.spec.ts; do [ -f "$s" ] && m "$T/$s" "$E/$s.txt"; done
for s in zz-frac.mjs zz-frac2.mjs zz-pdfprobe.mjs zz-pdfdump.mjs tests/integration/zz-v13to15.test.ts tests/integration/zz-backup-variants.test.ts tests/integration/zz-deadline.test.ts; do [ -f "$T/export-probe/$s" ] && m "$T/export-probe/$s" "$E/$(basename $s).txt"; done
cp $T/shots-shots/ux5gate-*-synthetic.png $E/
ls $E | wc -l
ls $E/*.png | wc -l
