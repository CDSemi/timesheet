async function settle(page: Page): Promise<void> {
  await page.evaluate('new Promise((done) => requestAnimationFrame(() => requestAnimationFrame(() => setTimeout(done, 200))))');
}

async function tokens(page: Page): Promise<{ pad: number; editor: number; gap: number; sheet: number }> {
  return page.evaluate(() => {
    const probe = document.createElement('div');
    probe.style.cssText = 'position:absolute;visibility:hidden;width:0;height:0;overflow:hidden';
    const kids = ['--page-pad', '--editor-width', '--editor-gap', '--sheet-min-width'].map((t) => {
      const c = document.createElement('div');
      c.style.width = `var(${t})`;
      probe.append(c);
      return c;
    });
    document.body.append(probe);
    const [pad, editor, gap, sheet] = kids.map((c) => c.getBoundingClientRect().width);
    probe.remove();
    return { pad: pad ?? 0, editor: editor ?? 0, gap: gap ?? 0, sheet: sheet ?? 0 };
  });
}

async function clientWidthTo(page: Page, target: number): Promise<number> {
  let width = target;
  for (let i = 0; i < 4; i += 1) {
    await page.setViewportSize({ width: Math.round(width), height: 900 });
    await page.evaluate('new Promise((d) => requestAnimationFrame(() => requestAnimationFrame(d)))');
    const client = await page.evaluate<number>('document.documentElement.clientWidth');
    if (client === target) return client;
    width += target - client;
  }
  return page.evaluate<number>('document.documentElement.clientWidth');
}

const HIDDEN_UNDER = `(() => {
  const el = document.activeElement; const ed = document.querySelector('[data-day-editor]');
  if (!el || !ed || ed.contains(el)) return null;
  const r = el.getBoundingClientRect(); const p = ed.getBoundingClientRect();
  if (r.width === 0 || r.height === 0) return null;
  const wholly = r.left >= p.left - 0.5 && r.right <= p.right + 0.5 && r.top >= p.top - 0.5 && r.bottom <= p.bottom + 0.5;
  return { wholly, name: (el.getAttribute('aria-label') || el.textContent || el.tagName).replace(/\\s+/g, ' ').trim().slice(0, 40) };
})()`;

for (const width of [768, 1024, 1280]) {
  test(`keyboard walk at ${width}px never leaves a focused sheet control wholly under the drawer (seeded long content)`, async ({ page, privateServer }, testInfo) => {
    test.skip(testInfo.project.name !== 'desktop', 'desktop');
    test.setTimeout(300_000);
    const data = await seedLongContent(privateServer, `Synthetic Walk ${width}`);
    await page.setViewportSize({ width, height: 800 });
    await signInTo(page, privateServer, data.person.account);
    await expect(page.locator('[data-day]')).toHaveCount(14);
    await namedDayButton(page, data.dates[0] ?? '', 'Edit').click();
    const editor = page.getByRole('dialog', { name: /Day editor/ });
    await expect(editor).toBeVisible();
    for (const key of ['Tab', 'Shift+Tab']) {
      if (key === 'Tab') await page.getByRole('switch', { name: 'Worked hours' }).focus();
      else await page.locator('.sheet-footer a, .sheet-footer button').last().focus();
      let hiddenAfterSettle = 0;
      let transient = 0;
      let stops = 0;
      const sides = new Set<string>();
      for (let step = 0; step < 200; step += 1) {
        await page.keyboard.press(key);
        await page.waitForTimeout(60);
        let r = await page.evaluate<{ wholly: boolean; name: string } | null>(HIDDEN_UNDER);
        if (r?.wholly) {
          transient += 1;
          await settle(page);
          r = await page.evaluate<{ wholly: boolean; name: string } | null>(HIDDEN_UNDER);
        }
        if (r !== null) {
          stops += 1;
          if (r.wholly) {
            hiddenAfterSettle += 1;
            note(OUTFILE, `  HIDDEN ${width} ${key}: ${r.name}`);
          }
        }
        sides.add((await editor.getAttribute('data-side')) ?? '-');
      }
      note(OUTFILE, `WALK ${width}px ${key}: ${stops} sheet stops over 200 presses, wholly under the drawer after settling ${hiddenAfterSettle}, transient before the drawer re-docked ${transient}; sides seen ${[...sides].join(',')}`);
      expect(hiddenAfterSettle).toBe(0);
    }
  });
}

test('discard prompt (switch, Next period, Today, Escape) and Escape never leaks from a nested dialog', async ({ page, privateServer }, testInfo) => {
  test.skip(testInfo.project.name !== 'desktop', 'desktop');
  test.setTimeout(180_000);
  const admin = await new SeedClient(privateServer.origin).signIn(privateServer.credentials.admin.email, privateServer.credentials.admin.password);
  const person = await newPerson(admin, privateServer, { displayName: 'Synthetic Discard Gate', signature: 'none' });
  const free = await person.api.displayedPeriodFreeWorkdays();
  const [a = '', b = ''] = free;
  await page.setViewportSize({ width: 1440, height: 800 });
  await signInTo(page, privateServer, person.account);
  await expect(page.locator('[data-day]')).toHaveCount(14);
  const editor = page.getByRole('dialog', { name: /Day editor/ });
  const prompt = page.getByRole('alertdialog', { name: 'Discard your changes?' });
  const keep = prompt.getByRole('button', { name: 'Keep editing', exact: true });
  const discard = prompt.getByRole('button', { name: 'Discard', exact: true });
  const periodName = () => page.locator('.period-name').innerText();
  const first = await periodName();
  const record = async (trigger: string) => {
    await expect(prompt).toBeVisible();
    const focused = await keep.evaluate((el) => el === document.activeElement);
    const modal = await prompt.evaluate((el) => el.matches(':modal'));
    note(OUTFILE, `DISCARD ${trigger}: prompt shown, modal ${modal}, Keep editing focused ${focused}, buttons ${await prompt.getByRole('button').allInnerTexts()}`);
    expect(modal && focused).toBe(true);
    await keep.click();
    await expect(prompt).toHaveCount(0);
    const kept = (await editor.getByLabel('Notes').inputValue()) === 'Synthetic unsaved note';
    note(OUTFILE, `DISCARD ${trigger}: Keep editing -> editor still open on ${await editor.getAttribute('data-day-editor')}, typed note kept ${kept}`);
    expect(kept).toBe(true);
  };
  await namedDayButton(page, a, 'Edit').click();
  await expect(editor).toHaveAttribute('data-day-editor', a);
  await editor.getByLabel('Notes').fill('Synthetic unsaved note');
  await namedDayButton(page, b, 'Edit').click();
  await record('switch to another day');
  await namedDayButton(page, b, 'Edit').click();
  await discard.click();
  await expect(editor).toHaveAttribute('data-day-editor', b);
  note(OUTFILE, `DISCARD switch: Discard -> editor on the other day, note now "${await editor.getByLabel('Notes').inputValue()}"`);
  await editor.getByLabel('Notes').fill('Synthetic unsaved note');
  await page.getByRole('button', { name: 'Next period' }).click();
  await record('Next period');
  await page.getByRole('button', { name: 'Next period' }).click();
  await discard.click();
  await expect.poll(periodName).not.toBe(first);
  note(OUTFILE, `DISCARD Next period: Discard -> period "${await periodName()}"`);
  const other = await periodName();
  const today = page.locator('[data-period-today]');
  if ((await editor.count()) === 0) {
    const dayInOther = (await page.locator('[data-day]').first().getAttribute('data-day')) ?? '';
    await namedDayButton(page, dayInOther, 'Edit').click().catch(async () => page.locator('[data-day-button]').first().click());
  }
  await expect(editor).toBeVisible();
  await editor.getByLabel('Notes').fill('Synthetic unsaved note');
  await today.click();
  await record('Today');
  await today.click();
  await discard.click();
  await expect.poll(periodName).not.toBe(other);
  note(OUTFILE, `DISCARD Today: Discard -> period "${await periodName()}" (the displayed period again: ${(await periodName()) === first})`);
  // Escape with an unsaved edit
  if ((await editor.count()) === 0) await page.locator('[data-day-button]').first().click();
  await expect(editor).toBeVisible();
  await editor.getByLabel('Notes').fill('Synthetic unsaved note');
  await page.keyboard.press('Escape');
  await record('Escape');
  await page.keyboard.press('Escape');
  await discard.click();
  await expect(editor).toHaveCount(0);
  note(OUTFILE, 'DISCARD Escape: Discard -> editor closed');

  // Escape in a nested dialog: only the top one closes. Add session dialog over the editor, then the Break sub-dialog is phone-only; use the save confirmation.
  await namedDayButton(page, a, 'Edit').click();
  await expect(editor).toBeVisible();
  await editor.getByRole('button', { name: 'Add session', exact: true }).click();
  const session = page.getByRole('dialog', { name: 'New session', exact: true });
  await expect(session).toBeVisible();
  await page.keyboard.press('Escape');
  const stillEditor = await editor.isVisible();
  const sessionGone = (await session.count()) === 0;
  note(OUTFILE, `ESCAPE nested: first Escape with the session dialog on top: session dialog closed ${sessionGone}, editor still open ${stillEditor}`);
  expect(sessionGone).toBe(true);
  expect(stillEditor).toBe(true);
  await page.keyboard.press('Escape');
  await expect(editor).toHaveCount(0);
  note(OUTFILE, 'ESCAPE nested: the second Escape closes the editor');
});

test('Current period: returns from another period and a Keep editing answer leaves focus on that button (needs a window day; reports "not in the window" otherwise)', async ({ page, privateServer }, testInfo) => {
  test.skip(testInfo.project.name !== 'desktop', 'desktop');
  test.setTimeout(120_000);
  const admin = await new SeedClient(privateServer.origin).signIn(privateServer.credentials.admin.email, privateServer.credentials.admin.password);
  const person = await newPerson(admin, privateServer, { displayName: 'Synthetic Current Gate', signature: 'none' });
  const periods = await person.api.periodsCurrent();
  const windowed = periods.current.payroll_date !== periods.in_progress.payroll_date;
  note(OUTFILE, `CURRENT period test: server today ${periods.today_local}, displayed ${periods.current.period_start}..${periods.current.period_end}, in progress ${periods.in_progress.period_start}..${periods.in_progress.period_end}, window ${windowed ? 'inside' : 'outside'}`);
  if (!windowed) return;
  const free = await person.api.displayedPeriodFreeWorkdays();
  await page.setViewportSize({ width: 1440, height: 800 });
  await signInTo(page, privateServer, person.account);
  await expect(page.locator('[data-day]')).toHaveCount(14);
  const editor = page.getByRole('dialog', { name: /Day editor/ });
  const prompt = page.getByRole('alertdialog', { name: 'Discard your changes?' });
  const current = page.locator('[data-period-current]');
  await page.locator('[data-period-today]').click();
  await expect(current).toBeVisible();
  const day2 = (await page.locator('[data-day-button]').first().getAttribute('aria-label')) ?? '';
  await page.locator('[data-day-button]').first().click();
  await expect(editor).toBeVisible();
  await editor.getByLabel('Notes').fill('Synthetic unsaved note');
  await current.click();
  await expect(prompt).toBeVisible();
  await prompt.getByRole('button', { name: 'Keep editing', exact: true }).click();
  const focusedOnCurrent = await current.evaluate((el) => el === document.activeElement);
  const focusName = await page.evaluate("document.activeElement ? (document.activeElement.getAttribute('aria-label') || document.activeElement.textContent || document.activeElement.tagName).trim().slice(0, 40) : 'none'");
  note(OUTFILE, `CURRENT period + Keep editing: focus is on "${focusName}" (the Current period button: ${focusedOnCurrent}); editor still open ${await editor.isVisible()} for ${day2.slice(0, 30)}`);
  expect(focusedOnCurrent).toBe(true);
  await current.click();
  await prompt.getByRole('button', { name: 'Discard', exact: true }).click();
  await expect(page.locator('.period-name')).toContainText(/\d/);
  void free;
});
