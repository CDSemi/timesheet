import { readFileSync } from 'node:fs';
import { getDocument, OPS } from 'pdfjs-dist/legacy/build/pdf.mjs';

const bytes = readFileSync(process.argv[2]);
const pdf = await getDocument({ data: new Uint8Array(bytes), useSystemFonts: false, verbosity: 0 }).promise;
const page = await pdf.getPage(1);
const ops = await page.getOperatorList();
let shown = 0;
ops.fnArray.forEach((fn, index) => {
  if (fn === OPS.constructPath && shown < 6) {
    const args = ops.argsArray[index];
    console.log(`constructPath #${index}: arg0 ${JSON.stringify(args[0])} type ${typeof args[0]}; arg1 ${args[1] === null ? 'null' : Object.prototype.toString.call(args[1])} ${JSON.stringify(Array.from(args[1]?.[0] ?? args[1] ?? []).slice(0, 14))}; arg2 ${JSON.stringify(Array.from(args[2] ?? []))}`);
    shown += 1;
  }
});
console.log(`OPS.constructPath=${OPS.constructPath} moveTo=${OPS.moveTo} lineTo=${OPS.lineTo} rectangle=${OPS.rectangle} stroke=${OPS.stroke} fill=${OPS.fill}`);
const counts = {};
for (const fn of ops.fnArray) counts[fn] = (counts[fn] ?? 0) + 1;
console.log(JSON.stringify(counts));
