. <task>/env.sh
export NODE_OPTIONS="--trace-deprecation --pending-deprecation"
O=$T/out
cd $T/export
# AC-13 three times
for i in 1 2 3; do
  node node_modules/vitest/vitest.mjs run tests/integration/ac13-two-week.test.ts > $O/05-ac13-$i.txt 2>&1; echo "ac13 run $i exit $? dep=$(grep -ci deprecat $O/05-ac13-$i.txt)" >> $O/05-ac13-exits.txt
done
# AC-13 under zones
for z in Asia/Tokyo America/New_York; do
  zn=$(echo $z | tr '/' '_')
  node $T/tzprobe.mjs $z > $O/06-tzprobe-$zn.txt 2>&1
  node $T/run-tz.mjs $z tests/integration/ac13-two-week.test.ts > $O/06-ac13-$zn.txt 2>&1; echo "ac13 zone $z exit $? $(cat $O/06-tzprobe-$zn.txt)" >> $O/06-ac13-exits.txt
done
# npm test three times
for i in 1 2 3; do
  npm test > $O/12-npmtest-$i.txt 2>&1; echo "npm test run $i exit $? dep=$(grep -ci deprecat $O/12-npmtest-$i.txt) crash=$(grep -ciE 'worker|crash|0xC0000005|exited' $O/12-npmtest-$i.txt)" >> $O/12-npmtest-exits.txt
done
# audit and validators
npm audit --omit=dev > $O/10-audit.txt 2>&1; echo "audit exit $?" >> $O/10-audit.txt
$PY handoff/delivery/validate_package.py --preflight > $O/11-validate_package.txt 2>&1; echo "validate_package exit $?" >> $O/11-exits.txt
$PY handoff/delivery/validate_orchestration.py > $O/11-validate_orchestration.txt 2>&1; echo "validate_orchestration exit $?" >> $O/11-exits.txt
$PY handoff/delivery/check_recovery.py > $O/11-check_recovery.txt 2>&1; echo "check_recovery exit $?" >> $O/11-exits.txt
node scripts/precommit-check.mjs --self-test > $O/11-precommit-selftest.txt 2>&1; echo "precommit self-test exit $?" >> $O/11-exits.txt
node $T/contrast5.mjs src/client/styles.css > $O/40-contrast.txt 2>&1; echo "contrast exit $?" >> $O/11-exits.txt
cat $O/05-ac13-exits.txt $O/06-ac13-exits.txt $O/12-npmtest-exits.txt $O/11-exits.txt
echo rest-done > $O/rest.done
