async function seededReviewer(server: Parameters<typeof newPerson>[1], name: string, options: { signature?: 'saved' | 'none' } = {}) {
  const admin = await new SeedClient(server.origin).signIn(server.credentials.admin.email, server.credentials.admin.password);
  const person = await newPerson(admin, server, { displayName: name, to: [TO], cc: [CC], signature: options.signature ?? 'saved' });
  const free = await person.api.displayedPeriodFreeWorkdays();
  await person.api.seedCompleteDay(free[0] ?? '');
  await person.api.seedCompleteDay(free[1] ?? '');
  return { admin, person };
}

const settle = (page: Page) => page.evaluate('new Promise((d) => requestAnimationFrame(() => requestAnimationFrame(() => setTimeout(d, 200))))');

async function openReview(page: Page, server: { origin: string }, account: { email: string; password: string }, payrollDate: string): Promise<void> {
  await signInTo(page, server as never, account, `#/review/${payrollDate}`);
  await expect(page.getByRole('heading', { name: 'Review and sign off', level: 1 })).toBeVisible();
  await expect(page.locator('[data-paper="review"]')).toBeVisible();
}

async function signOff(page: Page): Promise<void> {
  await page.locator('[data-sign-off]').click();
  const dialog = page.getByRole('dialog', { name: 'Sign off confirmation' });
  await expect(dialog).toBeVisible();
  const ack = dialog.getByRole('checkbox', { name: /I acknowledge the incomplete evidence/ });
  if (await ack.count()) await ack.check();
  await dialog.getByRole('button', { name: 'Sign off', exact: true }).click();
  await expect(page.locator('[data-result="created"]')).toContainText('Signed.');
}

test('Review layout facts before sign-off: title, Back label, company line, one column at 1440, signature geometry, Send disabled', async ({ page, privateServer }, testInfo) => {
  test.setTimeout(240_000);
  const project = testInfo.project.name;
  const { person } = await seededReviewer(privateServer, 'Synthetic Review Gate');
  if (project === 'desktop') await page.setViewportSize({ width: 1440, height: 900 });
  await openReview(page, privateServer, person.account, person.payrollDate);
  await settle(page);
  const title = await page.getByRole('heading', { name: 'Review and sign off', level: 1 }).innerText();
  const headText = (await page.locator('.review-head').innerText()).replace(/\s+/g, ' ');
  const hasId = /[0-9a-f]{8}-[0-9a-f]{4}|submission id|sub[_-]?[0-9a-f]{6}/i.test(`${title} ${headText}`);
  note(OUTFILE, `[${project}] TITLE h1 "${title}"; header text "${headText.slice(0, 140)}"; contains a submission id ${hasId}`);
  expect(hasId).toBe(false);
  // Send disabled before sign-off
  const sendDisabled = await page.locator('[data-send]').isDisabled();
  const chip = await page.locator('[data-send-card]').getByText('Sign off first').count();
  note(OUTFILE, `[${project}] SEND card before sign-off: Send button disabled ${sendDisabled}, "Sign off first" chip count ${chip}`);
  expect(sendDisabled).toBe(true);
  if (project === 'desktop') {
    // Back label below 480
    for (const width of [320, 390, 479, 480, 481, 768]) {
      await page.setViewportSize({ width, height: 900 });
      await settle(page);
      const back = await page.evaluate(() => {
        const full = document.querySelector('.back-full') as HTMLElement;
        const short = document.querySelector('.back-short') as HTMLElement;
        const link = document.querySelector('.back-link') as HTMLElement;
        const vis = (e: HTMLElement) => getComputedStyle(e).display !== 'none' && e.getBoundingClientRect().width > 0;
        return { full: vis(full), short: vis(short), name: link.getAttribute('aria-label'), text: link.innerText.trim() };
      });
      note(OUTFILE, `[desktop] BACK at ${width}px: visible text "${back.text}" (long ${back.full}, short ${back.short}), accessible name "${back.name}"`);
      expect(back.short).toBe(width < 480);
      expect(back.full).toBe(width >= 480);
    }
    // Company line hides rather than wraps
    const problems: string[] = [];
    const states = new Set<string>();
    for (let width = 320; width <= 1440; width += 20) {
      await page.setViewportSize({ width, height: 900 });
      await page.waitForTimeout(80);
      const c = await page.evaluate(() => {
        const e = document.querySelector('.review-company') as HTMLElement | null;
        if (e === null) return null;
        const r = e.getBoundingClientRect();
        const s = getComputedStyle(e);
        return { fits: e.getAttribute('data-fits'), h: r.height, vis: s.visibility, lh: parseFloat(s.lineHeight) || parseFloat(s.fontSize) * 1.4, ws: s.whiteSpace, sw: e.scrollWidth, cw: e.clientWidth };
      });
      if (c === null) {
        problems.push(`${width}: no company element`);
        continue;
      }
      const shown = c.fits !== 'false';
      states.add(shown ? 'shown' : 'hidden');
      const oneLine = c.h <= c.lh * 1.3 + 1;
      const ok = shown ? oneLine && c.sw <= c.cw + 1 : c.h <= 1 && c.vis === 'hidden';
      if (!ok) problems.push(`${width}: fits=${c.fits} h=${c.h} vis=${c.vis} sw=${c.sw} cw=${c.cw}`);
    }
    note(OUTFILE, `[desktop] COMPANY line 320-1440 step 20: states seen ${[...states].join('/')}; shown = one line without clipping, hidden = zero height; problems ${JSON.stringify(problems)}`);
    expect(problems).toEqual([]);
    // One column at 1440
    await page.setViewportSize({ width: 1440, height: 900 });
    await settle(page);
    const col = await page.evaluate(() => {
      const rect = (s: string) => { const e = document.querySelector(s); if (!e) return null; const r = e.getBoundingClientRect(); return { left: r.left, right: r.right, top: r.top, bottom: r.bottom }; };
      return { check: rect('.check-card'), preview: rect('[data-paper="review"]'), send: rect('[data-send-card]'), scroll: document.documentElement.scrollWidth - document.documentElement.clientWidth };
    });
    const parts = [col.check, col.preview, col.send];
    const sameLeft = parts.every((p) => p !== null && Math.abs((p?.left ?? 0) - (parts[0]?.left ?? 0)) < 1);
    const stacked = (col.check?.bottom ?? 0) <= (col.preview?.top ?? 0) + 1 && (col.preview?.bottom ?? 0) <= (col.send?.top ?? 0) + 1;
    note(OUTFILE, `[desktop] ONE COLUMN at 1440: check card, preview and Send card share the left edge ${sameLeft} (lefts ${parts.map((p) => p?.left.toFixed(1)).join('/')}), stacked top to bottom ${stacked}, page scroll ${col.scroll}`);
    expect(sameLeft && stacked).toBe(true);
    // Signature geometry in the DOM (before sign-off): fractions of the container content width
    const geo = await page.evaluate(() => {
      const box = document.querySelector('.review-signatures') as HTMLElement;
      const bs = getComputedStyle(box);
      const x0 = box.getBoundingClientRect().left + parseFloat(bs.paddingLeft);
      const w = box.clientWidth - parseFloat(bs.paddingLeft) - parseFloat(bs.paddingRight);
      const lines = (name: string) => { const e = document.querySelector(`[data-sig-line="${name}"]`) as HTMLElement; const r = e.getBoundingClientRect(); return { left: (r.left - x0) / w, width: r.width / w, top: r.top }; };
      const caps = [...document.querySelectorAll('.rsig-cap')].map((e) => { const r = e.getBoundingClientRect(); return { text: e.textContent, center: ((r.left + r.right) / 2 - x0) / w }; });
      return { w, employee: lines('employee'), date: lines('employee-date'), manager: lines('manager'), mdate: lines('manager-date'), caps };
    });
    note(OUTFILE, `[desktop] SIG geometry (DOM, fractions of the content width ${geo.w.toFixed(1)}px): employee line left ${(geo.employee.left * 100).toFixed(3)}% width ${(geo.employee.width * 100).toFixed(3)}%; date line left ${(geo.date.left * 100).toFixed(3)}% width ${(geo.date.width * 100).toFixed(3)}%; manager line left ${(geo.manager.left * 100).toFixed(3)}% width ${(geo.manager.width * 100).toFixed(3)}%; manager date left ${(geo.mdate.left * 100).toFixed(3)}% width ${(geo.mdate.width * 100).toFixed(3)}%; caption centres ${geo.caps.map((c) => `${c.text}@${(c.center * 100).toFixed(3)}%`).join(', ')}; row tops employee ${geo.employee.top.toFixed(1)}/${geo.date.top.toFixed(1)}, manager ${geo.manager.top.toFixed(1)}/${geo.mdate.top.toFixed(1)}`);
    if (OUT !== '') writeFileSync(join(OUT, 'sig-dom.txt'), JSON.stringify(geo) + '\n');
  }
});

const header = (eml: string, name: string): string => {
  const match = new RegExp(`^${name}: (.*(?:\\r?\\n[ \\t].*)*)`, 'im').exec(eml);
  return (match?.[1] ?? '').replace(/\r?\n[ \t]+/g, ' ').trim();
};

  const [download] = await Promise.all([page.waitForEvent('download'), exportRegion.getByRole('button', { name: 'Export settings' }).click()]);
  const parsed = JSON.parse(readFileSync(await download.path(), 'utf8')) as { data: { submission: Record<string, unknown> } };
  parsed.data.submission.show_time_on_pdf = true;
  await page.getByLabel('Import settings from a file (.json)').setInputFiles({ name: 'timesheet-settings-synthetic.json', mimeType: 'application/json', buffer: Buffer.from(JSON.stringify(parsed)) });
  const importDialog = page.getByRole('dialog', { name: 'Import settings' });
  await expect(importDialog).toBeVisible();

  await importDialog.getByRole('button', { name: 'Cancel' }).click();
  void admin;
});
