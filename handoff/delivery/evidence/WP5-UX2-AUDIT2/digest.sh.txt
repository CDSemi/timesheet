# Usage: bash digest.sh <tag>   (tag = before|after). Three digest forms of a643794.
. <task>/env.sh
TAG=$1
EX=${2:-$T/export}
OUTF=$T/out/01-digest-$TAG.txt
: > $OUTF
echo "date_utc $(date -u +%Y-%m-%dT%H:%M:%SZ)" >> $OUTF
echo "node $(node --version)" >> $OUTF
cd $EX || exit 2
find . -type f -not -path './handoff/*' -not -path './node_modules/*' -not -path './dist/*' -not -path './.git/*' -not -path './test-results/*' -not -path './playwright-report/*' | sed 's|^\./||' | LC_ALL=C sort > $T/out/ex-files-$TAG.txt
: > $T/out/ex-hashes-$TAG.txt
while IFS= read -r f; do
  h=$(git hash-object --no-filters -- "$f")
  printf '%s %s\n' "$f" "$h" >> $T/out/ex-hashes-$TAG.txt
done < $T/out/ex-files-$TAG.txt
LC_ALL=C sort $T/out/ex-hashes-$TAG.txt > $T/out/ex-sorted-$TAG.txt
echo "form 1 clean export (git archive of $SHA; git hash-object --no-filters per file): $(sha256sum $T/out/ex-sorted-$TAG.txt | cut -d' ' -f1) files $(wc -l < $T/out/ex-sorted-$TAG.txt)" >> $OUTF
cd $REPO
git ls-tree -r --format='%(path) %(objectname)' $SHA | grep -v '^handoff/' | LC_ALL=C sort > $T/out/lstree-sorted-$TAG.txt
echo "form 2 git ls-tree $SHA: $(sha256sum $T/out/lstree-sorted-$TAG.txt | cut -d' ' -f1) files $(wc -l < $T/out/lstree-sorted-$TAG.txt)" >> $OUTF
if cmp -s $T/out/ex-sorted-$TAG.txt $T/out/lstree-sorted-$TAG.txt; then echo "export list byte-identical to ls-tree list: yes" >> $OUTF; else echo "export list byte-identical to ls-tree list: NO" >> $OUTF; fi
echo "form 3 npm run digest in the working repository (HEAD $(git rev-parse HEAD), origin/main $(git rev-parse origin/main)):" >> $OUTF
node.exe "$NPM_CLI" run digest >> $OUTF 2>&1
echo "non-handoff status lines: $(git status --short -- . ':!handoff' | wc -l)" >> $OUTF
git status --short -- . ':!handoff' >> $OUTF
