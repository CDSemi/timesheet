"""Summarises the AUDIT2 mutation logs out/72-<label>.txt. Usage: mutsum.py <out dir> <label>..."""
import re
import sys
from pathlib import Path

sys.stdout.reconfigure(encoding="utf-8")
out = Path(sys.argv[1])
for label in sys.argv[2:]:
    text = (out / f"72-{label}.txt").read_text(encoding="utf-8", errors="replace")
    applied = [line for line in text.splitlines() if line.endswith(": applied")]
    exits = re.findall(r"^(?:exit|vitest exit)=(\d+)", text, re.MULTILINE)
    summary = [line.strip() for line in text.splitlines() if re.match(r"^\s+\d+ (passed|failed|skipped|flaky)", line) or re.match(r"^\s+Tests\s+", line)]
    failing = sorted({re.sub(r"\s+\(\d.*$", "", line.strip()) for line in text.splitlines() if re.match(r"^\s+(x|✘)\s+\d+\s+\[", line) or line.strip().startswith("×")})
    restored = [line for line in text.splitlines() if line.startswith("restored ")]
    identical = all("identical" in line for line in restored) and len(restored) == 5
    rebuild = re.findall(r"rebuild after restore exit=(\d+)", text)
    print(f"{label}: applied={len(applied)} exit={','.join(exits)} {' / '.join(summary)} restored_identical={identical} rebuild={','.join(rebuild) or '-'}")
    for line in failing[:12]:
        print(f"    failed: {line[:220]}")
