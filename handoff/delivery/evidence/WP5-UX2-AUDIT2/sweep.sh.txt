# Sequential simulated-clock runs (one Playwright run at a time, each with its own output folder). Log per run: out/6x-<label>.txt
. <task>/env.sh
P=$T/out/60-sweep-progress.txt
run() {
  L=$1; TARGET=$2; EXD=$3; shift 3
  echo "start $L $(date -u +%Y-%m-%dT%H:%M:%SZ)" >> $P
  bash $T/run-shift.sh $L $TARGET $EXD "$@" > $T/out/$L.txt 2>&1
  echo "end $L $(date -u +%Y-%m-%dT%H:%M:%SZ) $(grep -E '^exit=' $T/out/$L.txt | tr '\n' ' ')" >> $P
}
: > $P
