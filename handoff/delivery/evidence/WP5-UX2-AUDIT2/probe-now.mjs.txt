// Prints the (possibly shifted) clock of this process and the America/Los_Angeles wall time.
const now = new Date();
const la = new Intl.DateTimeFormat('en-CA', {
  timeZone: 'America/Los_Angeles', year: 'numeric', month: '2-digit', day: '2-digit', hour: '2-digit', minute: '2-digit', second: '2-digit', hourCycle: 'h23', weekday: 'short',
}).format(now);
console.log(`process clock ${now.toISOString()} (Date.now ${Date.now()}); America/Los_Angeles ${la}; shift ms ${process.env.AB8_SHIFT_MS ?? 'none'}; instanceof ${now instanceof Date}`);
