"""Per-project ok / skipped / failed counts of a Playwright list-reporter log. Usage: counts.py <log>..."""
import re
import sys

sys.stdout.reconfigure(encoding="utf-8")
LINE = re.compile(r"^\s+(ok|-|x|✘)\s+\d+\s+\[(\w+)\]")
for name in sys.argv[1:]:
    counts: dict[tuple[str, str], int] = {}
    text = open(name, encoding="utf-8", errors="replace").read()
    for line in text.splitlines():
        match = LINE.match(line)
        if match:
            key = (match.group(2), {"ok": "ok", "-": "skipped"}.get(match.group(1), "failed"))
            counts[key] = counts.get(key, 0) + 1
    summary = " ".join(f"{project}:{kind}={count}" for (project, kind), count in sorted(counts.items()))
    tail = [line.strip() for line in text.splitlines() if re.match(r"^\s+\d+ (passed|failed|skipped|flaky|did not run)", line) or line.startswith("exit=")]
    print(f"{name.split('/')[-1]}: {summary} | {' / '.join(tail)}")
