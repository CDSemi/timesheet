"""Expected contrasts of the inactive Today from the tokens (WCAG relative luminance). Usage: ringcalc.py"""


def channel(value: float) -> float:
    c = value / 255
    return c / 12.92 if c <= 0.04045 else ((c + 0.055) / 1.055) ** 2.4


def luminance(rgb: tuple[float, float, float]) -> float:
    r, g, b = (channel(v) for v in rgb)
    return 0.2126 * r + 0.7152 * g + 0.0722 * b


def contrast(a, b) -> float:
    la, lb = luminance(a), luminance(b)
    hi, lo = max(la, lb), min(la, lb)
    return (hi + 0.05) / (lo + 0.05)


def hexrgb(text: str) -> tuple[int, int, int]:
    return (int(text[1:3], 16), int(text[3:5], 16), int(text[5:7], 16))


def over(top, alpha, bottom):
    return tuple(t * alpha + b * (1 - alpha) for t, b in zip(top, bottom))


THEMES = {
    "light": {"card": "#ffffff", "muted": "#5b6675", "line": "#d9dee5", "accent": "#1f5fbf", "accent_line": ((31, 95, 191), 0.45)},
    "dark": {"card": "#1a2029", "muted": "#9aa5b4", "line": "#2c3440", "accent": "#6ea8ff", "accent_line": ((110, 168, 255), 0.5)},
}
for name, t in THEMES.items():
    card = hexrgb(t["card"])
    accent = hexrgb(t["accent"])
    muted = hexrgb(t["muted"])
    line = hexrgb(t["line"])
    accent_line = over(t["accent_line"][0], t["accent_line"][1], card)
    print(f"{name}: ring (accent) on card {contrast(accent, card):.2f}:1 (full strength, opacity 1)")
    print(f"{name}: ring at opacity 0.6 on card {contrast(over(accent, 0.6, card), card):.2f}:1 (the old rule, for reference)")
    print(f"{name}: inactive text (muted) on card {contrast(muted, card):.2f}:1; active text (accent) on card {contrast(accent, card):.2f}:1")
    print(f"{name}: inactive border (line) on card {contrast(line, card):.2f}:1; active border (accent-line over card) {contrast(accent_line, card):.2f}:1")
    print(f"{name}: active text vs inactive text {contrast(accent, muted):.2f}:1; active border vs inactive border {contrast(accent_line, line):.2f}:1")
