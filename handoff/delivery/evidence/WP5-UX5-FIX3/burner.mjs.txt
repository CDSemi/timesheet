// CPU burner for load reproduction: N busy threads for S seconds, then exits by itself. Usage: node burner.mjs <threads> <seconds>
import { isMainThread, Worker, workerData } from 'node:worker_threads';

if (isMainThread) {
  const threads = Number(process.argv[2] ?? '8');
  const seconds = Number(process.argv[3] ?? '60');
  const until = Date.now() + seconds * 1000;
  for (let i = 0; i < threads; i += 1) new Worker(new URL(import.meta.url), { workerData: { until } });
  console.log(`burner: ${threads} threads for ${seconds}s`);
} else {
  let x = 0;
  while (Date.now() < workerData.until) x += Math.sqrt(x + 1);
}
