async function call(method, path, { cookie, body, raw, type, withOrigin = true, headers = {} } = {}) {
  const h = { ...headers };
  if (withOrigin) h.origin = origin;
  if (cookie) h.cookie = cookie;
  let payload;
  if (raw !== undefined) {
    payload = raw;
    if (type !== undefined) h['content-type'] = type;
  } else if (body !== undefined) {
    h['content-type'] = 'application/json';
    payload = JSON.stringify(body);
  }
  const t0 = performance.now();
  const init = { method, headers: h, body: payload };
  if (payload instanceof ReadableStream) init.duplex = 'half';
  let res;
  try {
    res = await fetch(`${base}${path}`, init);
  } catch (error) {
    // The server may answer 413 and close the connection while the client is still sending.
    return { status: 0, json: null, text: String(error.cause?.message ?? error.message), bytes: Buffer.alloc(0), headers: new Headers(), ms: Math.round(performance.now() - t0) };
  }
  const buf = Buffer.from(await res.arrayBuffer());
  const ms = Math.round(performance.now() - t0);
  let json = null;
  try {
    json = JSON.parse(buf.toString('utf8'));
  } catch {}
  return { status: res.status, json, text: buf.toString('utf8'), bytes: buf, headers: res.headers, ms };
}
