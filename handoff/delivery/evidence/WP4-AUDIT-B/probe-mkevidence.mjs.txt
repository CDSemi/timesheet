// Writes masked LF evidence copies for WP4-AUDIT-B (logs and probe sources). Run with Node 24 from the task folder.
import { readFileSync, writeFileSync, mkdirSync } from 'node:fs';
import { join } from 'node:path';

const task = '<task>';
const out = 'D:\\Dropbox\\Work.CDSemi\\timesheet\\handoff\\delivery\\evidence\\WP4-AUDIT-B';
mkdirSync(out, { recursive: true });

function mask(text) {
  let t = text.replace(/\x1b\[[0-9;]*m/g, '').replace(/\r\n?/g, '\n');
  for (const variant of [task, task.replaceAll('\\', '/'), task.replaceAll('\\', '\\\\'), '<task>', '<task>']) {
    t = t.split(variant).join('<task>');
  }
  t = t.replace(/C:(\\\\|\\|\/)Users(\\\\|\\|\/)[^\\/\s"']+/gi, 'C:$1Users$2<user>').replace(/\/c\/Users\/[^/\s"']+/gi, '/c/Users/<user>');
  t = t.replace(/<user>/gi, '<user>');
  t = t.replace(/[A-Za-z0-9._%+-]+@(?![A-Za-z0-9.-]*example\.invalid)[A-Za-z0-9.-]+\.[A-Za-z]{2,}/g, '<email>');
  t = t.split('\n').map((line) => line.replace(/[ \t]+$/, '')).join('\n').replace(/\n+$/, '');
  return `${t}\n`;
}
