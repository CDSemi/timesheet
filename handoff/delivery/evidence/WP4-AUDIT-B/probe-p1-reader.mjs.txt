const MiB = 1024 * 1024;
const results = [];

function run(label, build, { preview = true, expect } = {}) {
  let bytes;
  try {
    bytes = build();
  } catch (error) {
    results.push({ label, outcome: `BUILD ERROR ${error.message}` });
    return;
  }
  const t0 = performance.now();
  let outcome;
  let extra = '';
  try {
    const out = preview ? previewWorkbook(bytes) : readWorkbook(bytes);
    outcome = 'accepted';
    if (preview) {
      const codes = [...new Set(out.findings.map((f) => f.code))].sort();
      extra = `findings=${codes.join(',')}`;
    } else {
      extra = `sheets=${out.sheets.length} notes=${out.notes.map((n) => n.code).join(',')}`;
    }
  } catch (error) {
    outcome = error.name === 'WorkbookRejectedError' ? `rejected:${error.code}` : `THROWN ${error.name}: ${String(error.message).slice(0, 80)}`;
  }
  const ms = Math.round(performance.now() - t0);
  const verdict = expect === undefined ? '' : outcome.startsWith(expect) ? ' [as expected]' : ` [EXPECTED ${expect}]`;
  results.push({ label, bytes: bytes.length, ms, outcome: outcome + verdict, extra });
}
