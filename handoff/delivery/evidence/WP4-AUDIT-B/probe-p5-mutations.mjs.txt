const results = [];
for (const mutation of MUTATIONS.filter((m) => process.argv[2] === undefined || m.id === process.argv[2])) {
  const originals = new Map();
  let applied = true;
  for (const [file, from, to] of mutation.edits) {
    const text = originals.get(file) ?? readFileSync(file, 'utf8');
    if (!originals.has(file)) originals.set(file, text);
    const current = readFileSync(file, 'utf8');
    const occurrences = current.split(from).length - 1;
    if (occurrences < 1) {
      applied = false;
      break;
    }
    writeFileSync(file, current.replace(from, to));
  }
  let outcome = 'NOT APPLIED (pattern missing)';
  if (applied) {
    const run = spawnSync(process.execPath, ['node_modules/vitest/vitest.mjs', 'run', ...mutation.tests], { encoding: 'utf8', env: process.env });
    const output = `${run.stdout}${run.stderr}`.replace(/\x1b\[[0-9;]*m/g, '');
    const summary = (output.match(/Tests\s+[^\n]+/) ?? ['?'])[0].trim();
    const failed = [...output.matchAll(/^\s*(?:×|✗|FAIL)\s+(.+)$/gm)].map((m) => m[1].trim()).filter((t) => !t.startsWith('tests/')).slice(0, 3);
    outcome = `${run.status === 0 ? 'SURVIVED (tests pass)' : 'KILLED (tests fail)'} | ${summary}${failed.length ? ' | e.g. ' + failed.join(' ; ').slice(0, 300) : ''}`;
  }
  for (const [file, text] of originals) writeFileSync(file, text);
  const restored = [...originals].every(([file, text]) => sha(readFileSync(file, 'utf8')) === sha(text));
  results.push(`${mutation.id} ${mutation.what}: ${outcome} | restored=${restored}`);
}
console.log(results.join('\n'));
