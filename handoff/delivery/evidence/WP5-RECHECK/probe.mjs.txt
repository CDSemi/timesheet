function cli(env, args, label) {
  const r = spawnSync(process.execPath, [CLI, ...args], { env, cwd: X, encoding: 'utf8', stdio: ['ignore', 'pipe', 'pipe'], timeout: 120_000 });
  const out = `${r.stdout ?? ''}${r.stderr ?? ''}`;

  return { status: r.status, stdout: r.stdout ?? '', stderr: r.stderr ?? '', out };
}

class Server {
  constructor(name, env) {
    this.name = name;
    this.env = env;
    this.output = '';
    this.child = null;
    this.exit = null;
  }
  start(waitListening = true) {
    this.output = '';
    this.exit = null;
    this.child = spawn(process.execPath, [SERVER], { env: this.env, cwd: X, stdio: ['ignore', 'pipe', 'pipe'] });
    this.child.stdout.on('data', (d) => (this.output += d));
    this.child.stderr.on('data', (d) => (this.output += d));
    this.exited = new Promise((resolve) => this.child.on('exit', (code) => resolve((this.exit = code))));
    if (!waitListening) return this.exited;
    return (async () => {
      for (let i = 0; i < 100; i += 1) {
        if (this.output.includes('listening')) return true;
        if (this.exit !== null) throw new Error(`server ${this.name} exited ${this.exit}: ${mask(this.output)}`);
        await sleep(200);
      }
      throw new Error(`server ${this.name} did not start`);
    })();
  }
  async stop() {
    if (this.child === null || this.exit !== null) return this.exit;
    this.child.kill();
    const code = await this.exited;
    log(`INFO  server ${this.name} stopped through its own handle (exit ${code})`);
    return code;
  }
}

class Client {
  constructor(who) {
    this.who = who;
    this.cookie = '';
  }
  async raw(method, path, body, headers = {}) {
    const h = { ...headers, origin: ORIGIN, connection: 'close' };
    if (this.cookie !== '') h.cookie = this.cookie;
    let payload;
    if (body instanceof Uint8Array) payload = body;
    else if (body !== undefined) {
      h['content-type'] = 'application/json';
      payload = JSON.stringify(body);
    }
    const res = await fetch(`${ORIGIN}${path}`, { method, headers: h, body: payload });
    const buffer = Buffer.from(await res.arrayBuffer());
    let json = null;
    try {
      json = JSON.parse(buffer.toString('utf8'));
    } catch {
      json = null;
    }
    return { status: res.status, headers: res.headers, json, buffer, text: buffer.toString('utf8') };
  }
  async call(method, path, body, headers) {
    const r = await this.raw(method, path, body, headers);
    log(`HTTP  ${this.who} ${method} ${path} -> ${r.status}${r.status >= 400 ? ` ${r.text.slice(0, 200)}` : ''}`);
    return r;
  }
  async login(email, password) {
    const r = await this.raw('POST', '/api/auth/login', { email, password });
    this.cookie = (r.headers.get('set-cookie') ?? '').split(';')[0] ?? '';
    secrets.add(this.cookie.split('=')[1]);
    log(`HTTP  ${this.who} login -> ${r.status}`);
    return r;
  }
}

function captures() {
  const root = join(dataDir, 'mail-capture');
  if (!existsSync(root)) return [];
  return readdirSync(root, { withFileTypes: true })
    .filter((e) => e.isDirectory() && !e.name.startsWith('.tmp-'))
    .map((e) => {
      const folder = join(root, e.name);
      const files = readdirSync(folder).sort();
      const eml = files.includes('message.eml') ? readFileSync(join(folder, 'message.eml'), 'utf8') : '';
      const pdf = files.includes('attachment.pdf') ? readFileSync(join(folder, 'attachment.pdf')) : null;
      const metadata = files.includes('metadata.json') ? JSON.parse(readFileSync(join(folder, 'metadata.json'), 'utf8')) : null;
      return { id: e.name, files, eml, pdf, metadata };
    });
}
function decodePart(head, body) {
  if (/Content-Transfer-Encoding:\s*base64/i.test(head)) return Buffer.from(body.replace(/\s+/g, ''), 'base64').toString('utf8');
  if (/Content-Transfer-Encoding:\s*quoted-printable/i.test(head)) {
    const soft = body.replace(/=\r?\n/g, '');
    const bytes = [];
    for (let i = 0; i < soft.length; i += 1) {
      if (soft[i] === '=' && /^[0-9A-F]{2}$/i.test(soft.slice(i + 1, i + 3))) {
        bytes.push(parseInt(soft.slice(i + 1, i + 3), 16));
        i += 2;
      } else bytes.push(...Buffer.from(soft[i], 'utf8'));
    }
    return Buffer.from(bytes).toString('utf8');
  }
  return body;
}
/** Headers of the message and the decoded text of every text/plain and text/html part. */
function readMessage(eml) {
  const top = eml.split(/\r?\n\r?\n/)[0] ?? '';
  const header = (name) => top.match(new RegExp(`^${name}: (.*)$`, 'mi'))?.[1]?.trim() ?? null;
  const texts = [];
  for (const part of eml.split(/\r?\n--/)) {
    const [head, ...rest] = part.split(/\r?\n\r?\n/);
    if (/Content-Type:\s*text\/(plain|html)/i.test(head ?? '')) texts.push(decodePart(head, rest.join('\n\n')));
  }
  if (texts.length === 0) {
    const [head, ...rest] = eml.split(/\r?\n\r?\n/);
    texts.push(decodePart(head, rest.join('\n\n')));
  }
  return { from: header('From'), to: header('To'), cc: header('Cc'), bcc: header('Bcc'), subject: header('Subject'), autoSubmitted: header('Auto-Submitted'), text: texts.join('\n----\n') };
}

  await lp.goto(`${ORIGIN}/#/history`);
  await lp.reload();
  await lp.locator('[data-attempt-badge="accepted"]').first().waitFor({ timeout: 15_000 });
  const badge = (await lp.locator('[data-attempt-badge="accepted"]').first().innerText()).trim();
  const historyText = await lp.locator('main').innerText();
  check('R-A2-1 screen', 'the owner\'s history shows the captured attempt as "Accepted by the mail server" and never shows "captured" or a "capture-" id', badge === 'Accepted by the mail server' && !/captur/i.test(historyText), `badge "${badge}"`);

  // ---------------------------------------------------------------- the documented console call (s13 step 5) and R-F5
  const set = await consoleRun(page, fill(`'${ACTIVATION}'`, 'Synthetic recheck activation'));
  log(`INFO  console set -> ${JSON.stringify({ active_from: set?.automation?.active_from })}`);
  check('S13-5 set', 'the verbatim console call records the activation instant', set?.automation?.active_from === ACTIVATION);
  const storageState = await adminContext.storageState();
  const shown = {};
  for (const zone of ['America/Los_Angeles', 'Asia/Tokyo', 'America/New_York']) {
    const ctx = await browser.newContext({ timezoneId: zone, storageState });
    const zp = await ctx.newPage();
    const s = await screen(zp, `activation set, zone ${zone}`);
    shown[zone] = { text: s.facts.activation, expected: zoneText(ACTIVATION, zone), browserZone: s.zone };
    await ctx.close();
  }
  log(`INFO  activation instant ${ACTIVATION} as shown: ${JSON.stringify(shown)}`);
  check('R-F5', 'the status shows the instant in the browser\'s zone as "YYYY-MM-DD HH:MM" without a zone label', Object.entries(shown).every(([zone, v]) => v.browserZone === zone && v.text === v.expected && /^\d{4}-\d{2}-\d{2} \d{2}:\d{2}$/.test(v.text)) && shown['Asia/Tokyo'].text !== shown['America/New_York'].text);
  const opsSet = await admin.call('GET', '/api/admin/operations');
  check('S13-5 json', 'the operations JSON carries the recorded instant', opsSet.json?.operations?.activation?.active_from === ACTIVATION, JSON.stringify(opsSet.json?.operations?.activation));

  // ---------------------------------------------------------------- the first reminder after activation (s13 step 7)
  await server.stop();
  const capsBefore = new Set(captures().map((c) => c.id));
  for (const instant of ['2027-01-01T00:01:00Z', '2027-01-01T00:03:00Z']) {
    const r = cli(devCliEnv, ['run-jobs', '--once', '--now', instant], `run-jobs --once --now ${instant} (NODE_ENV=development, same database)`);
    log(`INFO  run-jobs ${instant}: ${r.stdout.trim()}`);
  }
  const reminders = captures().filter((c) => !capsBefore.has(c.id)).map((c) => ({ id: c.id, files: c.files, envelopeTo: c.metadata?.envelope?.to, ...readMessage(c.eml) }));
  const reminderA = reminders.find((m) => (m.envelopeTo ?? []).includes(alice.email));
  const links = reminderA ? urlsIn(reminderA.text) : [];
  log(`INFO  new captures after activation: ${JSON.stringify(reminders.map((m) => ({ to: m.envelopeTo, subject: m.subject, autoSubmitted: m.autoSubmitted, pdf: m.files.includes('attachment.pdf'), links: urlsIn(m.text) })))}`);
  save('reminder-alice.txt', reminderA ? `Subject: ${reminderA.subject}\nTo: ${reminderA.to}\nAuto-Submitted: ${reminderA.autoSubmitted}\n\n${reminderA.text}` : 'none');
  const occurrences = db.prepare('SELECT kind, occurrence_key, disposition FROM reminder_occurrences ORDER BY kind, occurrence_key').all();
  log(`INFO  reminder occurrences (read-only): ${JSON.stringify(occurrences)}`);
  const expectedLink = reviewLink(PUBLIC, '2027-01-01');
  const runbookForm = /https:\/\/<nas-host>\/#\/review\/<payroll-date>/.test(runbook.split('\n').find((l) => l.includes('The link: once the activation instant is set')) ?? '');
  check('A2-02 step 7', 'docs/11 s13 step 7 has the first-reminder link check in the form https://<nas-host>/#/review/<payroll-date>', runbookForm);
  check('A2-02 link', 'the first reminder to the employee carries exactly PUBLIC_BASE_URL + /#/review/<payroll-date> (reviewLink), the runbook form', reminderA !== undefined && links.length === 1 && links[0] === expectedLink && /^https:\/\/[^/]+\/#\/review\/\d{4}-\d{2}-\d{2}$/.test(links[0]) && reminderA.autoSubmitted === 'auto-generated', `links ${JSON.stringify(links)}; expected ${expectedLink}`);

  await server.start();
  const lc2 = await browser.newContext({ timezoneId: ZONE });
  const lp2 = await lc2.newPage();
  const loopbackLink = (links[0] ?? expectedLink).replace(PUBLIC, ORIGIN);
  await lp2.goto(loopbackLink);
  await lp2.getByLabel('Email').waitFor({ timeout: 15_000 });
  const h1 = await lp2.evaluate('window.location.hash');
  await lp2.getByLabel('Email').fill(alice.email);
  await lp2.getByLabel('Password').fill(credentials.alice);
  await lp2.getByRole('button', { name: 'Sign in' }).click();
  await lp2.getByRole('heading', { name: 'Review and sign off' }).waitFor({ timeout: 15_000 });
  const h2 = await lp2.evaluate('window.location.hash');
  const rt2 = (await lp2.locator('main').innerText()).replace(/\s+/g, ' ');
  check('A2-02 link opens', 'the reminder link (host replaced by the loopback origin) asks for sign-in, then opens the review of that payroll date', h1 === '#/review/2027-01-01' && h2 === '#/review/2027-01-01' && rt2.includes('payroll date 2027-01-01'), `${h1} -> ${h2}; ${rt2.slice(0, 160)}`);
  await lc2.close();

  // ---------------------------------------------------------------- clear (s14 step 1)
  const page2 = await adminContext.newPage();
  await page2.goto(`${ORIGIN}/#/admin`);
  await page2.locator('[data-fact="activation"]').waitFor({ timeout: 15_000 });
  const clear = await consoleRun(page2, fill('null', 'Synthetic recheck deactivation'));
  const s2 = await screen(page2, 'after clear');
  check('S14-1', 'the same call with active_from: null clears the instant; the screen shows "Not activated" and still "Capture only (nothing leaves the server)"', clear?.automation?.active_from === null && s2.facts.activation === 'Not activated' && s2.facts.mode === 'Capture only (nothing leaves the server)', JSON.stringify(s2.facts));
  log(`INFO  jobs by kind and state after the 2027 run-jobs passes (time-shifted rows): ${JSON.stringify(jobRows())}`);

  // ---------------------------------------------------------------- pre-activation backup (s13 step 3) in capture mode
  const backup = cli(prodEnv, ['backup', '--to', backupDir], 'backup --to <run>/backups (capture mode, server running)');
  const backupName = readdirSync(backupDir).sort().at(-1);
  log(`INFO  backup: ${backup.stdout.trim()}; ls -1: ${JSON.stringify(readdirSync(backupDir))}; manifest sha256 ${sha256(readFileSync(join(backupDir, backupName, 'manifest.json')))}`);
  check('S13-3', 'the backup prints "outcome":"succeeded" and the newest folder holds manifest.json', backup.status === 0 && backup.stdout.includes('"outcome":"succeeded"') && existsSync(join(backupDir, backupName, 'manifest.json')));
  await server.stop();
  db.close();

  // ---------------------------------------------------------------- the refusal sentence (s13 step 4), SMTP without the flag
  const smtpUser = secret();
  const smtpPassword = secret();
  const smtpEnv = { ...prodEnv, OUTBOUND_MODE: 'smtp', SMTP_HOST: 'smtp.example.invalid', SMTP_PORT: '587', SMTP_SECURITY: 'starttls', SMTP_USER: smtpUser, SMTP_PASSWORD: smtpPassword };
  log(`INFO  refusal env: ${envKeys(smtpEnv)} (PRODUCTION_SENDING_ENABLED absent)`);
  const refusal = new Server('smtp-no-flag', smtpEnv);
  const serverExit = await Promise.race([refusal.start(false), sleep(20_000).then(() => 'timeout')]);
  if (serverExit === 'timeout') await refusal.stop();
  const serverOut = refusal.output;
  const named = (out) => out.includes(FLAG_MESSAGE);
  const leaks = (out) => out.includes(smtpUser) || out.includes(smtpPassword);
  check('REFUSE server', 'the server refuses to start in SMTP mode without the flag; the message names the flag, never a value; it never listens', serverExit !== 'timeout' && serverExit !== 0 && named(serverOut) && !leaks(serverOut) && !serverOut.includes('listening'), `exit ${serverExit}; first line naming the flag: ${serverOut.split('\n').find((l) => l.includes('PRODUCTION_SENDING_ENABLED'))?.trim()}`);
  const restoreTarget = join(RUN, 'restore-refused');
  const rows = [
    ['migrate', ['migrate']],
    ['bootstrap --new-token', ['bootstrap', '--new-token']],
    ['backup --to <run>/backups', ['backup', '--to', backupDir]],
    ['backup prune --in <run>/backups --dry-run', ['backup', 'prune', '--in', backupDir, '--dry-run']],
    ['restore --from <backup> --to <run>/restore-refused', ['restore', '--from', join(backupDir, backupName), '--to', restoreTarget]],
    ['outbound release', ['outbound', 'release']],
    ['outbound resume', ['outbound', 'resume']],
    ['run-jobs --once --now 2027-01-02T00:00:00Z', ['run-jobs', '--once', '--now', '2027-01-02T00:00:00Z']],
    ['seed', ['seed']],
  ];
  const table = [];
  for (const [label, args] of rows) {
    const r = cli(smtpEnv, args, `${label} (SMTP mode, no flag)`);
    const firstErr = r.stderr.split('\n').find((l) => l.trim() !== '' && !/^\s+at /.test(l))?.trim() ?? '';
    table.push({ command: label, exit: r.status, names_flag: named(r.out), leaks_value: leaks(r.out), stderr_first_line: mask(firstErr).slice(0, 160), stdout: mask(r.stdout.trim()).slice(0, 160) });
  }
  save('refusal-table.json', table);
  for (const row of table) log(`INFO  ${JSON.stringify(row)}`);
  const byCmd = Object.fromEntries(table.map((row) => [row.command.split(' ')[0] + (row.command.includes('prune') ? '-prune' : '') + (row.command.startsWith('outbound') ? `-${row.command.split(' ')[1]}` : ''), row]));
  check('REFUSE backup/restore', 'backup --to and restore refuse with the flag message and print no value; restore leaves no target', byCmd.backup?.exit !== 0 && byCmd.backup?.names_flag && byCmd.restore?.exit !== 0 && byCmd.restore?.names_flag && !existsSync(restoreTarget) && table.every((row) => !row.leaks_value), JSON.stringify({ backup: byCmd.backup?.exit, restore: byCmd.restore?.exit }));
  check('REFUSE others', 'other commands do not refuse for the flag: migrate 0, backup prune --dry-run 0, outbound previews 2, bootstrap/run-jobs/seed refuse for their own reasons', byCmd.migrate?.exit === 0 && !byCmd.migrate?.names_flag && byCmd['backup-prune']?.exit === 0 && !byCmd['backup-prune']?.names_flag && byCmd['outbound-release']?.exit === 2 && byCmd['outbound-resume']?.exit === 2 && !byCmd['outbound-release']?.names_flag && !byCmd.bootstrap?.names_flag && !byCmd['run-jobs']?.names_flag && !byCmd.seed?.names_flag, JSON.stringify(table.map((row) => [row.command, row.exit, row.names_flag])));
  // Control in capture mode: the same restore runs.
  const control = cli(prodEnv, ['restore', '--from', join(backupDir, backupName), '--to', join(RUN, 'restore-capture')], 'restore --from <backup> --to <run>/restore-capture (capture mode, control)');
  check('REFUSE control', 'in capture mode the same restore runs (exit 0)', control.status === 0, control.stdout.trim().slice(0, 160));
} catch (error) {
  failures += 1;
  log(`ERROR ${error instanceof Error ? error.stack : String(error)}`);
} finally {
  if (browser !== null) await browser.close();
  await server.stop();
  log(`INFO  server output (masked): ${server.output.trim().replace(/\r?\n/g, ' | ')}`);
  log(`RESULT ${passes} PASS, ${failures} FAIL`);
  writeFileSync(join(RUN, 'probe-log.txt'), `${lines.join('\n')}\n`);
}
process.exitCode = failures === 0 ? 0 : 1;
