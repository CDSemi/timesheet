// Masks a text file for the evidence folder (LF, no ANSI, task/repo/profile paths and the user name masked, only
// @example.invalid addresses kept). Usage: node mask.mjs <in> <out>
import { readFileSync, writeFileSync } from 'node:fs';

const [input, output] = process.argv.slice(2);
let text = readFileSync(input, 'utf8');
text = text.replace(/\x1b\[[0-9;]*[A-Za-z]/g, '');
text = text.replace(/\r\n/g, '\n').replace(/\r/g, '\n');
const bs = String.fromCharCode(92);
const profile = process.env.USERPROFILE ?? 'C:\\Users\\<user>';
const forms = (win, label) => {
  const parts = win.split('/');
  return [parts.join(bs + bs + bs + bs), parts.join(bs + bs), parts.join(bs), win, '/' + win[0].toLowerCase() + win.slice(2), win.toLowerCase(), parts.join(bs).toLowerCase()].map((f) => [f, label]);
};
const pairs = [
  ...forms('<task>', '<task>'),
  ...forms('<scratch>', '<scratch>'),
  ...forms('<repo>', '<repo>'),
  ...forms(profile.split(bs).join('/'), 'C:/Users/<user>'),
];
for (const [f, label] of pairs) text = text.split(f).join(label);
const user = process.env.USERNAME ?? '';
if (user) text = text.split(user).join('<user>');
text = text.replace(/[A-Za-z0-9._%+-]+@[A-Za-z0-9.-]+\.[A-Za-z]{2,}/g, (match) => (/@(example|timesheet)\.invalid$/i.test(match) ? match : '<email>'));
if (!text.endsWith('\n')) text += '\n';
writeFileSync(output, text);
