// Extracts each request-body builder (export [async] function NAME ... matching brace) from two
// trees and compares the sources with whitespace collapsed; prints the differing lines.
// Usage: node body-builders.mjs <treeA> <treeB> <labelA> <labelB>
import { existsSync, readFileSync } from 'node:fs';
import { join } from 'node:path';

const [a, b, labelA, labelB] = process.argv.slice(2);
const BUILDERS = [
  ['src/client/components/sessionModel.ts', 'buildSessionRequest'],
  ['src/client/components/sessionModel.ts', 'buildClockOutRequest'],
  ['src/client/components/sessionModel.ts', 'buildDayEntryRequest'],
  ['src/client/components/dayModel.ts', 'batchEntries'],
  ['src/client/components/reviewModel.ts', 'modeEndpoint'],
  ['src/client/components/reviewModel.ts', 'buildSubmitBody'],
  ['src/client/components/dayEditorModel.ts', 'quickBreaksRequest'],
  ['src/client/components/labelPickerModel.ts', 'labelEntry'],
  ['src/client/components/labelPickerModel.ts', 'labelPreviewOutcome'],
  ['src/client/components/leaveInputModel.ts', 'parseLeaveInput'],
  ['src/client/components/errors.ts', 'isStaleVersion'],
  ['src/client/api.ts', 'api'],
];

function extract(root, file, name) {
  const path = join(root, file);
  if (!existsSync(path)) return null;
  const text = readFileSync(path, 'utf8').replace(/\r\n/g, '\n');
  const start = text.search(new RegExp(`export (async )?function ${name}\\b`));
  if (start < 0) return null;
  // The body starts at the first "{" after the closing ")" of the parameter list.
  let depth = 0;
  let i = text.indexOf('(', start);
  for (; i < text.length; i += 1) {
    if (text[i] === '(') depth += 1;
    else if (text[i] === ')') {
      depth -= 1;
      if (depth === 0) break;
    }
  }
  let open = text.indexOf('{', i);
  // Skip an object-typed return annotation such as ": { ok: true }" by checking for "=>"-free header.
  const header = text.slice(i, open);
  if (/:\s*$/.test(header)) {
    let d = 0;
    let k = open;
    for (; k < text.length; k += 1) {
      if (text[k] === '{') d += 1;
      else if (text[k] === '}') {
        d -= 1;
        if (d === 0) break;
      }
    }
    open = text.indexOf('{', k + 1);
  }
  depth = 0;
  let end = open;
  for (; end < text.length; end += 1) {
    if (text[end] === '{') depth += 1;
    else if (text[end] === '}') {
      depth -= 1;
      if (depth === 0) break;
    }
  }
  return text.slice(start, end + 1);
}

const norm = (s) => s.replace(/\s+/g, ' ').trim();
for (const [file, name] of BUILDERS) {
  const x = extract(a, file, name);
  const y = extract(b, file, name);
  if (x === null || y === null) {
    console.log(`${name} (${file}): ${labelA} ${x === null ? 'absent' : 'present'}, ${labelB} ${y === null ? 'absent' : 'present'}`);
    continue;
  }
  const same = norm(x) === norm(y);
  console.log(`${name} (${file}): ${same ? 'IDENTICAL' : 'DIFFERENT'} (${x.split('\n').length} -> ${y.split('\n').length} lines)`);
  if (!same) {
    const setX = new Set(x.split('\n').map(norm));
    const setY = new Set(y.split('\n').map(norm));
    for (const l of x.split('\n')) if (!setY.has(norm(l))) console.log(`   - ${l}`);
    for (const l of y.split('\n')) if (!setX.has(norm(l))) console.log(`   + ${l}`);
  }
}
