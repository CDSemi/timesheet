// Prints the steps of one dialogs-*.json probe result: label, writes (call + body), focus after, stored state.
// Usage: node peek-dialogs.mjs <file>
import { readFileSync } from 'node:fs';

const r = JSON.parse(readFileSync(process.argv[2], 'utf8'));
console.log(`build ${r.build} project ${r.project} ${r.days ? JSON.stringify(r.days) : ''}`);
for (const s of r.steps) {
  console.log(`- ${s.label}`);
  console.log(`    writes: ${JSON.stringify(s.writes)}`);
  console.log(`    focus: ${s.focus}; stored: ${JSON.stringify(s.stored)}`);
}
