// WP5-UX-AUDIT-A6 (A5 probe adapted to the new day-button names, selected by data-day) - AC-16, AC-01 and
// A2 risk R5 on bf954c0 in three layouts: a
// view-only grantee has no edit, clock, batch, picker, Open-a-day, review or signature control (also
// in the modal read-only editor); an edit grantee edits through the shared requester (picker, the
// editor, and "Open a day" - on the phone at 320px wide, where the FIX5 tokens apply); a third user
// sees nothing. On the phone the owner bar, the first day row and the tab bar are measured (R5).
// Synthetic data only.
import { expect, newPerson, statusInPage, test } from '../export/tests/e2e/fixtures.ts';
import { dayButtonBf9, dayNameBf9, editorLayout, inspectControl, recordApi, signIn, writeResult } from './common.ts';

/** Any bf954c0 day-button name of a verb: "Edit 09/28 (2026-09-28)" or, on a phone, "Edit Mon 09/28 (2026-09-28)". */
const anyDayName = (verb: 'Edit' | 'View') => new RegExp(`^${verb} (?:(?:Mon|Tue|Wed|Thu|Fri|Sat|Sun) )?\\d{2}/\\d{2} \\(\\d{4}-\\d{2}-\\d{2}\\)$`);

test.use({ locale: 'en-US', timezoneId: 'Asia/Ho_Chi_Minh' });

const LA = 'America/Los_Angeles';

async function firstScreen(page: any) {
  return page.evaluate(`(() => {
    const box = (el) => { if (el === null) return null; const r = el.getBoundingClientRect(); return { top: Math.round(r.top * 10) / 10, bottom: Math.round(r.bottom * 10) / 10, height: Math.round(r.height * 10) / 10 }; };
    const bar = document.querySelector('[data-shared-owner] > :first-child');
    const tabbar = document.querySelector('.shell-tabs');
    return { scrollY: window.scrollY, viewport: { width: window.innerWidth, height: window.innerHeight }, owner_bar: box(bar), owner_bar_text: bar === null ? null : bar.textContent.replace(/\\s+/g, ' ').trim(), first_day_row: box(document.querySelector('[data-day]')), tab_bar: box(tabbar) };
  })()`);
}

test('AC-16 / AC-01 in each layout: view-only, edit grantee through the picker, the editor and Open a day, third user; R5 phone measurements', async ({ page, browser, adminSeed, builtServer }, testInfo) => {
  test.setTimeout(300_000);
  const ownerP = await newPerson(adminSeed, builtServer, { displayName: 'Synthetic Owner A' });
  const grantee = await newPerson(adminSeed, builtServer, { displayName: 'Synthetic Grantee B', signature: 'none' });
  const other = await newPerson(adminSeed, builtServer, { displayName: 'Synthetic Other C', signature: 'none' });
  const [ownerDay = '', editDay = '', fieldsDay = '', openDayDate = ''] = await ownerP.api.displayedPeriodFreeWorkdays();
  expect(openDayDate, 'four free workdays').not.toBe('');
  await ownerP.api.seedCompleteDay(ownerDay);
  const note = `private-synthetic-${Date.now().toString(36)}`;
  await ownerP.api.putDay(ownerDay, { notes: note });
  const ownerSession = (await ownerP.api.dayView(ownerDay)).sessions[0]?.id ?? '';
  await ownerP.api.call('POST', '/api/shares', { grantee_email: grantee.account.email, items: { timesheets: 'view', ot_read: false, pdf_download: false } }, 201);
  const ownerId = ownerP.account.id;
  const result: Record<string, unknown> = { project: testInfo.project.name };
  const count = async (locator: any) => locator.count();
  const visibleCount = async (locator: any) => {
    let n = 0;
    for (const item of await locator.all()) if (await item.isVisible()) n += 1;
    return n;
  };

  const requests = recordApi(page);
  await signIn(page, grantee.account, `#/shared/${ownerId}/timesheet`);
  await expect(page.locator('[data-day]')).toHaveCount(14);
  await expect(page.locator(`[data-range="${ownerSession}"]`)).toHaveCount(1);
  await page.evaluate('window.scrollTo(0, 0)');
  result.view_only_first_screen = await firstScreen(page);
  const identity = {
    sharing_bar_text: (await page.locator('[data-shared-owner] > :first-child').innerText()).replace(/\s+/g, ' ').trim(),
    sharing_bar_visible: await page.locator('[data-shared-owner] > :first-child').isVisible(),
    form_name: (await page.locator('.form-fields').first().innerText()).replace(/\s+/g, ' ').trim(),
  };
  result.view_only_identity = identity;
  expect(identity.sharing_bar_visible, 'the owner bar is visible').toBe(true);
  expect(identity.sharing_bar_text).toContain('Synthetic Owner A');
  expect(identity.sharing_bar_text).toContain('view only');
  expect(identity.form_name).toContain('Synthetic Owner A');
  const viewControls = {
    open_a_day: await count(page.getByLabel('Open a day')),
    open_day_form: await count(page.locator('.open-day')),
    change_several_days: await count(page.getByRole('button', { name: 'Change several days' })),
    clock_buttons: await count(page.getByRole('button', { name: /^Clock (in|out)$/ })),
    clock_panel: await count(page.locator('.clock-panel')),
    label_pickers: await count(page.locator('[data-label-picker]')),
    checkboxes: await count(page.getByRole('checkbox')),
    edit_buttons: await count(page.getByRole('button', { name: anyDayName('Edit') })),
    view_buttons: await count(page.getByRole('button', { name: anyDayName('View') })),
    view_buttons_visible: await visibleCount(page.getByRole('button', { name: anyDayName('View') })),
    day_buttons_by_data_day: await count(page.locator('[data-day] [data-day-button]')),
    signature_strip: await count(page.getByRole('region', { name: 'Signatures' })),
    review_links: await count(page.locator('[data-sheet-review-link], [data-review-link], [data-review-link-disabled]')),
    images: await count(page.locator('img')),
  };
  result.view_only_controls = viewControls;
  expect(viewControls).toMatchObject({ open_a_day: 0, open_day_form: 0, change_several_days: 0, clock_buttons: 0, clock_panel: 0, label_pickers: 0, checkboxes: 0, edit_buttons: 0, view_buttons: 14, day_buttons_by_data_day: 14, signature_strip: 0, review_links: 0, images: 0 });
  await expect(dayButtonBf9(page, ownerDay)).toHaveAccessibleName(dayNameBf9('View', ownerDay));
  await dayButtonBf9(page, ownerDay).click();
  await expect(page.locator(`dialog[data-day-editor="${ownerDay}"]`)).toHaveCount(1);
  const viewer = page.getByRole('dialog', { name: /^Day / });
  await expect(viewer).toBeVisible();
  result.view_only_editor_layout = await editorLayout(page);
  const viewerControls = {
    add_session: await count(viewer.getByRole('button', { name: 'Add session' })),
    edit_or_delete: await count(viewer.getByRole('button', { name: /^(Edit|Delete) session/ })),
    save_day_fields: await count(viewer.getByRole('button', { name: 'Save day fields' })),
    quick_breaks: await count(viewer.getByRole('button', { name: /Confirm (suggested )?breaks|No breaks taken/ })),
    textareas: await count(viewer.locator('textarea')),
    inputs: await count(viewer.locator('input, select')),
  };
  result.view_only_editor = viewerControls;
  expect(viewerControls).toEqual({ add_session: 0, edit_or_delete: 0, save_day_fields: 0, quick_breaks: 0, textareas: 0, inputs: 0 });
  await viewer.getByRole('button', { name: 'Close' }).click();
  const outside = requests.filter((entry) => {
    const path = entry.slice(entry.indexOf(' ') + 1);
    return !path.startsWith(`/api/shared/${ownerId}/`) && !['/api/shares', '/api/auth/me', '/api/auth/login', '/api/auth/setup'].includes(path);
  });
  result.view_only_outside_requests = outside;
  expect(outside).toEqual([]);
  expect(requests.filter((entry) => !entry.startsWith('GET ') && entry !== 'POST /api/auth/login'), 'the view-only grantee wrote nothing').toEqual([]);

  // Edit share.
  const { given } = await ownerP.api.call<any>('GET', '/api/shares');
  await ownerP.api.call('PUT', `/api/shares/${given[0].id}`, { items: { timesheets: 'edit', ot_read: false, pdf_download: false } });
  requests.length = 0;
  await page.reload();
  await expect(page.locator('[data-day]')).toHaveCount(14);
  const editControls = {
    open_a_day: await count(page.getByLabel('Open a day')),
    change_several_days: await count(page.getByRole('button', { name: 'Change several days' })),
    clock_buttons: await count(page.getByRole('button', { name: /^Clock (in|out)$/ })),
    label_pickers: await count(page.locator('[data-label-picker]')),
    signature_strip: await count(page.getByRole('region', { name: 'Signatures' })),
    review_links: await count(page.locator('[data-sheet-review-link], [data-review-link], [data-review-link-disabled]')),
  };
  result.edit_controls = editControls;
  expect(editControls).toMatchObject({ open_a_day: 1, change_several_days: 1, clock_buttons: 0, signature_strip: 0, review_links: 0 });
  await page.getByRole('button', { name: new RegExp(`^Label for ${editDay}`) }).click();
  await page.getByRole('listbox', { name: `Label for ${editDay}` }).getByRole('option', { name: 'Vacation', exact: true }).click();
  await expect(page.getByRole('status')).toContainText('Saved 1 day.');
  expect((await ownerP.api.dayView(editDay)).category).toBe('Vacation');

  (result.edit_controls as any).edit_day_buttons = await count(page.getByRole('button', { name: anyDayName('Edit') }));
  (result.edit_controls as any).view_day_buttons = await count(page.getByRole('button', { name: anyDayName('View') }));
  await expect(dayButtonBf9(page, fieldsDay)).toHaveAccessibleName(dayNameBf9('Edit', fieldsDay));
  await dayButtonBf9(page, fieldsDay).click();
  await expect(page.locator(`dialog[data-day-editor="${fieldsDay}"]`)).toHaveCount(1);
  const editor = page.getByRole('dialog', { name: /Day editor/ });
  await expect(editor).toBeVisible();
  result.edit_editor_layout = await editorLayout(page);
  const form = editor.getByRole('form', { name: 'Day fields' });
  await form.getByLabel('Worked from home (WFH)').check();
  await form.getByRole('button', { name: 'Save day fields' }).click();
  await expect(editor.getByText('Day fields saved.')).toBeVisible();
  await editor.getByRole('button', { name: 'Close' }).click();

  const use = testInfo.project.use as any;
  if (use.isMobile) await page.setViewportSize({ width: 320, height: 844 });
  await page.evaluate('window.scrollTo(0, 0)');
  await page.waitForTimeout(300);
  result.edit_open_day_row = {
    width: (await page.evaluate('window.innerWidth')) as number,
    field: await inspectControl(page, 'Open a day field', page.getByLabel('Open a day')),
    button_before_value: await inspectControl(page, 'Open day button', page.getByRole('button', { name: 'Open day', exact: true })),
  };
  await page.getByLabel('Open a day').fill(openDayDate);
  (result.edit_open_day_row as any).field_value = await page.getByLabel('Open a day').inputValue();
  (result.edit_open_day_row as any).button = await inspectControl(page, 'Open day button', page.getByRole('button', { name: 'Open day', exact: true }));
  await page.getByRole('button', { name: 'Open day', exact: true }).click();
  const opened = page.getByRole('dialog', { name: /Day editor/ });
  await expect(opened).toBeVisible();
  const openedForm = opened.getByRole('form', { name: 'Day fields' });
  await openedForm.getByLabel('Notes').fill('note by the grantee through Open a day (synthetic)');
  await openedForm.getByRole('button', { name: 'Save day fields' }).click();
  await expect(opened.getByText('Day fields saved.')).toBeVisible();
  await opened.getByRole('button', { name: 'Close' }).click();
  if (use.isMobile) await page.setViewportSize(use.viewport);

  const writes = requests.filter((entry) => !entry.startsWith('GET '));
  result.edit_writes = writes;
  expect(writes).toEqual([
    `POST /api/shared/${ownerId}/days/batch`,
    `POST /api/shared/${ownerId}/days/batch`,
    `PUT /api/shared/${ownerId}/days/${fieldsDay}`,
    `PUT /api/shared/${ownerId}/days/${openDayDate}`,
  ]);
  expect((await ownerP.api.dayView(fieldsDay)).wfh).toBe(true);
  expect((await ownerP.api.dayView(openDayDate)).entry?.notes).toBe('note by the grantee through Open a day (synthetic)');
  expect((await grantee.api.dayView(openDayDate)).entry?.notes ?? null, 'the grantee own day untouched').not.toBe('note by the grantee through Open a day (synthetic)');
  const ownerHistory = await ownerP.api.call<any>('GET', '/api/history');
  const attributed = ownerHistory.audit_events.filter((event: any) => event.via_share === true);
  result.owner_history_via_share = attributed.map((event: any) => ({ operation: event.operation, entity_type: event.entity_type, actor_display_name: event.actor_display_name }));
  expect(attributed.filter((event: any) => event.actor_display_name === 'Synthetic Grantee B').length).toBeGreaterThanOrEqual(3);

  // A third user: nothing of the owner, by screen or by id swap.
  const context = await browser.newContext({ baseURL: builtServer.origin, timezoneId: 'Asia/Ho_Chi_Minh', locale: 'en-US', viewport: use.viewport, isMobile: use.isMobile ?? false, hasTouch: use.hasTouch ?? false });
  const third = await context.newPage();
  await signIn(third, other.account, '#/timesheet');
  await expect(third.locator('[data-day]')).toHaveCount(14);
  const isolation: Record<string, unknown> = {
    note_on_page: (await third.locator('body').innerText()).includes(note),
    owner_session_in_dom: await third.locator(`[data-range="${ownerSession}"]`).count(),
    shared_timesheet_status: await statusInPage(third, 'GET', `/api/shared/${ownerId}/timesheets/${ownerP.payrollDate}`),
    shared_day_status: await statusInPage(third, 'GET', `/api/shared/${ownerId}/days/${ownerDay}`),
    shared_day_put_status: await statusInPage(third, 'PUT', `/api/shared/${ownerId}/days/${ownerDay}`, { category: 'Off', leave_minutes: 0, leave_kind: null, wfh: false, notes: '', expected_version: 1 }),
    own_day_sessions: (await other.api.dayView(ownerDay)).sessions.length,
    owner_session_put_status: await statusInPage(third, 'PUT', `/api/sessions/${ownerSession}`, {
      start: { local: `${ownerDay}T09:00`, zone: LA },
      end: { local: `${ownerDay}T10:00`, zone: LA },
      input_zone: LA,
      breaks: [],
      breaks_confirmed: true,
      expected_version: 1,
    }),
  };
  await third.evaluate(`window.location.hash = '#/shared/${ownerId}/timesheet'`);
  await third.waitForTimeout(1_500);
  isolation.shared_route_shows_owner_note = (await third.locator('body').innerText()).includes(note);
  isolation.shared_route_owner_session = await third.locator(`[data-range="${ownerSession}"]`).count();
  result.isolation = isolation;
  expect(isolation).toMatchObject({ note_on_page: false, owner_session_in_dom: 0, own_day_sessions: 0, shared_route_shows_owner_note: false, shared_route_owner_session: 0 });
  for (const key of ['shared_timesheet_status', 'shared_day_status', 'shared_day_put_status', 'owner_session_put_status']) expect([403, 404], key).toContain(isolation[key]);
  const ownerAfter = await ownerP.api.dayView(ownerDay);
  expect(ownerAfter.sessions[0]?.id).toBe(ownerSession);
  expect(ownerAfter.category).not.toBe('Off');
  await context.close();
  writeResult(`sharing-${testInfo.project.name}.json`, result);
});
