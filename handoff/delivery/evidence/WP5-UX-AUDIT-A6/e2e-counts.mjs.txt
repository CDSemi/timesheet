// Counts a Playwright JSON report per project and per spec file: expected (passed), skipped, unexpected (failed), flaky.
// Usage: node e2e-counts.mjs <report.json>
import { readFileSync } from 'node:fs';

const report = JSON.parse(readFileSync(process.argv[2], 'utf8'));
const rows = new Map();
const add = (key, status) => {
  const row = rows.get(key) ?? { expected: 0, skipped: 0, unexpected: 0, flaky: 0 };
  row[status] = (row[status] ?? 0) + 1;
  rows.set(key, row);
};
function walk(suite, file) {
  const here = suite.file ?? file;
  for (const spec of suite.specs ?? []) {
    for (const test of spec.tests ?? []) {
      add(`${test.projectName} | ${here}`, test.status);
      add(`${test.projectName} | (all)`, test.status);
      add('(all) | (all)', test.status);
    }
  }
  for (const child of suite.suites ?? []) walk(child, here);
}
for (const suite of report.suites ?? []) walk(suite, suite.file);
for (const [key, row] of [...rows.entries()].sort()) {
  console.log(`${key}: passed ${row.expected}, skipped ${row.skipped}, failed ${row.unexpected}, flaky ${row.flaky}`);
}
console.log(`stats: ${JSON.stringify(report.stats)}`);
