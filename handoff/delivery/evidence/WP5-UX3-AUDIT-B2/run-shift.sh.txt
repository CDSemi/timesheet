# Usage: bash run-shift.sh <label> <export dir name> <target instant UTC ISO or "real"> [playwright args...]
# One Playwright run in an export under the FIX10 simulated wall clock (shiftclock.mjs unchanged, copied into the task folder only).
. <task>/env.sh
L=$1; EXP=$T/$2; TARGET_I=$3; shift 3
mkdir -p $T/pw-$L $T/shots-$L
export E2E_SCREENSHOT_DIR="$TW\\shots-$L"
if [ "$TARGET_I" = "real" ]; then
  export NODE_OPTIONS="--trace-deprecation"
  export AB8_SHIFT_MS=0
else
  TARGET_MS=$(date -u -d "$TARGET_I" +%s%3N)
  NOW_MS=$(date -u +%s%3N)
  export AB8_SHIFT_MS=$((TARGET_MS - NOW_MS))
  export NODE_OPTIONS="--trace-deprecation --import file:///$(cygpath -m $T)/shiftclock.mjs"
fi
echo "== $L: real start $(date -u +%Y-%m-%dT%H:%M:%SZ), simulated start $TARGET_I, AB8_SHIFT_MS=$AB8_SHIFT_MS"
node.exe $T/probe-now.mjs
cd "$EXP" || exit 2
echo "\$ node node_modules/@playwright/test/cli.js test $* --output=<task>/pw-$L"
node.exe node_modules/@playwright/test/cli.js test "$@" --output="$TW\\pw-$L"
echo "exit=$?"
node.exe $T/probe-now.mjs
echo "== $L real end $(date -u +%Y-%m-%dT%H:%M:%SZ)"
