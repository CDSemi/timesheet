"""EN/VI parity of the area-B docs lines: numbers, code tokens and quoted English strings. Usage: python parity.py <export dir> <out.txt>"""
import re
import sys

root, out = sys.argv[1:3]
PAIRS = [
    ('docs/04_UX_AND_SETTINGS', 65, 65),
    ('docs/04_UX_AND_SETTINGS', 66, 66),
    ('docs/04_UX_AND_SETTINGS', 70, 70),
    ('docs/10_DECISIONS_AND_SOURCES', 101, 102),
    ('docs/10_DECISIONS_AND_SOURCES', 238, 239),
    ('docs/10_DECISIONS_AND_SOURCES', 245, 246),
    ('docs/12_RELEASE_NOTES', 71, 71),
]


def tokens(text):
    code = sorted(re.findall(r'`[^`]+`', text))
    quoted = sorted(q for q in re.findall(r'"([^"]+)"', text) if re.search(r'[A-Za-z]', q))
    numbers = sorted(re.findall(r'\d+(?:[.:]\d+)*', text))
    ids = sorted(set(re.findall(r'\b(?:[A-Z]{1,3}-\d+[a-z]?|U-\d+|E-\d+|D-\d+|R-\d+|G-\d+|WP\d[\w-]*)\b', text)))
    return code, quoted, numbers, ids


lines = []
ok_all = True
for base, en_no, vi_no in PAIRS:
    en = open(f'{root}/{base}.md', encoding='utf-8').read().split('\n')[en_no - 1]
    vi = open(f'{root}/{base}.vi.md', encoding='utf-8').read().split('\n')[vi_no - 1]
    te, tv = tokens(en), tokens(vi)
    names = ('code tokens', 'quoted strings', 'numbers', 'ids')
    lines.append(f'=== {base} EN line {en_no} / VI line {vi_no}')
    for name, a, b in zip(names, te, tv):
        same = a == b
        ok_all = ok_all and same
        lines.append(f'  {name}: {"equal" if same else "DIFFER"} ({len(a)} / {len(b)})')
        if not same:
            lines.append(f'    EN only: {sorted(set(a) - set(b))}')
            lines.append(f'    VI only: {sorted(set(b) - set(a))}')
lines.append(f'ALL EQUAL: {ok_all}')
open(out, 'w', encoding='utf-8', newline='\n').write('\n'.join(lines) + '\n')
print('\n'.join(lines))
