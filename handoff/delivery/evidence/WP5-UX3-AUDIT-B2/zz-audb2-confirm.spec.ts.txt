import { appendFileSync } from 'node:fs';
import { join } from 'node:path';
import type { Locator, Page } from '@playwright/test';
import { expect, newPerson, type Person, test } from './fixtures.ts';

/*
 * WP5-UX3-AUDIT-B2 task-local probe (never part of the repository): F-UX3B-2 recheck. Ended sessions with unconfirmed breaks and
 * no saved row are seeded through the API; "Confirm breaks" is pressed in the Day editor; the rows the session dialog lists are
 * read (date, start, minutes, end text, tag), then Save goes through the notice when nothing is listed or straight to "Save this
 * session?" otherwise; the PUT body is captured raw and the stored session read back. AUDB_LABEL names the build. Desktop only.
 */
const LABEL = process.env.AUDB_LABEL ?? 'unknown';
const OUT = process.env.AUDB_OUT ?? '';
const SHOTS = process.env.AUDB_SHOTS ?? '';
const LA = 'America/Los_Angeles';
const HCM = 'Asia/Ho_Chi_Minh';
const SYD = 'Australia/Sydney';
const REASON = 'audit probe reason';
test.use({ locale: 'en-US', timezoneId: LA });
test.skip(({ isMobile }) => isMobile, 'Desktop project only');

function rec(item: Record<string, unknown>) {
  const line = JSON.stringify({ build: LABEL, ...item });
  console.log(line);
  if (OUT !== '') appendFileSync(join(OUT, `confirm-${LABEL}.jsonl`), `${line}\n`);
}

interface Write {
  method: string;
  path: string;
  status: number;
  raw: string;
}

function watch(page: Page): Write[] {
  const writes: Write[] = [];
  page.on('response', (response) => {
    const request = response.request();
    const path = new URL(request.url()).pathname;
    if (!['POST', 'PUT'].includes(request.method())) return;
    if (!/^\/api\/(days\/[^/]+\/sessions|sessions\/[^/]+)$/.test(path)) return;
    writes.push({ method: request.method(), path: path.replace(/sessions\/[0-9a-f-]+$/, 'sessions/<id>'), status: response.status(), raw: request.postData() ?? '' });
  });
  return writes;
}

async function openEditor(page: Page, workDate: string): Promise<Locator> {
  await page.getByLabel('Open a day').fill(workDate);
  await page.getByRole('button', { name: 'Open day', exact: true }).click();
  const editor = page.getByRole('dialog', { name: /Day editor/ });
  await expect(editor).toBeVisible();
  await expect(editor.getByRole('heading', { name: 'Figures (computed by the server)' })).toBeVisible();
  const reason = editor.getByLabel('Reason for editing an old or finalized period');
  if ((await reason.count()) > 0) await reason.first().fill(REASON);
  return editor;
}

async function closeEditor(page: Page) {
  const editor = page.getByRole('dialog', { name: /Day editor/ });
  if ((await editor.count()) > 0) {
    await editor.getByRole('button', { name: 'Close', exact: true }).first().click();
    await expect(editor).toHaveCount(0);
  }
}

async function readRows(form: Locator) {
  const rows = form.locator('[data-break]');
  const out: Array<Record<string, unknown>> = [];
  for (let i = 1; i <= (await rows.count()); i += 1) {
    const date = form.getByLabel(`Break ${i} date`, { exact: true });
    out.push({
      date: (await date.count()) > 0 ? await date.inputValue() : null,
      time: await form.getByLabel(`Break ${i} start time`, { exact: true }).inputValue(),
      minutes: await form.getByLabel(`Break ${i} minutes`, { exact: true }).inputValue(),
      origin: await form.locator(`[data-break="${i}"]`).getAttribute('data-origin'),
      text: (await form.locator(`[data-break="${i}"]`).innerText()).replace(/\s+/g, ' ').trim(),
    });
  }
  return out;
}

async function stored(person: Person, date: string) {
  const day = await person.api.dayView(date);
  return day.sessions.map((s) => ({ start: s.start_utc, end: s.end_utc, zone: s.input_zone, confirmed: s.breaks_confirmed, breaks: s.breaks.map((b) => `${b.start_utc}..${b.end_utc}`) }));
}

/** Saves the dialog: through the notice when nothing is listed, else straight to the confirmation. */
async function save(page: Page, form: Locator, noticeChoice: string) {
  await form.getByRole('button', { name: 'Save', exact: true }).click();
  const notice = page.getByRole('alertdialog', { name: 'No breaks listed' });
  const confirm = page.getByRole('alertdialog', { name: 'Save this session?' });
  await expect(notice.or(confirm)).toBeVisible();
  const askedNotice = await notice.isVisible();
  if (askedNotice) await notice.getByRole('button', { name: noticeChoice, exact: true }).click();
  await expect(confirm).toBeVisible();
  const facts = (await confirm.locator('.confirm-facts').innerText()).replace(/\s+/g, ' ');
  await confirm.getByRole('button', { name: 'Save', exact: true }).click();
  await expect(confirm).toHaveCount(0);
  await expect(form).toHaveCount(0);
  return { askedNotice, facts };
}

interface Case {
  id: string;
  date: string;
  zone: string;
  start: string;
  end: string;
  expect: number | null;
}

test('F-UX3B-2 "Confirm breaks" lists the suggestions inside the session', async ({ page, adminSeed, builtServer, signInPageAs }) => {
  test.setTimeout(1_200_000);
  const person = await newPerson(adminSeed, builtServer, { signature: 'none' });
  const { todayLocal } = await person.api.today();
  const writes = watch(page);
  const errors: string[] = [];
  rec({ case: 'context', todayLocal, startedAt: new Date().toISOString() });
  await signInPageAs(person.account);
  await expect(page.getByRole('button', { name: 'Open day', exact: true })).toBeVisible();

  const cases: Case[] = [
    { id: 'F01 09:00-12:00', date: '2026-06-08', zone: LA, start: '2026-06-08T09:00', end: '2026-06-08T12:00', expect: 1 },
    { id: 'F02 09:00-14:00', date: '2026-06-09', zone: LA, start: '2026-06-09T09:00', end: '2026-06-09T14:00', expect: 2 },
    { id: 'F03 09:00-18:00', date: '2026-06-10', zone: LA, start: '2026-06-10T09:00', end: '2026-06-10T18:00', expect: 3 },
    { id: 'F04 09:00-10:30 none fits', date: '2026-06-11', zone: LA, start: '2026-06-11T09:00', end: '2026-06-11T10:30', expect: 0 },
    { id: 'F05 09:00-11:15 first ends exactly at the end', date: '2026-06-12', zone: LA, start: '2026-06-12T09:00', end: '2026-06-12T11:15', expect: 1 },
    { id: 'F06 09:00-11:14 first ends one minute after', date: '2026-06-15', zone: LA, start: '2026-06-15T09:00', end: '2026-06-15T11:14', expect: 0 },
    { id: 'F07 09:00-15:45 third ends exactly at the end', date: '2026-06-16', zone: LA, start: '2026-06-16T09:00', end: '2026-06-16T15:45', expect: 3 },
    { id: 'F08 09:00-15:44 third ends one minute after', date: '2026-06-17', zone: LA, start: '2026-06-17T09:00', end: '2026-06-17T15:44', expect: 2 },
    { id: 'F09 overnight 22:00-03:00+1', date: '2026-06-18', zone: LA, start: '2026-06-18T22:00', end: '2026-06-19T03:00', expect: 2 },
    { id: 'F10 overnight 22:00-06:00+1', date: '2026-06-19', zone: LA, start: '2026-06-19T22:00', end: '2026-06-20T06:00', expect: 3 },
    // The calendar has no version for 2025-11-02 (run 1: calendar_missing), so the fall-back day is Sydney's 2026-04-05 (03:00 AEDT -> 02:00 AEST).
    { id: 'F11 DST fall-back Sydney 00:00-05:45 (6h45m elapsed)', date: '2026-04-04', zone: SYD, start: '2026-04-05T00:00', end: '2026-04-05T05:45', expect: 3 },
    { id: 'F12 DST fall-back Sydney 00:00-05:44', date: '2026-04-04', zone: SYD, start: '2026-04-05T00:00', end: '2026-04-05T05:44', expect: 2 },
    { id: 'F13 DST spring-forward 00:00-07:45 (6h45m elapsed)', date: '2026-03-08', zone: LA, start: '2026-03-08T00:00', end: '2026-03-08T07:45', expect: 3 },
    { id: 'F14 DST spring-forward 00:00-07:44 (6h44m elapsed)', date: '2026-03-08', zone: LA, start: '2026-03-08T00:00', end: '2026-03-08T07:44', expect: 2 },
    { id: 'F15 input zone Ho Chi Minh 08:00-13:00 (LA 18:00-23:00)', date: '2026-06-22', zone: HCM, start: '2026-06-23T08:00', end: '2026-06-23T13:00', expect: 2 },
    { id: 'F16 today ended 07:00-12:00', date: todayLocal, zone: LA, start: `${todayLocal}T07:00`, end: `${todayLocal}T12:00`, expect: 2 },
  ];
  // F12 and F14 end one minute earlier than F11 and F13 on the same DST day: the third suggestion no longer fits. F11 (fall-back,
  // 5h45m of wall clock, 6h45m elapsed) and F14 (spring-forward, 7h44m of wall clock, 6h44m elapsed) tell elapsed time from wall time.
  for (const item of cases) {
    const before = writes.length;
    try {
      await person.api.call(
        'POST',
        `/api/days/${item.date}/sessions`,
        { start: { local: item.start, zone: item.zone }, end: { local: item.end, zone: item.zone }, input_zone: item.zone, breaks_confirmed: false, breaks: [], reason: REASON },
        201,
      );
      const seeded = await stored(person, item.date);
      const editor = await openEditor(page, item.date);
      await editor.getByRole('button', { name: /^Confirm breaks of/ }).click();
      const form = page.getByRole('dialog', { name: 'Edit session', exact: true });
      await expect(form).toBeVisible();
      const rows = await readRows(form);
      const box = await form.getByRole('checkbox', { name: 'Confirm no breaks taken' }).isChecked();
      if (SHOTS !== '' && item.id.startsWith('F02')) await form.screenshot({ path: join(SHOTS, `audb2-confirm-breaks-0900-1400-${LABEL}-synthetic.png`) });
      const saved = await save(page, form, 'Confirm no breaks taken');
      rec({ case: item.id, date: item.date, expectRows: item.expect, seeded, rows, box, saved, writes: writes.slice(before), stored: await stored(person, item.date) });
      await closeEditor(page);
    } catch (caught) {
      errors.push(`${item.id}: ${String(caught).slice(0, 400)}`);
      rec({ case: item.id, error: String(caught).slice(0, 800), writes: writes.slice(before) });
      await page.keyboard.press('Escape').catch(() => undefined);
      await page.goto('/');
    }
    await person.api.clearSessions(item.date, REASON);
  }

  /* Pinned like the new-session pre-fill: "Confirm breaks" on 09:00-14:00 against a new session typed 09:00-14:00 on the same past date. */
  const pinDate = '2026-06-24';
  try {
    await person.api.call(
      'POST',
      `/api/days/${pinDate}/sessions`,
      { start: { local: `${pinDate}T09:00`, zone: LA }, end: { local: `${pinDate}T14:00`, zone: LA }, input_zone: LA, breaks_confirmed: false, breaks: [], reason: REASON },
      201,
    );
    let before = writes.length;
    let editor = await openEditor(page, pinDate);
    await editor.getByRole('button', { name: /^Confirm breaks of/ }).click();
    let form = page.getByRole('dialog', { name: 'Edit session', exact: true });
    const confirmRows = await readRows(form);
    await save(page, form, 'Confirm no breaks taken');
    const confirmWrites = writes.slice(before);
    const confirmStored = await stored(person, pinDate);
    await closeEditor(page);
    await person.api.clearSessions(pinDate, REASON);
    before = writes.length;
    editor = await openEditor(page, pinDate);
    await editor.getByRole('button', { name: 'Add session' }).click();
    form = page.getByRole('dialog', { name: 'New session', exact: true });
    await expect(form).toBeVisible();
    const prefillRows = await readRows(form);
    await form.getByLabel('Start time', { exact: true }).fill('09:00');
    await form.getByLabel('End time', { exact: true }).fill('14:00');
    const shiftedRows = await readRows(form);
    await save(page, form, 'Confirm no breaks taken');
    const newWrites = writes.slice(before);
    const newStored = await stored(person, pinDate);
    const breaksOf = (raw: string) => JSON.stringify((JSON.parse(raw) as { breaks: unknown }).breaks);
    rec({
      case: 'P01 pinned like the pre-fill',
      confirmRows,
      prefillRows,
      shiftedRows,
      confirmWrites,
      newWrites,
      breaksIdentical: confirmWrites.length === 1 && newWrites.length === 1 && breaksOf(confirmWrites[0]?.raw ?? '{}') === breaksOf(newWrites[0]?.raw ?? '{}'),
      storedBreaksIdentical: JSON.stringify(confirmStored.map((s) => s.breaks)) === JSON.stringify(newStored.map((s) => s.breaks)),
      confirmStored,
      newStored,
    });
    await closeEditor(page);
  } catch (caught) {
    errors.push(`P01: ${String(caught).slice(0, 400)}`);
    rec({ case: 'P01 pinned like the pre-fill', error: String(caught).slice(0, 800) });
    await page.keyboard.press('Escape').catch(() => undefined);
    await page.goto('/');
  }
  await person.api.clearSessions(pinDate, REASON);

  /* When "Confirm breaks" appears: an ended unconfirmed session (yes), a confirmed one (no), a running one (no). */
  try {
    const gateDate = '2026-06-25';
    await person.api.call('POST', `/api/days/${gateDate}/sessions`, { start: { local: `${gateDate}T09:00`, zone: LA }, end: { local: `${gateDate}T12:00`, zone: LA }, input_zone: LA, breaks_confirmed: true, breaks: [], reason: REASON }, 201);
    let editor = await openEditor(page, gateDate);
    const confirmedButtons = await editor.getByRole('button', { name: /^Confirm breaks of/ }).count();
    await closeEditor(page);
    await person.api.clearSessions(gateDate, REASON);
    await person.api.call('POST', `/api/days/${todayLocal}/sessions`, { start: { local: `${todayLocal}T13:00`, zone: LA }, end: null, input_zone: LA, breaks_confirmed: false, breaks: [] }, 201);
    await page.goto('/');
    editor = await openEditor(page, todayLocal);
    const runningButtons = await editor.getByRole('button', { name: /^Confirm breaks of/ }).count();
    await closeEditor(page);
    rec({ case: 'G01 when Confirm breaks appears', confirmedButtons, runningButtons });
    const open = (await person.api.dayView(todayLocal)).sessions.find((s) => s.end_utc === null);
    if (open !== undefined) await person.api.call('POST', '/api/clock/out', { breaks: [], breaks_confirmed: false, expected_version: open.version });
    await person.api.clearSessions(todayLocal, REASON);
  } catch (caught) {
    errors.push(`G01: ${String(caught).slice(0, 400)}`);
    rec({ case: 'G01 when Confirm breaks appears', error: String(caught).slice(0, 800) });
  }
  rec({ case: 'errors', errors });
  expect(errors).toEqual([]);
});
