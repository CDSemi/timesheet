// WP5-UX-AUDIT-A3 scope 2 (AC-04) and scope 3 (A2 risk R2), on a2ea7a4 in three layouts: 1280
// (non-modal panel), 1024 (modal side panel) and the 390 phone (modal bottom sheet). Reason-required
// corrections, stale versions and the conflict confirmation through the picker and the editor (the
// editor opened from the sheet and from the "Open a day" row); then R2: a picker review open over the
// non-modal editor when the window shrinks below 1200px - what each Escape closes, whether anything
// is written, and whether the review still demands its reason. Synthetic accounts only.
import { expect, newPerson, test } from '../export/tests/e2e/fixtures.ts';
import { editorLayout, openDay, signIn, writeResult } from './common.ts';

test.use({ locale: 'en-US', timezoneId: 'Asia/Ho_Chi_Minh' });

const LA = 'America/Los_Angeles';

test('AC-04: reason, stale version and conflict through the picker and the editor in each layout', async ({ page, adminSeed, builtServer }, testInfo) => {
  test.setTimeout(300_000);
  const person = await newPerson(adminSeed, builtServer, { displayName: 'Synthetic Edit Probe', signature: 'none' });
  const free = await person.api.displayedPeriodFreeWorkdays();
  expect(free.length, 'four past free workdays').toBeGreaterThanOrEqual(4);
  const [staleDay = '', quickDay = '', conflictDay = '', fieldsStaleDay = ''] = free;
  const batch: any[] = [];
  const dayPuts: any[] = [];
  page.on('request', (request: any) => {
    const { pathname } = new URL(request.url());
    if (pathname === '/api/days/batch') batch.push(request.postDataJSON());
    if (request.method() === 'PUT' && pathname.startsWith('/api/days/')) dayPuts.push({ path: pathname, body: request.postDataJSON() });
  });
  const result: Record<string, unknown> = { project: testInfo.project.name };
  const picker = (workDate: string) => page.getByRole('button', { name: new RegExp(`^Label for ${workDate}`) });
  const pick = async (workDate: string, label: string) => {
    await picker(workDate).click();
    await page.getByRole('listbox', { name: `Label for ${workDate}` }).getByRole('option', { name: label, exact: true }).click();
  };

  await signIn(page, person.account, '#/timesheet');
  await expect(page.locator('[data-day]')).toHaveCount(14);

  // 1. Reason-required correction through the in-cell picker of an old period.
  await page.getByRole('button', { name: 'Previous period' }).click();
  await expect(page.locator('.period-title .badge.old')).toBeVisible();
  const oldDay = (await page.locator('[data-day]').nth(2).getAttribute('data-day')) ?? '';
  const oldBefore = await person.api.dayView(oldDay);
  await pick(oldDay, 'Vacation');
  const dialog = page.getByRole('dialog', { name: `Review label change for ${oldDay}` });
  await expect(dialog).toBeVisible();
  const reasonBox = dialog.getByLabel(/Reason for editing an old or finalized period/);
  await expect(reasonBox).toBeVisible();
  await expect(dialog.getByRole('button', { name: 'Commit changes' })).toBeDisabled();
  expect(batch.map((body) => body.mode), 'only the preview before the reason').toEqual(['preview']);
  result.picker_preview_body = batch[0];
  expect(batch[0]).toEqual({ mode: 'preview', entries: [{ work_date: oldDay, expected_version: oldBefore.entry?.version ?? null, category: 'Vacation' }] });
  expect((await person.api.dayView(oldDay)).category, 'nothing saved before the reason').toBe(oldBefore.category);
  await reasonBox.fill('Audit A3 synthetic correction (picker)');
  await dialog.getByRole('button', { name: 'Commit changes' }).click();
  await expect(page.getByRole('status')).toContainText('Saved 1 day.');
  result.picker_reason_commit = batch.at(-1);
  expect(batch.at(-1)).toEqual({ mode: 'commit', entries: [{ work_date: oldDay, expected_version: oldBefore.entry?.version ?? null, category: 'Vacation' }], reason: 'Audit A3 synthetic correction (picker)' });

  // 2. Reason-required correction through the day editor of this layout (opened from the sheet).
  await page.locator(`[data-day="${oldDay}"]`).getByRole('button', { name: `Edit ${oldDay}`, exact: true }).click();
  const editor = page.getByRole('dialog', { name: /Day editor/ });
  await expect(editor).toBeVisible();
  result.editor_layout = await editorLayout(page);
  const fields = editor.getByRole('form', { name: 'Day fields' });
  await fields.getByLabel('Worked from home (WFH)').check();
  await expect(fields.getByRole('button', { name: 'Save day fields' })).toBeDisabled();
  expect(dayPuts, 'no day PUT before the reason').toEqual([]);
  await editor.getByLabel('Reason for editing an old or finalized period').fill('Audit A3 synthetic correction (editor)');
  await fields.getByRole('button', { name: 'Save day fields' }).click();
  await expect(editor.getByText('Day fields saved.')).toBeVisible();
  result.editor_reason_put = dayPuts.at(-1);
  expect(dayPuts.at(-1)?.body).toMatchObject({ reason: 'Audit A3 synthetic correction (editor)', wfh: true, category: 'Vacation' });
  expect(Object.keys(dayPuts.at(-1)?.body ?? {}).sort()).toEqual(['category', 'expected_version', 'leave_kind', 'leave_minutes', 'notes', 'reason', 'wfh']);
  await editor.getByRole('button', { name: 'Close' }).click();
  await expect(editor).toHaveCount(0);
  const history = await person.api.call<any>('GET', '/api/history');
  result.history_reasons = history.audit_events
    .filter((event: any) => typeof event.reason === 'string' && event.reason.startsWith('Audit A3 synthetic'))
    .map((event: any) => ({ operation: event.operation, entity_type: event.entity_type, reason: event.reason }));
  expect((result.history_reasons as any[]).map((event) => event.reason)).toEqual(
    expect.arrayContaining(['Audit A3 synthetic correction (picker)', 'Audit A3 synthetic correction (editor)']),
  );

  // 3. Stale version through the picker.
  await page.getByRole('button', { name: 'Next period' }).click();
  await expect(page.locator(`[data-day="${staleDay}"]`)).toBeVisible();
  await person.api.putDay(staleDay, { notes: 'changed elsewhere (synthetic)' });
  const staleBefore = await person.api.dayView(staleDay);
  batch.length = 0;
  await pick(staleDay, 'Sick');
  const staleDialog = page.getByRole('dialog', { name: `Review label change for ${staleDay}` });
  await expect(staleDialog).toBeVisible();
  await expect(staleDialog.getByRole('alert')).toContainText('Some dates changed elsewhere');
  await expect(staleDialog.getByRole('button', { name: 'Commit changes' })).toBeDisabled();
  result.picker_stale_modes = batch.map((body) => body.mode);
  expect(batch.map((body) => body.mode)).toEqual(['preview']);
  await staleDialog.getByRole('button', { name: 'Cancel' }).click();
  expect((await person.api.dayView(staleDay)).entry?.version).toBe(staleBefore.entry?.version);

  // 4. Stale version through the editor's day fields form, the editor opened from the "Open a day" row.
  await page.reload();
  await expect(page.locator('[data-day]')).toHaveCount(14);
  dayPuts.length = 0;
  const fieldsEditor = await openDay(page, fieldsStaleDay);
  await person.api.putDay(fieldsStaleDay, { notes: 'changed behind the editor (synthetic)' });
  const behind = await person.api.dayView(fieldsStaleDay);
  const fieldsForm = fieldsEditor.getByRole('form', { name: 'Day fields' });
  await fieldsForm.getByLabel('Notes').fill('typed in the editor (synthetic)');
  await fieldsForm.getByRole('button', { name: 'Save day fields' }).click();
  const fieldsStale = fieldsEditor.getByRole('alert').filter({ hasText: 'This day changed since it was loaded' });
  await expect(fieldsStale).toBeVisible();
  const fieldsAfter = await person.api.dayView(fieldsStaleDay);
  result.editor_fields_stale = { sent_expected_version: dayPuts.at(-1)?.body?.expected_version ?? 'none', server_version: behind.entry?.version, after_version: fieldsAfter.entry?.version, after_notes: fieldsAfter.entry?.notes };
  expect(dayPuts.at(-1)?.body?.expected_version ?? null, 'the editor sent the version it loaded').toBeNull();
  expect(fieldsAfter.entry?.version).toBe(behind.entry?.version);
  expect(fieldsAfter.entry?.notes).toBe('changed behind the editor (synthetic)');
  await fieldsStale.getByRole('button', { name: 'Reload day' }).click();
  await expect(fieldsStale).toHaveCount(0);
  await expect(fieldsEditor.getByRole('form', { name: 'Day fields' }).getByLabel('Notes')).toHaveValue('changed behind the editor (synthetic)');
  await fieldsEditor.getByRole('button', { name: 'Close' }).click();

  // 5. Stale version through the one-tap break confirmation of the editor.
  await person.api.seedUnconfirmedBreaksDay(quickDay);
  await page.reload();
  await expect(page.locator('[data-day]')).toHaveCount(14);
  await page.locator(`[data-day="${quickDay}"]`).getByRole('button', { name: `Edit ${quickDay}`, exact: true }).click();
  const quickEditor = page.getByRole('dialog', { name: /Day editor/ });
  const quick = quickEditor.getByRole('group', { name: /^Breaks of the session/ });
  await expect(quick.getByRole('button', { name: 'Confirm suggested breaks' })).toBeVisible();
  const session = (await person.api.dayView(quickDay)).sessions[0];
  await person.api.call('PUT', `/api/sessions/${session?.id}`, {
    start: { local: `${quickDay}T09:00`, zone: LA },
    end: { local: `${quickDay}T18:30`, zone: LA },
    input_zone: LA,
    breaks: [],
    breaks_confirmed: false,
    expected_version: session?.version,
  });
  const changed = (await person.api.dayView(quickDay)).sessions[0];
  const sessionPuts: any[] = [];
  const sessionListener = (request: any) => {
    if (request.method() === 'PUT' && new URL(request.url()).pathname.startsWith('/api/sessions/')) sessionPuts.push(request.postDataJSON());
  };
  page.on('request', sessionListener);
  await quick.getByRole('button', { name: 'Confirm suggested breaks' }).click();
  const staleAlert = quickEditor.getByRole('alert').filter({ hasText: 'This day changed since it was loaded' });
  await expect(staleAlert).toBeVisible();
  page.off('request', sessionListener);
  const quickAfter = (await person.api.dayView(quickDay)).sessions[0];
  result.quick_stale = { loaded_version: session?.version, changed_version: changed?.version, after_version: quickAfter?.version, after_confirmed: quickAfter?.breaks_confirmed, sent_body_keys: Object.keys(sessionPuts[0] ?? {}).sort(), sent_expected_version: sessionPuts[0]?.expected_version };
  expect(sessionPuts[0]?.expected_version, 'the one-tap confirmation sends the loaded version').toBe(session?.version);
  expect(Object.keys(sessionPuts[0] ?? {}).sort()).toEqual(['breaks', 'breaks_confirmed', 'end', 'expected_version', 'input_zone', 'start']);
  expect(quickAfter?.version).toBe(changed?.version);
  expect(quickAfter?.breaks_confirmed).toBe(false);

  // 6. The delete confirmation in this layout's editor: Delete, then Escape, deletes nothing.
  await staleAlert.getByRole('button', { name: 'Reload day' }).click();
  await quickEditor.getByRole('button', { name: /^Delete session/ }).click();
  await expect(quickEditor.getByRole('button', { name: 'Confirm delete' })).toBeVisible();
  await page.keyboard.press('Escape');
  await expect(quickEditor).toHaveCount(0);
  result.delete_then_escape_sessions = (await person.api.dayView(quickDay)).sessions.length;
  expect(result.delete_then_escape_sessions, 'Escape after Delete deletes nothing').toBe(1);

  // 7. Conflict confirmation through the picker.
  await person.api.seedCompleteDay(conflictDay);
  await page.reload();
  await expect(page.locator('[data-day]')).toHaveCount(14);
  batch.length = 0;
  await pick(conflictDay, 'Off');
  const conflictDialog = page.getByRole('dialog', { name: `Review label change for ${conflictDay}` });
  await conflictDialog.getByRole('button', { name: 'Review conflicts' }).click();
  await expect(page.getByRole('dialog', { name: 'Recorded work conflicts with the new label' })).toBeVisible();
  const confirmButton = page.getByRole('button', { name: 'Confirm and commit' });
  await expect(confirmButton).toBeDisabled();
  expect((await person.api.dayView(conflictDay)).category).toBe('Worked');
  await page.getByLabel('I confirm the label change for these dates').check();
  await confirmButton.click();
  await expect(page.getByRole('status')).toContainText('Saved 1 day.');
  result.picker_conflict_commit = batch.at(-1);
  expect(batch.at(-1)).toMatchObject({ mode: 'commit', confirm_conflicts: true });
  expect((await person.api.dayView(conflictDay)).sessions).toHaveLength(1);
  writeResult(`ac04-${testInfo.project.name}.json`, { oldDay, staleDay, quickDay, conflictDay, fieldsStaleDay, ...result });
});

test('R2: a picker review over the non-modal editor when the window shrinks below 1200px - nothing written, nothing bypassed', async ({ page, adminSeed, builtServer }, testInfo) => {
  test.skip(testInfo.project.name !== 'desktop', 'Resizes a desktop browser across 1200px itself');
  test.setTimeout(300_000);
  const person = await newPerson(adminSeed, builtServer, { displayName: 'Synthetic Risk Probe', signature: 'none' });
  const [day = ''] = await person.api.displayedPeriodFreeWorkdays();
  const writes: Array<{ call: string; body: unknown }> = [];
  page.on('request', (request: any) => {
    const { pathname } = new URL(request.url());
    if (request.method() !== 'GET' && pathname.startsWith('/api/') && !pathname.startsWith('/api/auth/')) writes.push({ call: `${request.method()} ${pathname}`, body: request.postDataJSON() });
  });
  const result: Record<string, unknown> = {};
  const openDialogs = () =>
    page.evaluate(`[...document.querySelectorAll('dialog')].filter((d) => d.open).map((d) => ({ kind: d.hasAttribute('data-day-editor') ? 'editor' : (d.getAttribute('aria-labelledby') ? document.getElementById(d.getAttribute('aria-labelledby'))?.textContent ?? 'dialog' : 'dialog'), modal: d.matches(':modal') }))`);
  await page.setViewportSize({ width: 1280, height: 800 });
  await signIn(page, person.account, '#/timesheet');
  await expect(page.locator('[data-day]')).toHaveCount(14);

  const runScenario = async (label: string, afterShrink: 'escape-twice' | 'escape-once-then-reason') => {
    await page.setViewportSize({ width: 1280, height: 800 });
    await page.reload();
    await expect(page.locator('[data-day]')).toHaveCount(14);
    const wideEditor = await openDay(page, day);
    const notes = wideEditor.getByRole('form', { name: 'Day fields' }).getByLabel('Notes');
    await notes.fill(`unsaved note ${label} (synthetic)`);
    await page.getByRole('button', { name: 'Previous period' }).click();
    await expect(page.locator('.period-title .badge.old')).toBeVisible();
    const oldDay = (await page.locator('[data-day]').nth(3).getAttribute('data-day')) ?? '';
    const oldBefore = await person.api.dayView(oldDay);
    writes.length = 0;
    await page.getByRole('button', { name: new RegExp(`^Label for ${oldDay}`) }).click();
    await page.getByRole('listbox', { name: `Label for ${oldDay}` }).getByRole('option', { name: 'Sick', exact: true }).click();
    const review = page.getByRole('dialog', { name: `Review label change for ${oldDay}` });
    await expect(review).toBeVisible();
    const before = await openDialogs();
    await page.setViewportSize({ width: 1024, height: 800 });
    await page.waitForTimeout(600);
    const stacked = await openDialogs();
    const commit = review.getByRole('button', { name: 'Commit changes' });
    const reviewReason = review.getByLabel(/Reason for editing an old or finalized period/);
    const reasonFocusable = await reviewReason.evaluate((el: HTMLElement) => {
      el.focus();
      return document.activeElement === el;
    });
    const commitDisabledStacked = await commit.isDisabled();
    await page.keyboard.press('Escape');
    await page.waitForTimeout(400);
    const afterFirst = await openDialogs();
    const record: Record<string, unknown> = { oldDay, dialogs_at_1280: before, dialogs_after_shrink: stacked, review_reason_focusable_while_editor_on_top: reasonFocusable, commit_disabled_while_stacked: commitDisabledStacked, after_first_escape: afterFirst };
    if (afterShrink === 'escape-twice') {
      if ((await review.count()) === 1) {
        record.commit_disabled_after_first_escape = await commit.isDisabled();
        expect(record.commit_disabled_after_first_escape, 'the review still needs its reason').toBe(true);
      }
      await page.keyboard.press('Escape');
      await page.waitForTimeout(400);
      record.after_second_escape = await openDialogs();
      record.writes = writes.map((w) => ({ call: w.call, mode: (w.body as any)?.mode ?? null }));
      expect(writes.filter((w) => w.call !== 'POST /api/days/batch'), 'no write other than the preview').toEqual([]);
      expect(writes.map((w) => (w.body as any)?.mode), 'only the preview').toEqual(['preview']);
      const after = await person.api.dayView(oldDay);
      record.stored_category = { before: oldBefore.category, after: after.category, version_before: oldBefore.entry?.version ?? null, version_after: after.entry?.version ?? null };
      expect(after.category, 'nothing committed').toBe(oldBefore.category);
      const dayAfter = await person.api.dayView(day);
      record.editor_day_notes_after = dayAfter.entry?.notes ?? null;
      expect(dayAfter.entry?.notes ?? null, 'the unsaved editor note was not written').not.toBe(`unsaved note ${label} (synthetic)`);
    } else {
      record.review_open_after_first_escape = await review.count();
      if (record.review_open_after_first_escape !== 1) {
        // The first Escape closed the review itself: nothing may have been committed.
        record.writes = writes.map((w) => ({ call: w.call, mode: (w.body as any)?.mode ?? null }));
        expect(writes.map((w) => (w.body as any)?.mode)).toEqual(['preview']);
        expect((await person.api.dayView(oldDay)).category).toBe(oldBefore.category);
        return record;
      }
      record.commit_disabled_without_reason = await commit.isDisabled();
      expect(record.commit_disabled_without_reason).toBe(true);
      await reviewReason.fill(`Audit A3 R2 reason ${label}`);
      await commit.click();
      await expect(page.getByRole('status')).toContainText('Saved 1 day.');
      const commitBody = writes.map((w) => w.body as any).find((b) => b?.mode === 'commit');
      record.commit_body = commitBody;
      record.writes = writes.map((w) => ({ call: w.call, mode: (w.body as any)?.mode ?? null }));
      expect(commitBody).toEqual({ mode: 'commit', entries: [{ work_date: oldDay, expected_version: oldBefore.entry?.version ?? null, category: 'Sick' }], reason: `Audit A3 R2 reason ${label}` });
      expect(writes.filter((w) => w.call !== 'POST /api/days/batch'), 'no other write').toEqual([]);
      const history = await person.api.call<any>('GET', '/api/history');
      record.history_reason = history.audit_events.filter((e: any) => e.reason === `Audit A3 R2 reason ${label}`).map((e: any) => ({ operation: e.operation, entity_type: e.entity_type, reason: e.reason }));
      expect((record.history_reason as any[]).length).toBeGreaterThanOrEqual(1);
      const dayAfter = await person.api.dayView(day);
      record.editor_day_notes_after = dayAfter.entry?.notes ?? null;
      expect(dayAfter.entry?.notes ?? null, 'the unsaved editor note was not written').not.toBe(`unsaved note ${label} (synthetic)`);
    }
    return record;
  };

  result.escape_twice = await runScenario('e', 'escape-twice');
  result.escape_once_then_reason = await runScenario('f', 'escape-once-then-reason');
  writeResult(`r2-${testInfo.project.name}.json`, { day, ...result });
});
