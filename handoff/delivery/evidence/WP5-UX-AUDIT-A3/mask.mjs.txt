// Masks task-local paths, the user profile and the project path, strips ANSI codes and writes LF
// text. Usage: node mask.mjs <outDir> <suffix> <file>...  (each file is written to
// <outDir>/<basename><suffix>, for example ".txt" for JSON results)
import { readFileSync, writeFileSync } from 'node:fs';
import { basename, join } from 'node:path';

const [outDir, suffix, ...files] = process.argv.slice(2);
const rules = [
  [/[A-Za-z]:[\\/]+\.claude-tmp[\\/]+timesheet[\\/]+WP5-UX-AUDIT-A3/gi, '<task>'],
  [/\/d\/\.claude-tmp\/timesheet\/WP5-UX-AUDIT-A3/gi, '<task>'],
  [/[A-Za-z]:\\\\+\.claude-tmp\\\\+timesheet\\\\+WP5-UX-AUDIT-A3/gi, '<task>'],
  [/[A-Za-z]:[\\/]+\.claude-tmp[\\/]+claude[\\/]+[^\s"']+/gi, '<scratch>'],
  [/[A-Za-z]:[\\/]+Users[\\/]+[^\\/\s"']+/gi, '<user-home>'],
  [/\/c\/Users\/[^/\s"']+/gi, '<user-home>'],
  [/[A-Za-z]:[\\/]+Dropbox[\\/]+Work\.CDSemi[\\/]+timesheet/gi, '<project>'],
  [/\/d\/Dropbox\/Work\.CDSemi\/timesheet/gi, '<project>'],
  [/[A-Za-z0-9._%+-]+@(?!example\.invalid)[A-Za-z0-9.-]+\.[A-Za-z]{2,}/g, '<email>'],
];
const ansi = new RegExp(`${String.fromCharCode(27)}\\[[0-9;]*[A-Za-z]`, 'g');
for (const file of files) {
  let text = readFileSync(file, 'utf8').replace(ansi, '').replace(/\r\n?/g, '\n');
  for (const [pattern, replacement] of rules) text = text.replace(pattern, replacement);
  if (!text.endsWith('\n')) text += '\n';
  const name = basename(file).replace(/\.raw\.txt$/, '.txt') + suffix;
  writeFileSync(join(outDir, name), text, 'utf8');
  console.log(`masked ${name} (${Buffer.byteLength(text)} bytes)`);
}
