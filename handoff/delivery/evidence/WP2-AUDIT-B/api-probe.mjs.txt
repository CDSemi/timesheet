class Client {
  constructor() { this.cookie = ''; }
  async req(method, path, body, raw = false) {
    const headers = { origin };
    if (this.cookie) headers.cookie = this.cookie;
    if (body !== undefined) headers['content-type'] = 'application/json';
    const res = await fetch(`${origin}${path}`, { method, headers, body: body === undefined ? undefined : JSON.stringify(body) });
    const text = await res.text();
    if (raw) return { status: res.status, text, headers: res.headers };
    let json = null;
    try { json = JSON.parse(text); } catch {}
    return { status: res.status, json, text };
  }
  async login(email, password) {
    const res = await fetch(`${origin}/api/auth/login`, { method: 'POST', headers: { origin, 'content-type': 'application/json' }, body: JSON.stringify({ email, password }) });
    this.cookie = (res.headers.get('set-cookie') ?? '').split(';')[0];
    return res.status;
  }
}
