function cutEffect(text, label) {
  const start = text.indexOf(EFFECT_START);
  const end = text.indexOf(EFFECT_END, start);
  if (start < 0 || end < 0) {
    fail(`${label}: effect block not found`);
    return { rest: text, block: '' };
  }
  return { rest: text.slice(0, start) + '<EFFECT>\n' + text.slice(end + EFFECT_END.length), block: text.slice(start, end + EFFECT_END.length) };
}

function rules(css) {
  const out = [];
  const text = css.replace(/\/\*[\s\S]*?\*\//g, '');
  const stack = [];
  let buffer = '';
  for (const ch of text) {
    if (ch === '{') {
      stack.push(buffer.trim());
      buffer = '';
    } else if (ch === '}') {
      const body = buffer.trim();
      const selector = stack.pop();
      if (body !== '') out.push({ context: stack.join(' > '), selector, declarations: body.split(';').map((d) => d.trim()).filter((d) => d !== '') });
      buffer = '';
    } else buffer += ch;
  }
  return out;
}
