// Summarize run-cost / cost output files: per case worst ms and worst +MiB, overall worst, outcome counts, flags.
import { readFileSync } from 'node:fs';
const files = process.argv.slice(2);
const byCase = new Map();
let lines = 0;
for (const f of files) {
  for (const line of readFileSync(f, 'utf8').split('\n')) {
    const m = /^(.*?) \| upload=(\d+) B? ?\| (?:preview )?ms=(\d+).*?maxRSS (\d+) -> (\d+) MiB(?: \(\+(\d+)\))? \| (\S+)/.exec(line);
    if (!m) continue;
    lines += 1;
    const [, name, , ms, a, b, plus, outcome] = m;
    const d = plus !== undefined ? Number(plus) : Number(b) - Number(a);
    const e = byCase.get(name) ?? { runs: 0, ms: 0, mib: 0, outcomes: new Set() };
    e.runs += 1;
    e.ms = Math.max(e.ms, Number(ms));
    e.mib = Math.max(e.mib, d);
    e.outcomes.add(outcome.replace(/\(.*/, ''));
    byCase.set(name, e);
  }
}
let worstMs = ['', 0];
let worstMib = ['', 0];
let over = 0;
const outcomes = new Map();
for (const [name, e] of byCase) {
  if (e.ms > worstMs[1]) worstMs = [name, e.ms];
  if (e.mib > worstMib[1]) worstMib = [name, e.mib];
  if (e.ms > 500 || e.mib > 150) over += 1;
  for (const o of e.outcomes) outcomes.set(o, (outcomes.get(o) ?? 0) + 1);
}
console.log(`lines parsed ${lines}, cases ${byCase.size}`);
console.log(`worst time ${worstMs[1]} ms: ${worstMs[0].slice(0, 120)}`);
console.log(`worst memory +${worstMib[1]} MiB: ${worstMib[0].slice(0, 120)}`);
console.log(`cases over budget (500 ms or +150 MiB): ${over}`);
console.log(`outcomes: ${[...outcomes].map(([k, v]) => `${k}=${v}`).join(' ')}`);
