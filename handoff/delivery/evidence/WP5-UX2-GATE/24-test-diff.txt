+import { readFileSync } from 'node:fs';
+import { join } from 'node:path';
+import { describe, expect, it } from 'vitest';

+/*
+ * WP5-UX2-T01: the dark token set is written twice in src/client/styles.css, once under the device's
+ * preference (`@media (prefers-color-scheme: dark)`, when no manual choice is stored) and once under the
+ * manual choice (`:root[data-theme='dark']`). CSS has no way to share the two, so this test parses the
+ * stylesheet and fails when the sets differ, when a dark token has no light value in `:root`, or when a
+ * dark rule exists outside the token sets (it would not follow the manual choice).
+ */
+
+const STYLES = join(import.meta.dirname, '..', '..', 'src', 'client', 'styles.css');
+
+const MEDIA_SELECTOR = '@media (prefers-color-scheme: dark)';
+const MEDIA_INNER_SELECTOR = ":root:not([data-theme='light'])";
+const MANUAL_DARK_SELECTOR = ":root[data-theme='dark']";
+const MANUAL_LIGHT_SELECTOR = ":root[data-theme='light']";
+
+interface Block {
+  selector: string;
+  body: string;
+}
+
+function stripComments(css: string): string {
+  return css.replace(/\/\*[\s\S]*?\*\//g, '');
+}
+
+/** The blocks at the top nesting level of `css`: a rule or an at-rule with its raw body. */
+function topLevelBlocks(css: string): Block[] {
+  const blocks: Block[] = [];
+  let depth = 0;
+  let selectorStart = 0;
+  let bodyStart = 0;
+  let selector = '';
+  for (let index = 0; index < css.length; index += 1) {
+    const character = css[index];
+    if (character === '{') {
+      if (depth === 0) {
+        selector = css.slice(selectorStart, index).trim().replace(/\s+/g, ' ');
+        bodyStart = index + 1;
+      }
+      depth += 1;
+    } else if (character === '}') {
+      depth -= 1;
+      if (depth === 0) {
+        blocks.push({ selector, body: css.slice(bodyStart, index) });
+        selectorStart = index + 1;
+      }
+    }
+  }
+  return blocks;
+}
+
+/** The declarations of a rule body as name to whitespace-normalised value (`;` inside parentheses is not a separator). */
+function declarations(body: string): Record<string, string> {
+  const result: Record<string, string> = {};
+  let depth = 0;
+  let start = 0;
+  const take = (end: number) => {
+    const text = body.slice(start, end).trim();
+    const colon = text.indexOf(':');
+    if (text !== '' && colon > 0) result[text.slice(0, colon).trim()] = text.slice(colon + 1).trim().replace(/\s+/g, ' ');
+  };
+  for (let index = 0; index < body.length; index += 1) {
+    const character = body[index];
+    if (character === '(') depth += 1;
+    else if (character === ')') depth -= 1;
+    else if (character === ';' && depth === 0) {
+      take(index);
+      start = index + 1;
+    }
+  }
+  take(body.length);
+  return result;
+}
+
+function onlyTokens(set: Record<string, string>): Record<string, string> {
+  return Object.fromEntries(Object.entries(set).filter(([name]) => name.startsWith('--')));
+}
+
+interface ThemeSets {
+  light: Record<string, string>;
+  mediaDark: Record<string, string>;
+  manualDark: Record<string, string>;
+  manualLight: Record<string, string>;
+}
+
+function themeSets(css: string): ThemeSets {
+  const blocks = topLevelBlocks(stripComments(css));
+  const only = (selector: string, from: Block[]): Block => {
+    const matches = from.filter((block) => block.selector === selector);
+    expect(matches, `exactly one top-level ${selector}`).toHaveLength(1);
+    return matches[0] ?? { selector, body: '' };
+  };
+  const media = only(MEDIA_SELECTOR, blocks);
+  const inner = only(MEDIA_INNER_SELECTOR, topLevelBlocks(media.body));
+  return {
+    light: declarations(only(':root', blocks).body),
+    mediaDark: declarations(inner.body),
+    manualDark: declarations(only(MANUAL_DARK_SELECTOR, blocks).body),
+    manualLight: declarations(only(MANUAL_LIGHT_SELECTOR, blocks).body),
+  };
+}
+
+/** The names whose values differ between two token sets, or that exist in one only. */
+function differences(left: Record<string, string>, right: Record<string, string>): string[] {
+  const names = new Set([...Object.keys(left), ...Object.keys(right)]);
+  return [...names].filter((name) => left[name] !== right[name]).sort();
+}
+
+const css = readFileSync(STYLES, 'utf8');
+
+describe('the stylesheet parser used by the parity test', () => {
+  const sample = `
+    :root { --a: 1; --shadow: 0 0 0 1px rgb(1 2 3 / 0.4),
+      0 1px 2px rgb(1 2 3 / 0.3); }
+    @media (prefers-color-scheme: dark) { :root:not([data-theme='light']) { --a: 2; } }
+    /* a comment { with a brace */
+    :root[data-theme='dark'] { color-scheme: dark; --a: 3; }
+    :root[data-theme='light'] { color-scheme: light; }
+  `;
+
+  it('reads nested blocks and multi-line values', () => {
+    const sets = themeSets(sample);
+    expect(sets.light).toEqual({ '--a': '1', '--shadow': '0 0 0 1px rgb(1 2 3 / 0.4), 0 1px 2px rgb(1 2 3 / 0.3)' });
+    expect(sets.mediaDark).toEqual({ '--a': '2' });
+    expect(sets.manualDark).toEqual({ 'color-scheme': 'dark', '--a': '3' });
+  });
+
+  it('reports a changed, a missing and an extra token', () => {
+    expect(differences({ '--a': '1', '--b': '2' }, { '--a': '1', '--b': '2' })).toEqual([]);
+    expect(differences({ '--a': '1' }, { '--a': '2' })).toEqual(['--a']);
+    expect(differences({ '--a': '1', '--b': '2' }, { '--a': '1' })).toEqual(['--b']);
+    expect(differences({ '--a': '1' }, { '--a': '1', '--c': '3' })).toEqual(['--c']);
+  });
+});
+
+describe('src/client/styles.css theme sets', () => {
+  const sets = themeSets(css);
+
+  it('has a non-empty dark set', () => {
+    expect(Object.keys(onlyTokens(sets.mediaDark)).length).toBeGreaterThan(20);
+  });
+
+  it('writes the same dark tokens under the device preference and under the manual choice', () => {
+    expect(differences(onlyTokens(sets.mediaDark), onlyTokens(sets.manualDark))).toEqual([]);
+  });
+
+  it('holds only custom properties in the media set, and the colour scheme in the manual sets', () => {
+    expect(Object.keys(sets.mediaDark).filter((name) => !name.startsWith('--'))).toEqual([]);
+    expect(sets.manualDark['color-scheme']).toBe('dark');
+    expect(sets.manualLight).toEqual({ 'color-scheme': 'light' });
+    expect(sets.light['color-scheme']).toBe('light dark');
+  });
+
+  it('has a light value in :root for every dark token', () => {
+    const missing = Object.keys(onlyTokens(sets.mediaDark)).filter((name) => !(name in sets.light));
+    expect(missing).toEqual([]);
+  });
+
+  it('keeps every other dark rule inside the sets: no further prefers-color-scheme query', () => {
+    expect(stripComments(css).match(/prefers-color-scheme/g) ?? []).toHaveLength(1);
+  });
+});

+ * after today) and `in_progress` (the period that contains today) are built with the production functions, and the

+ * For the default schedule `current` differs from `in_progress` on five days of every cycle (the Monday after a period
+ * ends through its payroll Friday); a payroll exception that moves a payroll date moves that window with it.

+const LA = 'America/Los_Angeles';
+const SCHEDULE: PayrollScheduleRules = {
+  anchorPayrollDate: '2026-10-02',
+  cycleDays: 14,
+  periodStartOffsetDays: -18,
+  periodEndOffsetDays: -5,
+  dueOffsetDays: -3,
+  dueLocalTime: '17:00',

+const WALL_TIMES = ['00:00', '16:59', '17:00', '23:59'] as const;
+
+/** The period as the API returns it (`periodJson`): ISO dates as text. */
+function apiPeriod(period: PayPeriod): Period {
+  return {
+    payroll_date: period.payrollDate,
+    period_start: period.periodStart,
+    period_end: period.periodEnd,
+    due_local_date: period.dueLocalDate,
+    due_local_time: period.dueLocalTime,
+    due_at_utc: formatUtcInstant(period.dueAtUtc),
+  };
+}

+interface Step {
+  /** The reporting-zone date at that wall time: the server's `today_local`. */
+  today: string;
+  time: string;
+  periods: Period[];
+  current: Period;
+  inProgress: Period;
+}
+
+/** The server's answer for every wall time of every day in `[from, from + days)`, plus the periods around it. */
+function steps(from: string, days: number, exceptions: readonly PayrollException[] = []): Step[] {
+  const out: Step[] = [];
+  for (let offset = 0; offset < days; offset += 1) {
+    const date = addDays(from, offset);
+    for (const time of WALL_TIMES) {
+      const today = localDateOf(LA, resolveLocalDateTimeCompatible(`${date}T${time}`, LA));
+      const inProgress = payPeriodContaining(SCHEDULE, today, exceptions);
+      const current = currentPayPeriod(SCHEDULE, today, exceptions);
+      const around = [-1, 0, 1].map((shift) => payPeriodAt(SCHEDULE, inProgress.index + shift, exceptions));
+      out.push({ today, time, periods: around.map(apiPeriod), current: apiPeriod(current), inProgress: apiPeriod(inProgress) });
+    }
+  }
+  return out;
+}

+/*
+ * Colour maths and computed-style probes for the contrast checks of the e2e specs (WCAG 2.x relative
+ * luminance; SC 1.4.3 text, 1.4.11 non-text, 2.4.7 focus visible). The formulas are copied from
+ * focus-ring.spec.ts (WP5-UX-B4-01) so that spec stays untouched in this slice. Nothing here is
+ * computed by the application: every number is read from the browser's computed style.
+ */
+
+export type Rgba = [number, number, number, number];
+
+export interface Layer {
+  colour: Rgba;
+  inset: boolean;
+  spread: number;
+  /** The horizontal offset in px: the width of a one-sided rail such as `inset 3px 0 0 colour`. */
+  offsetX: number;
+  /** The vertical offset in px: the height of a one-sided bar such as `inset 0 3px 0 colour` (negative from the bottom). */
+  offsetY: number;
+}
+
+export function parseColour(text: string): Rgba {
+  const numbers = (/rgba?\(([^)]*)\)/.exec(text)?.[1] ?? '').split(/[\s,/]+/).filter((part) => part !== '').map(Number);
+  const [r = 0, g = 0, b = 0, a = 1] = numbers;
+  return [r, g, b, a];
+}
+
+export function parseLayers(boxShadow: string): Layer[] {
+  if (boxShadow === 'none') return [];
+  return boxShadow.split(/,\s*(?=rgba?\()/).map((layer) => {
+    const lengths = (layer.replace(/rgba?\([^)]*\)/, '').match(/-?\d+(\.\d+)?px/g) ?? []).map(parseFloat);
+    return { colour: parseColour(layer), inset: layer.includes('inset'), spread: lengths[3] ?? 0, offsetX: lengths[0] ?? 0, offsetY: lengths[1] ?? 0 };
+  });
+}
+
+/** The layer that touches the outside edge of the indicator: the widest outset layer, or the thinnest inset one. */
+export function outermost(layers: Layer[]): Layer | undefined {
+  const outset = layers.filter((layer) => !layer.inset);
+  if (outset.length > 0) return outset.reduce((a, b) => (b.spread > a.spread ? b : a));
+  return layers.length === 0 ? undefined : layers.reduce((a, b) => (b.spread < a.spread ? b : a));
+}
+
+export function over(top: Rgba, bottom: Rgba): Rgba {
+  const alpha = top[3] + bottom[3] * (1 - top[3]);
+  if (alpha === 0) return [0, 0, 0, 0];
+  const mix = (index: 0 | 1 | 2) => (top[index] * top[3] + bottom[index] * bottom[3] * (1 - top[3])) / alpha;
+  return [mix(0), mix(1), mix(2), alpha];
+}
+
+export function luminance([r, g, b]: Rgba): number {
+  const channel = (value: number) => {
+    const s = value / 255;
+    return s <= 0.03928 ? s / 12.92 : ((s + 0.055) / 1.055) ** 2.4;
+  };
+  return 0.2126 * channel(r) + 0.7152 * channel(g) + 0.0722 * channel(b);
+}
+
+export function contrast(a: Rgba, b: Rgba): number {
+  const [light, dark] = [luminance(a), luminance(b)].sort((x, y) => y - x);
+  return ((light ?? 0) + 0.05) / ((dark ?? 0) + 0.05);
+}

+/** The focus ring of a control as painted: ring and gap layers read from the computed shadow. */
+async function ringOf(locator: Locator): Promise<{ outside: number; insideGap: number; gapOnFill: number }> {
+  const boxShadow = await settledStyle(locator, 'boxShadow');
+  const surface = backdropOf(await backdropsAbove(locator));
+  const fill = await flattenedBackground(locator);
+  const outset = parseLayers(boxShadow).filter((layer) => !layer.inset && layer.spread > 0);
+  expect(outset.length, `a focus ring is painted (${boxShadow})`).toBeGreaterThanOrEqual(2);
+  const ring = outset.reduce((a, b) => (b.spread > a.spread ? b : a));
+  const gap = outset.reduce((a, b) => (b.spread < a.spread ? b : a));
+  const ringColour = over(ring.colour, surface);
+  const gapColour = over(gap.colour, surface);
+  return { outside: contrast(ringColour, surface), insideGap: contrast(ringColour, gapColour), gapOnFill: contrast(gapColour, fill) };
+}

+interface EditingSurfaces {
+  /** The colour of the rails (the frame on a phone). */
+  rail: Rgba;
+  /** What lies under a body cell: the card or the non-working tint, with the editing fill over it. */
+  bodies: Rgba[];
+  /** Opaque cells that the rails cross: the day head (desktop). */
+  heads: Rgba[];
+  /** The amber cell, with the fill over it on a phone (the fill is behind the cell on a desktop). */
+  attention: Rgba[];
+}
+
+/** The surfaces of the day open in the editor, composited from the computed styles of the cells and layers. */
+async function editingSurfaces(page: Page, desktop: boolean, workDate: string): Promise<EditingSurfaces> {
+  await page.mouse.move(0, 0);
+  if (desktop) {
+    const host = page.locator(`[data-day="${workDate}"]`);
+    const base = over(parseColour(await computedStyle(host, 'backgroundColor')), backdropOf(await backdropsAbove(host)));
+    const fill = parseColour((await readPseudo(host, '::before')).backgroundColor);
+    const rail = parseColour((await readPseudo(host, '::after')).borderStartColor);
+    const head = over(parseColour(await computedStyle(host.locator('.d-day'), 'backgroundColor')), base);
+    const attention = host.locator('.d-time.attention');
+    return {
+      rail,
+      bodies: [over(fill, base)],
+      heads: [head],
+      attention: (await attention.count()) === 0 ? [] : [over(parseColour(await computedStyle(attention, 'backgroundColor')), base)],
+    };
+  }
+  const cells = page.locator(`tr[data-day="${workDate}"] > td`);
+  const rails = parseLayers(await computedStyle(cells.first(), 'boxShadow')).filter((layer) => layer.colour[3] > 0);
+  const surfaces: Rgba[] = [];
+  const amber: Rgba[] = [];
+  for (const cell of await cells.all()) {
+    const under = over(parseColour(await computedStyle(cell, 'backgroundColor')), backdropOf(await backdropsAbove(cell)));
+    const surface = over(parseColour(await computedStyle(cell, 'backgroundImage')), under);
+    if (((await cell.getAttribute('class')) ?? '').includes('attention')) amber.push(surface);
+    else surfaces.push(surface);
+  }
+  // On a phone every cell that is not amber is a body cell (the fill is a background layer of the cell itself).
+  return { rail: rails[0]?.colour ?? [0, 0, 0, 0], bodies: surfaces, heads: [], attention: amber };
+}

+async function visibleLabel(review: Locator): Promise<string> {

+      const computed = (element.ownerDocument.defaultView ?? (() => { throw new Error('no window'); })()).getComputedStyle(target);
+      return { colour: computed.color, weight: computed.fontWeight };

+  const payrollValue = await style(payroll, 'b');
