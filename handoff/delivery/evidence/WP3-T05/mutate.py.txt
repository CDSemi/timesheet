"""Mutation checks for WP3-T05.

Each mutation is computed in memory first (every pattern must match exactly once); only then
is the file opened for writing. The two new test files are run, and the original bytes are
restored in a finally block and compared with the saved SHA-256. Usage (repository root,
NODE_EXE pointing at the Node 24 binary):
python mutate.py.txt <out-file>
"""

import os

import sys
from pathlib import Path

ROOT = Path.cwd()
TESTS = ['tests/integration/finalization.test.ts', 'tests/integration/finalization-concurrency.test.ts']

]


def sha(data: bytes) -> str:
    return hashlib.sha256(data).hexdigest()


def run_tests() -> tuple[int, str]:
    node = os.environ.get('NODE_EXE', 'node')
    result = subprocess.run([node, 'node_modules/vitest/vitest.mjs', 'run', *TESTS], capture_output=True, text=True, encoding='utf-8', errors='replace', cwd=ROOT)
    out = result.stdout + result.stderr
    summary = [line.strip() for line in out.splitlines() if line.strip().startswith(('Tests ', 'Test Files'))]
    failing = sorted({line.strip().split(' > ')[-1][:110] for line in out.splitlines() if line.strip().startswith('FAIL ') and ' > ' in line})
    detail = ' | '.join(summary)
    if failing:
        detail += '\n    failing: ' + '\n    failing: '.join(failing[:6])
    return result.returncode, detail


def main() -> int:
    out_path = Path(sys.argv[1])
    lines = []
    caught_total = 0
    names = sorted({name for _, name, _ in MUTATIONS})
    originals = {name: (ROOT / name).read_bytes() for name in names}
    baseline = {name: sha(data) for name, data in originals.items()}
    for name in names:
        lines.append(f'baseline sha256 {name} {baseline[name]}')
    code, summary = run_tests()
    lines.append(f'baseline exit={code} {summary}')
    if code != 0:
        out_path.write_text('\n'.join(lines) + '\n', encoding='utf-8', newline='\n')
        return 1
    for label, name, edits in MUTATIONS:
        original = originals[name].decode('utf-8')
        mutated = original
        ok = True
        for old, new in edits:
            if mutated.count(old) != 1:
                lines.append(f'{label}: PATTERN COUNT {mutated.count(old)} (not run)')
                ok = False
                break
            mutated = mutated.replace(old, new)
        if not ok:
            continue
        mutated_bytes = mutated.encode('utf-8')
        target = ROOT / name
        try:
            target.write_bytes(mutated_bytes)
            code, summary = run_tests()
        finally:
            target.write_bytes(originals[name])
        restored = sha(target.read_bytes()) == baseline[name]
        caught = code != 0
        caught_total += 1 if caught else 0
        lines.append(f'{label}: exit={code} {"CAUGHT" if caught else "SURVIVED"} restored={"byte-identical" if restored else "MISMATCH"}\n    {summary}')
        if not restored:
            out_path.write_text('\n'.join(lines) + '\n', encoding='utf-8', newline='\n')
            return 2
    code, summary = run_tests()
    lines.append(f'after restore exit={code} {summary}')
    lines.append(f'caught {caught_total} of {len(MUTATIONS)}')
    for name in names:
        lines.append(f'final sha256 {name} {sha((ROOT / name).read_bytes())}')
    out_path.write_text('\n'.join(lines) + '\n', encoding='utf-8', newline='\n')
    return 0 if caught_total == len(MUTATIONS) and code == 0 else 3


if __name__ == '__main__':
    sys.exit(main())
