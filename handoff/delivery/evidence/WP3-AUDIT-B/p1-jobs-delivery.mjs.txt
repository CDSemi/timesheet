// WP3-AUDIT-B probe 1: durable jobs, PDF/send fault injection and the uncertain-send decision
// on a real SQLite file. Synthetic data only; capture outbound only; children use process.execPath.
// Usage: node p1-jobs-delivery.mjs <repoDir> <workDir>            (main)
//        node p1-jobs-delivery.mjs <repoDir> <workDir> <role>     (child; started by main)
import { spawn, spawnSync } from 'node:child_process';
import { randomBytes } from 'node:crypto';
import { existsSync, mkdirSync, readdirSync, readFileSync, statSync, writeFileSync } from 'node:fs';
import { join } from 'node:path';

const [repoDir, workDir, role] = process.argv.slice(2);
const src = (p) => pathToFileURL(join(repoDir, 'src', p)).href;
const DB = join(workDir, 'p1.db');
const DATA = join(workDir, 'data');
const SENDER = '<email>';
const log = [];
const out = (line) => {
  log.push(line);
  console.log(line);
};
let failures = 0;
const check = (ok, label) => {
  if (!ok) failures += 1;
  out(`${ok ? 'PASS' : 'FAIL'} ${label}`);
};

const { openDatabase } = await import(src('server/db/database.ts'));
const { FileStore } = await import(src('server/files/fileStore.ts'));
const { runJobsOnce } = await import(src('server/jobs/runner.ts'));
const jobStore = await import(src('server/jobs/jobStore.ts'));
const { createPdfJobHandler } = await import(src('server/jobs/pdfJob.ts'));
const { createSendJobHandler } = await import(src('server/jobs/sendJob.ts'));
const { createOutboundAdapter } = await import(src('server/mail/outbound.ts'));

const fixed = (iso) => ({ now: () => new Date(iso) });
const kill = () => process.kill(process.pid, 'SIGKILL');

/* ------------------------------------------------------------------ children ---- */
if (role !== undefined) {
  const db = openDatabase(DB);
  const clock = fixed(process.env.P_NOW);
  const files = new FileStore(DATA);
  const hook = process.env.P_HOOK ?? '';
  const capture = createOutboundAdapter({ mode: 'capture' }, { dataDir: DATA });
  const handlers = {};
  if (role === 'pdf') {
    handlers.render_pdf = createPdfJobHandler({
      db,
      clock,
      files,
      hooks: {
        beforeWrite: () => hook === 'beforeWrite' && kill(),
        beforeRename: () => hook === 'beforeRename' && kill(),
        afterRename: () => hook === 'afterRename' && kill(),
      },
    });
  } else if (role === 'send') {
    const outbound =
      hook === 'afterCapture'
        ? { mode: 'capture', send: async (message, context) => { const result = await capture.send(message, context); kill(); return result; } }
        : capture;
    handlers.send_email = createSendJobHandler({
      db,
      clock,
      files,
      outbound,
      senderAddress: SENDER,
      hooks: { afterPrepare: () => hook === 'afterPrepare' && kill(), beforeSend: () => hook === 'beforeSend' && kill() },
    });
  } else if (role === 'enqueue') {
    for (let round = 0; round < 200; round += 1) {
      for (let k = 0; k < 10; k += 1) jobStore.enqueueJob(db, clock, { kind: 'probe_dup', businessKey: `probe:dup:${k}` });
    }
    db.close();
    process.exit(0);
  } else if (role === 'claimer') {
    const mine = [];
    handlers.probe_noop = async ({ job }) => {
      mine.push(job.id);
      await new Promise((resolve) => setTimeout(resolve, 5));
    };
    await runJobsOnce({ db, clock, handlers, maxJobs: 1000 });
    writeFileSync(join(workDir, `claims-${process.pid}.txt`), mine.join('\n'));
    db.close();
    process.exit(0);
  } else if (role === 'hang') {
    handlers.probe_hang = async () => kill();
  }
  const summary = await runJobsOnce({ db, clock, handlers });
  console.log(JSON.stringify(summary));
  db.close();
  process.exit(0);
}

/* ---------------------------------------------------------------------- main ---- */
mkdirSync(DATA, { recursive: true });
const runChild = (childRole, now, hook = '') => {
  const result = spawnSync(process.execPath, [process.argv[1], repoDir, workDir, childRole], {
    env: { ...process.env, P_NOW: now, P_HOOK: hook },
    encoding: 'utf8',
  });
  return { status: result.status, signal: result.signal, stdout: (result.stdout ?? '').trim(), stderr: (result.stderr ?? '').trim() };
};
const addSeconds = (iso, s) => new Date(new Date(iso).getTime() + s * 1000).toISOString().replace('.000Z', 'Z');

const passwords = { admin: randomBytes(18).toString('base64url'), employee: randomBytes(18).toString('base64url') };
await seedSynthetic(db, clock, { passwords });
const ORIGIN = 'http://localhost:4519';

const app = createApp({ db, clock, config, loginLimiter: new LoginRateLimiter(), staticDir: null, delivery }, { dataDir: DATA });
const request = async (method, path, { cookie, body, type } = {}) => {
  const headers = { origin: ORIGIN };
  if (cookie) headers.cookie = cookie;
  const init = { method, headers };
  if (body !== undefined) {
    headers['content-type'] = type ?? 'application/json';
    init.body = type === undefined ? JSON.stringify(body) : body;
  }
  const response = await app.request(path, init);
  const text = await response.text();
  let json = null;
  try {
    json = text === '' ? null : JSON.parse(text);
  } catch {
    json = text;
  }
  return { status: response.status, body: json, headers: response.headers };
};

// Saturday 2026-09-26 work: 3 h off-calendar OT; Sunday 2026-09-27: 2 h.

/* ---- A. PDF job: kill before write, before rename, after rename; then a normal run ---- */
let t = '2026-09-29T20:01:00Z';
for (const hook of ['beforeWrite', 'beforeRename', 'afterRename']) {
  const child = runChild('pdf', t, hook);
  const job = db.prepare("SELECT state, attempts FROM jobs WHERE kind = 'render_pdf'").get();
  check(child.status !== 0 && job.state === 'leased', `PDF child killed at ${hook} (exit ${child.status}); job left leased, attempts ${job.attempts}`);
  const early = runChild('pdf', addSeconds(t, 119));
  check(JSON.parse(early.stdout).claimed === 0, `PDF job not reclaimed before lease expiry (119 s after ${hook})`);
  t = addSeconds(t, 121);
}
const pdfRun = runChild('pdf', t);
check(JSON.parse(pdfRun.stdout).succeeded === 1, `PDF job succeeds after three kills: ${pdfRun.stdout}`);
const pdfRows = db.prepare("SELECT count(*) AS n FROM attachments WHERE kind = 'pdf'").get().n;
const fileRow = db.prepare("SELECT state FROM revision_files WHERE revision_id = ?").get(revisionId);
check(pdfRows === 1 && fileRow.state === 'ready', `one PDF attachment row (${pdfRows}) and revision file ready (${fileRow.state})`);
const walk = (dir) => readdirSync(dir, { withFileTypes: true }).flatMap((e) => (e.isDirectory() ? walk(join(dir, e.name)) : [join(dir, e.name)]));
const pdfKey = db.prepare("SELECT storage_key FROM attachments WHERE kind = 'pdf'").get().storage_key;
const allFiles = walk(DATA);
const tmpLeft = allFiles.filter((f) => f.endsWith('.tmp'));
check(allFiles.filter((f) => f.endsWith(pdfKey)).length === 1, 'exactly one stored PDF file for the revision key');

/* ---- B. Send: kill after prepare (nothing sent), restart sends once ---- */
const captureCount = () => (existsSync(join(DATA, 'mail-capture')) ? readdirSync(join(DATA, 'mail-capture')).filter((n) => !n.startsWith('.tmp')).length : 0);
t = addSeconds(t, 10);

child = runChild('send', t);
attempts = db.prepare('SELECT attempt_no, state FROM delivery_attempts ORDER BY attempt_no').all();

// Capture content equals the frozen snapshot.
const snapshot = JSON.parse(db.prepare('SELECT payload_json FROM timesheet_revisions WHERE id = ?').get(revisionId).payload_json);
const firstAttempt = db.prepare("SELECT id FROM delivery_attempts WHERE attempt_no = 1").get().id;

const eml = readFileSync(join(folder, 'message.eml'), 'utf8');
const pdfBytes = readFileSync(join(folder, 'attachment.pdf'));

check(JSON.stringify(meta.envelope.to) === JSON.stringify([...snapshot.recipients.to, ...snapshot.recipients.cc]) && meta.envelope.from === SENDER, 'captured envelope = snapshot To+Cc and MAIL_FROM');
check(eml.includes(`Subject: ${snapshot.recipients.subject}`), `captured subject equals the snapshot subject ("${snapshot.recipients.subject}")`);

/* ---- C. Resend, kill after `sending` committed -> uncertain, restart sends nothing ---- */
const resend1 = await request('POST', `/api/revisions/${revisionId}/resend`, { cookie, body: {} });

check(child.status !== 0 && a2.state === 'sending' && captureCount() === 1, `send killed after 'sending' commit: attempt sending, captures still 1`);
t = addSeconds(t, 121);
child = runChild('send', t);
a2 = db.prepare('SELECT id, state, provider_response FROM delivery_attempts WHERE id = ?').get(a2.id);
const j2 = db.prepare('SELECT state, last_error FROM jobs WHERE id = ?').get(resend1.body.job.id);

t = addSeconds(t, 3600);

/* ---- G. Retry delays 1/5/15/60 minutes, then intervention ---- */
let rt = '2026-10-02T00:00:00Z';
const failClock = { now: () => new Date(rt) };
jobStore.enqueueJob(db, failClock, { kind: 'probe_fail', businessKey: 'probe:fail:1' });
const failHandlers = { probe_fail: async () => { throw new jobStore.JobError('probe_failure'); } };
const delays = [];
for (let i = 0; i < 5; i += 1) {
  await runJobsOnce({ db, clock: failClock, handlers: failHandlers });
  const row = db.prepare("SELECT state, attempts, next_run_at, last_error FROM jobs WHERE business_key = 'probe:fail:1'").get();
  if (row.state === 'intervention') {
    delays.push(`intervention@${row.attempts}:${row.last_error}`);
    break;
  }
  const delay = (new Date(row.next_run_at) - new Date(rt)) / 1000;
  delays.push(delay);
  rt = addSeconds(row.next_run_at, -1);
  const early = await runJobsOnce({ db, clock: failClock, handlers: failHandlers });
  if (early.claimed !== 0) delays.push('claimed-early');
  rt = row.next_run_at;
}
check(JSON.stringify(delays) === JSON.stringify([60, 300, 900, 3600, 'intervention@5:probe_failure']), `retry delays ${JSON.stringify(delays)}`);

/* ---- H. Lease expiry: a runner that dies is reclaimed after the lease; the fifth expiry goes to intervention ---- */
let ht = '2026-10-03T00:00:00Z';
jobStore.enqueueJob(db, fixed(ht), { kind: 'probe_hang', businessKey: 'probe:hang:1' });
const hangLog = [];
for (let i = 0; i < 6; i += 1) {
  runChild('hang', ht);
  const row = db.prepare("SELECT state, attempts, last_error FROM jobs WHERE business_key = 'probe:hang:1'").get();
  hangLog.push(`${row.state}/${row.attempts}${row.last_error ? `/${row.last_error}` : ''}`);
  if (row.state === 'intervention') break;
  ht = addSeconds(ht, 120);
}
check(hangLog.at(-1) === 'intervention/5/lease_expired', `dying runner: ${hangLog.join(' -> ')}`);
