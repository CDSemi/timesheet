const request = async (method, path, { cookie, body, type } = {}) => {
  const headers = { origin: ORIGIN };
  if (cookie) headers.cookie = cookie;
  const init = { method, headers };
  if (body !== undefined) {
    headers['content-type'] = type ?? 'application/json';
    init.body = type === undefined ? JSON.stringify(body) : body;
  }
  const response = await app.request(path, init);
  const text = await response.text();
  let json = null;
  try {
    json = text === '' ? null : JSON.parse(text);
  } catch {
    json = text;
  }
  return { status: response.status, body: json, headers: response.headers };
};

const INDICATOR = /automat|pending|unsigned|not signed|review|deadline|system/i;
const readPdf = async (bytes) => {
  const task = pdfjs.getDocument({ data: new Uint8Array(bytes), verbosity: 0 });
  const doc = await task.promise;
  let text = '';
  let images = 0;
  for (let n = 1; n <= doc.numPages; n += 1) {
    const page = await doc.getPage(n);
    const content = await page.getTextContent();
    text += `${content.items.map((i) => i.str).join('\n')}\n`;
    const ops = await page.getOperatorList();
    images += ops.fnArray.filter((f) => f === pdfjs.OPS.paintImageXObject || f === pdfjs.OPS.paintInlineImageXObject).length;
  }
  await task.destroy();
  return { text, images };
};

const qpDecode = (text) => {
  const joined = text.replace(/=\r?\n/g, "");
  const bytes = [];
  for (let i = 0; i < joined.length; i += 1) {
    if (joined[i] === "=" && /^[0-9A-F]{2}$/.test(joined.slice(i + 1, i + 3))) {
      bytes.push(parseInt(joined.slice(i + 1, i + 3), 16));
      i += 2;
    } else bytes.push(...Buffer.from(joined[i], "utf8"));
  }
  return Buffer.from(bytes).toString("utf8");
};
const headerOf = (eml, name) => {
  const head = eml.split(/\r?\n\r?\n/)[0].replace(/\r?\n[ \t]+/g, " ");
  const raw = new RegExp("^" + name + ": (.*)$", "m").exec(head)?.[1] ?? "";
  return raw
    .replace(/\?= =\?UTF-8\?/gi, "?==?UTF-8?")
    .replace(/=\?UTF-8\?([QB])\?([^?]*)\?=/gi, (_, enc, data) =>
      enc.toUpperCase() === "B" ? Buffer.from(data, "base64").toString("utf8") : qpDecode(data.replace(/_/g, " ")));
};
