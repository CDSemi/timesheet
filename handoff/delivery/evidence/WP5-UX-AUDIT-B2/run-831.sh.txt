#!/usr/bin/env bash
# WP5-UX-AUDIT-B2: do the new e2e assertions of the fix round fail on the 831f760 behaviour?
# Scratch clone at 831f760 (task folder only), the 589bcff spec files copied over it, build, then only the new tests.
source <task>/env.sh
S="$T/run-831-summary.txt"
{
  echo "start: $(date -u +%Y-%m-%dT%H:%M:%SZ)"
  echo "node: $("$N24" --version)"
} > "$S"
if [ ! -d "$T/clone831/.git" ]; then
  git clone --no-hardlinks <project> "$T/clone831" > "$T/clone831.log" 2>&1
  echo "clone exit=$?" >> "$S"
fi
cd "$T/clone831" || exit 9
git checkout --detach 831f760838950a59f0e5c880f0bbefda15fe0c61 >> "$T/clone831.log" 2>&1
echo "checkout 831f760 exit=$? HEAD=$(git rev-parse HEAD)" >> "$S"
npmr ci > "$T/clone831-npm-ci.txt" 2>&1
echo "npm ci exit=$?" >> "$S"
for spec in timesheet.spec.ts day-editor.spec.ts import.spec.ts; do
  git show 589bcff5541a603abad696a3303dbea11cccb4a7:tests/e2e/$spec > "tests/e2e/$spec"
done
echo "copied 589bcff specs; git status --short:" >> "$S"
git status --short >> "$S"
npmr run build > "$T/clone831-build.txt" 2>&1
echo "build exit=$?" >> "$S"
export E2E_SCREENSHOT_DIR="$T/e2e-screens-831"
mkdir -p "$E2E_SCREENSHOT_DIR"
# The new fix-round tests only: B-01 (phone first screen), the period-card tab order, B-02 (768/1024/1280), B-03 (import).
"$N24" node_modules/@playwright/test/cli.js test \
  tests/e2e/timesheet.spec.ts tests/e2e/day-editor.spec.ts tests/e2e/import.spec.ts \
  -g "phone first screen|period card keeps its tab order|WP5-UX-B-02|the Import entry is for the signed-in person" \
  --retries=0 > "$T/run-831-e2e.txt" 2>&1
echo "e2e (589bcff tests on 831f760 client) exit=$?" >> "$S"
echo "end: $(date -u +%Y-%m-%dT%H:%M:%SZ)" >> "$S"
