/*
 * WP5-UX-AUDIT-B2 independent probes (auditor-written; not part of the reviewed source; kept out of git
 * through .git/info/exclude of the scratch clone). Synthetic data only. Each project starts its own built
 * server (dist/) on a fixed port of the task range (desktop 48001, mobile 48002) with DATA_DIR and
 * DATABASE_PATH inside the task folder, and stops it through the child handle. Each probe records what
 * it measured as JSON in PROBE_OUT; the few hard expectations only guard the measurement itself.
 */

const REPO = resolve(import.meta.dirname, '..');
const OUT = process.env.PROBE_OUT ?? join(process.cwd(), 'probe-out');
const WORK = process.env.PROBE_WORK ?? join(process.cwd(), 'probe-work');
const SHOTS = process.env.PROBE_SHOTS ?? OUT;
mkdirSync(OUT, { recursive: true });
mkdirSync(WORK, { recursive: true });
mkdirSync(SHOTS, { recursive: true });

function record(name: string, data: unknown) {
  writeFileSync(join(OUT, `${name}.json`), `${JSON.stringify(data, null, 2)}\n`);
}
function shot(name: string): string {
  if (!name.endsWith('-synthetic.png')) throw new Error('screenshot names end in -synthetic.png');
  return join(SHOTS, name);
}

interface ProbeServer {
  origin: string;
  employee: { email: string; password: string };
  admin: { email: string; password: string };
}

async function waitHealthy(origin: string, child: ChildProcess) {
  for (let attempt = 0; attempt < 150; attempt += 1) {
    if (child.exitCode !== null) throw new Error(`server exited early with ${child.exitCode}`);
    try {
      if ((await fetch(`${origin}/api/health`)).ok) return;
    } catch {
      // not listening yet
    }
    await new Promise((done) => setTimeout(done, 100));
  }
  throw new Error('server did not become healthy');
}

    async ({}, use, workerInfo) => {
      const port = workerInfo.project.name === 'desktop' ? 48001 : 48002;
      const work = mkdtempSync(join(WORK, `${workerInfo.project.name}-`));
      const origin = `http://127.0.0.1:${port}`;
      const employee = { email: 'employee@example.invalid', password: randomBytes(18).toString('base64url') };
      const admin = { email: 'admin@example.invalid', password: randomBytes(18).toString('base64url') };
      const env: NodeJS.ProcessEnv = {
        ...process.env,
        NODE_ENV: 'development',
        HOST: '127.0.0.1',
        PORT: String(port),
        DATABASE_PATH: join(work, 'timesheet.db'),
        DATA_DIR: join(work, 'private-data'),
        APP_ORIGINS: origin,
        OUTBOUND_MODE: 'capture',
        MAIL_FROM: 'timesheet-capture@example.invalid',

      for (const command of ['migrate', 'seed']) {
        const result = spawnSync(process.execPath, ['dist/server/cli.js', command], { cwd: REPO, env, encoding: 'utf8' });
        if (result.status !== 0) throw new Error(`cli ${command} failed: ${result.status}`);
      }
      const child = spawn(process.execPath, ['dist/server/index.js'], { cwd: REPO, env, stdio: 'ignore' });

/* ---- measurement helpers (run in the page) ---------------------------------------------------- */

function layout(page: Page) {
  return page.evaluate(() => {
    const r = (selector: string) => {
      const element = document.querySelector(selector);
      if (element === null) return null;
      const box = element.getBoundingClientRect();
      return { top: +box.top.toFixed(1), bottom: +box.bottom.toFixed(1), left: +box.left.toFixed(1), right: +box.right.toFixed(1), height: +box.height.toFixed(1), width: +box.width.toFixed(1) };
    };
    const firstDay = document.querySelector('[data-day]');
    return {
      viewport: { width: window.innerWidth, height: window.innerHeight },
      scrollY: window.scrollY,
      scrollWidth: document.documentElement.scrollWidth,
      clientWidth: document.documentElement.clientWidth,
      topBar: r('.shell-bar'),
      pageHeadH1: r('.page-head h1'),
      pageHeadVisible: (() => {
        const head = document.querySelector('.page-head');
        if (head === null) return null;
        const box = head.getBoundingClientRect();
        return box.width > 2 && box.height > 2;
      })(),
      periodCard: r('.period-bar'),
      periodPrev: r('.period-bar [aria-label="Previous period"]'),
      periodNext: r('.period-bar [aria-label="Next period"]'),
      periodTitle: r('.period-title'),
      periodMeta: r('.period-meta'),
      zoneNote: r('[data-zone-note]'),
      reviewLink: r('.period-bar [data-review-link]'),
      clockCard: r('.clock-panel'),
      clockText: document.querySelector('.clock-panel')?.textContent?.trim().replace(/\s+/g, ' ') ?? null,
      toolsCard: r('.sheet-area > .card'),
      tools: r('.tools'),
      openDayInput: r('.tools .open-day input'),
      formHead: r('.form-head'),
      formTitleVisible: (() => {
        const title = document.querySelector('.sheet .form-title');
        if (title === null) return null;
        const box = title.getBoundingClientRect();
        return box.width > 2 && box.height > 2;
      })(),
      weekBar: r('.week-bar'),
      firstDay: r('[data-day]'),
      firstDayName: firstDay?.getAttribute('aria-label') ?? null,
      secondDay: r('[data-day]:nth-of-type(2)'),
      tabBar: r('.shell-tabs'),
      visibleDayRowsAboveTabBar: (() => {
        const tabs = document.querySelector('.shell-tabs')?.getBoundingClientRect();
        const limit = tabs === undefined || tabs.height === 0 ? window.innerHeight : tabs.top;
        return [...document.querySelectorAll('[data-day]')].filter((row) => {
          const box = row.getBoundingClientRect();
          return box.top >= 0 && box.bottom <= limit + 0.5;
        }).length;
      })(),
    };
  });
}

interface Stop {
  tag: string;
  name: string;
  day: string | null;
  region: string;
  inDialog: boolean;
  box: { top: number; left: number; bottom: number; right: number } | null;
  hiddenByDialog: boolean;
}

function activeStop(page: Page): Promise<Stop> {
  return page.evaluate(() => {
    const active = document.activeElement as HTMLElement | null;
    if (active === null || active === document.body) return { tag: 'BODY', name: '', day: null, region: '', inDialog: false, box: null, hiddenByDialog: false };
    const dialog = document.querySelector('dialog[open]');
    const host = active.closest('[data-day], [data-review-day]');
    const region = active.closest('.shell-tabs') !== null ? 'tabbar' : active.closest('.shell-bar') !== null ? 'topbar' : active.closest('dialog') !== null ? 'dialog' : active.closest('.period-bar') !== null ? 'period' : active.closest('.clock-panel') !== null ? 'clock' : active.closest('.tools, .batch-bar') !== null ? 'tools' : active.closest('.sheet') !== null ? 'sheet' : 'page';
    const b = active.getBoundingClientRect();
    let hidden = false;
    if (dialog !== null && !dialog.contains(active)) {
      const p = dialog.getBoundingClientRect();
      hidden = b.left >= p.left && b.right <= p.right && b.top >= p.top && b.bottom <= p.bottom;
    }
    return {
      tag: active.tagName,
      name: (active.getAttribute('aria-label') ?? active.textContent ?? '').trim().replace(/\s+/g, ' ').slice(0, 60),
      day: host?.getAttribute('data-day') ?? host?.getAttribute('data-review-day') ?? null,
      region,
      inDialog: dialog !== null && dialog.contains(active),
      box: { top: +b.top.toFixed(1), left: +b.left.toFixed(1), bottom: +b.bottom.toFixed(1), right: +b.right.toFixed(1) },
      hiddenByDialog: hidden,
    };
  });
}

/** Tabs from the start of the document until the sequence repeats; page coordinates (scroll-corrected). */
async function tabSequence(page: Page, max = 260): Promise<Array<Stop & { pageTop: number }>> {
  await page.evaluate(() => {
    (document.activeElement as HTMLElement | null)?.blur();
    window.scrollTo(0, 0);
  });
  const stops: Array<Stop & { pageTop: number }> = [];
  for (let step = 0; step < max; step += 1) {
    await page.keyboard.press('Tab');
    const stop = await activeStop(page);
    const scrollY = await page.evaluate(() => window.scrollY);
    if (stops.length > 3 && stop.name === stops[0]?.name && stop.tag === stops[0]?.tag) break;
    stops.push({ ...stop, pageTop: +((stop.box?.top ?? 0) + scrollY).toFixed(1) });
  }
  return stops;
}

/* ---- PB2: phone keyboard order (tab order follows the visual reading order) ------------------- */

test.describe('PB2 keyboard order', () => {
  test.use({ locale: 'en-US', timezoneId: 'America/Los_Angeles' });
  test('PB2 full Tab sequence on the Timesheet page with positions', async ({ page, signIn }, testInfo) => {
    await signIn();
    await expect(page.locator('[data-day]')).toHaveCount(14);
    const stops = await tabSequence(page);
    // Reading order inside main: each stop starts at or below the previous one (8px tolerance for one row),
    // and on one row its left edge is to the right of the previous one.
    const inMain = stops.filter((stop) => stop.region !== 'tabbar' && stop.region !== 'topbar');
    const inversions: string[] = [];
    for (let index = 1; index < inMain.length; index += 1) {
      const previous = inMain[index - 1];
      const current = inMain[index];
      if (previous === undefined || current === undefined || previous.box === null || current.box === null) continue;
      const sameRow = Math.abs(current.pageTop - previous.pageTop) <= 8;
      const backwardsUp = current.pageTop + 8 < previous.pageTop;
      const backwardsLeft = sameRow && current.box.left + 1 < previous.box.left;
      // Within one day the picker and the day button share the row; on the desktop sheet the days are columns,
      // so a step back up to the next day's column top is the expected column order, recorded separately.
      if (backwardsUp || backwardsLeft) inversions.push(`${index}: ${previous.region}/${previous.name} (${previous.pageTop},${previous.box.left}) -> ${current.region}/${current.name} (${current.pageTop},${current.box.left})`);
    }
    const dayOrder = stops.filter((stop) => stop.day !== null).map((stop) => stop.day ?? '');
    const dayOrderNondecreasing = dayOrder.every((date, index) => index === 0 || date >= (dayOrder[index - 1] ?? ''));
    const periodStops = stops.filter((stop) => stop.region === 'period').map((stop) => stop.name);
    record(`PB2-tab-order-${testInfo.project.name}`, {
      count: stops.length,
      regions: [...new Set(stops.map((stop) => stop.region))],
      periodStops,
      dayOrderNondecreasing,
      firstStops: stops.slice(0, 14).map((stop) => `${stop.region} ${stop.tag} ${stop.name} @${stop.pageTop},${stop.box?.left}`),
      inversions,
      all: stops.map((stop) => `${stop.region}\t${stop.tag}\t${stop.name}\t${stop.pageTop}\t${stop.box?.left}`),
    });
  });
});

test.describe('PB3 phone widths', () => {
  test.skip(({ isMobile }) => !isMobile, 'phone only');
  test.use({ locale: 'en-US', timezoneId: 'Asia/Ho_Chi_Minh' });
  test('PB3 390, 360 and 320px: sideways scroll, overflowing elements, small targets, the tools row', async ({ page, signIn }) => {
    await signIn();
    await expect(page.locator('[data-day]')).toHaveCount(14);
    const result: Record<string, unknown> = {};
    for (const width of [390, 360, 320]) {
      await page.setViewportSize({ width, height: 844 });
      for (const state of ['default', 'batch']) {
        if (state === 'batch') await page.getByRole('button', { name: 'Change several days' }).click();

        const measured = await page.evaluate(() => {
          const vw = document.documentElement.clientWidth;

            .filter((element) => {
              const style = getComputedStyle(element);
              if (style.clipPath !== 'none' || style.display === 'contents') return false;
              const box = element.getBoundingClientRect();
              return box.width > 0 && box.right > vw + 0.5;
            })
            .slice(0, 12)
            .map((element) => `${element.tagName}.${(element.getAttribute('class') ?? '').replace(/\s+/g, '.')} right=${element.getBoundingClientRect().right.toFixed(1)}`);
          const controls = [...document.querySelectorAll('main button, main a[href], main input:not([type="checkbox"]), main select, main label.pick, nav.shell-tabs a, nav.shell-tabs button')].filter((element) => {
            const box = element.getBoundingClientRect();
            return box.width > 0 && box.height > 0 && getComputedStyle(element).visibility !== 'hidden';
          });
          const small = controls
            .filter((element) => {
              const box = element.getBoundingClientRect();
              return box.width < 44 || box.height < 44;
            })
            .map((element) => `${element.tagName} ${(element.getAttribute('aria-label') ?? element.textContent ?? '').trim().slice(0, 30)} ${element.getBoundingClientRect().width.toFixed(1)}x${element.getBoundingClientRect().height.toFixed(1)}`);
          const input = document.querySelector('.tools .open-day input')?.getBoundingClientRect();
          const toolsRight = [...document.querySelectorAll('.tools-right button')].map((button) => {
            const box = button.getBoundingClientRect();
            return `${(button.textContent ?? '').trim()} ${box.left.toFixed(1)}-${box.right.toFixed(1)} sw=${(button as HTMLElement).scrollWidth} cw=${(button as HTMLElement).clientWidth}`;
          });
          return {
            scrollWidth: document.documentElement.scrollWidth,
            clientWidth: vw,
            overflowing,
            controls: controls.length,
            small,
            openDayInputWidth: input === undefined ? null : +input.width.toFixed(1),
            toolsRight,
          };
        });
        result[`${width}-${state}`] = measured;
        if (width === 320) await page.screenshot({ path: shot(`pb3-phone-w320-${state}-synthetic.png`), animations: 'disabled' });
        if (state === 'batch') await page.getByRole('button', { name: 'Done' }).click();
      }
    }
    record('PB3-phone-widths', result);
  });
});

          panel: d === undefined ? null : { left: +d.left.toFixed(1), right: +d.right.toFixed(1), top: +d.top.toFixed(1), bottom: +d.bottom.toFixed(1) },
          sheetCard: c === undefined ? null : { left: +c.left.toFixed(1), right: +c.right.toFixed(1) },
          panelOverlapsSheetCard: d !== undefined && c !== undefined && d.left < c.right && d.right > c.left,
          firstDayHit: hit === null ? null : hit.closest('[data-day]') !== null ? 'sheet day' : hit.closest('dialog') !== null ? 'dialog' : hit.tagName,
          scrollWidth: document.documentElement.scrollWidth,
          clientWidth: document.documentElement.clientWidth,
        };
      });
      await page.screenshot({ path: shot(`pb4-editor-w${width}-synthetic.png`), animations: 'disabled' });
      // Keyboard walk both ways.
      const walk = { outside: 0, hidden: 0, hiddenNames: [] as string[], outsideRegions: [] as string[] };
      for (const key of ['Tab', 'Shift+Tab']) {
        for (let step = 0; step < 80; step += 1) {
          await page.keyboard.press(key);
          const stop = await activeStop(page);
          if (stop.tag !== 'BODY' && !stop.inDialog) {
            walk.outside += 1;
            if (!walk.outsideRegions.includes(stop.region)) walk.outsideRegions.push(stop.region);
          }
          if (stop.hiddenByDialog) {
            walk.hidden += 1;
            walk.hiddenNames.push(stop.name);
          }
        }
      }
      // Escape from inside the panel (a field), then focus return.
      await editor.getByLabel('Notes').focus();
      await page.keyboard.press('Escape');
      await expect(editor).toHaveCount(0);
      const returnAfterInside = await dayButton(dayA).evaluate((element) => element === document.activeElement);
      // Escape with focus on the sheet (another day's button), and on the "Open a day" field.
      await dayButton(dayA).click();
      await expect(editor).toBeVisible();
      await dayButton(dayB).focus();
      const focusMovedToSheet = await dayButton(dayB).evaluate((element) => element === document.activeElement);
      await page.keyboard.press('Escape');
      await expect(editor).toHaveCount(0);
      const returnAfterSheet = await dayButton(dayA).evaluate((element) => element === document.activeElement);
      await dayButton(dayA).click();
      await expect(editor).toBeVisible();
      await page.getByLabel('Open a day').focus();
      const focusMovedToOpenDay = await page.getByLabel('Open a day').evaluate((element) => element === document.activeElement);
      await page.keyboard.press('Escape');
      await expect(editor).toHaveCount(0);
      const returnAfterOpenDay = await dayButton(dayA).evaluate((element) => element === document.activeElement);
      // Close button.
      await dayButton(dayA).click();
      await expect(editor).toBeVisible();
      await editor.getByRole('button', { name: 'Close', exact: true }).click();
      await expect(editor).toHaveCount(0);
      const returnAfterClose = await dayButton(dayA).evaluate((element) => element === document.activeElement);
      result[String(width)] = { geometry, walk, focusMovedToSheet, focusMovedToOpenDay, returnAfterInside, returnAfterSheet, returnAfterOpenDay, returnAfterClose };
    }

    // Resize while open: 1280 -> 1024 -> 1280.
    await page.setViewportSize({ width: 1280, height: 800 });
    await dayButton(dayA).click();
    await expect(editor).toBeVisible();
    const before = await editor.evaluate((element) => element.matches(':modal'));
    await page.setViewportSize({ width: 1024, height: 800 });
    await page.waitForTimeout(300);
    const at1024 = await page.evaluate(() => {
      const dialog = document.querySelector('dialog[open]');
      return { open: dialog !== null, modal: dialog?.matches(':modal') ?? null, focusInDialog: dialog?.contains(document.activeElement) ?? false, active: document.activeElement?.tagName ?? null };
    });
    let outsideAfterResize = 0;
    for (let step = 0; step < 30; step += 1) {
      await page.keyboard.press('Tab');
      const stop = await activeStop(page);
      if (stop.tag !== 'BODY' && !stop.inDialog) outsideAfterResize += 1;
    }
    await page.setViewportSize({ width: 1280, height: 800 });
    await page.waitForTimeout(300);
    const back = await page.evaluate(() => {
      const dialog = document.querySelector('dialog[open]');
      return { open: dialog !== null, modal: dialog?.matches(':modal') ?? null, withEditor: document.querySelector('.sheet-area.with-editor') !== null };
    });
    await page.keyboard.press('Escape');
    await expect(editor).toHaveCount(0);
    const returnAfterResize = await dayButton(dayA).evaluate((element) => element === document.activeElement);
    result.resize = { before1280Modal: before, at1024, outsideAfterResize, back, returnAfterResize };
    record('PB4-editor-modes', { dayA, dayB, result });
  });
});

/* ---- PB7: rendered contrast of the compact phone text, light and dark --------------------------- */

test.describe('PB7 contrast', () => {
  test.skip(({ isMobile }) => !isMobile, 'phone only');
  test.use({ locale: 'en-US', timezoneId: 'Asia/Ho_Chi_Minh' });
  test('PB7 period card, clock, tools and form header text', async ({ page, signIn }) => {
    await signIn();
    await expect(page.locator('[data-day]')).toHaveCount(14);
    const result: Record<string, unknown> = {};
    for (const scheme of ['light', 'dark'] as const) {
      await page.emulateMedia({ colorScheme: scheme });
      await page.waitForTimeout(400);
      result[scheme] = await page.evaluate(() => {
        const parse = (value: string) => {
          const m = /rgba?\(([\d.]+)[, ]+([\d.]+)[, ]+([\d.]+)(?:[,/ ]+([\d.]+))?\)/.exec(value);
          return m === null ? null : { r: +m[1]!, g: +m[2]!, b: +m[3]!, a: m[4] === undefined ? 1 : +m[4] };
        };
        const lum = (c: { r: number; g: number; b: number }) => {
          const f = (v: number) => {
            const s = v / 255;
            return s <= 0.03928 ? s / 12.92 : ((s + 0.055) / 1.055) ** 2.4;
          };
          return 0.2126 * f(c.r) + 0.7152 * f(c.g) + 0.0722 * f(c.b);
        };
        const background = (element: Element | null): { r: number; g: number; b: number } => {
          for (let node = element; node !== null; node = node.parentElement) {
            const bg = parse(getComputedStyle(node).backgroundColor);
            if (bg !== null && bg.a >= 0.99) return bg;
          }
          return { r: 255, g: 255, b: 255 };
        };
        const rows: Array<{ text: string; ratio: number; size: string; weight: string }> = [];
        const walker = document.createTreeWalker(document.querySelector('main') ?? document.body, NodeFilter.SHOW_TEXT);
        for (let node = walker.nextNode(); node !== null; node = walker.nextNode()) {
          const text = (node.textContent ?? '').trim();
          const parent = node.parentElement;
          if (text === '' || parent === null) continue;
          if (parent.closest('.period-bar, .clock-panel, .tools, .form-head, .week-bar') === null) continue;
          const style = getComputedStyle(parent);
          const box = parent.getBoundingClientRect();
          if (box.width <= 2 || box.height <= 2 || style.visibility === 'hidden') continue;
          const fg = parse(style.color);
          if (fg === null) continue;
          const bg = background(parent);
          const blended = { r: fg.r * fg.a + bg.r * (1 - fg.a), g: fg.g * fg.a + bg.g * (1 - fg.a), b: fg.b * fg.a + bg.b * (1 - fg.a) };
          const [l1, l2] = [lum(blended), lum(bg)].sort((x, y) => y - x);
          rows.push({ text: text.slice(0, 40), ratio: +((l1! + 0.05) / (l2! + 0.05)).toFixed(2), size: style.fontSize, weight: style.fontWeight });
        }
        rows.sort((x, y) => x.ratio - y.ratio);
        return { count: rows.length, min: rows[0]?.ratio ?? null, below45: rows.filter((row) => row.ratio < 4.5).length, lowest: rows.slice(0, 8) };
      });
    }
    record('PB7-contrast-phone', result);
  });
});
