let lastInstant = 0;
const runs = [];
/** run-jobs --once --now at a monotonic instant. */
function runJobsAt(date, label = '') {
  if (date.getTime() < lastInstant) throw new Error(`non-monotonic run-jobs instant ${utc(date)}`);
  lastInstant = date.getTime();
  const r = cli(['run-jobs', '--once', '--now', utc(date)], `run-jobs ${utc(date)} ${label}`);
  let summary = null;
  try {
    summary = JSON.parse(r.stdout);
  } catch {
    summary = null;
  }
  runs.push({ at: utc(date), label, status: r.status, summary });
  return { status: r.status, summary };
}
function drain(start, offsets = [0, 120, 240, 600, 720, 840], label = '') {
  return offsets.map((offset) => runJobsAt(new Date(start.getTime() + offset * 1000), label));
}

async function session(c, date, start, end, breaks, endDate = date, reason) {
  const body = {
    start: at(date, start),
    end: at(endDate, end),
    input_zone: ZONE,
    breaks_confirmed: breaks !== null,
    breaks: (breaks ?? []).map(([from, to]) => ({ start: at(date, from), end: at(date, to), counts_as_work: false })),
    ...(reason === undefined ? {} : { reason }),
  };
  const r = await c.call('POST', `/api/days/${date}/sessions`, body);
  if (r.status !== 201) throw new Error(`session ${date} ${start}-${end}: ${r.status} ${r.text}`);
  return r.json;
}
function dbState() {
  const tables = monitor.prepare("SELECT name FROM sqlite_master WHERE type = 'table' AND name NOT LIKE 'sqlite_%' ORDER BY name").pluck().all();
  return JSON.stringify({ v: monitor.pragma('data_version', { simple: true }), rows: tables.map((n) => [n, monitor.prepare(`SELECT count(*) FROM "${n}"`).pluck().get()]) });
}
const count = (sql, ...args) => monitor.prepare(sql).pluck().get(...args);

  rec.check('AC14-5', 'server log and CLI outputs contain no password, session cookie or setup token (token only in the bootstrap terminal output)', leaked.length === 0 && !tokenOutsideBootstrap && !/password/i.test(server.output), `leaked ${leaked.length}; server log lines ${server.output.split('\n').length}`);
  writeFileSync(join(runDir, 'server-log.txt'), rec.mask(server.output));
  writeFileSync(join(runDir, 'cli-outputs.txt'), rec.mask(cliOutputs.join('\n')));

  // Capture listing (synthetic recipients only).
  const listing = listCaptures(dataDir).map((c) => {
    const m = c.eml === '' ? null : plainBody(c.eml);
    return { attempt: c.id, files: c.files, from: c.metadata?.envelope?.from ?? null, to: c.metadata?.envelope?.to ?? null, subject: m?.subject ?? null, headers: m?.headers ?? null, pdf_sha256: c.pdfSha256, body: m?.text ?? null };
  });
  writeFileSync(join(runDir, 'capture-listing.json'), `${JSON.stringify(listing, null, 2)}\n`);
  writeFileSync(join(runDir, 'runs.json'), `${JSON.stringify(runs, null, 2)}\n`);
}

try {
  await main();
} catch (error) {
  const cause = error instanceof Error && error.cause !== undefined ? ` cause: ${error.cause instanceof Error ? `${error.cause.name} ${error.cause.code ?? ''} ${error.cause.message}` : String(error.cause)}` : '';
  rec.check('PROBE', 'probe ran to the end without an unexpected error', false, error instanceof Error ? `${error.message}${cause}\n${error.stack}` : String(error));
} finally {
  try {
    monitor?.close();
  } catch {
    // closed
  }
  await server.stop();
  writeFileSync(join(runDir, 'server-log-final.txt'), rec.mask(server.output));
  rec.check('NET-1', 'every client fetch error was a keep-alive reset (ECONNRESET) on an idempotent read after an idle gap, retried once successfully', Client.fetchErrors.every((e) => e.code === 'ECONNRESET' && (e.method === 'GET' || e.method === 'HEAD')), JSON.stringify(Client.fetchErrors));
  const { pass, fail } = rec.write();
  console.log(`probe ${runDir.replace(T, '<task>')}: pass ${pass} fail ${fail}`);
  process.exitCode = fail === 0 ? 0 : 1;
}
