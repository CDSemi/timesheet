// Self-stopping CPU burner (the WP5-UX5-DIAG10 one, unchanged). Usage: node burn.mjs <threads> <seconds>
// Each worker thread spins until the deadline and then ends by itself; the main thread exits when all have ended.
import { Worker, isMainThread, workerData } from 'node:worker_threads';

if (isMainThread) {
  const threads = Number(process.argv[2]);
  const seconds = Number(process.argv[3]);
  if (!Number.isInteger(threads) || threads < 1 || !Number.isFinite(seconds) || seconds < 1 || seconds > 1800) {
    throw new Error('usage: node burn.mjs <threads 1..> <seconds 1..1800>');
  }
  const deadline = Date.now() + seconds * 1000;
  console.log(`burn: ${threads} threads until ${new Date(deadline).toISOString()}`);
  let alive = threads;
  for (let i = 0; i < threads; i += 1) {
    const worker = new Worker(new URL(import.meta.url), { workerData: { deadline } });
    worker.on('exit', () => {
      alive -= 1;
      if (alive === 0) console.log('burn: done');
    });
  }
} else {
  const { deadline } = workerData;
  let x = 0;
  while (Date.now() < deadline) {
    for (let i = 0; i < 1e6; i += 1) x += Math.sqrt(i);
  }
  if (x < 0) console.log(x);
}
