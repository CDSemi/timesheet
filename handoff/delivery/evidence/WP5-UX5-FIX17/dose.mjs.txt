// One controlled-load run: a self-stopping CPU burner (when threads > 0), two census samples, the Playwright run, one census sample.
// Usage: EXP=<export> node dose.mjs <label> <burnerThreads> <burnerSeconds> <instant|real> <specFile> [playwright args...]
// Env GREP selects the tests, PROJECT the project, REPS the repeat count. The burner ends by itself; this driver waits for it.
import { spawn } from 'node:child_process';

const [label, threads, seconds, instant, spec, ...extra] = process.argv.slice(2);
const T = '/d/.claude-tmp/timesheet/WP5-UX5-FIX17';
const bash = 'C:\\Program Files\\Git\\bin\\bash.exe';

function run(command, args, env = process.env) {
  return new Promise((resolve) => {
    const child = spawn(command, args, { stdio: 'inherit', env });
    child.on('exit', (code) => resolve(code ?? -1));
  });
}

let burnerDone = Promise.resolve(0);
if (Number(threads) > 0) {
  burnerDone = run(process.execPath, [`${T.replace('/d/', 'D:/')}/burn.mjs`, threads, seconds]);
}
await run(bash, [`${T}/census.sh`, `${label}-p1`]);
await run(bash, [`${T}/census.sh`, `${label}-p2`]);
const args = [`${T}/run.sh`, label, instant, spec];
if (process.env.PROJECT) args.push(`--project=${process.env.PROJECT}`);
if (process.env.GREP) args.push('--grep', process.env.GREP);
if (process.env.REPS) args.push(`--repeat-each=${process.env.REPS}`);
args.push('--workers=6', ...extra);
const out = await new Promise((resolve) => {
  const child = spawn(bash, args, { stdio: 'inherit', env: process.env });
  child.on('exit', (code) => resolve(code ?? -1));
});
await run(bash, [`${T}/census.sh`, `${label}-p3`]);
await burnerDone;
console.log(`dose ${label}: wrapper exit ${out}`);
