# WP3-T12 mutation checks for src/client/components/reviewModel.ts.
# Usage: python mutate.py.txt <vitest.mjs path>
# Each mutation is computed in memory before the file is opened for writing; the file is restored
# byte-identical (SHA-256 compared) after every run.
import hashlib, subprocess, sys

TARGET = 'src/client/components/reviewModel.ts'
VITEST = sys.argv[1]
original = open(TARGET, 'rb').read()
base_hash = hashlib.sha256(original).hexdigest()
text = original.decode('utf-8')

NL = chr(10)
MUTATIONS = [
    ('M1 signer name not required', "if (name === '') blockers.push", "if (name === '\u0000') blockers.push"),
    ('M2 stale_review no longer a stale review', "['stale_version', 'stale_review', 'not_finalized', 'no_pending_review']", "['stale_version', 'not_finalized', 'no_pending_review']"),
    ('M3 reviewed hash not sent', "    reviewed_hash: review.payload_hash,\n", ""),
    ('M4 automatic review-pending shown as manual', "revision.review_state === 'pending') {" + NL + "    return { key: 'submitted_auto_pending'", "revision.review_state === 'signed') {" + NL + "    return { key: 'submitted_auto_pending'"),
    ('M5 every deficit mode needs a choice', "item.mode === 'choose_at_signoff');\n}", "item.mode !== '');\n}"),
    ('M6 deep link accepts trailing text', r"/^\d{4}-\d{2}-\d{2}$/.test(date) && isCivilDate(date)", r"/^\d{4}-\d{2}-\d{2}/.test(date) && isCivilDate(date.slice(0, 10))"),
    ('M7 acknowledgement not required', "payload.unresolved_inputs.length > 0 && !form.acknowledged", "payload.unresolved_inputs.length > 0 && form.acknowledged === undefined"),
    ('M8 stale reset keeps old choices', "return { ...form, acknowledged: false, choices: {} };", "return { ...form, acknowledged: false };"),
]

caught = 0
for name, old, new in MUTATIONS:
    assert text.count(old) == 1, f'{name}: pattern found {text.count(old)} times'
    mutated = text.replace(old, new, 1).encode('utf-8')
    assert mutated != original
    try:
        open(TARGET, 'wb').write(mutated)
        run = subprocess.run(['node.exe', VITEST, 'run', 'tests/client/reviewModel.test.ts'], capture_output=True, text=True, encoding='utf-8', errors='replace')
    finally:
        open(TARGET, 'wb').write(original)
    restored = hashlib.sha256(open(TARGET, 'rb').read()).hexdigest() == base_hash
    failed = [l.strip() for l in run.stdout.splitlines() if l.strip().startswith('FAIL')]
    summary = [l.strip() for l in run.stdout.splitlines() if l.strip().startswith('Tests')]
    ok = run.returncode != 0
    caught += ok
    print(f'{name}: {"CAUGHT" if ok else "SURVIVED"}; restored byte-identical: {restored}; {summary[0] if summary else ""}')
    for line in failed[:3]:
        print('    ' + line)
print(f'caught {caught} of {len(MUTATIONS)}')
sys.exit(0 if caught == len(MUTATIONS) else 1)
