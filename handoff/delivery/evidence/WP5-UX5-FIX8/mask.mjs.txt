// Usage: node mask.mjs <outDir> <file>...   Copies each file to <outDir>/<basename>.txt as masked LF text.
import { mkdirSync, readFileSync, writeFileSync } from 'node:fs';
import { basename, join } from 'node:path';

const [outDir, ...files] = process.argv.slice(2);
mkdirSync(outDir, { recursive: true });
const rules = [
  [/D:[\\/]+\.claude-tmp[\\/]+timesheet[\\/]+WP5-UX5-FIX8/gi, '<task>'],
  [/[\/]d[\/]\.claude-tmp[\/]timesheet[\/]WP5-UX5-FIX8/g, '<task>'],
  [/D:[\\/]+Dropbox[\\/]+Work\.CDSemi[\\/]+timesheet/gi, '<repo>'],
  [/[\/]d[\/]Dropbox[\/]Work\.CDSemi[\/]timesheet/g, '<repo>'],
  [/C:[\\/]+Users[\\/]+[^\\/\s'"]+/gi, '<home>'],
  [/\u001b\[[0-9;]*m/g, ''],
  [/[A-Za-z0-9._%+-]+@(?!example\.invalid)[A-Za-z0-9.-]+\.[A-Za-z]{2,}/g, '<email>'],
];
for (const file of files) {
  let text = readFileSync(file, 'utf8').replace(/\r\n?/g, '\n');
  for (const [pattern, replacement] of rules) text = text.replace(pattern, replacement);
  const name = basename(file).replace(/\.txt$/, '');
  writeFileSync(join(outDir, `${name}.txt`), text);
}
