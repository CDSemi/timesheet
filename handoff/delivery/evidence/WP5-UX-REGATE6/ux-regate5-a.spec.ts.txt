/* ---- colour maths ---------------------------------------------------------------------------- */
type Rgba = [number, number, number, number];
interface Layer {
  colour: Rgba;
  inset: boolean;
  spread: number;
}
function parseColour(text: string): Rgba {
  const numbers = (/rgba?\(([^)]*)\)/.exec(text)?.[1] ?? '').split(/[\s,/]+/).filter((part) => part !== '').map(Number);
  const [r = 0, g = 0, b = 0, a = 1] = numbers;
  return [r, g, b, a];
}
function parseLayers(boxShadow: string): Layer[] {
  if (boxShadow === 'none') return [];
  return boxShadow.split(/,\s*(?=rgba?\()/).map((layer) => {
    const lengths = (layer.replace(/rgba?\([^)]*\)/, '').match(/-?\d+(\.\d+)?px/g) ?? []).map(parseFloat);
    return { colour: parseColour(layer), inset: layer.includes('inset'), spread: lengths[3] ?? 0 };
  });
}
function outermost(layers: Layer[]): Layer | undefined {
  const outset = layers.filter((layer) => !layer.inset);
  if (outset.length > 0) return outset.reduce((a, b) => (b.spread > a.spread ? b : a));
  return layers.length === 0 ? undefined : layers.reduce((a, b) => (b.spread < a.spread ? b : a));
}
function over(top: Rgba, bottom: Rgba): Rgba {
  const alpha = top[3] + bottom[3] * (1 - top[3]);
  if (alpha === 0) return [0, 0, 0, 0];
  const mix = (index: 0 | 1 | 2) => (top[index] * top[3] + bottom[index] * bottom[3] * (1 - top[3])) / alpha;
  return [mix(0), mix(1), mix(2), alpha];
}
function luminance([r, g, b]: Rgba): number {
  const channel = (value: number) => {
    const s = value / 255;
    return s <= 0.03928 ? s / 12.92 : ((s + 0.055) / 1.055) ** 2.4;
  };
  return 0.2126 * channel(r) + 0.7152 * channel(g) + 0.0722 * channel(b);
}
function contrast(a: Rgba, b: Rgba): number {
  const [light, dark] = [luminance(a), luminance(b)].sort((x, y) => y - x);
  return ((light ?? 0) + 0.05) / ((dark ?? 0) + 0.05);
}
function composite(nearestFirst: string[]): Rgba {
  return nearestFirst.reduceRight<Rgba>((below, css) => over(parseColour(css), below), [255, 255, 255, 1]);
}
function hex([r, g, b]: Rgba): string {
  return `#${[r, g, b].map((v) => Math.round(v).toString(16).padStart(2, '0')).join('')}`;
}

async function settledShadow(locator: Locator): Promise<string> {
  const read = async () => locator.evaluate((element) => element.ownerDocument.defaultView?.getComputedStyle(element).boxShadow ?? 'none');
  let settled = await read();
  await expect
    .poll(
      async () => {
        const first = await read();
        await locator.page().waitForTimeout(400);
        settled = await read();
        return first === settled;
      },
      { message: 'the box-shadow is settled', timeout: 7000 },
    )
    .toBe(true);
  return settled;
}
async function backdropsOf(locator: Locator, includeSelf: boolean): Promise<string[]> {
  return locator.evaluate((element, self) => {
    const view = element.ownerDocument.defaultView;
    const backdrops: string[] = [];
    for (let node = self ? element : element.parentElement; node; node = node.parentElement) backdrops.push(view?.getComputedStyle(node).backgroundColor ?? '');
    return backdrops;
  }, includeSelf);
}
/** Outer ring (outset) or edge ring (inset) against the adjacent surface(s); returns the lowest ratio and the description. */
function ringRatio(boxShadow: string, surfaces: string[][], mode: 'outer' | 'inset' = 'outer'): { ratio: number; text: string } {
  const layers = parseLayers(boxShadow);
  const insetLayers = layers.filter((layer) => layer.inset && layer.spread > 0);
  const ring = mode === 'inset' ? (insetLayers.length === 0 ? undefined : insetLayers.reduce((a, b) => (b.spread < a.spread ? b : a))) : outermost(layers);
  if (ring === undefined || ring.spread <= 0) return { ratio: 0, text: `no ring (${boxShadow})` };
  let lowest = Number.POSITIVE_INFINITY;
  let shown = '';
  for (const backdrops of surfaces) {
    const surface = composite(backdrops);
    const painted = over(ring.colour, over(surface, [255, 255, 255, 1]));
    const ratio = contrast(painted, surface);
    if (ratio < lowest) {
      lowest = ratio;
      shown = `${ring.inset ? 'inset' : 'outset'} ${ring.spread}px ${hex(painted)} on ${hex(surface)}`;
    }
  }
  return { ratio: lowest, text: shown };
}

const ME = 'Synthetic Keyboard User';
const OWNER = 'Synthetic Sharing Owner';
interface World {
  me: Person;
  owner: Person;
  complete: string;
}
async function newWorld(adminSeed: SeedClient, server: BuiltServer, options: { mutual?: boolean } = {}): Promise<World> {
  const me = await newPerson(adminSeed, server, { displayName: ME, signature: 'none' });
  const owner = await newPerson(adminSeed, server, { displayName: OWNER, signature: 'none' });
  const [complete = '', unconfirmed = ''] = await me.api.displayedPeriodFreeWorkdays();
  expect(complete, 'a past free workday').not.toBe('');
  await me.api.seedCompleteDay(complete);
  if (unconfirmed !== '') await me.api.seedUnconfirmedBreaksDay(unconfirmed);
  const [ownerDay = ''] = await owner.api.displayedPeriodFreeWorkdays();
  if (ownerDay !== '') await owner.api.seedCompleteDay(ownerDay);
  const items = { timesheets: 'view', ot_read: false, pdf_download: false };

  if (options.mutual === true) await me.api.call('POST', '/api/shares', { grantee_email: owner.account.email, items }, 201);
  return { me, owner, complete };
}

/* ---- AX-01: Tab and Shift+Tab walks ---------------------------------------------------------- */
interface Stop {
  key: number;
  name: string;
  visible: number;
  inside: boolean;
  covers: string[];
}
async function focusedStop(page: Page, scope: string): Promise<Stop | null> {
  return page.evaluate((scopeSelector) => {
    const w = window as unknown as { walkKeys?: WeakMap<Element, number>; walkNext?: number };
    const active = document.activeElement;
    if (!(active instanceof HTMLElement) || active === document.body) return null;
    w.walkKeys = w.walkKeys ?? new WeakMap();
    let key = w.walkKeys.get(active);
    if (key === undefined) {
      w.walkNext = (w.walkNext ?? 0) + 1;
      key = w.walkNext;
      w.walkKeys.set(active, key);
    }
    const labels = active instanceof HTMLInputElement || active instanceof HTMLSelectElement ? [...(active.labels ?? [])] : [];
    const rect = active.getBoundingClientRect();
    let visible = 0;
    const covers = new Set<string>();
    for (let i = 0; i < 5; i += 1) {
      for (let j = 0; j < 5; j += 1) {
        const x = rect.left + 1 + ((rect.width - 2) * (i + 0.5)) / 5;
        const y = rect.top + 1 + ((rect.height - 2) * (j + 0.5)) / 5;
        if (x < 0 || y < 0 || x >= window.innerWidth || y >= window.innerHeight) {
          covers.add('outside the viewport');
          continue;
        }
        const hit = document.elementFromPoint(x, y);
        if (hit !== null && (active.contains(hit) || labels.some((label) => label.contains(hit)))) visible += 1;
        else if (hit !== null) {
          const cover = hit.closest('.shell-bar, .shell-tabs, .share-bar, .editor-head, dialog') ?? hit;
          covers.add(`${cover.tagName.toLowerCase()}.${String(cover.className).split(' ')[0]}`);
        }
      }
    }
    const name = (active.getAttribute('aria-label') ?? active.textContent ?? '').replace(/\s+/g, ' ').trim().slice(0, 60);
    return { key, name: `${active.tagName.toLowerCase()} "${name}"`, visible, covers: [...covers], inside: active.closest(scopeSelector) !== null };
  }, scope);
}
async function focusEdge(page: Page, scope: string, which: 'first' | 'last'): Promise<void> {
  const found = await page.evaluate(
    ([scopeSelector, edge]) => {
      const root = document.querySelector(scopeSelector as string);
      const candidates = root === null ? [] : [...root.querySelectorAll<HTMLElement>('a[href], button:not([disabled]), input:not([disabled]), select:not([disabled]), textarea:not([disabled])')];
      const visibleOnes = candidates.filter((element) => element.getClientRects().length > 0);
      const target = edge === 'first' ? visibleOnes[0] : visibleOnes.at(-1);
      if (target === undefined) return false;
      target.focus();
      return document.activeElement === target;
    },
    [scope, which],
  );
  expect(found, `a focusable control in ${scope}`).toBe(true);
}
async function walk(page: Page, scope: string, label: string, direction: 'forward' | 'backward'): Promise<{ stops: number; hidden: string[]; minVisible: number }> {
  await focusEdge(page, scope, direction === 'forward' ? 'first' : 'last');
  const seen = new Set<number>();
  const hidden: string[] = [];
  let minVisible = 25;
  let last: number | undefined;
  let repeats = 0;
  const key = direction === 'forward' ? 'Tab' : 'Shift+Tab';
  for (let index = 0; index < 200; index += 1) {
    const stop = await focusedStop(page, scope);
    if (stop === null || !stop.inside) break;
    if (stop.key === last && repeats < 6) {
      repeats += 1;
    } else {
      if (stop.key !== last && seen.has(stop.key)) break;
      repeats = 0;
      last = stop.key;
      seen.add(stop.key);
      minVisible = Math.min(minVisible, stop.visible);
      if (stop.visible === 0) hidden.push(`${label} ${direction}: ${stop.name} under ${stop.covers.join(', ')}`);
    }
    await page.keyboard.press(key);
  }
  return { stops: seen.size, hidden, minVisible };
}

async function barGeometry(page: Page): Promise<string> {
  return page.evaluate(() => {
    const rect = (selector: string) => {
      const element = document.querySelector(selector);
      if (element === null) return null;
      const box = element.getBoundingClientRect();
      return { top: Math.round(box.top * 10) / 10, bottom: Math.round(box.bottom * 10) / 10, position: getComputedStyle(element).position };
    };
    return JSON.stringify({ shell: rect('.shell-bar'), share: rect('.share-bar,[data-share-bar]'), scrollY: window.scrollY });
  });
}

test('AX-09 Complete text contrast, light', async ({ page, adminSeed, builtServer, signInPageAs }, testInfo) => {
  const project = testInfo.project.name;
  const world = await newWorld(adminSeed, builtServer);
  await page.emulateMedia({ colorScheme: 'light', reducedMotion: 'reduce' });
  await signInPageAs(world.me.account, '#/timesheet');
  await expect(page.locator('[data-day]')).toHaveCount(14);
  const measure = async (label: string) => {
    const status = page.locator(`[data-day="${world.complete}"] .status-complete`).first();
    await expect(status).toHaveText(/Complete/);
    await page.waitForTimeout(800); // the shared transition (300 ms) must have finished before the backdrop is read
    const rendered = await status.evaluate((element) => {
      const view = element.ownerDocument.defaultView;
      const chain = [element];
      for (let node = element.parentElement; node; node = node.parentElement) chain.push(node);
      return { colour: view?.getComputedStyle(element).color ?? '', backdrops: chain.map((node) => view?.getComputedStyle(node).backgroundColor ?? '') };
    });
    const background = composite(rendered.backdrops);
    const ratio = contrast(over(parseColour(rendered.colour), background), background);
    note('obs-ax09.txt', `${project} light ${label} row: "Complete" ${rendered.colour} on ${hex(background)} = ${ratio.toFixed(2)}:1`);
    expect(ratio).toBeGreaterThanOrEqual(4.5);
  };
  await measure('normal');
  await page.getByRole('button', { name: 'Change several days' }).click();
  await page.getByRole('checkbox', { name: `Select ${world.complete}` }).check();
  await measure('selected');
});

test.describe('AX-10 phone', () => {
  test.skip(({ isMobile }) => !isMobile, 'phone only');
  for (const size of [{ width: 390, height: 844 }, { width: 320, height: 640 }]) {
    test(`batch mode at ${size.width}`, async ({ page, adminSeed, builtServer, signInPageAs }) => {
      test.setTimeout(120_000);
      const world = await newWorld(adminSeed, builtServer);
      await page.setViewportSize(size);
      await page.emulateMedia({ reducedMotion: 'reduce' });
      await signInPageAs(world.me.account, '#/timesheet');
      await expect(page.locator('[data-day]')).toHaveCount(14);
      await page.getByRole('button', { name: 'Change several days' }).click();
      await expect(page.getByRole('group', { name: 'Batch category edit' })).toBeVisible();
      const result = await page.evaluate(() => {
        const rows = [...document.querySelectorAll('tr.sheet-row[data-day]')].map((row) => {
          const box = (selector: string) => {
            const rect = row.querySelector(selector)?.getBoundingClientRect();
            return rect === undefined ? null : { left: rect.left, right: rect.right, width: rect.width, height: rect.height };
          };
          return { day: row.getAttribute('data-day'), button: box('button.sheet-day-button'), cell: box('td.t-day'), label: box('td.t-label'), pick: box('label.pick') };
        });
        const small = [...document.querySelectorAll('main button, main a[href], main input:not([type="hidden"]), main select, main label.pick')]
          .filter((element) => (element as HTMLElement).getClientRects().length > 0)
          .map((element) => ({ tag: element.tagName.toLowerCase(), text: (element.getAttribute('aria-label') ?? element.textContent ?? '').trim().slice(0, 30), w: element.getBoundingClientRect().width, h: element.getBoundingClientRect().height }))
          .filter((item) => item.w < 43.5 || item.h < 43.5);
        return { rows, small, scroll: document.documentElement.scrollWidth, inner: window.innerWidth, count: document.querySelectorAll('main button, main a[href], main input, main select').length };
      });
      const problems: string[] = [];
      for (const { day, button, cell, label, pick } of result.rows) {
        if (button === null || cell === null || label === null || pick === null) {
          problems.push(`${day}: missing part`);
          continue;
        }
        if (button.right > cell.right + 0.5) problems.push(`${day}: button ${(button.right - cell.right).toFixed(1)}px outside its cell`);
        if (button.right > label.left + 0.5) problems.push(`${day}: button ${(button.right - label.left).toFixed(1)}px over the label cell`);
        if (pick.width < 44 || pick.height < 44) problems.push(`${day}: selection target ${pick.width}x${pick.height}`);
      }
      note('obs-ax10.txt', `${size.width}px batch mode: rows ${result.rows.length}, problems ${problems.length}, scrollWidth ${result.scroll} vs innerWidth ${result.inner}, controls ${result.count}, targets under 44x44: ${result.small.length} ${JSON.stringify(result.small.slice(0, 6))}`);
      expect(problems).toEqual([]);
      expect(result.scroll).toBeLessThanOrEqual(result.inner);
      expect(result.rows).toHaveLength(14);
      if (size.width === 320) await page.screenshot({ path: screenshotPath('regate5-batch-mode-w320-synthetic.png'), animations: 'disabled' });
    });
  }
});
