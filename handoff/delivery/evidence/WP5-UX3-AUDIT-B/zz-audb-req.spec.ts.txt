function rec(item: Record<string, unknown>) {
  const line = JSON.stringify({ build: BUILD, ...item });
  console.log(line);
  if (OUT !== '') appendFileSync(join(OUT, `req-${BUILD}.jsonl`), `${line}\n`);
}

interface Brk {
  date: string;
  time: string;
  minutes: number;
  /** What a person types as the end in the old form. */
  endDate: string;
  endTime: string;
}

/** Answers to the prompts, in order: old build (server answers) and new build (row checks). */
type Answer = { kind: 'radio'; name: string } | { kind: 'button'; name: string };

interface Write {
  method: string;
  path: string;
  status: number;
  body: unknown;
}

function watch(page: Page): Write[] {
  const writes: Write[] = [];
  page.on('response', (response) => {
    const request = response.request();
    const path = new URL(request.url()).pathname;
    if (!['POST', 'PUT'].includes(request.method())) return;
    if (!/^\/api\/(days\/[^/]+\/sessions|sessions\/[^/]+|clock\/out)$/.test(path)) return;
    writes.push({ method: request.method(), path, status: response.status(), body: request.postDataJSON() as unknown });
  });
  return writes;
}

async function openEditor(page: Page, workDate: string): Promise<Locator> {
  await page.getByLabel('Open a day').fill(workDate);
  await page.getByRole('button', { name: 'Open day', exact: true }).click();
  const editor = page.getByRole('dialog', { name: /Day editor/ });
  await expect(editor).toBeVisible();
  await expect(editor.getByRole('heading', { name: 'Figures (computed by the server)' })).toBeVisible();
  const reason = editor.getByLabel('Reason for editing an old or finalized period');
  if ((await reason.count()) > 0) await reason.first().fill(REASON);
  return editor;
}

async function closeEditor(page: Page) {
  const editor = page.getByRole('dialog', { name: /Day editor/ });
  if ((await editor.count()) > 0) {
    await editor.getByRole('button', { name: 'Close', exact: true }).first().click();
    await expect(editor).toHaveCount(0);
  }
}

async function stored(person: Person, date: string) {
  const day = await person.api.dayView(date);
  return day.sessions.map((s) => ({
    start_utc: s.start_utc,
    end_utc: s.end_utc,
    input_zone: s.input_zone,
    breaks_confirmed: s.breaks_confirmed,
    breaks: s.breaks.map((b) => [b.start_utc, b.end_utc]),
    source: s.source,
  }));
}

interface Times {
  zone: string;
  startDate: string;
  startTime: string;
  endDate: string;
  endTime: string;
}

async function fillTimes(form: Locator, times: Partial<Times>) {
  if (times.zone !== undefined) await form.getByLabel('Input zone').fill(times.zone);
  if (times.startDate !== undefined) await form.getByLabel('Start date', { exact: true }).fill(times.startDate);
  if (times.startTime !== undefined) await form.getByLabel('Start time', { exact: true }).fill(times.startTime);
  if (times.endDate !== undefined) await form.getByLabel('End date', { exact: true }).fill(times.endDate);
  if (times.endTime !== undefined) await form.getByLabel('End time', { exact: true }).fill(times.endTime);
}

/* ---- old build (a643794) ---------------------------------------------------------------- */

async function oldRows(form: Locator, rows: readonly Brk[]) {
  for (const [i, row] of rows.entries()) {
    await form.getByRole('button', { name: 'Add break', exact: true }).click();
    const n = i + 1;
    await form.getByLabel(`Break ${n} from date`, { exact: true }).fill(row.date);
    await form.getByLabel(`Break ${n} from time`, { exact: true }).fill(row.time);
    await form.getByLabel(`Break ${n} to date`, { exact: true }).fill(row.endDate);
    await form.getByLabel(`Break ${n} to time`, { exact: true }).fill(row.endTime);
  }
}

async function oldSave(page: Page, editor: Locator, form: Locator, label: string, answers: readonly Answer[]) {
  await form.getByRole('button', { name: label, exact: true }).click();
  for (const answer of answers) {
    if (answer.kind === 'radio') {
      const radio = form.getByRole('radio', { name: answer.name, exact: true });
      await expect(radio).toBeVisible();
      await radio.check();
    } else {
      const button = form.getByRole('button', { name: answer.name, exact: true });
      await expect(button).toBeVisible();
      await button.click();
    }
    await form.getByRole('button', { name: label, exact: true }).click();
  }
  await expect(editor.getByText('Session saved.')).toBeVisible();
}

/* ---- new build (target) ------------------------------------------------------------------ */

async function newRemoveAll(form: Locator) {
  const rows = form.locator('[data-break]');
  for (let count = await rows.count(); count > 0; count -= 1) {
    await form.getByRole('button', { name: `Remove break ${count}`, exact: true }).click();
  }
  await expect(rows).toHaveCount(0);
}

async function newRows(form: Locator, rows: readonly Brk[]) {
  for (const [i, row] of rows.entries()) {
    await form.getByRole('button', { name: 'Add break', exact: true }).click();
    const n = i + 1;
    const date = form.getByLabel(`Break ${n} date`, { exact: true });
    if ((await date.count()) > 0) await date.fill(row.date);
    await form.getByLabel(`Break ${n} start time`, { exact: true }).fill(row.time);
    await form.getByLabel(`Break ${n} minutes`, { exact: true }).fill(String(row.minutes));
  }
}

async function newSave(page: Page, form: Locator, answers: readonly Answer[], notice: string | null) {
  await form.getByRole('button', { name: 'Save', exact: true }).click();
  for (const answer of answers) {
    if (answer.kind === 'radio') {
      const radio = form.getByRole('radio', { name: answer.name, exact: true });
      await expect(radio).toBeVisible();
      await radio.check();
    } else {
      const button = form.getByRole('button', { name: answer.name, exact: true });
      await expect(button).toBeVisible();
      await button.click();
    }
    await form.getByRole('button', { name: 'Save', exact: true }).click();
  }
  if (notice !== null) {
    const box = page.getByRole('alertdialog', { name: 'No breaks listed' });
    await expect(box).toBeVisible();
    await box.getByRole('button', { name: notice, exact: true }).click();
  }
  const confirm = page.getByRole('alertdialog', { name: 'Save this session?' });
  await expect(confirm).toBeVisible();
  const facts = (await confirm.locator('.confirm-facts').innerText()).replace(/\s+/g, ' ');
  const message = await confirm.locator('.confirm-message').innerText();
  await confirm.getByRole('button', { name: 'Save', exact: true }).click();
  await expect(confirm).toHaveCount(0);
  await expect(form).toHaveCount(0);
  return { facts, message };
}

function lc(date: string, time: string) {
  return { local: `${date}T${time}`, zone: LA };
}

            input_zone: LA,
            breaks_confirmed: false,
            breaks: item.seed === 'manual-row' ? [{ start: lc(todayLocal, breakAt), end: lc(todayLocal, breakEnd), counts_as_work: false }] : [],
          },
          201,
        );
        await page.goto('/');
      }
      const seeded = await stored(person, todayLocal);
      await page.getByRole('button', { name: 'Check out', exact: true }).click();
      const dialog = page.getByRole('dialog', { name: 'Check out' });
      await expect(dialog).toBeVisible();
      const submit = dialog.locator('button[type="submit"]');
      const ui: Record<string, unknown> = { defaultLabel: await submit.innerText(), defaultRows: await dialog.locator('[data-break], .break-row').count() };
      if (item.act === 'box') {
        if (isNew) await dialog.getByRole('checkbox', { name: 'Confirm no breaks taken' }).check();
        else await dialog.getByRole('radio', { name: /No breaks taken/ }).check();
      }
      if (item.act === 'row') {
        if (isNew) {
          await dialog.getByRole('button', { name: 'Add break', exact: true }).click();
          await dialog.getByLabel('Break 1 date', { exact: true }).fill(todayLocal);
          await dialog.getByLabel('Break 1 start time', { exact: true }).fill(breakAt);
          await dialog.getByLabel('Break 1 minutes', { exact: true }).fill('15');
        } else {
          await dialog.getByRole('radio', { name: /Breaks confirmed as listed/ }).check();
          await dialog.getByRole('button', { name: 'Add break', exact: true }).click();
          await dialog.getByLabel('Break 1 from date', { exact: true }).fill(todayLocal);
          await dialog.getByLabel('Break 1 from time', { exact: true }).fill(breakAt);
          await dialog.getByLabel('Break 1 to date', { exact: true }).fill(todayLocal);
          await dialog.getByLabel('Break 1 to time', { exact: true }).fill(breakEnd);
        }
      }
      if (item.act === 'remove-saved') {
        await dialog.getByRole('button', { name: 'Remove break 1', exact: true }).click();
        ui.afterRemoveText = (await dialog.innerText()).replace(/\s+/g, ' ').slice(0, 900);
      }
      ui.submitLabel = await submit.innerText();
      await submit.click();
      await expect(dialog).toHaveCount(0);
      await expect.poll(async () => (await person.api.dayView(todayLocal)).sessions.every((s) => s.end_utc !== null)).toBe(true);
      rec({ case: item.id, date: todayLocal, seeded, writes: writes.slice(before), stored: await stored(person, todayLocal), ui });
    } catch (caught) {
      errors.push(`${item.id}: ${String(caught).slice(0, 400)}`);
      rec({ case: item.id, error: String(caught).slice(0, 800), writes: writes.slice(before) });
      await page.keyboard.press('Escape').catch(() => undefined);
    }
    // End any running session left by a failed step, then clear the day.
    const open = (await person.api.dayView(todayLocal)).sessions.find((s) => s.end_utc === null);
    if (open !== undefined) await person.api.call('POST', '/api/clock/out', { breaks_confirmed: false, expected_version: open.version }).catch(() => undefined);
    await person.api.clearSessions(todayLocal, REASON);
  }
  rec({ case: 'errors', errors });
  expect(errors).toEqual([]);
});
