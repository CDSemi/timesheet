// src/server/ops/backup.ts as it stood for the red run (RED-FIRST STUB, replaced by the implementation):
// it copies the live main file only (no WAL, no online backup API), lists no files and records no status.
export async function createBackup(request: BackupRequest): Promise<BackupResult> {
  const name = `timesheet-backup-stub-${Date.now()}`;
  const directory = join(resolve(request.targetDir), name);
  mkdirSync(directory, { recursive: true });
  copyFileSync(request.databasePath, join(directory, BACKUP_DATABASE_NAME));
  await request.hooks?.afterSnapshot?.();
  const manifest = buildManifest({ appVersion: '0.0.0', schemaVersion: 0, createdAt: nowUtc(request.clock), database: { sha256: '', sizeBytes: 0 }, files: [] });
  writeFileSync(join(directory, MANIFEST_FILE_NAME), serializeManifest(manifest));
  return { directory, name, manifest, durationMs: 0, statusRecorded: false };
}
